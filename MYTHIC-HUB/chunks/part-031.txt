function CustomSounds.asset(state, name)
	if state.assets[name] then return state.assets[name] end
	local environment = state.context.environment
	if type(environment.writefile) ~= "function" or type(environment.getcustomasset) ~= "function" then return "" end
	if not table.find(LOCAL_SOUNDS, name) then return "" end

	pcall(function()
		if type(environment.isfolder) == "function" and not environment.isfolder("PulseHub") then environment.makefolder("PulseHub") end
		if type(environment.isfolder) == "function" and not environment.isfolder("PulseHub/mm2sounds") then environment.makefolder("PulseHub/mm2sounds") end
	end)
	local path = "PulseHub/mm2sounds/" .. name .. ".ogg"
	if not (type(environment.isfile) == "function" and environment.isfile(path)) then
		local data = rawget(environment.getgenv(), "__pulseData")
		local encoded = data and data.take and data.take("snd_" .. name)
		if encoded then pcall(function() environment.writefile(path, buffer.tostring(OfflineFiles._decodeBuffer(encoded))) end) end
	end
	local ok, asset = pcall(environment.getcustomasset, path)
	if not ok or type(asset) ~= "string" or #asset == 0 then return "" end
	state.assets[name] = asset
	return asset
end

function CustomSounds.resolve(state, catalog, selection)
	for _, entry in catalog do
		if entry[1] == selection then return entry[2] end
	end
	if type(selection) == "string" and selection:sub(1, 1) == "~" then return CustomSounds.asset(state, selection:sub(2)) end
	return ""
end

function CustomSounds.play(state, asset)
	if not asset or asset == "" then return end
	local context, environment = state.context, state.context.environment
	local sound = environment.Instance.new("Sound")
	sound.SoundId = asset
	sound.Volume = (tonumber(state.settings.sndVol) or 60) / 100 * 3
	sound.PlayOnRemove, sound.Parent = false, context.soundService
	pcall(function() sound:Play() end)
	environment.task.delay(8, function() pcall(function() sound:Destroy() end) end)
	pcall(function() context.connect(sound.Ended, function() pcall(function() sound:Destroy() end) end) end)
end

function CustomSounds.shouldMute(state, sound)
	local name = string.lower(sound.Name)
	if state.settings.shotSnd ~= "Off" and (name == "gunshot" or name == "reload" or name == "gunkill") then return true end
	return state.settings.killSnd ~= "Off" and name == "gunkill"
end

function CustomSounds.watch(state, sound)
	if not sound:IsA("Sound") then return end
	if CustomSounds.shouldMute(state, sound) then
		if not state.muted[sound] then state.muted[sound] = {vol = sound.Volume} end
		if not state.watched[sound] then
			state.watched[sound] = true
			pcall(function()
				state.context.connect(sound:GetPropertyChangedSignal("Volume"), function()
					local saved = state.muted[sound]
					if saved and sound.Volume ~= 0 then saved.vol, sound.Volume = sound.Volume, 0 end
				end)
			end)
		end
		if sound.Volume ~= 0 then pcall(function() sound.Volume = 0 end) end
		return
	end

	local saved = state.muted[sound]
	if saved then
		state.muted[sound] = nil
		pcall(function() sound.Volume = saved.vol end)
	end
end

function CustomSounds.refresh(state)
	for sound, saved in state.muted do
		if not sound.Parent then state.muted[sound], state.watched[sound] = nil, nil; continue end
		if not CustomSounds.shouldMute(state, sound) then
			state.muted[sound] = nil
			pcall(function() sound.Volume = saved.vol end)
		end
	end
	for sound in state.watched do if not sound.Parent then state.watched[sound] = nil end end
	local gun = SheriffRuntime.gun(state.context)
	if not gun then return end

	if state.gun ~= gun then
		if state.gunConnection then pcall(function() state.gunConnection:Disconnect() end) end
		state.gun = gun
		state.gunConnection = state.context.connect(gun.DescendantAdded, function(sound)
			if sound:IsA("Sound") then CustomSounds.watch(state, sound) end
		end)
	end
	for _, sound in gun:GetDescendants() do if sound:IsA("Sound") then CustomSounds.watch(state, sound) end end
end

function CustomSounds.characterOf(environment, instance)
	local character = environment.typeof(instance) == "Instance" and instance or nil
	while character and not (character:IsA("Model") and character:FindFirstChildOfClass("Humanoid")) do character = character.Parent end
	return character
end

function CustomSounds.onFired(state, tool, origin, endpoint, hit)
	local context, environment = state.context, state.context.environment
	if CustomSounds.characterOf(environment, tool) ~= context.player.Character then return end
	local now, settings = environment.os.clock(), state.settings
	if settings.shotSnd ~= "Off" and now - state.lastShot > 0.15 then
		state.lastShot = now
		CustomSounds.play(state, CustomSounds.resolve(state, SHOT_SOUNDS, settings.shotSnd))
	end
	if settings.killSnd ~= "Off" and environment.typeof(hit) == "Instance" then
		local victim = CustomSounds.characterOf(environment, hit)
		if victim and victim ~= context.player.Character and context.players:GetPlayerFromCharacter(victim) then
			environment.task.delay(0.06, function() CustomSounds.play(state, CustomSounds.resolve(state, KILL_SOUNDS, settings.killSnd)) end)
		end
	end
	pcall(CustomSounds.refresh, state)
end

function GameSounds.cap(state, sound, duration)
	local environment = state.context.environment
	state.capTokens[sound] = (state.capTokens[sound] or 0) + 1
	local token = state.capTokens[sound]
	environment.task.delay(duration, function()
		if state.capTokens[sound] ~= token or not sound.Parent or not sound.IsPlaying then return end
		local volume = sound.Volume
		pcall(function() state.context.tweenService:Create(sound, environment.TweenInfo.new(0.6), {Volume = 0}):Play() end)
		environment.task.wait(0.65)
		if state.capTokens[sound] ~= token or not sound.Parent then return end
		pcall(function() sound:Stop() end)
		sound.Volume = volume
	end)
end

function GameSounds.apply(state, sound)
	local original = state.original[sound]
	if not original then return end
	local selected = state.cfg[original.kind]
	if selected == "Game" or selected == nil then
		if sound.SoundId ~= original.id then sound.SoundId = original.id end
		sound.Volume = original.vol
		sound:SetAttribute("PulseOrigSound", nil)
		sound:SetAttribute("PulseOrigVol", nil)
		return
	end

	sound:SetAttribute("PulseOrigSound", original.id)
	sound:SetAttribute("PulseOrigVol", original.vol)
	if selected == "Off" then sound.Volume = 0; return end
	local soundOption = GameSounds.find(GAME_SOUND_LIBRARY_OF[original.kind], selected)
	if not soundOption then return end
	if sound.SoundId ~= soundOption[2] then sound.SoundId = soundOption[2] end
	sound.Volume = original.vol * state.cfg.vol / 100
end

function GameSounds.track(state, sound)
	if not sound:IsA("Sound") then return end
	if state.original[sound] then GameSounds.apply(state, sound); return end
	local id = sound:GetAttribute("PulseOrigSound") or sound.SoundId
	local kind = GAME_SOUND_KINDS[GameSounds.id(id) or ""]
	if not kind then return end

	state.original[sound] = {id = id, vol = tonumber(sound:GetAttribute("PulseOrigVol")) or sound.Volume, kind = kind}
	state.context.connect(sound.Played, function()
		local original = state.original[sound]
		local selected = original and GameSounds.find(GAME_SOUND_LIBRARY_OF[original.kind], state.cfg[original.kind])
		if selected and selected[3] then GameSounds.cap(state, sound, selected[3]) end
	end)
	GameSounds.apply(state, sound)
end

function GameSounds.refresh(state)
	for sound in state.original do
		if sound.Parent then pcall(GameSounds.apply, state, sound); continue end
		state.original[sound] = nil
	end
end

	sound.Name, sound.SoundId, sound.Volume = "PulseEndMusic", GameSounds.asset(state, asset), 0.4 * state.cfg.vol / 100

	end
	sound.Parent, state.own = context.soundService, sound
	local function destroy() pcall(function() sound:Destroy() end) end
	sound.Ended:Once(destroy)
	sound:Play()
	environment.task.delay(10, destroy)
	return sound
end

	pcall(function() sound.Volume = 0; sound:Stop() end)
	state.context.environment.task.defer(function() pcall(function() sound:Stop() end) end)
	if selected then GameSounds.playOwn(state, selected[2], false) end
end

function GameSounds.onEndSound(state, sound)
	if not sound:IsA("Sound") then return end
	local now = state.context.environment.os.clock()
	local music = sound.SoundGroup ~= nil and sound.SoundGroup.Name == "Music"
	if not (now - state.wonAt < 2 and music) and not GAME_END_SOUND_IDS[GameSounds.id(sound.SoundId) or ""] then return end
	state.endSound, state.endAt = sound, now
	if now - state.wonAt < 3 then state.endPending = false; pcall(GameSounds.playEnd, state, sound); return end
	state.endPending = true
end

function GameSounds.onVictory(state, role, reason)
	local results = GAME_WIN_RESULTS[tostring(role)]
	state.won = results ~= nil and results[tostring(reason)] == true
	state.wonAt = state.context.environment.os.clock()
	local sound = state.endSound
	if state.endPending and sound and sound.Parent and state.wonAt - state.endAt < 1.5 then
		state.endPending = false
		pcall(GameSounds.playEnd, state, sound)
	end
end

local SkinModel = {}

function SkinModel.decode(environment, value)
	if type(value) ~= "table" then return value end
	local kind = value[1]
	if kind == "V" then return environment.Vector3.new(value[2], value[3], value[4]) end
	if kind == "C" then return environment.Color3.new(value[2], value[3], value[4]) end
	if kind == "F" then return environment.CFrame.new(table.unpack(value, 2, 13)) end
	if kind == "NR" then return environment.NumberRange.new(value[2], value[3]) end
	if kind == "E" then
		local ok, item = pcall(function() return environment.Enum[value[2]][value[3]] end)
		return ok and item or nil
	end
	if kind == "CS" then
		local points = {}
		for index = 2, #value do
			local point = value[index]
			table.insert(points, environment.ColorSequenceKeypoint.new(point[1], environment.Color3.new(point[2], point[3], point[4])))
		end
		local ok, sequence = pcall(environment.ColorSequence.new, points)
		return ok and sequence or nil
	end
	if kind == "NS" then
		local points = {}
		for index = 2, #value do
			local point = value[index]
			table.insert(points, environment.NumberSequenceKeypoint.new(point[1], point[2], point[3] or 0))
		end
		local ok, sequence = pcall(environment.NumberSequence.new, points)
		return ok and sequence or nil
	end
	return nil
end

function SkinModel.frame(environment, values)
	return environment.CFrame.new(table.unpack(values, 1, 12))
end

function SkinModel.mesh(state, meshId)
	if state.meshes[meshId] then return state.meshes[meshId]:Clone() end
	while state.loading[meshId] do state.context.environment.task.wait() end
	if state.meshes[meshId] then return state.meshes[meshId]:Clone() end

	state.loading[meshId] = true
	local environment = state.context.environment
	local ok, result = pcall(function()
		return state.context.insertService:CreateMeshPartAsync(meshId, environment.Enum.CollisionFidelity.Box, environment.Enum.RenderFidelity.Automatic)
	end)
	state.loading[meshId] = nil
	if not ok or not result then return nil end
	state.meshes[meshId] = result
	return result:Clone()
end

function SkinModel.node(state, data, root, references, unresolved, options)
	if options and options.skip and options.skip(data) then return nil end
	local context, environment = state.context, state.context.environment
	local object
	if data.c == "MeshPart" then
		object = SkinModel.mesh(state, data.p.MeshId or "")
		if not object then object = environment.Instance.new("Part"); object.Transparency = 1 end
	end
	if not object then
		local ok, created = pcall(environment.Instance.new, data.c)
		if not ok or not created then return nil end
		object = created
	end
	object.Name = data.n
	for property, value in data.p or {} do
		if property == "MeshId" and data.c == "MeshPart" then continue end
		if type(value) == "table" and value[1] == "R" then
			table.insert(unresolved, {object, property, value[2]})
			continue
		end
		local decoded = SkinModel.decode(environment, value)
		if decoded ~= nil then pcall(function() object[property] = decoded end) end
	end
	if data.i then references[data.i] = object end
	if object:IsA("BasePart") then
		object.Anchored, object.CanCollide, object.CanQuery, object.CanTouch, object.Massless = false, false, false, false, true
		pcall(function() object.AudioCanCollide = false end)
		if root and data.r then
			local weld = environment.Instance.new("Weld")
			weld.Name, weld.Part0, weld.Part1, weld.C0, weld.Parent = "PW", root, object, SkinModel.frame(environment, data.r), object
		end
	end
	for key, value in data.a or {} do pcall(object.SetAttribute, object, key, SkinModel.decode(environment, value)) end
	for _, tag in data.t or {} do pcall(context.collection.AddTag, context.collection, object, tag) end
	local parentRoot = root or object
	for _, child in data.k or {} do
		local built = SkinModel.node(state, child, parentRoot, references, unresolved, options)
		if built then built.Parent = object end
	end
	return object
end

function SkinModel.build(state, data, options)
	local references, unresolved = {}, {}
	local root = SkinModel.node(state, data, nil, references, unresolved, options)
	if not root then return nil end
	for _, reference in unresolved do
		local target = references[reference[3]]
		if target then pcall(function() reference[1][reference[2]] = target end) end
	end
	return root
end

function SkinModel.hasChroma(data)
	for _, tag in data.t or {} do
		if tag == "ChromaPart" or tag == "ChromaDecal" or tag == "ChromaFire" then return true end
	end
	for _, child in data.k or {} do if SkinModel.hasChroma(child) then return true end end
	return false
end

function SkinModel.applyChroma(state, skin, model)
	if not skin.ch or SkinModel.hasChroma(skin.d) then return end
	for _, object in model:GetDescendants() do
		if object:IsA("Decal") and object.Name:lower():find("chroma") then pcall(state.context.collection.AddTag, state.context.collection, object, "ChromaDecal") end
		if object:IsA("Fire") then pcall(state.context.collection.AddTag, state.context.collection, object, "ChromaFire") end
	end
end

local SkinCatalog = {}
local SKIN_RARITY_ORDER = {Pulse = 0, Ancient = 1, Unique = 2, Godly = 3, Classic = 4, Legendary = 5, Rare = 6}

function SkinCatalog.list(database)
	local result = {}
	for key, entry in database do
		local name = entry.nm or key
		table.insert(result, {key = key, name = name, short = entry.ch and name:gsub("^Chroma ", "") or name,
			chroma = entry.ch == true, type = entry.t, rarity = entry.r, img = entry.img})
	end
	table.sort(result, function(first, second)
		local firstRank = (SKIN_RARITY_ORDER[first.rarity] or 9) + (first.chroma and 0.5 or 0)
		local secondRank = (SKIN_RARITY_ORDER[second.rarity] or 9) + (second.chroma and 0.5 or 0)
		if firstRank ~= secondRank then return firstRank < secondRank end
		if first.type ~= second.type then return first.type < second.type end
		return first.name:lower() < second.name:lower()
	end)
	return result
end

function SkinCatalog.filter(items, kind, rarity, query)
	local result = {}
	query = (query or ""):gsub("^%s+", ""):gsub("%s+$", ""):lower()
	for _, item in items do
		if kind ~= "All" and item.type ~= kind then continue end
		if rarity ~= "Any" and not (rarity == "Chroma" and item.chroma) and not (item.rarity == rarity and not item.chroma) then continue end
		if query ~= "" and not (tostring(item.name) .. " " .. tostring(item.key)):lower():find(query, 1, true) then continue end
		table.insert(result, item)
	end
	return result
end

function PulseSkin.template(state, asset)
	if state.templates[asset] ~= nil then return state.templates[asset] or nil end
	local environment = state.context.environment
	while state.loading[asset] do environment.task.wait() end
	if state.templates[asset] ~= nil then return state.templates[asset] or nil end
	state.loading[asset] = true
	local result
	local ok, objects = pcall(function() return environment.game:GetObjects("rbxassetid://" .. asset) end)
	if ok and type(objects) == "table" and #objects > 0 then
		result = environment.Instance.new("Folder")
		for _, root in objects do
			for _, object in root:GetDescendants() do
				if object:IsA("LuaSourceContainer") then pcall(function() object.Disabled = true end) end
			end
			for _, object in root:GetDescendants() do
				if PulseSkin.isAny(object, PULSE_REMOVED) then pcall(function() object:Destroy() end) end
			end
			if root:IsA("LuaSourceContainer") then root:Destroy() else root.Parent = result end
		end
	end
	state.templates[asset], state.loading[asset] = result or false, nil
	return result
end

function PulseSkin.scaleSequence(environment, sequence, scale)
	local points = {}
	for _, point in sequence.Keypoints do table.insert(points, environment.NumberSequenceKeypoint.new(point.Time, point.Value * scale, point.Envelope * scale)) end
	return environment.NumberSequence.new(points)
end

function PulseSkin.scaleEffect(environment, object, scale)
	pcall(function()
		if object:IsA("ParticleEmitter") then
			object.Size = PulseSkin.scaleSequence(environment, object.Size, scale)
			object.Speed = environment.NumberRange.new(object.Speed.Min * scale, object.Speed.Max * scale)
			object.Acceleration *= scale
			return
		end
		if object:IsA("Beam") then
			object.Width0, object.Width1 = object.Width0 * scale, object.Width1 * scale
			object.CurveSize0, object.CurveSize1 = object.CurveSize0 * scale, object.CurveSize1 * scale
			return
		end
		if object:IsA("Trail") then object.WidthScale = PulseSkin.scaleSequence(environment, object.WidthScale, 1); return end
		if object:IsA("Light") then object.Range *= math.max(scale, 0.5); return end
		if object:IsA("Fire") then object.Size *= scale end
	end)
end

function PulseSkin.build(state, skin)
	local spec = skin.cx
	local template = spec and PulseSkin.template(state, spec.asset)
	if not template then return nil end
	local environment, scale = state.context.environment, spec.k
	local origin = SkinModel.frame(environment, spec.X)
	local root = environment.Instance.new("Part")
	root.Name, root.Size, root.Transparency = "PulseCX", environment.Vector3.new(0.2, 0.2, 0.2), 1
	root.CanCollide, root.CanQuery, root.CanTouch, root.Massless, root.CastShadow = false, false, false, true, false
	local mesh = environment.Instance.new("BlockMesh")
	mesh.Scale, mesh.Parent = environment.Vector3.zero, root
	local copy = template:Clone()
	local parts = {}
	for _, object in copy:GetDescendants() do if object:IsA("BasePart") then table.insert(parts, object) end end
	for _, part in parts do
		local visible = part.Transparency < 0.999 and math.max(part.Size.X, part.Size.Y, part.Size.Z) <= 30
		for _, child in part:GetChildren() do
			if child:IsA("DataModelMesh") and not (child:IsA("SpecialMesh") and child.MeshType == environment.Enum.MeshType.FileMesh) then
				local size = child.Scale
				if math.max(size.X, size.Y, size.Z) > 20 then visible = false end
			end
		end
		if not visible then
			for _, object in part:GetDescendants() do if PulseSkin.isAny(object, PULSE_EFFECTS) then visible = true; break end end
		end
		if not visible then continue end
		local relative = origin:ToObjectSpace(part.CFrame)
		for _, child in part:GetChildren() do
			if PulseSkin.isAny(child, {"JointInstance", "WeldConstraint", "NoCollisionConstraint", "RigidConstraint"}) then child:Destroy(); continue end
			if child:IsA("SpecialMesh") then
				if child.MeshType == environment.Enum.MeshType.FileMesh then child.Scale *= scale end
				child.Offset *= scale
				continue
			end
			if child:IsA("BlockMesh") or child:IsA("CylinderMesh") then child.Offset *= scale; continue end
			if child:IsA("Attachment") then child.Position *= scale end
		end
		part.Anchored, part.CanCollide, part.CanQuery, part.CanTouch, part.Massless, part.CastShadow = false, false, false, false, true, false
		pcall(function() part.AudioCanCollide = false end)
		part.Size *= scale
		local weld = environment.Instance.new("Weld")
		weld.Name, weld.Part0, weld.Part1, weld.C0, weld.Parent = "PW", root, part, environment.CFrame.new(relative.Position * scale) * relative.Rotation, part
		part.Parent = root
	end
	copy:Destroy()
	for _, object in root:GetDescendants() do
		if PulseSkin.isAny(object, PULSE_EFFECTS) then PulseSkin.scaleEffect(environment, object, scale) end
		if spec.glowOnly and PulseSkin.isAny(object, {"ParticleEmitter", "Fire", "Smoke", "Sparkles"}) then object:Destroy(); continue end
		if spec.burst and object:IsA("ParticleEmitter") then
			for _, name in spec.burst do
				if object.Name:find(name) then object:SetAttribute("PulseBurst", object.Rate); object.Enabled = false; break end
			end
		end
	end
	root:SetAttribute("PulseCX", true)
	return root
end

function PulseSkin.vector(environment, values)
	return environment.Vector3.new(values[1], values[2], values[3])
end

function PulseSkin.joint(character, name)
	local motor
	for _, object in character:GetDescendants() do
		if object.Name ~= name then continue end
		if object:IsA("AnimationConstraint") and object.Enabled and object.Attachment0 and object.Attachment1 then
			return {joint = object, c0 = object.Attachment0.CFrame, c1 = object.Attachment1.CFrame}
		end
		if object:IsA("Motor6D") and object.Enabled and object.Part0 and object.Part1 then
			motor = {joint = object, c0 = object.C0, c1 = object.C1}
		end
	end
	return motor
end

function PulseSkin.rotationBetween(environment, from, to)
	local dot = math.clamp(from:Dot(to), -1, 1)
	local axis = from:Cross(to)
	if axis.Magnitude >= 1e-6 then return environment.CFrame.fromAxisAngle(axis.Unit, math.acos(dot)) end
	if dot > 0 then return environment.CFrame.identity end
	local fallback = math.abs(from.X) < 0.9 and environment.Vector3.xAxis or environment.Vector3.yAxis
	return environment.CFrame.fromAxisAngle(from:Cross(fallback).Unit, math.pi)
end

function PulseSkin.solveArm(environment, arm, torso, shift, target, bend)
	local shoulder, elbow, wrist = arm.shoulder, arm.elbow, arm.wrist
	local initial = torso * shoulder.c0 * shoulder.c1:Inverse() + torso:VectorToWorldSpace(shift)
	local shoulderPosition = (initial * shoulder.c1).Position
	local upperLength = (elbow.c0.Position - shoulder.c1.Position).Magnitude
	local lowerLength = (wrist.c0.Position - elbow.c1.Position).Magnitude
	local direction = (target * wrist.c1).Position - shoulderPosition
	local distance = math.clamp(direction.Magnitude, math.abs(upperLength - lowerLength) + 0.02, upperLength + lowerLength - 0.002)
	local axis = direction.Magnitude > 1e-4 and direction.Unit or environment.Vector3.new(0, -1, 0)
	local wristPosition = shoulderPosition + axis * distance
	local along = (upperLength ^ 2 - lowerLength ^ 2 + distance ^ 2) / (2 * distance)
	local height = math.sqrt(math.max(upperLength ^ 2 - along ^ 2, 0))
	local perpendicular = bend - axis * bend:Dot(axis)
	perpendicular = perpendicular.Magnitude > 1e-4 and perpendicular.Unit or environment.Vector3.new(0, -1, 0)
	local elbowPosition = shoulderPosition + axis * along + perpendicular * height
	local currentElbow = (initial * elbow.c0).Position
	local cframe = environment.CFrame
	local upper = cframe.new(shoulderPosition) * PulseSkin.rotationBetween(environment, (currentElbow - shoulderPosition).Unit, (elbowPosition - shoulderPosition).Unit) * cframe.new(-shoulderPosition) * initial
	local lowerBase = upper * elbow.c0 * elbow.c1:Inverse()
	local elbowOrigin, currentWrist = (lowerBase * elbow.c1).Position, (lowerBase * wrist.c0).Position
	local lower = cframe.new(elbowOrigin) * PulseSkin.rotationBetween(environment, (currentWrist - elbowOrigin).Unit, (wristPosition - elbowOrigin).Unit) * cframe.new(-elbowOrigin) * lowerBase
	local handOrigin = (lower * wrist.c0).Position
	local hand = cframe.new(handOrigin - target:VectorToWorldSpace(wrist.c1.Position)) * target.Rotation
	shoulder.joint.Transform = shoulder.c0:Inverse() * torso:Inverse() * upper * shoulder.c1
	elbow.joint.Transform = elbow.c0:Inverse() * upper:Inverse() * lower * elbow.c1
	wrist.joint.Transform = wrist.c0:Inverse() * lower:Inverse() * hand * wrist.c1
end

function PulseSkin.arm(character, side)
	local arm = {shoulder = PulseSkin.joint(character, side .. "Shoulder"), elbow = PulseSkin.joint(character, side .. "Elbow"), wrist = PulseSkin.joint(character, side .. "Wrist")}
	if not arm.shoulder or not arm.elbow or not arm.wrist then return nil end
	return arm
end

function PulseSkin.hold(state, tool, entry, spec, weld)
	local rig = PULSE_RIGS[spec.rig]
	local character = state.context.player.Character
	local torso = character and character:FindFirstChild("UpperTorso")
	if not rig or not torso then return nil end

	local environment = state.context.environment
	local vector, cframe = environment.Vector3, environment.CFrame
	local grip = rig.grip
	local base = cframe.new(vector.new(grip[1] * torso.Size.X / 1.4, grip[2] * torso.Size.Y / 1.65, grip[3] * torso.Size.X / 1.4))
		* cframe.Angles(math.rad(rig.pitch), 0, 0) * cframe.new(-PulseSkin.vector(environment, spec.g))
	weld.Part0, weld.C0 = torso, base

	local step
	local function stop()
		if step then step:Disconnect(); step = nil end
	end
	local function start()
		stop()
		local current = state.context.player.Character
		local upper = current and current:FindFirstChild("UpperTorso")
		local hand = current and current:FindFirstChild("RightHand")
		local attachment = hand and hand:FindFirstChild("RightGripAttachment")
		if not upper or not attachment then return end
		local right, left = PulseSkin.arm(current, "Right"), PulseSkin.arm(current, "Left")
		if not right or not left then return end
		local gripFrame = cframe.new(PulseSkin.vector(environment, spec.g))
		local fore = spec.fore or {spec.g[1], spec.g[2] + 0.05, spec.g[3] - 0.7}
		local forward = -vector.new(0.45, 0.35, -0.82).Unit
		local up = (vector.yAxis - forward * vector.yAxis:Dot(forward)).Unit
		local foreFrame = cframe.fromMatrix(PulseSkin.vector(environment, fore), up, forward)
		local handFrame = attachment.CFrame:Inverse()
		step = state.context.connect(state.context.runService.Stepped, function()
			if not weld.Parent or weld.Part0 ~= upper or not upper.Parent or tool.Parent ~= current then return end
			local torsoFrame = upper.CFrame
			local visualFrame = torsoFrame * weld.C0
			pcall(PulseSkin.solveArm, environment, right, torsoFrame, PulseSkin.vector(environment, rig.right), visualFrame * gripFrame * handFrame, torsoFrame:VectorToWorldSpace(vector.new(0.7, -1, 0.6)))
			pcall(PulseSkin.solveArm, environment, left, torsoFrame, PulseSkin.vector(environment, rig.left), visualFrame * foreFrame, torsoFrame:VectorToWorldSpace(vector.new(-0.7, -1, 0.3)))
		end)
		table.insert(entry.connections, step)
	end
	table.insert(entry.connections, state.context.connect(tool.Equipped, start))
	table.insert(entry.connections, state.context.connect(tool.Unequipped, stop))
	if tool.Parent == character then start() end
	return base
end

function PulseSkin.easeBack(value)
	return 1 + 2.70158 * (value - 1) ^ 3 + 1.70158 * (value - 1) ^ 2
end

function PulseSkin.easeOut(value)
	return 1 - (1 - value) ^ 3
end

local PULSE_DRAWS = {
	iai = {0.42, function(cf, value, pivot)
		local remaining = 1 - PulseSkin.easeBack(value)
		return pivot * cf.Angles(math.rad(-20) * remaining, math.rad(-150) * remaining, math.rad(-70) * remaining) * pivot:Inverse()
	end},
	twirl = {0.62, function(cf, value, pivot)
		local remaining, back = 1 - PulseSkin.easeOut(value), 1 - PulseSkin.easeBack(value)
		return pivot * cf.Angles(math.rad(-50) * back, 0, 0) * cf.Angles(0, -4 * math.pi * remaining, 0) * pivot:Inverse()
	end},
	toss = {0.58, function(cf, value, pivot)
		return cf.new(0, 4.6 * value * (1 - value), 0) * pivot * cf.Angles(-2 * math.pi * (1 - PulseSkin.easeOut(value)), 0, 0) * pivot:Inverse()
	end},
	spin = {0.6, function(cf, value, pivot)
		local shifted = pivot * cf.new(0, 0.12, -0.12)
		return shifted * cf.Angles(4 * math.pi * (1 - PulseSkin.easeOut(value)), 0, 0) * shifted:Inverse()
	end},
	raise = {0.7, function(cf, value, pivot)
		local remaining = 1 - PulseSkin.easeBack(math.min(value / 0.7, 1))
		local settle = value > 0.72 and value < 0.95 and math.sin((value - 0.72) / 0.23 * math.pi) or 0
		return cf.new(0, -0.5 * remaining, 0.12 * settle) * pivot * cf.Angles(math.rad(-55) * remaining, math.rad(20) * remaining, 0) * pivot:Inverse()
	end},
	sideroll = {0.5, function(cf, value, pivot)
		local remaining = 1 - PulseSkin.easeBack(value)
		return pivot * cf.Angles(0, math.rad(35) * remaining, math.rad(-90) * remaining) * pivot:Inverse()
	end},
	aktilt = {0.7, function(cf, value, pivot)
		local remaining = 1 - PulseSkin.easeOut(math.min(value / 0.62, 1))
		local settle = value > 0.62 and math.sin((value - 0.62) / 0.38 * math.pi * 3) * (1 - value) * 2.2 or 0
		return pivot * cf.Angles(math.rad(20) * remaining + math.rad(5) * settle, 0, math.rad(75) * remaining) * pivot:Inverse()
	end},
	heavy = {0.8, function(cf, value, pivot)
		local eased = value < 0.5 and 4 * value ^ 3 or 1 - (-2 * value + 2) ^ 3 / 2
		local remaining = 1 - PulseSkin.easeBack(eased)
		return cf.new(0, -0.9 * remaining, 0.3 * remaining) * pivot * cf.Angles(math.rad(-75) * remaining, 0, math.rad(15) * remaining) * pivot:Inverse()
	end},
}

function PulseSkin.effects(environment, visual, skin)
	local spec = skin.cx
	local first, second = environment.Color3.fromRGB(table.unpack(spec.col)), environment.Color3.fromRGB(table.unpack(spec.col2))
	local function attachment(name, position)
		local result = environment.Instance.new("Attachment")
		result.Name, result.Position, result.Parent = name, PulseSkin.vector(environment, position), visual
		return result
	end
	local function sequence(a, b)
		return environment.NumberSequence.new({environment.NumberSequenceKeypoint.new(0, a), environment.NumberSequenceKeypoint.new(1, b)})
	end
	if skin.t == "Knife" then
		local trail = environment.Instance.new("Trail")
		trail.Name, trail.Attachment0, trail.Attachment1 = "PulseSlash", attachment("PulseT0", spec.trail[1]), attachment("PulseT1", spec.trail[2])
		trail.Lifetime, trail.MinLength, trail.LightEmission, trail.LightInfluence = 0.24, 0.02, 1, 0
		trail.Color, trail.Transparency, trail.WidthScale = environment.ColorSequence.new(first, second), sequence(0.1, 1), sequence(1, 0.25)
		trail.Enabled, trail.Parent = false, visual
		return trail
	end
	local muzzle = attachment("PulseMuzzle", spec.muzzle)
	local flash = environment.Instance.new("ParticleEmitter")
	flash.Name, flash.Texture, flash.Color = "PulseFlash", "rbxasset://textures/particles/sparkles_main.dds", environment.ColorSequence.new(second, first)
	flash.LightEmission, flash.LightInfluence, flash.Rate = 1, 0, 0
	flash.Lifetime, flash.Speed, flash.SpreadAngle = environment.NumberRange.new(0.08, 0.18), environment.NumberRange.new(6, 14), environment.Vector2.new(22, 22)
	flash.EmissionDirection, flash.Size, flash.Enabled, flash.Parent = environment.Enum.NormalId.Front, sequence(0.7, 0), false, muzzle
	local light = environment.Instance.new("PointLight")
	light.Name, light.Color, light.Range, light.Brightness, light.Shadows, light.Parent = "PulseMuzzleLight", first, 9, 0, false, muzzle
	return nil, flash, light
end

function PulseSkin.animate(state, tool, handle, visual, entry, skin)
	local context, environment = state.context, state.context.environment
	local spec, weld = skin.cx, visual:FindFirstChild("PulseGrip")
	if not spec or not weld then return end
	local base = PulseSkin.hold(state, tool, entry, spec, weld) or weld.C0
	local pivot = environment.CFrame.new(PulseSkin.vector(environment, spec.g))
	local trail, flash, light = PulseSkin.effects(environment, visual, skin)
	local parts = {}
	for _, object in visual:GetDescendants() do if object:IsA("BasePart") then table.insert(parts, object) end end
	local animation, slash = 0, 0
	local function set(frame)
		if weld.Parent then weld.C0 = frame end
	end
	local function play(duration, update, complete)
		animation += 1
		local token, start = animation, environment.os.clock()
		local connection
		connection = context.connect(context.runService.RenderStepped, function()
			if token ~= animation or not weld.Parent then connection:Disconnect(); return end
			local progress = math.clamp((environment.os.clock() - start) / duration, 0, 1)
			update(progress)
			if progress < 1 then return end
			connection:Disconnect()
			set(base)
			if complete then complete(token) end
		end)
		table.insert(entry.connections, connection)
	end
	local draw = PULSE_DRAWS[spec.draw] or PULSE_DRAWS[skin.t == "Knife" and "iai" or "toss"]
	local function drawWeapon()
		if skin.t == "Knife" then
			pcall(function()
				local sound = environment.Instance.new("Sound")
				sound.SoundId, sound.Volume, sound.PlaybackSpeed = "rbxasset://sounds/unsheath.wav", 0.4, spec.draw == "twirl" and 0.9 or 1.05
				sound.Parent = handle
				sound:Play()
				environment.game:GetService("Debris"):AddItem(sound, 2)
			end)
		end
		if trail and spec.draw == "iai" then trail.Enabled = true end
		play(draw[1], function(progress)
			set(base * draw[2](environment.CFrame, progress, pivot))
			local transparency = 1 - math.clamp(progress / 0.15, 0, 1)
			for _, part in parts do if part.Parent then part.LocalTransparencyModifier = transparency end end
		end, function(token)
			for _, part in parts do if part.Parent then part.LocalTransparencyModifier = 0 end end
			if trail then environment.task.delay(0.15, function() if animation == token and slash == 0 then trail.Enabled = false end end) end
		end)
	end
	local function activate()
		if trail then
			slash += 1
			local token = slash
			trail.Enabled = true
			environment.task.delay(0.42, function() if slash == token then slash = 0; trail.Enabled = false end end)
			return
		end
		pcall(function() flash:Emit(12) end)
		environment.task.spawn(function()
			local start = environment.os.clock()
			while light.Parent and environment.os.clock() - start < 0.14 do
				light.Brightness = 5 * (1 - (environment.os.clock() - start) / 0.14)
				context.runService.RenderStepped:Wait()
			end
			light.Brightness = 0
		end)
		local angle, distance = spec.rig and 7 or 16, spec.rig and 0.14 or 0.2
		play(0.24, function(progress)
			local amount = math.sin(math.min(progress * 4, 1) * math.pi / 2) * (1 - progress) ^ 1.5
			set(base * pivot * environment.CFrame.Angles(math.rad(angle) * amount, 0, 0) * pivot:Inverse() * environment.CFrame.new(0, 0, distance * amount))
		end)
	end
	table.insert(entry.connections, context.connect(tool.Equipped, function() environment.task.defer(drawWeapon) end))
	table.insert(entry.connections, context.connect(tool.Activated, activate))
	if tool.Parent == context.player.Character then environment.task.defer(drawWeapon) end
end

local WeaponSkins = {}
local SKIN_EFFECTS = {ParticleEmitter = true, Fire = true, Smoke = true, Sparkles = true, PointLight = true, SpotLight = true, SurfaceLight = true}

function WeaponSkins.toolType(state, tool)
	if not tool:IsA("Tool") then return nil end
	if tool.Name == "Gun" or state.context.collection:HasTag(tool, "Weapon_Gun") then return "Gun" end
	if tool.Name == "Knife" or state.context.collection:HasTag(tool, "Weapon_Knife") then return "Knife" end
	return nil
end

function WeaponSkins.disconnect(entry)
	for _, connection in entry.connections do pcall(function() connection:Disconnect() end) end
	if entry.track then pcall(function() entry.track:Stop(0) end) end
	if entry.visual then pcall(function() entry.visual:Destroy() end) end
	if entry.beam then pcall(function() entry.beam:Destroy() end) end
	for _, saved in entry.restore do
		pcall(function()
			if saved[4] then saved[1]:SetAttribute(saved[2], saved[3]); return end
			saved[1][saved[2]] = saved[3]
		end)
	end
end

function WeaponSkins.clearTool(state, tool, remote)
	local tools = remote and state.rm.tools or state.tools
	local entry = tools[tool]
	if not entry then return end
	tools[tool] = nil
	WeaponSkins.disconnect(entry)
	if entry.handle then pcall(function() entry.handle.LocalTransparencyModifier = 0 end) end
	if not remote and tool.Parent and entry.oldTexture ~= nil then
		pcall(function() tool.TextureId = entry.oldTexture end)
		WeaponSkins.updateBackpack(state, entry.newTexture, entry.oldTexture, tool.Name)
	end
end

function WeaponSkins.clearDisplay(state, kind)
	state.displayTokens[kind] = (state.displayTokens[kind] or 0) + 1
	local entry = state.displays[kind]
	if not entry then return end
	state.displays[kind] = nil
	WeaponSkins.disconnect(entry)
end

function WeaponSkins.hide(state, entry, object, keepHidden)
	local function set(property, value)
		table.insert(entry.restore, {object, property, object[property]})
		object[property] = value
	end
	if object:IsA("BasePart") then
		set("LocalTransparencyModifier", 1)
		if keepHidden then
			table.insert(entry.connections, state.context.connect(object:GetPropertyChangedSignal("LocalTransparencyModifier"), function()
				if object.LocalTransparencyModifier ~= 1 then object.LocalTransparencyModifier = 1 end
			end))
		end
	end
	if object:IsA("Decal") or object:IsA("Texture") then set("Transparency", 1) end
	if (object:IsA("ParticleEmitter") or object:IsA("Fire") or object:IsA("Smoke") or object:IsA("Sparkles")
		or object:IsA("Light") or object:IsA("Trail") or object:IsA("Highlight")) and not object:IsA("Beam") then set("Enabled", false) end
end

function WeaponSkins.copyEffects(state, from, to, realName)
	local existing = {}
	local original = state.DB[realName]
	local function collect(data)
		if not data then return end
		if SKIN_EFFECTS[data.c] then existing[data.c .. "|" .. data.n] = true end
		for _, child in data.k or {} do collect(child) end
	end
	if original then collect(original.d); collect(original.h) end
	for _, child in from:GetChildren() do
		if not SKIN_EFFECTS[child.ClassName] or existing[child.ClassName .. "|" .. child.Name] then continue end
		if child:GetAttribute("PulseFx") or child:GetAttribute("PulseFX") then continue end
		local copy = child:Clone()
		copy:SetAttribute("PulseFx", true)
		copy.Parent = to
	end
end

function WeaponSkins.restoreVisual(visual)
	local objects = visual:GetDescendants()
	table.insert(objects, visual)
	for _, object in objects do
		local transparency, enabled = object:GetAttribute("PulseBase"), object:GetAttribute("PulseOn")
		if transparency ~= nil then object.Transparency = transparency end
		if enabled ~= nil then object.Enabled = enabled end
	end
end

function WeaponSkins.attachDual(state, entry, dual)
	if not dual:IsA("BasePart") or dual:GetAttribute("PulseDual") or not entry.visual then return end
	dual:SetAttribute("PulseDual", true)
	local visual = entry.visual:Clone()
	local weld = visual:FindFirstChild("PulseGrip")
	if weld then weld.Part0 = dual end
	WeaponSkins.restoreVisual(visual)
	visual.CFrame = dual.CFrame * (weld and weld.C0 or state.context.environment.CFrame.identity)
	WeaponSkins.hide(state, entry, dual, true)
	for _, object in dual:GetDescendants() do WeaponSkins.hide(state, entry, object, object:IsA("BasePart")) end
	visual.Parent = dual
	table.insert(entry.restore, {visual, "Parent", nil})
	table.insert(entry.restore, {dual, "PulseDual", nil, true})
end

function WeaponSkins.hideLinked(state, entry, part)
	if not part:IsA("BasePart") or part.Name ~= "Handle" then return end
	local constraint = part:FindFirstChild("VisualConstraint")
	if not constraint or not constraint:IsA("RigidConstraint") or not constraint.Attachment0 or constraint.Attachment0.Parent ~= entry.handle then return end
	part.LocalTransparencyModifier = 1
	table.insert(entry.connections, state.context.connect(part:GetPropertyChangedSignal("LocalTransparencyModifier"), function()
		if part.LocalTransparencyModifier ~= 1 then part.LocalTransparencyModifier = 1 end
	end))
	for _, object in part:GetDescendants() do
		if PulseSkin.isAny(object, PULSE_EFFECTS) then
			table.insert(entry.restore, {object, "Enabled", object.Enabled})
			object.Enabled = false
			continue
		end
		if object:IsA("Decal") or object:IsA("Texture") then object.LocalTransparencyModifier = 1 end
	end
end

function WeaponSkins.mountFrame(state, kind, player)
	local character = (player or state.context.player).Character
	local reference = character and character:FindFirstChild("DisplayRef" .. kind)
	local display = reference and reference.Value
	local rigid = display and display:FindFirstChildWhichIsA("RigidConstraint")
	return rigid and rigid.Attachment1 and rigid.Attachment1.CFrame or nil
end

local SKIN_SOUND_PROPERTIES = {"SoundId", "Volume", "PlaybackSpeed", "RollOffMaxDistance", "RollOffMinDistance", "RollOffMode", "Looped"}

function WeaponSkins.sound(state, entry, sound, spec)
	if not sound:IsA("Sound") or entry.sounds[sound] then return end
	if entry.visual and sound:IsDescendantOf(entry.visual) then return end
	entry.sounds[sound] = true
	local values = {}
	for _, property in SKIN_SOUND_PROPERTIES do
		if spec.p[property] == nil then continue end
		local value = SkinModel.decode(state.context.environment, spec.p[property])
		values[property] = value
		table.insert(entry.restore, {sound, property, sound[property]})
		pcall(function() sound[property] = value end)
	end
	if values.SoundId then
		table.insert(entry.connections, state.context.connect(sound:GetPropertyChangedSignal("SoundId"), function()
			if sound.SoundId == values.SoundId then return end
			for property, value in values do pcall(function() sound[property] = value end) end
		end))
	end
	for _, effect in spec.fx or {} do
		local ok, object = pcall(state.context.environment.Instance.new, effect.c)
		if not ok or not object then continue end
		for property, value in effect.p or {} do
			pcall(function() object[property] = SkinModel.decode(state.context.environment, value) end)
		end
		object.Name, object.Parent = "PulseFx", sound
		table.insert(entry.restore, {object, "Parent", nil})
	end
end

function WeaponSkins.applySounds(state, entry, tool, skin)
	local specs = {}
	for _, spec in skin.sounds or {} do specs[spec.n] = spec end
	if not next(specs) then return end
	entry.sounds = {}
	local function apply(object)
		local spec = specs[object.Name]
		if spec then WeaponSkins.sound(state, entry, object, spec) end
	end
	for _, object in tool:GetDescendants() do apply(object) end
	table.insert(entry.connections, state.context.connect(tool.DescendantAdded, apply))
end

function WeaponSkins.findBeam(node)
	if not node then return nil end
	if node.c == "Beam" and node.n == "CustomBeam" then return node end
	for _, child in node.k or {} do
		local beam = WeaponSkins.findBeam(child)
		if beam then return beam end
	end
	return nil
end

function WeaponSkins.applyBeam(state, entry, handle, skin)
	local previous = handle:FindFirstChild("CustomBeam")
	if previous then
		table.insert(entry.restore, {previous, "Name", previous.Name})
		previous.Name = "CustomBeamOff"
	end
	local data = WeaponSkins.findBeam(skin.h or skin.d) or WeaponSkins.findBeam(skin.d)
	if not data then return end
	local beam = SkinModel.build(state.model, data)
	if not beam then return end
	beam.Name, beam.Parent, entry.beam = "CustomBeam", handle, beam
end

function WeaponSkins.updateBackpack(state, oldImage, newImage, name)
	local playerGui = state.context.player:FindFirstChild("PlayerGui")
	local backpack = playerGui and playerGui:FindFirstChild("BackpackUI")
	local frame = backpack and backpack:FindFirstChild("BackpackFrame")
	if not frame then return end
	for _, item in frame:GetChildren() do
		local container = item:FindFirstChild("Container")
		local icon = container and container:FindFirstChild("ToolIcon")
		local label = container and container:FindFirstChild("NameLabel")
		if not icon or not label or icon.Image ~= oldImage then continue end
		if oldImage == "" and label.Text ~= name then continue end
		icon.Image, label.Text = newImage, ""
	end
end

function WeaponSkins.applyDetails(state, entry, tool, skin, kind)
	if kind == "Gun" then WeaponSkins.applyBeam(state, entry, entry.handle, skin) end
	if skin.img and skin.img ~= "" then
		entry.oldTexture, entry.newTexture = tool.TextureId, skin.img
		tool.TextureId = skin.img
		WeaponSkins.updateBackpack(state, entry.oldTexture, entry.newTexture, tool.Name)
		state.context.environment.task.delay(0.5, function()
			if state.tools[tool] == entry then WeaponSkins.updateBackpack(state, entry.oldTexture, entry.newTexture, tool.Name) end
		end)
	end
	local item = state.weapons and state.weapons[entry.name]
	if not item then return end
	for _, property in {"ItemName", "Image", "Rarity"} do
		local value = tool:GetAttribute(property)
		if value == nil or item[property] == nil then continue end
		table.insert(entry.restore, {tool, property, value, true})
		pcall(tool.SetAttribute, tool, property, item[property])
	end
end

function WeaponSkins.animator(state)
	local character = state.context.player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	return humanoid and humanoid:FindFirstChildOfClass("Animator")
end

function WeaponSkins.stopOriginalHold(state, entry, tool, skin)
	local hold = tool:FindFirstChild("CustomHold")
	if not hold or not hold:IsA("Animation") or hold.AnimationId == skin.hold then return end
	local originalId = hold.AnimationId
	local function stop()
		local animator = WeaponSkins.animator(state)
		if not animator then return end
		for _, track in animator:GetPlayingAnimationTracks() do
			if track.Animation and track.Animation.AnimationId == originalId then track:Stop(0) end
		end
	end
	local environment = state.context.environment
	table.insert(entry.connections, state.context.connect(tool.Equipped, function()
		environment.task.delay(0.05, stop)
		environment.task.delay(0.3, stop)
	end))
	if tool.Parent == state.context.player.Character then stop() end
end

function WeaponSkins.applyPose(state, entry, tool, pose)
	local step
	local function stop()
		if step then step:Disconnect(); step = nil end
	end
	local function start()
		if step then return end
		local character = state.context.player.Character
		if not character then return end
		local joints = {}
		for _, joint in character:GetDescendants() do
			local part
			if joint:IsA("Motor6D") then part = joint.Part1 end
			if joint:IsA("AnimationConstraint") then part = joint.Attachment1 and joint.Attachment1.Parent end
			if part and pose[part.Name] then joints[joint] = SkinModel.frame(state.context.environment, pose[part.Name]) end
		end
		entry.poseMotors = 0
		for _ in joints do entry.poseMotors += 1 end
		step = state.context.connect(state.context.runService.Stepped, function()
			for joint, frame in joints do if joint.Parent then joint.Transform = frame end end
		end)
		table.insert(entry.connections, step)
	end
	table.insert(entry.connections, state.context.connect(tool.Equipped, start))
	table.insert(entry.connections, state.context.connect(tool.Unequipped, stop))
	if tool.Parent == state.context.player.Character then start() end
end

function WeaponSkins.playHold(state, entry, tool, animationId)
	local animation = state.context.environment.Instance.new("Animation")
	animation.AnimationId = animationId
	local function play()
		local animator = WeaponSkins.animator(state)
		if not animator then return end
		if not entry.track then
			local ok, track = pcall(animator.LoadAnimation, animator, animation)
			if ok then entry.track = track end
		end
		if entry.track then entry.track:Play() end
	end
	table.insert(entry.connections, state.context.connect(tool.Equipped, play))
	table.insert(entry.connections, state.context.connect(tool.Unequipped, function() if entry.track then entry.track:Stop() end end))
	if tool.Parent == state.context.player.Character then play() end
end

function WeaponSkins.applyHold(state, entry, tool, skin)
	WeaponSkins.stopOriginalHold(state, entry, tool, skin)
	local pose = skin.hold and SKIN_HOLD_POSES[skin.hold]
	entry.poseReached = pose and "joints" or "nojoints:" .. tostring(skin.hold)
	if pose then WeaponSkins.applyPose(state, entry, tool, pose); return end
	if skin.hold then WeaponSkins.playHold(state, entry, tool, skin.hold) end
end

function WeaponSkins.watchTool(state, entry, tool, remote)
	table.insert(entry.connections, state.context.connect(tool.AncestryChanged, function()
		if not tool.Parent or remote and tool.Parent ~= remote.player.Character then WeaponSkins.clearTool(state, tool, remote) end
	end))
end

function WeaponSkins.applyTool(state, tool, remote)
	local kind = WeaponSkins.toolType(state, tool)
	if not kind then return end
	local tools = remote and state.rm.tools or state.tools
	local equipped = remote and remote.name or not remote and state.profile.Weapons.Equipped[kind]
	local existing = tools[tool]
	if existing and existing.name == equipped then
		if not existing.visual or existing.visual.Parent and existing.visual:IsDescendantOf(tool) then return end
	end
	WeaponSkins.clearTool(state, tool, remote)
	local skin = equipped and state.DB[equipped]
	local real = tool:GetAttribute("ItemID") or not remote and state.realName[kind]
	if not state.enabled or not skin or equipped == real then return end

	local context, environment = state.context, state.context.environment
	local handle = tool:FindFirstChild("Handle") or tool:WaitForChild("Handle", 5)
	if not handle or not tool.Parent or not state.enabled then return end
	if remote and tool.Parent ~= remote.player.Character then return end
	if remote and (not state.rm.players[remote.player] or state.rm.players[remote.player].want[kind] ~= equipped) then return end
	if not remote and state.profile.Weapons.Equipped[kind] ~= equipped then return end
	local entry = {name = equipped, handle = handle, plr = remote and remote.player, connections = {}, restore = {}}
	tools[tool] = entry
	WeaponSkins.applySounds(state, entry, tool, skin)
	local data = skin.h or skin.d
	local visual
	if skin.cx then
		visual = PulseSkin.build(state.pulse, skin)
	end
	if not skin.cx and data then
		visual = SkinModel.build(state.model, data, {skip = function(node)
			return node.c == "Beam" and node.n == "CustomBeam" or node.c == "Sound"
		end})
	end
	if tools[tool] ~= entry then if visual then visual:Destroy() end; return end
	if not visual then
		if remote and kind == "Gun" then WeaponSkins.applyBeam(state, entry, handle, skin) end
		if not remote then WeaponSkins.applyDetails(state, entry, tool, skin, kind); WeaponSkins.applyHold(state, entry, tool, skin) end
		WeaponSkins.watchTool(state, entry, tool, remote)
		return
	end

	SkinModel.applyChroma(state.model, skin, visual)
	if not remote then WeaponSkins.copyEffects(state, handle, visual, real) end
	visual.Name = "VH"
	local offset = environment.CFrame.identity
	if skin.g then offset = tool.Grip * SkinModel.frame(environment, skin.g):Inverse() end
	if not skin.g then
		local mount = WeaponSkins.mountFrame(state, kind, remote and remote.player) or environment.CFrame.identity
		offset = mount * (skin.a1 and SkinModel.frame(environment, skin.a1) or environment.CFrame.identity):Inverse()
	end
	local weld = environment.Instance.new("Weld")
	weld.Name, weld.Part0, weld.Part1, weld.C0, weld.Parent = "PulseGrip", handle, visual, offset, visual
	visual.CFrame, visual.Parent, entry.visual = handle.CFrame * offset, handle, visual

	local appearance, effects = {}, {}
	local function capture(object)
		if object:IsA("BasePart") or object:IsA("Decal") or object:IsA("Texture") then
			appearance[object] = object.Transparency
			object:SetAttribute("PulseBase", object.Transparency)
			return
		end
		if (object:IsA("ParticleEmitter") or object:IsA("Fire") or object:IsA("Smoke") or object:IsA("Sparkles")
			or object:IsA("Light") or object:IsA("Trail") or object:IsA("Highlight")) and not object:IsA("Beam") then
			effects[object] = object.Enabled
			object:SetAttribute("PulseOn", object.Enabled)
		end
	end
	capture(visual)
	for _, object in visual:GetDescendants() do capture(object) end
	if skin.cx and not remote then pcall(PulseSkin.animate, state, tool, handle, visual, entry, skin) end
	local function updateVisibility()
		local hidden = handle.Transparency >= 0.99
		for object, transparency in appearance do if object.Parent then object.Transparency = hidden and 1 or transparency end end
		for object, enabled in effects do if object.Parent then object.Enabled = enabled and not hidden end end
	end
	updateVisibility()
	table.insert(entry.connections, context.connect(handle:GetPropertyChangedSignal("Transparency"), updateVisibility))
	WeaponSkins.hide(state, entry, handle, true)
	for _, object in handle:GetDescendants() do
		if object == visual or object:IsDescendantOf(visual) then continue end
		WeaponSkins.hide(state, entry, object, object:IsA("BasePart"))
		if object:IsA("BasePart") and object.Name == "VH" and object.Parent == handle then
			table.insert(entry.restore, {object, "Name", object.Name})
			object.Name = "VHOff"
		end
	end
	for _, child in tool:GetChildren() do
		if child.Name == "Dual" then WeaponSkins.attachDual(state, entry, child) end
	end
	if tool.ChildAdded then
		table.insert(entry.connections, context.connect(tool.ChildAdded, function(child)
			if child.Name == "Dual" then environment.task.defer(function() if tools[tool] == entry then WeaponSkins.attachDual(state, entry, child) end end) end
		end))
	end
	if context.workspace then
		for _, child in context.workspace:GetChildren() do WeaponSkins.hideLinked(state, entry, child) end
		table.insert(entry.connections, context.connect(context.workspace.ChildAdded, function(child)
			if child.Name ~= "Handle" then return end
			environment.task.defer(function() if tools[tool] == entry then WeaponSkins.hideLinked(state, entry, child) end end)
		end))
	end
	if remote and kind == "Gun" then WeaponSkins.applyBeam(state, entry, handle, skin) end
	if not remote then WeaponSkins.applyDetails(state, entry, tool, skin, kind); WeaponSkins.applyHold(state, entry, tool, skin) end
	WeaponSkins.watchTool(state, entry, tool, remote)
end

function WeaponSkins.hideDisplay(entry, object)
	if object:IsA("BasePart") or object:IsA("Decal") then
		table.insert(entry.restore, {object, "Transparency", object.Transparency})
		object.Transparency = 1
	end
	if PulseSkin.isAny(object, PULSE_EFFECTS) then
		table.insert(entry.restore, {object, "Enabled", object.Enabled})
		object.Enabled = false
	end
	if object:IsA("BasePart") then
		table.insert(entry.restore, {object, "LocalTransparencyModifier", object.LocalTransparencyModifier})
		object.LocalTransparencyModifier = 1
	end
end

function WeaponSkins.enforceHidden(entry)
	for _, saved in entry.restore do
		local object, property = saved[1], saved[2]
		if not object.Parent then continue end
		if property == "Enabled" then object.Enabled = false; continue end
		if property == "Transparency" or property == "LocalTransparencyModifier" then object[property] = 1 end
	end
end

function WeaponSkins.firstPerson(state)
	local context, environment = state.context, state.context.environment
	local character = context.player.Character
	local head = character and character:FindFirstChild("Head")
	local camera = context.workspace.CurrentCamera
	if context.player.CameraMode == environment.Enum.CameraMode.LockFirstPerson then return 1 end
	if camera and head and (camera.CFrame.Position - head.Position).Magnitude < 1.4 then return 1 end
	local invisible = rawget(environment.getgenv(), "__pulseInvis")
	if type(invisible) == "table" and invisible.active then return 0.5 end
	return 0
end

function WeaponSkins.renderDisplays(state)
	local transparency = WeaponSkins.firstPerson(state)
	for _, entry in state.displays do
		if entry.ltm ~= transparency then
			entry.ltm = transparency
			for _, part in entry.parts do if part.Parent then part.LocalTransparencyModifier = transparency end end
			if entry.visual and entry.visual.Parent then
				for _, object in entry.visual:GetDescendants() do
					if not PulseSkin.isAny(object, {"Beam", "Trail", "ParticleEmitter", "Light", "Fire", "Smoke", "Sparkles"}) then continue end
					local original = object:GetAttribute("PulseFpOff")
					if transparency >= 1 then
						if original == nil then object:SetAttribute("PulseFpOff", object.Enabled) end
						object.Enabled = false
						continue
					end
					if original ~= nil then object.Enabled = original; object:SetAttribute("PulseFpOff", nil) end
				end
			end
		end
		WeaponSkins.enforceHidden(entry)
	end
end

function WeaponSkins.waitMount(state, kind, display, token)
	local task = state.context.environment.task
	for _ = 1, 60 do
		if not display.Parent or state.displayTokens[kind] ~= token or not state.enabled then return nil end
		local rigid = display:FindFirstChildWhichIsA("RigidConstraint")
		if rigid and rigid.Attachment0 and rigid.Attachment1 then return rigid end
		task.wait(0.1)
	end
	return nil
end

function WeaponSkins.claimDisplay(state, kind, display)
	local parent = display.Parent or state.context.workspace
	local newer = false
	for _, object in parent:GetChildren() do
		if object.Name ~= "PulseSkin" .. kind or object:GetAttribute("PulseOwner") == state.owner then continue end
		if (object:GetAttribute("PulseBirth") or 0) > state.birth then
			newer = true
			continue
		end
		pcall(function() object:Destroy() end)
	end
	return not newer
end

function WeaponSkins.repairDisplays(state)
	if not state.enabled or not state.ready then return end
	local context, environment = state.context, state.context.environment
	local character = context.player.Character
	for _, kind in {"Knife", "Gun"} do
		local equipped = state.profile.Weapons.Equipped[kind]
		local reference = character and character:FindFirstChild("DisplayRef" .. kind)
		local display = reference and reference.Value
		local wanted = equipped and state.DB[equipped] and (equipped ~= state.realName[kind] or kind == "Knife" and state.knifeBelt)
		if not wanted or not display or not display.Parent then continue end
		if not WeaponSkins.claimDisplay(state, kind, display) then WeaponSkins.clearDisplay(state, kind); continue end
		if state.repairing[kind] or environment.os.clock() <= 7 then continue end
		local entry = state.displays[kind]
		local visual = entry and entry.visual
		local weld = visual and visual:FindFirstChild("PulseMount")
		local valid = entry and entry.name == equipped and entry.display == display and visual and visual.Parent and weld
			and (weld.Part0 == display or entry.mount and weld.Part0 == entry.mount and entry.mount.Parent)
			and weld.Part1 == visual and weld.Enabled and (visual.Position - display.Position).Magnitude < 8
		if valid then continue end
		state.repairing[kind] = true
		environment.task.spawn(function()
			pcall(WeaponSkins.clearDisplay, state, kind)
			pcall(WeaponSkins.applyDisplay, state, kind)
			state.repairing[kind] = nil
		end)
	end
	for tool, entry in state.tools do
		if not tool.Parent or entry.visual and (not entry.visual.Parent or not entry.visual:IsDescendantOf(tool)) then
			WeaponSkins.clearTool(state, tool)
		end
	end
	WeaponSkins.refreshTools(state)
end

function WeaponSkins.applyDisplay(state, kind)
	local context, environment = state.context, state.context.environment
	local character = context.player.Character
	local reference = character and character:FindFirstChild("DisplayRef" .. kind)
	local display = reference and reference.Value
	local equipped = state.profile.Weapons.Equipped[kind]
	local existing = state.displays[kind]
	if existing and existing.name == equipped and existing.display == display and existing.visual and existing.visual.Parent then return end
	WeaponSkins.clearDisplay(state, kind)
	local token = state.displayTokens[kind]
	local skin = state.DB[equipped]
	if not state.enabled or not skin or not display or not display.Parent then return end
	if equipped == state.realName[kind] and not (kind == "Knife" and state.knifeBelt) then return end
	local rigid = WeaponSkins.waitMount(state, kind, display, token)
	if not rigid or not WeaponSkins.claimDisplay(state, kind, display) then return end

	local visual
	if skin.cx then
		visual = PulseSkin.build(state.pulse, skin)
	end
	if not skin.cx and skin.d then
		visual = SkinModel.build(state.model, skin.d, {skip = function(node) return node.c == "Beam" and node.n == "CustomBeam" end})
	end
	if not visual then return end
	if token ~= state.displayTokens[kind] or not state.enabled or not display.Parent or context.player.Character ~= character
		or state.profile.Weapons.Equipped[kind] ~= equipped then visual:Destroy(); return end
	SkinModel.applyChroma(state.model, skin, visual)
	WeaponSkins.copyEffects(state, display, visual, state.realName[kind])
	visual.Name = "PulseSkin" .. kind
	visual:SetAttribute("PulseOwner", state.owner)
	visual:SetAttribute("PulseBirth", state.birth)
	local attachment = skin.a1 and SkinModel.frame(environment, skin.a1) or environment.CFrame.identity
	local offset, mount, anchor = rigid.Attachment1.CFrame * attachment:Inverse(), display, nil
	local torso = kind == "Knife" and state.knifeBelt and character:FindFirstChild("LowerTorso")
	local belt = torso and torso:FindFirstChild("KnifeBelt")
	if belt and belt:IsA("Attachment") then
		local original = rawget(environment.getgenv(), "__pulseMountOrig")
		offset, mount, anchor = (original and original(belt) or belt.CFrame) * attachment:Inverse(), torso, belt.Name
	end
	local upper = skin.cx and skin.cx.sling and character:FindFirstChild("UpperTorso")
	if upper and upper:IsA("BasePart") then
		local vector = environment.Vector3
		local depth = upper.Size.Z / 2 + skin.cx.sling / 2 + 0.06
		offset, mount, anchor = environment.CFrame.fromMatrix(vector.new(0, -0.15, depth), vector.new(0, 0, -1), vector.new(0.866, 0.5, 0)), upper, nil
	end
	visual.CFrame = mount.CFrame * offset
	local weld = environment.Instance.new("Weld")
	weld.Name, weld.Part0, weld.Part1, weld.C0 = "PulseMount", mount, visual, offset
	if anchor then weld:SetAttribute("PulseAnchor", anchor) end
	weld.Parent, visual.Parent = visual, display.Parent or context.workspace
	local parts = {visual}
	for _, object in visual:GetDescendants() do if object:IsA("BasePart") or object:IsA("Decal") then table.insert(parts, object) end end
	local entry = {name = equipped, display = display, visual = visual, mount = mount, parts = parts, ltm = -1, connections = {}, restore = {}}
	state.displays[kind] = entry
	WeaponSkins.hideDisplay(entry, display)
	for _, object in display:GetDescendants() do WeaponSkins.hideDisplay(entry, object) end
	table.insert(entry.connections, context.connect(display.DescendantAdded, function(object)
		environment.task.defer(function() if state.displays[kind] == entry then pcall(WeaponSkins.hideDisplay, entry, object) end end)
	end))
	table.insert(entry.connections, context.connect(display.AncestryChanged, function()
		if not display.Parent then environment.task.defer(function() pcall(WeaponSkins.applyDisplay, state, kind) end) end
	end))
end

local RemoteSkins = {}

function RemoteSkins.clearDisplay(remote, kind)
	remote.token[kind] = (remote.token[kind] or 0) + 1
	local entry = remote.lobby[kind]
	if not entry then return end
	remote.lobby[kind] = nil
	WeaponSkins.disconnect(entry)
end

function RemoteSkins.enforce(state)
	for _, remote in state.rm.players do
		for _, entry in remote.lobby do WeaponSkins.enforceHidden(entry) end
	end
end

function RemoteSkins.applyDisplay(state, player, remote, kind)
	local context, environment = state.context, state.context.environment
	local character, equipped = player.Character, remote.want[kind]
	local reference = character and character:FindFirstChild("DisplayRef" .. kind)
	local display = reference and reference.Value
	local belt = kind == "Knife" and remote.want.belt
	local existing = remote.lobby[kind]
	if existing and existing.name == equipped and existing.display == display and existing.belt == belt and existing.visual and existing.visual.Parent then return end
	RemoteSkins.clearDisplay(remote, kind)
	local token = remote.token[kind]
	local skin = equipped and state.DB[equipped]
	if not state.enabled or not skin or not display or not display.Parent then return end
	local rigid = display:FindFirstChildWhichIsA("RigidConstraint")
	if not rigid or not rigid.Attachment0 or not rigid.Attachment1 then return end
	local visual
	if skin.cx then
		visual = PulseSkin.build(state.pulse, skin)
	end
	if not skin.cx and skin.d then
		visual = SkinModel.build(state.model, skin.d, {skip = function(node) return node.c == "Beam" and node.n == "CustomBeam" end})
	end
	if not visual then return end
	if remote.token[kind] ~= token or remote.dead or not display.Parent or player.Character ~= character or remote.want[kind] ~= equipped then
		visual:Destroy()
		return
	end
	SkinModel.applyChroma(state.model, skin, visual)
	visual.Name = "PulseRSkin" .. kind
	local attachment = skin.a1 and SkinModel.frame(environment, skin.a1) or environment.CFrame.identity
	local offset, mount, anchor = rigid.Attachment1.CFrame * attachment:Inverse(), display, nil
	local torso = belt and character:FindFirstChild("LowerTorso")
	local beltAttachment = torso and torso:FindFirstChild("KnifeBelt")
	if beltAttachment and beltAttachment:IsA("Attachment") then
		local original = rawget(environment.getgenv(), "__pulseMountOrig")
		offset, mount, anchor = (original and original(beltAttachment) or beltAttachment.CFrame) * attachment:Inverse(), torso, beltAttachment.Name
	end
	local upper = skin.cx and skin.cx.sling and character:FindFirstChild("UpperTorso")
	if upper and upper:IsA("BasePart") then
		local vector = environment.Vector3
		local depth = upper.Size.Z / 2 + skin.cx.sling / 2 + 0.06
		offset, mount, anchor = environment.CFrame.fromMatrix(vector.new(0, -0.15, depth), vector.new(0, 0, -1), vector.new(0.866, 0.5, 0)), upper, nil
	end
	visual.CFrame = mount.CFrame * offset
	local weld = environment.Instance.new("Weld")
	weld.Name, weld.Part0, weld.Part1, weld.C0 = "PulseMount", mount, visual, offset
	if anchor then weld:SetAttribute("PulseAnchor", anchor) end
	weld.Parent, visual.Parent = visual, display.Parent or context.workspace
	local entry = {name = equipped, display = display, visual = visual, belt = belt, mount = mount, connections = {}, restore = {}}
	remote.lobby[kind] = entry
	WeaponSkins.hideDisplay(entry, display)
	for _, object in display:GetDescendants() do WeaponSkins.hideDisplay(entry, object) end
	table.insert(entry.connections, context.connect(display.DescendantAdded, function(object)
		environment.task.defer(function() if remote.lobby[kind] == entry then pcall(WeaponSkins.hideDisplay, entry, object) end end)
	end))
	table.insert(entry.connections, context.connect(display.AncestryChanged, function() if not display.Parent then remote.dirty = true end end))
end

function RemoteSkins.clear(state, player)
	local remote = state.rm.players[player]
	if not remote then return end
	remote.dead = true
	state.rm.players[player] = nil
	for _, connection in remote.connections do pcall(function() connection:Disconnect() end) end
	for _, kind in {"Knife", "Gun"} do RemoteSkins.clearDisplay(remote, kind) end
	for tool, entry in state.rm.tools do
		if entry.plr == player then WeaponSkins.clearTool(state, tool, {player = player}) end
	end
end

function RemoteSkins.refresh(state, player, remote)
	if remote.dead or state.rm.players[player] ~= remote then return end
	local context, environment = state.context, state.context.environment
	local character = player.Character
	if remote.character ~= character then
		remote.character = character
		for _, connection in remote.connections do pcall(function() connection:Disconnect() end) end
		remote.connections = {}
		if character then
			table.insert(remote.connections, context.connect(character.ChildAdded, function(child)
				local kind = WeaponSkins.toolType(state, child)
				if kind then
					environment.task.spawn(function()
						if state.rm.players[player] == remote then pcall(WeaponSkins.applyTool, state, child, {player = player, name = remote.want[kind]}) end
					end)
					return
				end
				if child:IsA("ObjectValue") and child.Name:find("^DisplayRef") then
					environment.task.delay(0.3, function()
						if state.rm.players[player] == remote then pcall(RemoteSkins.applyDisplay, state, player, remote, child.Name:sub(11)) end
					end)
				end
			end))
		end
	end
	for _, kind in {"Knife", "Gun"} do pcall(RemoteSkins.applyDisplay, state, player, remote, kind) end
	if character then
		for _, child in character:GetChildren() do
			local kind = WeaponSkins.toolType(state, child)
			if kind then pcall(WeaponSkins.applyTool, state, child, {player = player, name = remote.want[kind]}) end
		end
	end
	for tool, entry in state.rm.tools do
		if entry.plr == player and (tool.Parent ~= character or not entry.visual or not entry.visual.Parent) then
			WeaponSkins.clearTool(state, tool, {player = player})
		end
	end
	remote.dirty = false
end

function RemoteSkins.set(state, player, payload)
	if state.context.environment.typeof(player) ~= "Instance" or player == state.context.player then return end
	local valid = type(payload) == "table" and payload or {}
	local knife = type(valid.k) == "string" and state.DB[valid.k] and valid.k or nil
	local gun = type(valid.g) == "string" and state.DB[valid.g] and valid.g or nil
	if not knife and not gun then RemoteSkins.clear(state, player); return end
	local remote = state.rm.players[player]
	if not remote then
		remote = {lobby = {}, connections = {}, token = {}}
		state.rm.players[player] = remote
	end
	remote.want = {Knife = knife, Gun = gun, belt = knife ~= nil and (valid.b == 1 or valid.b == true)}
	if remote.busy then remote.again = true; return end
	remote.busy = true
	state.context.environment.task.spawn(function()
		repeat
			remote.again = false
			pcall(RemoteSkins.refresh, state, player, remote)
		until not remote.again or remote.dead
		remote.busy = false
	end)
end

function RemoteSkins.clearAll(state)
	for player in state.rm.players do RemoteSkins.clear(state, player) end
end

function WeaponSkins.refreshTools(state)
	local player = state.context.player
	for _, container in {player:FindFirstChildOfClass("Backpack"), player.Character} do
		if not container then continue end
		for _, tool in container:GetChildren() do
			if WeaponSkins.toolType(state, tool) then state.context.environment.task.spawn(function() pcall(WeaponSkins.applyTool, state, tool) end) end
		end
	end
end

function WeaponSkins.refresh(state)
	if not state.ready then return end
	for _, kind in {"Knife", "Gun"} do
		state.context.environment.task.spawn(function() pcall(WeaponSkins.applyDisplay, state, kind) end)
	end
	WeaponSkins.refreshTools(state)
	for _, listener in state.listeners do state.context.environment.task.defer(listener) end
end

function WeaponSkins.save(state)
	pcall(function()
		local context, environment = state.context, state.context.environment
		if not environment.isfolder("PulseHub") then environment.makefolder("PulseHub") end
		environment.writefile("PulseHub/mm2_skins.json", context.http:JSONEncode({Knife = state.selected.Knife or false, Gun = state.selected.Gun or false}))
	end)
end

function WeaponSkins.spawn(state, key, amount)
	local owned = state.profile.Weapons.Owned
	amount = amount or 1
	owned[key] = (owned[key] or 0) + amount
	local globals = state.context.environment.getgenv()
	globals.__pulseSpawned = type(globals.__pulseSpawned) == "table" and globals.__pulseSpawned or {}
	globals.__pulseSpawned[key] = (globals.__pulseSpawned[key] or 0) + amount
	pcall(function() state.inventoryChanged:Fire("Weapons", key, owned[key]) end)
end

function WeaponSkins.equip(state, key)
	if not state.ready then return end
	local item = state.weapons[key]
	if not item or item.ItemType ~= "Knife" and item.ItemType ~= "Gun" then return end
	local owned = state.profile.Weapons.Owned
	if not owned[key] or owned[key] < 1 then
		WeaponSkins.spawn(state, key, 1)
	end
	state.profile.Weapons.Equipped[item.ItemType] = key
	pcall(function() state.equipService.EquippedChanged:Fire(item.ItemType, key) end)
	state.selected[item.ItemType] = key ~= state.realName[item.ItemType] and key or nil
	WeaponSkins.save(state)
	WeaponSkins.refresh(state)
end

function WeaponSkins.resetSlot(state, kind)
	local original = state.realName[kind]
	if not original or not state.ready then return end
	state.profile.Weapons.Equipped[kind] = original
	pcall(function() state.equipService.EquippedChanged:Fire(kind, original) end)
	state.selected[kind] = nil
	WeaponSkins.save(state)
	WeaponSkins.refresh(state)
end

function WeaponSkins.setEnabled(state, enabled)
	state.enabled = not not enabled
	if state.inventory and state.inventory.refresh then state.context.environment.task.defer(state.inventory.refresh) end
	if not state.enabled then
		for kind in state.displays do WeaponSkins.clearDisplay(state, kind) end
		for tool in state.tools do WeaponSkins.clearTool(state, tool) end
		if state.remoteClearAll then state.remoteClearAll() end
		return
	end
	WeaponSkins.refresh(state)
end

function WeaponSkins.snapshot(state)
	local result = {}
	for _, kind in {"Knife", "Gun"} do
		local current = state.profile.Weapons.Equipped[kind]
		local selected = state.enabled and current and state.DB[current] and current or state.realName[kind] or current
		result[kind] = type(selected) == "string" and selected or false
	end
	return state.context.http:JSONEncode(result)
end

function WeaponSkins.applySnapshot(state, encoded)
	local ok, saved = pcall(function() return state.context.http:JSONDecode(encoded) end)
	if not ok or type(saved) ~= "table" then saved = {} end
	for _, kind in {"Knife", "Gun"} do
		local key = saved[kind]
		if type(key) ~= "string" or not state.DB[key] or key == state.realName[kind] then continue end
		if not state.enabled then
			local setter = state.context.setters and state.context.setters["Visuals/WSkins/Weapon Skin Changer"]
			if setter then pcall(setter, true) else pcall(WeaponSkins.setEnabled, state, true) end
		end
		break
	end
	for _, kind in {"Knife", "Gun"} do
		local key = saved[kind]
		local current = state.profile.Weapons.Equipped[kind]
		if type(key) == "string" and state.DB[key] then
			if current ~= key then pcall(WeaponSkins.equip, state, key) end
			state.selected[kind] = key ~= state.realName[kind] and key or nil
			continue
		end
		if current ~= state.realName[kind] then pcall(WeaponSkins.resetSlot, state, kind) end
		state.selected[kind] = nil
	end
	WeaponSkins.save(state)
	state.context.environment.task.defer(function() WeaponSkins.refresh(state) end)
end

function WeaponSkins.updateCatalog(state)
	if not state.catalog then return end
	local items = SkinCatalog.filter(state.list or {}, state.kind, state.rarity, state.query)
	local options = {}
	for _, item in items do table.insert(options, {code = item.key, label = item.name .. " В· " .. item.type .. " В· " .. item.rarity}) end
	state.catalog.SetOptions(options)
end

function WeaponSkins.watchCharacter(state, character)
	if not character then return end
	local context, environment = state.context, state.context.environment
	context.connect(character.ChildAdded, function(child)
		if WeaponSkins.toolType(state, child) then environment.task.spawn(function() pcall(WeaponSkins.applyTool, state, child) end) end
		if child:IsA("ObjectValue") and child.Name:find("^DisplayRef") then
			environment.task.spawn(function() WeaponSkins.readRealInventory(state) end)
			local kind = child.Name:sub(11)
			environment.task.delay(0.3, function() pcall(WeaponSkins.applyDisplay, state, kind) end)
			context.connect(child.Changed, function() environment.task.delay(0.3, function() pcall(WeaponSkins.applyDisplay, state, kind) end) end)
		end
	end)
	for _, kind in {"Knife", "Gun"} do
		local reference = character:FindFirstChild("DisplayRef" .. kind)
		if reference then
			context.connect(reference.Changed, function() environment.task.delay(0.3, function() pcall(WeaponSkins.applyDisplay, state, kind) end) end)
		end
	end
	WeaponSkins.refresh(state)
end

local ThrownSkins = {}

function ThrownSkins.signature(part)
	if part:IsA("MeshPart") then return part.MeshId .. "|" .. part.TextureID end
	local mesh = part:FindFirstChildWhichIsA("SpecialMesh")
	return mesh and mesh.MeshId .. "|" .. mesh.TextureId or ""
end

function ThrownSkins.cloneVisual(state, part, source)
	local visual = source:Clone()
	local weld = visual:FindFirstChild("PulseGrip")
	local offset = weld and weld.C0 or state.context.environment.CFrame.identity
	if weld then weld.Part0, weld.Part1 = part, visual end
	WeaponSkins.restoreVisual(visual)
	local objects = visual:GetDescendants()
	table.insert(objects, visual)
	for _, object in objects do
		if object:IsA("BasePart") or object:IsA("Decal") then pcall(function() object.LocalTransparencyModifier = 0 end) end
	end
	visual.CFrame, visual.Parent = part.CFrame * offset, part
	return visual
end

function ThrownSkins.hide(part, visual)
	part.LocalTransparencyModifier = 1
	part:GetPropertyChangedSignal("LocalTransparencyModifier"):Connect(function()
		if part.LocalTransparencyModifier ~= 1 then part.LocalTransparencyModifier = 1 end
	end)
	for _, object in part:GetDescendants() do
		if object == visual or object:IsDescendantOf(visual) then continue end
		if PulseSkin.isAny(object, PULSE_EFFECTS) then object.Enabled = false; continue end
		if object:IsA("Decal") or object:IsA("Texture") then object.LocalTransparencyModifier = 1 end
	end
end

function ThrownSkins.capture(state, projectile)
	local context, environment = state.context, state.context.environment
	local link = projectile:WaitForChild("HandleLink", 3)
	local handle = link and link.Value
	if not handle or state.alive == false then return end

	local character = context.player.Character
	local backpack = context.player:FindFirstChildOfClass("Backpack")
	if character and handle:IsDescendantOf(character) or backpack and handle:IsDescendantOf(backpack) then
		state.lastThrow = environment.os.clock()
	end
	local entry
	for _, tool in state.tools do if tool.handle == handle then entry = tool end end
	if not entry or not entry.visual then return end

	local blade = projectile:FindFirstChild("BladePosition")
	local directionValue = projectile:FindFirstChild("ThrowDirection")
	local direction = directionValue and directionValue:IsA("Vector3Value") and directionValue.Value or nil
	local attribute = projectile:GetAttribute("Direction")
	if environment.typeof(attribute) == "Vector3" and attribute.Magnitude > 0 then direction = attribute end
	table.insert(state.throws, {time = environment.os.clock(), origin = blade and blade:IsA("BasePart") and blade.Position or handle.Position,
		direction = direction and direction.Magnitude > 0 and direction.Unit or nil, signature = ThrownSkins.signature(handle), template = entry.visual:Clone()})

	local knife = projectile:WaitForChild("KnifeVisual", 3)
	if not knife or state.alive == false then return end
	local visual = knife:FindFirstChild("VH") or ThrownSkins.cloneVisual(state, knife, entry.visual)
	WeaponSkins.restoreVisual(visual)
	knife.Transparency = 1
	pcall(function() knife.LocalTransparencyModifier = 1 end)
	for _, object in knife:GetDescendants() do
		if object == visual or object:IsDescendantOf(visual) then continue end
		if object:IsA("Decal") or object:IsA("Texture") then pcall(function() object.LocalTransparencyModifier = 1 end) end
	end
end

function ThrownSkins.match(state, part)
	local now = state.context.environment.os.clock()
	local signature = ThrownSkins.signature(part)
	local best, bestThrow
	for index = #state.throws, 1, -1 do
		local throw = state.throws[index]
		if now - throw.time > 6 then
			throw.template:Destroy()
			table.remove(state.throws, index)
			continue
		end
		if throw.signature ~= signature then continue end
		local displacement = part.Position - throw.origin
		local distance = displacement.Magnitude * 0.05
		if throw.direction then
			local along = displacement:Dot(throw.direction)
			distance = along > -3 and (displacement - throw.direction * along).Magnitude or math.huge
		end
		if distance < 5 and (not best or distance < best) then best, bestThrow = distance, throw end
	end
	if not bestThrow then return false end
	table.remove(state.throws, table.find(state.throws, bestThrow))
	local template = bestThrow.template
	local visual = ThrownSkins.cloneVisual(state, part, template)
	template:Destroy()
	ThrownSkins.hide(part, visual)
	return true
end

function ThrownSkins.applyStuck(state, part)
	if part:FindFirstChild("VH") or ThrownSkins.match(state, part) then return true end
	if state.context.environment.os.clock() - state.lastThrow >= 4 then return false end
	for tool, entry in state.tools do
		if WeaponSkins.toolType(state, tool) ~= "Knife" or not entry.visual or not entry.visual.Parent or not entry.handle then continue end
		if ThrownSkins.signature(part) ~= ThrownSkins.signature(entry.handle) then continue end
		ThrownSkins.hide(part, ThrownSkins.cloneVisual(state, part, entry.visual))
		return true
	end
	return false
end

function ThrownSkins.install(state)
	local context, environment = state.context, state.context.environment
	state.throws, state.pendingThrows, state.lastThrow = {}, {}, 0
	context.connect(context.collection:GetInstanceAddedSignal("ThrowingKnife"), function(projectile)
		environment.task.spawn(function() pcall(ThrownSkins.capture, state, projectile) end)
	end)
	context.connect(context.workspace.ChildAdded, function(part)
		if state.enabled and part.Name == "StuckKnife" and part:IsA("BasePart") then
			table.insert(state.pendingThrows, {part = part, time = environment.os.clock()})
		end
	end)
	environment.task.spawn(function()
		while state.alive and context.isAlive() do
			local now = environment.os.clock()
			for index = #state.pendingThrows, 1, -1 do
				local pending = state.pendingThrows[index]
				if not pending.part.Parent or now - pending.time > 3 then table.remove(state.pendingThrows, index); continue end
				local ok, applied = pcall(ThrownSkins.applyStuck, state, pending.part)
				if ok and applied then table.remove(state.pendingThrows, index) end
			end
			environment.task.wait(0.05)
		end
		for _, throw in state.throws do pcall(function() throw.template:Destroy() end) end
	end)
end

function WeaponSkins.updateRealInventory(state, full)
	if type(full) ~= "table" or type(full.Weapons) ~= "table" or type(full.Weapons.Equipped) ~= "table" then return end
	state.realName.Knife, state.realName.Gun = full.Weapons.Equipped.Knife, full.Weapons.Equipped.Gun
	local owned = {}
	for key, value in type(full.Weapons.Owned) == "table" and full.Weapons.Owned or {} do
		if type(key) == "string" then owned[key] = true; continue end
		if type(value) == "string" then owned[value] = true end
	end
	for _, unique in type(full.Uniques) == "table" and full.Uniques or {} do
		if type(unique) == "table" and unique.BaseItem then owned[unique.BaseItem] = true end
	end
	state.realOwned = owned

	local globals = state.context.environment.getgenv()
	if globals.__pulseSpawnedMig then return end
	globals.__pulseSpawnedMig = true
	globals.__pulseSpawned = type(globals.__pulseSpawned) == "table" and globals.__pulseSpawned or {}
	for key, amount in state.profile.Weapons.Owned do
		if type(key) ~= "string" or type(amount) ~= "number" or amount <= 0 then continue end
		if not owned[key] and state.DB[key] and globals.__pulseSpawned[key] == nil then globals.__pulseSpawned[key] = amount end
	end
end

function WeaponSkins.readRealInventory(state)
	local ok, full = pcall(state.fullInventory.InvokeServer, state.fullInventory, state.context.player)
	if not ok or not state.alive or not state.context.isAlive() then return end
	WeaponSkins.updateRealInventory(state, full)
end

function WeaponSkins.withRealOwned(state, callback, ...)
	local globals = state.context.environment.getgenv()
	local spawned = globals.__pulseSpawned
	local owned = state.profile.Weapons and state.profile.Weapons.Owned
	local saved = {}
	if type(owned) == "table" and type(spawned) == "table" then
		for key, amount in spawned do
			local current = owned[key]
			if type(current) ~= "number" then continue end
			saved[key] = current
			local real = current - amount
			owned[key] = real > 0 and real or nil
		end
	end
	local result = table.pack(pcall(callback, ...))
	if type(owned) == "table" then for key, amount in saved do owned[key] = amount end end
	if not result[1] then error(result[2], 0) end
	return table.unpack(result, 2, result.n)
end

function WeaponSkins.scrubTrade(state)
	local inventory = state.tradeModule and state.tradeModule.TradeInventory
	local groups = type(inventory) == "table" and type(inventory.Data) == "table" and inventory.Data.Weapons
	if type(groups) ~= "table" then return end
	local spawned = state.context.environment.getgenv().__pulseSpawned or {}
	local owned = state.profile.Weapons and state.profile.Weapons.Owned or {}
	for _, group in groups do
		if type(group) ~= "table" then continue end
		for key, item in group do
			if not spawned[key] then continue end
			local amount = type(owned[key]) == "number" and owned[key] - spawned[key] or nil
			if not amount or amount <= 0 then
				if type(item) == "table" and item.Frame then pcall(function() item.Frame:Destroy() end) end
				group[key] = nil
				continue
			end
			if type(item) == "table" then item.Amount = amount end
		end
	end
end

local InventoryChanger = {}

function InventoryChanger.connect(changer, signal, callback)
	if not signal then return end
	local connection = changer.state.context.connect(signal, callback)
	table.insert(changer.connections, connection)
end

function InventoryChanger.drop(changer)
	for _, connection in changer.connections do pcall(function() connection:Disconnect() end) end
	for object, saved in changer.shifted do
		pcall(function()
			if type(saved) == "table" then object.Position, object.Size = saved[1], saved[2]; return end
			object.LayoutOrder = saved
		end)
	end
	if changer.page then changer.page:Destroy() end
	if changer.button then changer.button:Destroy() end
	changer.page, changer.button, changer.built, changer.building = nil, nil, false, false
	changer.frames, changer.connections, changer.shifted = {}, {}, {}
	changer.gradients, changer.gradientConnection = setmetatable({}, {__mode = "k"}), nil
end

function InventoryChanger.tag(changer, frame, visible, active, overlay)
	local environment = changer.state.context.environment
	local tagsName = overlay and "PulseTags" or "Tags"
	local tags = frame:FindFirstChild(tagsName)
	if not tags and visible then
		local template = frame:FindFirstChild("Tags") or changer.template:FindFirstChild("Tags")
		if not template then return end
		tags = template:Clone()
		for _, child in tags:GetChildren() do if child:IsA("GuiObject") then child:Destroy() end end
		tags.Name, tags.Parent = tagsName, frame
		if overlay and changer.tags then changer.tags[tags] = true end
		if overlay then tags.ZIndex = 6 end
	end
	if not tags then return end
	local tag = tags:FindFirstChild("PulseChanger")
	if not tag and visible then
		local templateTags = changer.template:FindFirstChild("Tags")
		local source = templateTags and templateTags:FindFirstChild("Chroma")
		if not source then return end
		tag = source:Clone()
		tag.Name, tag.LayoutOrder = "PulseChanger", -1
		for _, child in tag:GetDescendants() do if child:IsA("UIGradient") then child:Destroy() end end
		local collection = changer.state.context.collection
		for _, name in collection:GetTags(tag) do collection:RemoveTag(tag, name) end
		local background = tag:FindFirstChild("BG")
		if background then background:Destroy() end
		local label = tag:FindFirstChild("TagName")
		if label then label.Text = "Changer" end
		if overlay then
			tag.ZIndex = 6
			for _, child in tag:GetDescendants() do if child:IsA("GuiObject") then child.ZIndex = 7 end end
		end
		tag.Parent = tags
		if changer.tags then changer.tags[tag] = true end
	end
	if tag then tag.Visible, tag.BackgroundColor3 = visible, active and environment.Color3.fromRGB(60, 200, 110) or environment.Color3.fromRGB(150, 90, 255) end
end

function InventoryChanger.decorate(changer, frame, key)
	local state, environment = changer.state, changer.state.context.environment
	local spec = state.pulse and state.pulse.byKey[key]
	if not spec then return end
	pcall(function()
		local container = frame:FindFirstChild("Container")
		local icon = container and container:FindFirstChild("Icon")
		if icon then icon.Image = spec.img end
		local name = frame:FindFirstChild("ItemName")
		if not name or name:FindFirstChild("PulseGrad") then return end
		name.BackgroundColor3 = environment.Color3.new(1, 1, 1)
		local gradient = environment.Instance.new("UIGradient")
		gradient.Name = "PulseGrad"
		local first, second = environment.Color3.fromRGB(46, 242, 255), environment.Color3.fromRGB(150, 92, 255)
		gradient.Color = environment.ColorSequence.new({environment.ColorSequenceKeypoint.new(0, first), environment.ColorSequenceKeypoint.new(0.5, second), environment.ColorSequenceKeypoint.new(1, first)})
		gradient.Parent = name
		changer.gradients[gradient] = true
		if changer.gradientConnection or not state.context.runService then return end
		changer.gradientConnection = state.context.connect(state.context.runService.RenderStepped, function()
			local offset = environment.os.clock() * 0.35 % 2 - 1
			for item in changer.gradients do if item.Parent then item.Offset = environment.Vector2.new(offset, 0) end end
		end)
		table.insert(changer.connections, changer.gradientConnection)
	end)
end

function InventoryChanger.positionMobile(changer, nav, source, button)
	local environment = changer.state.context.environment
	local groups, byPosition = {}, {}
	for _, object in nav:GetChildren() do
		if not object:IsA("GuiObject") then continue end
		changer.shifted[object] = {object.Position, object.Size}
		local position = object.Position
		local key = string.format("%.3f|%.3f", position.Y.Scale, position.X.Scale)
		if not byPosition[key] then
			byPosition[key] = {y = position.Y.Scale, x = position.X.Scale, items = {}}
			table.insert(groups, byPosition[key])
		end
		table.insert(byPosition[key].items, object)
	end
	table.sort(groups, function(a, b) if math.abs(a.y - b.y) > 0.01 then return a.y < b.y end return a.x < b.x end)
	local rows = {}
	for _, group in groups do
		table.insert(rows, group.items)
		if table.find(group.items, source) then table.insert(rows, {button}) end
	end
	local columns = math.max(4, math.ceil(#rows / 2))
	for index, row in rows do
		local column, line = (index - 1) % columns, math.floor((index - 1) / columns)
		for _, object in row do
			object.Position = environment.UDim2.new(column / columns, 0, line * 0.5, 0)
			object.Size = environment.UDim2.new(1 / columns, 0, 0.5, 0)
		end
	end
end

function InventoryChanger.paint(changer)
	local state, page = changer.state, changer.page
	if not page then return end
	local knife, gun = state.active("Knife"), state.active("Gun")
	for key, frame in changer.frames do
		if frame.Parent then pcall(InventoryChanger.tag, changer, frame, true, key == knife or key == gun); InventoryChanger.decorate(changer, frame, key) end
	end
	local equipped = page:FindFirstChild("Equipped")
	local container = equipped and equipped:FindFirstChild("Container")
	if not container then return end
	for _, kind in {"Knife", "Gun"} do
		local slot = container:FindFirstChild(kind)
		local frame = slot and slot:FindFirstChild("Container")
		local key = state.equipped(kind)
		if not frame or not key or not state.weapons[key] then continue end
		pcall(changer.item.DisplayItem, frame, state.weapons[key])
		InventoryChanger.decorate(changer, frame, key)
		frame.Visible = true
		local active = state.enabled and state.active(kind) ~= nil
		pcall(InventoryChanger.tag, changer, frame, active, active, true)
	end
end

function InventoryChanger.filter(changer)
	local query = changer.query:lower()
	for key, frame in changer.frames do
		if not frame.Parent then continue end
		local item = changer.state.weapons[key]
		frame.Visible = query == "" or tostring(item and item.ItemName or key):lower():find(query, 1, true) ~= nil
	end
end

function InventoryChanger.selectTab(changer, kind)
	changer.tab = kind
	local page = changer.page
	if not page then return end
	local items = page:FindFirstChild("Items")
	local title = page:FindFirstChild("TitleBar")
	local tabs = items and items:FindFirstChild("Tabs") or title and title:FindFirstChild("Container")
	for _, tab in tabs and tabs:GetChildren() or {} do
		local view, border = tab:FindFirstChild("View"), tab:FindFirstChild("ViewBorder")
		if view and border then border.Visible, tab.BackgroundTransparency = tab.Name == kind, tab.Name == kind and 0.8 or 0.9 end
	end
	local container = items and items:FindFirstChild("Container")
	for _, section in container and container:GetChildren() or {} do
		if section:IsA("GuiObject") then section.Visible = section.Name == kind end
	end
end

function InventoryChanger.selectItem(changer, key)
	local state = changer.state
	local item = state.weapons[key]
	if not state.enabled or not item or item.ItemType ~= "Knife" and item.ItemType ~= "Gun" then return end
	if state.active(item.ItemType) == key then WeaponSkins.resetSlot(state, item.ItemType) else WeaponSkins.equip(state, key) end
	InventoryChanger.paint(changer)
end

function InventoryChanger.populate(changer)
	local state, page = changer.state, changer.page
	if changer.built or changer.building or not page then return end
	changer.building = true
	local items = page:FindFirstChild("Items")
	local sections = items and items:FindFirstChild("Container")
	for index, skin in state.list do
		if not state.alive or changer.page ~= page or not page.Parent then break end
		local section = sections and sections:FindFirstChild(skin.type)
		local target = section and section:FindFirstChild("Container")
		if not target or not state.weapons[skin.key] or changer.frames[skin.key] then continue end
		local ok, data = pcall(changer.inventory.CreateNewItemData, skin.key, 1, "Weapons")
		if not ok or not data then continue end
		local frame = changer.template:Clone()
		local displayed = pcall(changer.item.DisplayItem, frame, data, nil, true)
		if not displayed then frame:Destroy(); continue end
		frame.LayoutOrder, frame.Parent = index, target
		changer.frames[skin.key] = frame
		local container = frame:FindFirstChild("Container")
		local amount = container and container:FindFirstChild("Amount")
		if amount then amount.Text = "" end
		local button = container and container:FindFirstChild("ActionButton")
		if button then InventoryChanger.connect(changer, button.MouseButton1Click, function() InventoryChanger.selectItem(changer, skin.key) end) end
		if index % 25 == 0 then state.context.environment.task.wait() end
	end
	changer.built, changer.building = changer.page == page and page.Parent ~= nil, false
	InventoryChanger.filter(changer)
	InventoryChanger.paint(changer)
end

function InventoryChanger.preparePage(changer, source, mobile)
	local environment = changer.state.context.environment
	local page = source:Clone()
	page.Name = "Changer"
	if mobile then page.Position = environment.UDim2.new(1, 0, 0, 0) else page.Visible = false end
	for _, object in page:GetDescendants() do if object:IsA("LuaSourceContainer") then object:Destroy() end end
	for _, name in mobile and {"EvoMenu", "Actions"} or {"EvoMenu"} do
		local object = page:FindFirstChild(name)
		if object then object:Destroy() end
	end
	local items = page:FindFirstChild("Items")
	local sections = items and items:FindFirstChild("Container")
	local current = sections and (sections:FindFirstChild("Current") or sections:FindFirstChildWhichIsA("ScrollingFrame"))
	local container = current and current:FindFirstChild("Container")
	if not container then page:Destroy(); return nil end
	for _, object in sections:GetChildren() do if object ~= current and object:IsA("GuiObject") then object:Destroy() end end
	local grid
	for _, object in container:GetChildren() do
		if object:IsA("UIGridLayout") and not grid then grid = object; continue end
		object:Destroy()
	end
	current.CanvasPosition = environment.Vector2.zero
	local gun = current:Clone()
	current.Name, gun.Name, gun.Parent = "Knife", "Gun", sections
	for _, section in {current, gun} do
		local inner = section:FindFirstChild("Container")
		local layout = inner and inner:FindFirstChildWhichIsA("UIGridLayout")
		if not layout then continue end
		local function resize() section.CanvasSize = environment.UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + 5) end
		InventoryChanger.connect(changer, layout:GetPropertyChangedSignal("AbsoluteContentSize"), resize)
		resize()
	end
	local title = page:FindFirstChild("TitleBar")
	local tabs = mobile and title and title:FindFirstChild("Container") or items:FindFirstChild("Tabs")
	if tabs then
		local holiday = tabs:FindFirstChild("Holiday")
		if holiday then holiday:Destroy() end
		for index, pair in {{"Current", "Knife"}, {"Classic", "Gun"}} do
			local tab = tabs:FindFirstChild(pair[1])
			if not tab then continue end
			tab.Name, tab.LayoutOrder = pair[2], index
			for _, label in tab:GetDescendants() do if label:IsA("TextLabel") and (mobile or label.Name == "TabName") then label.Text = pair[2] end end
			local button = tab:FindFirstChild("View")
			if button then InventoryChanger.connect(changer, button.MouseButton1Click, function() InventoryChanger.selectTab(changer, pair[2]) end) end
		end
		local search = tabs:FindFirstChild("Search")
		local searchContainer = search and search:FindFirstChild("Container")
		local input = searchContainer and searchContainer:FindFirstChild("SearchText")
		if input then
			input.Text = ""
			InventoryChanger.connect(changer, input:GetPropertyChangedSignal("Text"), function() changer.query = input.Text; InventoryChanger.filter(changer) end)
		end
	end
	local equipped = page:FindFirstChild("Equipped")
	local equippedContainer = equipped and equipped:FindFirstChild("Container")
	for _, kind in {"Knife", "Gun"} do
		local slot = equippedContainer and equippedContainer:FindFirstChild(kind)
		local frame = slot and slot:FindFirstChild("Container")
		local inner = frame and frame:FindFirstChild("Container")
		local button = inner and inner:FindFirstChild("Button")
		if button then InventoryChanger.connect(changer, button.MouseButton1Click, function()
			if changer.state.enabled and changer.state.active(kind) then WeaponSkins.resetSlot(changer.state, kind); environment.task.defer(function() InventoryChanger.paint(changer) end) end
		end) end
	end
	return page
end

function InventoryChanger.build(changer)
	local state, inventory = changer.state, changer.inventory
	local gui = inventory.GUI and inventory.GUI.MyInventory
	local main = gui and gui.Main
	local root = main and main.Parent
	local nav = root and root:FindFirstChild("Nav")
	local source = main and main:FindFirstChild("Weapons")
	local desktopButton = nav and nav:FindFirstChild("Weapons")
	local mobileNav = nav and nav:FindFirstChild("Main")
	local mobileButton = mobileNav and mobileNav:FindFirstChild("Weapons")
	local mobile = desktopButton == nil and mobileButton ~= nil
	local navParent, sourceButton = mobile and mobileNav or nav, desktopButton or mobileButton
	if not main or not nav or not source or not sourceButton or not source:FindFirstChild("Items") then return end
	if changer.page and changer.page.Parent == main and changer.button and changer.button.Parent == navParent then return end
	InventoryChanger.drop(changer)
	changer.mobile = mobile
	local page = InventoryChanger.preparePage(changer, source, mobile)
	if not page then return end
	page.Parent, changer.page = main, page
	local button = sourceButton:Clone()
	button.Name = "Changer"
	for _, object in button:GetDescendants() do
		if object:IsA("LuaSourceContainer") then object:Destroy(); continue end
		if object:IsA("TextLabel") and (object.Name == "TabName" or object.Name == "FramName") then object.Text = "Changer" end
	end
	local icon = button:FindFirstChild("Icon")
	if icon and state.DB.Pulse_Oni then icon.Image = state.DB.Pulse_Oni.img end
	if mobile then InventoryChanger.positionMobile(changer, navParent, sourceButton, button) end
	if not mobile then
		for _, other in navParent:GetChildren() do
			if other:IsA("GuiButton") and other.LayoutOrder > sourceButton.LayoutOrder then
				changer.shifted[other], other.LayoutOrder = other.LayoutOrder, other.LayoutOrder + 1
			end
		end
		button.LayoutOrder = sourceButton.LayoutOrder + 1
	end
	button.Parent, changer.button = navParent, button
	InventoryChanger.connect(changer, button.MouseButton1Click, function()
		if mobile then
			pcall(function() nav:TweenPosition(state.context.environment.UDim2.new(-1, 0, 0, 0), "Out", "Quad", 0.2, true) end)
			pcall(function() page:TweenPosition(state.context.environment.UDim2.new(0, 0, 0, 0), "Out", "Quad", 0.2, true) end)
		end
		if not mobile then
			for _, item in nav:GetChildren() do if item:IsA("TextButton") then pcall(function() item.Style = item == button and state.context.environment.Enum.ButtonStyle.RobloxRoundDefaultButton or state.context.environment.Enum.ButtonStyle.RobloxRoundButton end) end end
			for _, child in main:GetChildren() do if child:IsA("GuiObject") then child.Visible = child == page end end
		end
		InventoryChanger.selectTab(changer, changer.tab)
		InventoryChanger.paint(changer)
		state.context.environment.task.spawn(function() InventoryChanger.populate(changer) end)
	end)
	if not mobile then
		for _, other in nav:GetChildren() do
			if other:IsA("TextButton") and other ~= button then
				InventoryChanger.connect(changer, other.MouseButton1Click, function()
					pcall(function() button.Style = state.context.environment.Enum.ButtonStyle.RobloxRoundButton end)
				end)
			end
		end
	end
	if mobile then
		local title = page:FindFirstChild("TitleBar")
		local titleContainer = title and title:FindFirstChild("Container")
		local back = titleContainer and titleContainer:FindFirstChild("Back") or title and title:FindFirstChild("Back") or page:FindFirstChild("Back")
		if back then InventoryChanger.connect(changer, back.MouseButton1Click, function()
			pcall(function() nav:TweenPosition(state.context.environment.UDim2.new(0, 0, 0, 0), "Out", "Quad", 0.2, true) end)
			pcall(function() page:TweenPosition(state.context.environment.UDim2.new(1, 0, 0, 0), "Out", "Quad", 0.2, true) end)
		end) end
	end
	InventoryChanger.selectTab(changer, changer.tab)
end

function InventoryChanger.isFake(changer, key)
	local state = changer.state
	return state.DB[key] ~= nil and state.realOwned ~= nil and not state.realOwned[key]
end

function InventoryChanger.bindInventoryItem(changer, key, entry)
	local frame = entry and entry.Frame
	if not frame or not frame.Parent then return end
	local container = frame:FindFirstChild("Container")
	local button = container and container:FindFirstChild("ActionButton")
	if not button then return end
	local state = changer.state
	if InventoryChanger.isFake(changer, key) then
		if entry.PulseConn == button then return end
		if entry.EquipConnection then pcall(function() entry.EquipConnection:Disconnect() end) end
		local connection = state.context.connect(button.MouseButton1Click, function() InventoryChanger.selectItem(changer, key) end)
		entry.EquipConnection, entry.PulseConn = connection, button
		changer.inventoryItems[entry] = connection
		table.insert(changer.inventoryConnections, connection)
		return
	end
	if entry.PulseReal == button then return end
	local connection = state.context.connect(button.MouseButton1Click, function()
		local item = state.weapons[key]
		if not item or item.EvoBaseID or item.ItemType ~= "Knife" and item.ItemType ~= "Gun" then return end
		if state.realOwned and state.realOwned[key] then state.realName[item.ItemType] = key end
	end)
	entry.PulseReal = button
	changer.inventoryItems[entry] = changer.inventoryItems[entry] or false
	table.insert(changer.inventoryConnections, connection)
end

function InventoryChanger.bindInventory(changer)
	local inventory = changer.inventory.MyInventory
	local groups = inventory and inventory.Data and inventory.Data.Weapons
	if type(groups) ~= "table" then return end
	for _, group in groups do
		if type(group) ~= "table" then continue end
		for key, entry in group do InventoryChanger.bindInventoryItem(changer, key, entry) end
	end
end

function InventoryChanger.paintInventory(changer)
	local state, inventory = changer.state, changer.inventory
	local current = inventory.MyInventory
	local groups = current and current.Data and current.Data.Weapons
	if type(groups) == "table" then
		local knife, gun = state.active("Knife"), state.active("Gun")
		for _, group in groups do
			if type(group) ~= "table" then continue end
			for key, entry in group do
				local frame = entry and entry.Frame
				if not frame or not frame.Parent then continue end
				local fake = state.enabled and InventoryChanger.isFake(changer, key)
				pcall(InventoryChanger.tag, changer, frame, fake, fake and (key == knife or key == gun))
			end
		end
	end
	local gui = inventory.GUI and inventory.GUI.MyInventory
	local main = gui and gui.Main
	local weapons = main and main:FindFirstChild("Weapons")
	local equipped = weapons and weapons:FindFirstChild("Equipped")
	local container = equipped and equipped:FindFirstChild("Container")
	if not container then return end
	for _, kind in {"Knife", "Gun"} do
		local slot = container:FindFirstChild(kind)
		local frame = slot and slot:FindFirstChild("Container")
		local key = state.equipped(kind)
		if not frame or not key or not state.weapons[key] then continue end
		pcall(changer.item.DisplayItem, frame, state.weapons[key])
		InventoryChanger.decorate(changer, frame, key)
		frame.Visible = true
		local active = state.enabled and state.active(kind) ~= nil
		pcall(InventoryChanger.tag, changer, frame, active, active, true)
		local inner = frame:FindFirstChild("Container")
		local button = inner and inner:FindFirstChild("Button")
		if not button or changer.inventoryPanels[button] then continue end
		local connection = state.context.connect(button.MouseButton1Click, function()
			if not state.enabled or not state.active(kind) then return end
			WeaponSkins.resetSlot(state, kind)
			state.context.environment.task.defer(function() pcall(InventoryChanger.paintInventory, changer) end)
		end)
		changer.inventoryPanels[button] = connection
		table.insert(changer.inventoryConnections, connection)
	end
end

function InventoryChanger.stopInventory(changer)
	for _, connection in changer.inventoryConnections do pcall(function() connection:Disconnect() end) end
	for entry, connection in changer.inventoryItems do
		if entry.EquipConnection == connection then entry.EquipConnection = nil end
		entry.PulseConn, entry.PulseReal = nil, nil
	end
	for tag in changer.tags do pcall(function() tag:Destroy() end) end
	changer.inventoryConnections, changer.inventoryItems, changer.inventoryPanels, changer.tags = {}, {}, {}, {}
end

function WeaponSkins.installInventory(state)
	local environment = state.context.environment
	local modules = state.context.storage:WaitForChild("Modules")
	local inventoryModule = modules:WaitForChild("InventoryModule", 10)
	local itemModule = modules:WaitForChild("ItemModule", 10)
	if not inventoryModule or not itemModule then return end
	local inventoryOk, inventory = pcall(environment.require, inventoryModule)
	local itemOk, item = pcall(environment.require, itemModule)
	if not state.alive or not inventoryOk or not itemOk or type(inventory) ~= "table" or type(item) ~= "table" then return end
	local tradeModule = modules:FindFirstChild("TradeModule")
	if tradeModule then
		local ok, trade = pcall(environment.require, tradeModule)
		if ok and type(trade) == "table" then state.tradeModule = trade end
	end
	local globals = environment.getgenv()
	local originals = globals.__pulseInvOrig
	if type(originals) ~= "table" or originals.mod ~= inventory then
		originals = {mod = inventory, ceb = inventory.ConnectEquipButtons, ue = inventory.UpdateEquip,
			gi = inventory.GenerateInventory, ui = inventory.UpdateInventory}
		globals.__pulseInvOrig = originals
	end
	originals.gi = originals.gi or inventory.GenerateInventory
	originals.ui = originals.ui or inventory.UpdateInventory
	local wrappers = {}
	if type(originals.gi) == "function" then
		wrappers.GenerateInventory = function(...)
			if state.alive then return WeaponSkins.withRealOwned(state, originals.gi, ...) end
			return originals.gi(...)
		end
		inventory.GenerateInventory = wrappers.GenerateInventory
	end
	if type(originals.ui) == "function" then
		wrappers.UpdateInventory = function(owner, target, ...)
			local trade = state.tradeModule and state.tradeModule.TradeInventory
			if state.alive and target ~= nil and (target == trade or target == inventory.MyInventory) then
				return WeaponSkins.withRealOwned(state, originals.ui, owner, target, ...)
			end
			return originals.ui(owner, target, ...)
		end
		inventory.UpdateInventory = wrappers.UpdateInventory
	end
	local template = inventoryModule:FindFirstChild("NewItem")
	local changer = {state = state, inventory = inventory, item = item, template = template, frames = {}, connections = {}, shifted = {},
		gradients = setmetatable({}, {__mode = "k"}), inventoryConnections = {}, inventoryItems = {}, inventoryPanels = {}, tags = {}, tab = "Knife", query = ""}
	state.inventory = {changer = changer, on = false}
	state.inventory.refresh = function()
		pcall(InventoryChanger.paintInventory, changer)
		pcall(InventoryChanger.paint, changer)
	end
	state.scrubTrade = function() WeaponSkins.scrubTrade(state) end
	if type(originals.ceb) == "function" then
		wrappers.ConnectEquipButtons = function(...)
			if state.alive then pcall(InventoryChanger.bindInventory, changer) end
			return originals.ceb(...)
		end
		inventory.ConnectEquipButtons = wrappers.ConnectEquipButtons
	end
	if type(originals.ue) == "function" then
		wrappers.UpdateEquip = function(owner, target, ...)
			local result = table.pack(originals.ue(owner, target, ...))
			if state.alive and target == inventory.MyInventory then environment.task.defer(state.inventory.refresh) end
			return table.unpack(result, 1, result.n)
		end
		inventory.UpdateEquip = wrappers.UpdateEquip
	end
	if template and environment.task then
		table.insert(state.listeners, state.inventory.refresh)
		environment.task.spawn(function()
			while state.alive and state.context.isAlive() do
				pcall(InventoryChanger.build, changer)
				environment.task.wait(1)
			end
		end)
	end
	state.stopInventory = function()
		InventoryChanger.drop(changer)
		InventoryChanger.stopInventory(changer)
		if inventory.ConnectEquipButtons == wrappers.ConnectEquipButtons then inventory.ConnectEquipButtons = originals.ceb end
		if inventory.UpdateEquip == wrappers.UpdateEquip then inventory.UpdateEquip = originals.ue end
		if type(originals.ceb) == "function" then
			if environment.task then environment.task.spawn(function() pcall(originals.ceb) end) else pcall(originals.ceb) end
		end
		if inventory.GenerateInventory == wrappers.GenerateInventory then inventory.GenerateInventory = originals.gi end
		if inventory.UpdateInventory == wrappers.UpdateInventory then inventory.UpdateInventory = originals.ui end
	end
	state.inventory.destroy = state.stopInventory
	local pending = false
	if state.inventoryChanged and state.inventoryChanged.Event and environment.task then
		state.context.connect(state.inventoryChanged.Event, function()
			if pending then return end
			pending = true
			environment.task.delay(2, function()
				pending = false
				if not state.alive then return end
				WeaponSkins.readRealInventory(state)
				state.inventory.refresh()
			end)
		end)
	end
	pcall(WeaponSkins.scrubTrade, state)
end

function WeaponSkins.load(state)
	local context, environment = state.context, state.context.environment
	local data = rawget(environment.getgenv(), "__pulseData")
	local encoded = data and data.take and data.take("skins_db") or "{}"
	local database = context.http:JSONDecode(encoded)
	if type(database) ~= "table" then return end
	for key, entry in database do if type(entry) ~= "table" then database[key] = nil end end
	if database.Hallowgun then database.Hallowgun.g = {-0.66, -0.24, 0, 0, 0, -1, 0, 1, 0, 1, 0, 0} end
	if database.ElderwoodGun then database.ElderwoodGun.g = {-0.52, -0.25, 0, 0, 0, -1, 0, 1, 0, 1, 0, 0} end

	local storage = context.storage
	local sync = environment.require(storage:WaitForChild("Database"):WaitForChild("Sync"))
	local profile = environment.require(storage:WaitForChild("Modules"):WaitForChild("ProfileData"))
	local equip = environment.require(storage:WaitForChild("ClientServices"):WaitForChild("EquipService"))
	local remotes = storage:WaitForChild("Remotes")
	local inventory = remotes:WaitForChild("Inventory"):WaitForChild("InventoryDataChanged")
	local fullInventory = remotes:WaitForChild("Extras"):WaitForChild("GetFullInventory")
	if not state.alive or not context.isAlive() then return end

	PulseSkin.register(state, database, sync)
	state.DB, state.weapons, state.profile, state.equipService, state.inventoryChanged = database, sync.Weapons, profile, equip, inventory
	state.fullInventory = fullInventory
	state.list = SkinCatalog.list(database)
	WeaponSkins.readRealInventory(state)
	if not state.alive or not context.isAlive() then return end
	pcall(function()
		local saved = context.http:JSONDecode(environment.readfile("PulseHub/mm2_skins.json"))
		if type(saved) ~= "table" then return end
		for _, kind in {"Knife", "Gun"} do
			if type(saved[kind]) == "string" and database[saved[kind]] then state.selected[kind] = saved[kind] end
		end
	end)
	state.ready = true
	environment.task.spawn(function() pcall(WeaponSkins.installInventory, state) end)
	if context.collection and context.workspace then ThrownSkins.install(state) end
	WeaponSkins.updateCatalog(state)
	for _, kind in {"Knife", "Gun"} do
		if state.selected[kind] and profile.Weapons.Equipped[kind] ~= state.selected[kind] then WeaponSkins.equip(state, state.selected[kind]) end
	end
	pcall(function()
		context.connect(equip.EquippedChanged.Event, function(kind)
			if kind ~= "Knife" and kind ~= "Gun" then return end
			environment.task.defer(function()
				local current = profile.Weapons.Equipped[kind]
				state.selected[kind] = database[current] and current ~= state.realName[kind] and current or nil
				WeaponSkins.save(state)
				WeaponSkins.refresh(state)
			end)
		end)
	end)
	local backpack = context.player:FindFirstChildOfClass("Backpack") or context.player:WaitForChild("Backpack", 10)
	if backpack then context.connect(backpack.ChildAdded, function(tool)
		if WeaponSkins.toolType(state, tool) then environment.task.spawn(function() pcall(WeaponSkins.applyTool, state, tool) end) end
	end) end
	context.connect(context.player.CharacterAdded, function(character)
		environment.task.wait(0.5)
		WeaponSkins.watchCharacter(state, character)
	end)
	if context.player.Character then WeaponSkins.watchCharacter(state, context.player.Character) end
	WeaponSkins.refresh(state)
	environment.task.spawn(function()
		while state.alive and context.isAlive() do
			environment.task.wait(1)
			if state.alive and context.isAlive() then pcall(WeaponSkins.repairDisplays, state) end
		end
	end)
	environment.task.spawn(function()
		local previous = {}
		while state.alive and context.isAlive() do
			for _, kind in {"Knife", "Gun"} do
				local equipped = profile.Weapons.Equipped[kind]
				if equipped ~= previous[kind] then
					previous[kind] = equipped
					state.selected[kind] = database[equipped] and equipped ~= state.realName[kind] and equipped or nil
					WeaponSkins.save(state)
					WeaponSkins.refresh(state)
				end
			end
			environment.task.wait(0.1)
		end
	end)
end

function WeaponSkins.install(context)
	local environment, globals = context.environment, context.environment.getgenv()
	for _, name in {"__pulseSkins", "__pulseWeaponSkins"} do
		local previous = globals[name]
		if previous and previous.destroy then pcall(previous.destroy); globals[name] = nil end
	end
	local connections, originalConnect = {}, context.connect
	context = table.clone(context)
	context.connect = function(signal, callback)
		local connection = originalConnect(signal, callback)
		table.insert(connections, connection)
		return connection
	end
	local state = {context = context, DB = {}, list = {}, realName = {}, selected = {}, displays = {}, displayTokens = {}, tools = {}, conns = connections,
		listeners = {}, repairing = {}, rm = {players = {}, tools = {}}, enabled = true, alive = true, ready = false, knifeBelt = false, kind = "All", rarity = "Any", query = "",
		model = {context = context, meshes = {}, loading = {}}}
	state.owner = context.http:GenerateGUID(false)
	state.birth = context.workspace:GetServerTimeNow()
	state.remoteSet = function(player, payload) RemoteSkins.set(state, player, payload) end
	state.rmClear = function(player) RemoteSkins.clear(state, player) end
	state.remoteClearAll = function() RemoteSkins.clearAll(state) end
	if context.players and context.players.PlayerRemoving then context.connect(context.players.PlayerRemoving, state.rmClear) end
	if context.runService and context.runService.RenderStepped then
		context.connect(context.runService.RenderStepped, function()
			if next(state.rm.players) then pcall(RemoteSkins.enforce, state) end
			if next(state.displays) then pcall(WeaponSkins.renderDisplays, state) end
		end)
	end
	state.equip = function(key) WeaponSkins.equip(state, key) end
	state.snapshot = function() return WeaponSkins.snapshot(state) end
	state.applySnapshot = function(value) WeaponSkins.applySnapshot(state, value) end
	state.spawn = function(key, amount) WeaponSkins.spawn(state, key, amount) end
	state.equipSkin = function(kind, key)
		if type(key) == "string" and state.DB[key] then
			if state.profile.Weapons.Equipped[kind] == key then return end
			state.profile.Weapons.Equipped[kind] = key
			pcall(function() state.equipService.EquippedChanged:Fire(kind, key) end)
			return
		end
		WeaponSkins.resetSlot(state, kind)
	end
	state.resetSlot = function(kind) WeaponSkins.resetSlot(state, kind) end
	state.reset = function() WeaponSkins.resetSlot(state, "Knife"); WeaponSkins.resetSlot(state, "Gun") end
	state.setEnabled = function(value) WeaponSkins.setEnabled(state, value) end
	state.setKnifeBelt = function(value)
		state.knifeBelt = not not value
		WeaponSkins.clearDisplay(state, "Knife")
		if state.ready then environment.task.spawn(function() pcall(WeaponSkins.applyDisplay, state, "Knife") end) end
	end
	state.equipped = function(kind) return state.ready and state.profile.Weapons.Equipped[kind] or nil end
	state.active = function(kind)
		local current = state.equipped(kind)
		return current and state.DB[current] and current ~= state.realName[kind] and current or nil
	end
	state.info = function(key) return key and state.weapons and state.weapons[key] or nil end
	state.onChange = function(callback) table.insert(state.listeners, callback) end
	state.knifeTargets = function()
		local result, display = {}, state.displays.Knife
		if display and display.visual and display.visual.Parent then result.display = display.visual end
		for tool, entry in state.tools do
			if WeaponSkins.toolType(state, tool) == "Knife" and entry.visual and entry.visual.Parent then result.handle = entry.visual end
		end
		return result
	end
	state.destroy = function()
		state.alive = false
		for _, connection in state.conns do pcall(function() connection:Disconnect() end) end
		WeaponSkins.setEnabled(state, false)
		for _, mesh in state.model.meshes do pcall(function() mesh:Destroy() end) end
		if state.pulse then
			for _, template in state.pulse.templates do if template then pcall(function() template:Destroy() end) end end
			if globals.__pulseCX == state.pulse then globals.__pulseCX = nil end
		end
	end
	globals.__pulseWeaponSkins = state
	WeaponSkins.controls(state)
	environment.task.spawn(function()
		local ok, raised = pcall(WeaponSkins.load, state)
		if not ok then state.lastError = tostring(raised) end
	end)
	return state, state.destroy
end

local GameEffects = {}
local GAME_EFFECT_CLASSES = {ParticleEmitter = true, Trail = true, Beam = true, Fire = true, Smoke = true, Sparkles = true, PointLight = true, SpotLight = true}
local GAME_EFFECT_FIELDS = {cls = true, nm = true, par = true, attCF = true, a0CF = true, a1CF = true}

function GameEffects.load(state)
	if state.data or state.loading then return end
	state.loading = true
	state.context.environment.task.spawn(function()
		local source = state.context.environment.getgenv().__pulseData
		local contents = source and source.take and source.take("fx_effects")
		if contents then state.data = {} end
		state.loading = false
	end)
end

function GameEffects.unlock(state)
	local profile = state.profile
	if not profile or not profile.Effects then return end
	pcall(function()
		profile.Effects.Owned = table.clone(state.keys)
		local inventory = state.context.storage.Remotes.Inventory
		local changed = inventory:FindFirstChild("InventoryDataChanged")
		if changed then pcall(changed.Fire, changed, "Effects"); pcall(changed.Fire, changed) end
		local profileChanged = inventory:FindFirstChild("ProfileDataChanged")
		if profileChanged then pcall(profileChanged.Fire, profileChanged) end
	end)
end

function GameEffects.clear(target)
	if not target then return end
	for _, object in target:GetDescendants() do
		if object:GetAttribute("PulseFX") then pcall(function() object:Destroy() end) end
	end
end

function GameEffects.enum(environment, value)
	if type(value) ~= "string" then return value end
	local group, name = value:match("^Enum%.([%w_]+)%.([%w_]+)$")
	if not group then return value end
	local ok, item = pcall(function() return environment.Enum[group][name] end)
	return ok and item or nil
end

function GameEffects.build(state, target, key)
	local data = state.data and state.data[key]
	if not target or not data then return 0 end
	local environment, count = state.context.environment, 0
	for _, spec in data do
		if not GAME_EFFECT_CLASSES[spec.cls] then continue end
		local ok = pcall(function()
			local effect = environment.Instance.new(spec.cls)
			for property, value in spec do
				if GAME_EFFECT_FIELDS[property] then continue end
				value = GameEffects.enum(environment, value)
				if value ~= nil then pcall(function() effect[property] = value end) end
			end
			effect:SetAttribute("PulseFX", true)
			if spec.cls == "Trail" or spec.cls == "Beam" then
				local first, second = environment.Instance.new("Attachment"), environment.Instance.new("Attachment")
				first:SetAttribute("PulseFX", true)
				second:SetAttribute("PulseFX", true)
				first.CFrame, second.CFrame = spec.a0CF or environment.CFrame.new(0, 0, 0.4), spec.a1CF or environment.CFrame.new(0, 0, -0.4)
				first.Parent, second.Parent = target, target
				pcall(function() effect.Attachment0, effect.Attachment1 = first, second end)
				effect.Parent = target
				return
			end
			if spec.attCF then
				local attachment = environment.Instance.new("Attachment")
				attachment:SetAttribute("PulseFX", true)
				attachment.CFrame, attachment.Parent, effect.Parent = spec.attCF, target, attachment
				return
			end
			effect.Parent = target
		end)
		if ok then count += 1 end
	end
	return count
end

function GameEffects.targets(state)
	local context = state.context
	local character = context.player.Character
	local targets = {}
	local knife = character and character:FindFirstChild("Knife")
	local handle = knife and knife:FindFirstChild("Handle")
	if handle then table.insert(targets, handle) end
	local reference = character and character:FindFirstChild("DisplayRefKnife")
	local display = reference and reference.Value
	local skins = context.environment.getgenv().__pulseWeaponSkins
	local replacement = skins and skins.knifeTargets and skins.knifeTargets() or {}
	if replacement.handle then
		if handle then
			GameEffects.clear(handle)
			targets[#targets] = replacement.handle
		else
			table.insert(targets, replacement.handle)
		end
	end
	if replacement.display then
		if display then GameEffects.clear(display) end
		display = replacement.display
	end
	if display then table.insert(targets, display) end
	return targets
end

function GameEffects.refresh(state)
	local targets, built = GameEffects.targets(state), 0
	for _, target in targets do
		GameEffects.clear(target)
		if state.current and state.current ~= "" then built += GameEffects.build(state, target, state.current) end
	end
	state.context.environment.getgenv().__pulseEffectState = {unlockOn = state.enabled, curKey = tostring(state.current),
		dbHasKey = state.data ~= nil and state.current ~= nil and state.data[state.current] ~= nil, handles = #targets, built = built, dbLoaded = state.data ~= nil}
end

function GameEffects.step(state)
	if not state.enabled or not state.alive then return end
	local now = state.context.environment.os.clock()
	if now - state.checked >= 0.35 then
		state.checked = now
		local equipped = state.profile and state.profile.Effects and state.profile.Effects.Equipped
		local current = equipped and equipped[1]
		if current ~= state.current then state.current = current; GameEffects.refresh(state) end
	end
	if now - state.repaired >= 0.6 then
		state.repaired = now
		if state.current and state.current ~= "" then
			for _, target in GameEffects.targets(state) do
				local present = false
				for _, object in target:GetDescendants() do if object:GetAttribute("PulseFX") then present = true; break end end
				if not present then GameEffects.build(state, target, state.current) end
			end
		end
	end
	if now - state.unlocked >= 3 then state.unlocked = now; GameEffects.unlock(state) end
end

function GameEffects.install(context)
	local environment = context.environment
	local now = environment.os.clock()
	local state = {context = context, keys = {}, enabled = false, alive = true, checked = now, repaired = now, unlocked = now}
	pcall(function()
		state.profile = environment.require(context.storage.Modules.ProfileData)
		local effects = environment.require(context.storage.Database.Sync).Effects
		for key, entry in effects do if type(entry) == "table" then table.insert(state.keys, tostring(key)) end end
	end)
	context.section:Header("Knife Effects")
	context.section:Toggle({Text = "Unlock All Effects (game menu)", Default = false, Callback = function(value)
		state.enabled = not not value
		if state.enabled then GameEffects.load(state); GameEffects.unlock(state); return end
		state.current = nil
		for _, target in GameEffects.targets(state) do GameEffects.clear(target) end
	end})
	context.connect(context.runService.Heartbeat, function() GameEffects.step(state) end)
	context.connect(context.player.CharacterAdded, function()
		environment.task.wait(0.9)
		if state.enabled and state.alive then GameEffects.refresh(state) end
	end)
	return state, function()
		state.alive, state.enabled = false, false
		for _, target in GameEffects.targets(state) do GameEffects.clear(target) end
	end
end

function GameEffects.Kill.effectContext(state, character, killer)
	local root = character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("UpperTorso") or character:FindFirstChild("Torso")
	if not root then return nil end

	local chest = character:FindFirstChild("UpperTorso") or character:FindFirstChild("Torso") or root
	local head = character:FindFirstChild("Head") or root
	return {char = character, root = root.Position, chest = chest.Position, head = head.Position, killer = killer,
		ignore = {a = character, b = state.context.player.Character, c = state.context.workspace.CurrentCamera}}
end

function GameEffects.Kill.play(state, name, options, character, killer)
	if name == "Random" or not state.effects[name] then name = options[math.random(1, #options)] end
	local effect = state.effects[name]
	local context = character and GameEffects.Kill.effectContext(state, character, killer)
	if not effect or not context then return end

	context.orig = state.originals[character]
	context.ignore.d = context.orig
	context.ghostTakes = GameEffects.Kill.ghostEffects[name] or false
	if context.orig then state.contexts[character] = context end

	local ok, message = pcall(effect, context)
	if not ok then state.lastError = tostring(message) end
end

function GameEffects.Kill.evidence(state, victim, character, position)
	local now = state.context.environment.os.clock()

	for index = #state.shots, 1, -1 do
		local shot = state.shots[index]
		if now - shot.time > 0.9 then break end
		if shot.shooter == character then continue end

		if shot.hit and shot.hit:IsDescendantOf(character) then return "gun", shot.shooter end
		if position and shot.position and (shot.position - position).Magnitude < 4.5 then return "gun", shot.shooter end
	end

	if not position then return nil end

	for _, player in state.context.players:GetPlayers() do
		local other = player ~= victim and player.Character
		local root = other and other:FindFirstChild("Knife") and other:FindFirstChild("HumanoidRootPart")
		if root and (root.Position - position).Magnitude < 12 then return "knife", other end
	end

	for index = #state.blades, 1, -1 do
		local blade = state.blades[index]
		if now - blade.time > 0.6 then break end
		if (blade.position - position).Magnitude < 7 then return "knife", nil end
	end

	return nil
end

function GameEffects.Kill.isMine(state, weapon, roles)
	local player = state.context.player
	local data = roles[player.Name]
	local role = data and data.Role
	if weapon == "knife" then return role == "Murderer" end

	local character = player.Character
	local backpack = player:FindFirstChild("Backpack")
	return character and character:FindFirstChild("Gun") ~= nil or backpack and backpack:FindFirstChild("Gun") ~= nil or role == "Sheriff"
end

function GameEffects.Kill.killer(state, victim, roles, weapon)
	local victimRoot = victim.Character and victim.Character:FindFirstChild("HumanoidRootPart")
	local nearest, distance

	for _, player in state.context.players:GetPlayers() do
		local character = player ~= victim and player.Character
		if not character then continue end

		local data = roles[player.Name]
		local matches = weapon == "knife" and data and data.Role == "Murderer" or weapon ~= "knife" and character:FindFirstChild("Gun") ~= nil
		if not matches then continue end

		local part = character:FindFirstChild("UpperTorso") or character:FindFirstChild("Torso") or character:FindFirstChild("HumanoidRootPart")
		if not part then continue end

		local current = victimRoot and (part.Position - victimRoot.Position).Magnitude or 0
		if not distance or current < distance then nearest, distance = part, current end
	end

	return nearest
end

function GameEffects.Kill.recordShot(state, shooter, position, hit)
	local environment = state.context.environment
	if environment.typeof(shooter) ~= "Instance" then shooter = nil end

	while shooter and not (shooter:IsA("Model") and shooter:FindFirstChildOfClass("Humanoid")) do
		shooter = shooter.Parent
	end

	table.insert(state.shots, {time = environment.os.clock(), shooter = shooter,
		position = environment.typeof(position) == "Vector3" and position or nil,
		hit = environment.typeof(hit) == "Instance" and hit or nil})
	if #state.shots > 16 then table.remove(state.shots, 1) end
end

function GameEffects.Kill.recordBlade(state, position)
	table.insert(state.blades, {time = state.context.environment.os.clock(), position = position})
	if #state.blades > 300 then table.remove(state.blades, 1) end
end

function GameEffects.Kill.waitEvidence(state, victim, character, position)
	local environment = state.context.environment
	local started = environment.os.clock()

	while state.alive do
		local weapon, shooter = GameEffects.Kill.evidence(state, victim, character, position)
		if weapon then return weapon, shooter end
		if environment.os.clock() - started > 0.35 then return nil end
		environment.task.wait()
	end

	return nil
end

function GameEffects.Kill.appearance(character)
	local ok, value = pcall(function()
		local torso = character:FindFirstChild("UpperTorso")
		local shirt = character:FindFirstChildOfClass("Shirt")
		local pants = character:FindFirstChildOfClass("Pants")
		return table.concat({torso and torso:IsA("MeshPart") and torso.MeshId or "", shirt and shirt.ShirtTemplate or "", pants and pants.PantsTemplate or ""}, "|")
	end)

	return ok and value or "||"
end

function GameEffects.Kill.recordRagdoll(state, model)
	if model.Name ~= "Raggy" or not model:IsA("Model") then return end

	local now, recent = state.context.environment.os.clock(), {}
	for _, ragdoll in state.ragdolls do
		if now - ragdoll.time < 6 and ragdoll.model.Parent then table.insert(recent, ragdoll) end
	end

	table.insert(recent, {model = model, time = now})
	state.ragdolls = recent
end

function GameEffects.Kill.findRagdoll(state, position, appearance, deathTime, timeout)
	local environment = state.context.environment
	local deadline = environment.os.clock() + (timeout or 0.8)

	repeat
		local nearest, distance
		for _, ragdoll in state.ragdolls do
			local model = ragdoll.model
			if not model.Parent or state.originals[model] or ragdoll.time < deathTime - 0.6 then continue end

			local root = model:FindFirstChild("LowerTorso") or model:FindFirstChild("HumanoidRootPart")
			if not root then continue end

			local current = (root.Position - position).Magnitude
			if current >= 14 or appearance ~= "||" and GameEffects.Kill.appearance(model) ~= appearance then continue end
			if not distance or current < distance then nearest, distance = model, current end
		end

		if nearest then return nearest end
		environment.task.wait()
	until environment.os.clock() > deadline

	return nil
end

function GameEffects.Kill.hide(character)
	for _, object in character:GetDescendants() do
		if object:IsA("BasePart") then
			pcall(function() object.LocalTransparencyModifier = 1 end)
			pcall(function() object.CollisionGroup = "Characters" end)
			continue
		end

		if object:IsA("Decal") then pcall(function() object.LocalTransparencyModifier = 1 end) end
	end
end

function GameEffects.Kill.onDeath(state, victim, character)
	if not state.alive or not state.on then return end

	local environment = state.context.environment
	local torso = character:FindFirstChild("LowerTorso") or character:FindFirstChild("HumanoidRootPart")
	local root = character:FindFirstChild("HumanoidRootPart") or torso
	local position = root and root.Position
	local torsoPosition = torso and torso.Position
	local appearance = torso and GameEffects.Kill.appearance(character)
	local deathTime = environment.os.clock()
	local weapon, shooter = GameEffects.Kill.waitEvidence(state, victim, character, position)
	if not weapon or not state.alive or not state.on then return end

	local roles = state.context.roles and state.context.roles() or {}
	if state.mine then
		if shooter and shooter ~= state.context.player.Character then return end
		if not shooter and not GameEffects.Kill.isMine(state, weapon, roles) then return end
	end

	local killer = shooter and (shooter:FindFirstChild("UpperTorso") or shooter:FindFirstChild("Torso") or shooter:FindFirstChild("HumanoidRootPart"))
		or GameEffects.Kill.killer(state, victim, roles, weapon)
	local target = character

	if torso then
		for _, tracked in {state.originals, state.taken, state.contexts} do
			for model in tracked do if not model.Parent then tracked[model] = nil end end
		end

		local ragdoll = GameEffects.Kill.findRagdoll(state, torsoPosition, appearance, deathTime)
		if ragdoll then
			state.originals[ragdoll] = character
			GameEffects.Kill.hide(character)
			target = ragdoll
		else
			environment.task.spawn(function()
				local delayed = GameEffects.Kill.findRagdoll(state, torsoPosition, appearance, deathTime, 4)
				if not delayed or not state.alive then return end
				state.originals[delayed], state.taken[delayed] = character, true
				GameEffects.Kill.hide(delayed)
			end)
		end
	end

	if not state.alive then return end

	local now = environment.os.clock()
	local slot = math.max(now, state.slotAt or 0)
	state.slotAt = slot + 0.03
	if slot > now then environment.task.wait(slot - now) end
	if not state.alive then return end

	state.play(weapon == "gun" and state.gun or state.knife, weapon, target, killer)
end

	state.folder = folder
	return folder
end

function GameEffects.Kill.delay(state, duration, callback)

end

function GameEffects.Kill.destroyLater(state, object, duration)
	GameEffects.Kill.delay(state, duration, function() pcall(function() object:Destroy() end) end)
end

function GameEffects.Kill.tween(state, object, duration, properties, style, direction)

		direction or environment.Enum.EasingDirection.Out)

	tween:Play()
	return tween

	local environment = state.context.environment
	local values, points = {...}, {}
	for index = 1, #values, 2 do
		table.insert(points, environment.NumberSequenceKeypoint.new(values[index], values[index + 1]))
	end
	return environment.NumberSequence.new(points)
end

function GameEffects.Kill.animate(state, duration, update, finish)
	local context, environment = state.context, state.context.environment
	local started = environment.os.clock()
	local connection

	connection = context.runService.RenderStepped:Connect(function()
		local progress = math.clamp((environment.os.clock() - started) * state.speed / duration, 0, 1)
		local ok, message = pcall(update, progress)
		if not ok then state.driveError = tostring(message) end
		if progress < 1 and ok and state.alive then return end

		connection:Disconnect()
		if not finish then return end
		local finished, failure = pcall(finish)
		if not finished then state.driveError = tostring(failure) end
	end)
	return connection
end

function GameEffects.Kill.particle(state, parent, properties, count, delay)
	local environment = state.context.environment
	local emitter = environment.Instance.new("ParticleEmitter")
	emitter.Enabled, emitter.LightEmission, emitter.LightInfluence, emitter.Brightness = false, 1, 0, 2
	emitter.Speed = environment.NumberRange.new(0)
	for key, value in properties do emitter[key] = value end
	if state.brightTextures and state.brightTextures[emitter.Texture] then emitter.LightEmission = math.max(emitter.LightEmission, 1) end
	emitter.TimeScale, emitter.Parent = state.speed, parent

	GameEffects.Kill.delay(state, delay or 0, function()
		if emitter.Parent then emitter:Emit(count) end
	end)
	return emitter
end

function GameEffects.Kill.sound(state, id, volume, speed, parent, fade)
	if not state.sound then return nil end

	local sound
	pcall(function()
		local environment = state.context.environment
		sound = environment.Instance.new("Sound")
		sound.SoundId, sound.Volume, sound.PlaybackSpeed = id, math.clamp(state.volume / 100, 0, 1) * (volume or 1), speed or 1
		sound.RollOffMaxDistance, sound.Parent = 200, parent or GameEffects.Kill.folder(state)
		sound:Play()
		if not fade then GameEffects.Kill.destroyLater(state, sound, 5); return end

		GameEffects.Kill.delay(state, fade, function()
			if sound.Parent then GameEffects.Kill.tween(state, sound, 0.4, {Volume = 0}) end
		end)
		GameEffects.Kill.destroyLater(state, sound, fade + 0.5)
	end)
	return sound
end

function GameEffects.Kill.scatter(state, position, count, color, material, distance, duration)
	local environment = state.context.environment
	for _ = 1, count do
		local part = environment.Instance.new("WedgePart")
		part.Anchored, part.CanCollide, part.CanQuery, part.CanTouch, part.CastShadow = true, false, false, false, false
		part.Material, part.Color, part.Transparency, part.Reflectance = material, color, 0.15, 0.25

		local size = math.random() * 0.6 + 0.25
		part.Size = environment.Vector3.new(size * 0.4, size, size * 0.8)
		local offset = environment.Vector3.new(math.random() - 0.5, math.random() * 1.4 - 0.7, math.random() - 0.5) * environment.Vector3.new(2, 4, 1.2)
		part.CFrame = environment.CFrame.new(position + offset) * environment.CFrame.Angles(math.random() * 6.3, math.random() * 6.3, math.random() * 6.3)
		part.Parent = GameEffects.Kill.folder(state)

		local direction = (offset.Magnitude > 0.05 and offset.Unit or environment.Vector3.new(0, 1, 0)) * (distance * (0.6 + math.random() * 0.6))
		local target = environment.CFrame.new(position + offset + direction + environment.Vector3.new(0, -2.5 - math.random() * 2, 0))
			* environment.CFrame.Angles(math.random() * 9, math.random() * 9, math.random() * 9)
		GameEffects.Kill.tween(state, part, duration, {CFrame = target}, environment.Enum.EasingStyle.Quad, environment.Enum.EasingDirection.Out)
		GameEffects.Kill.tween(state, part, duration, {Transparency = 1}, environment.Enum.EasingStyle.Quint, environment.Enum.EasingDirection.In)
		GameEffects.Kill.destroyLater(state, part, duration + 0.1)
	end
end

function GameEffects.Kill.line(state, start, finish, width, up, color, material, transparency)
	local environment = state.context.environment

function GameEffects.Kill.arcSegment(state, start, finish, normal, outward, width, color, hold, white)
	if not state.alive then return end

	local environment = state.context.environment
	local inner = GameEffects.Kill.line(state, start + outward * width * 0.3, finish + outward * width * 0.3,
		math.max(0.04, width * 0.55), normal, white or environment.Color3.new(1, 1, 1))
	local outer = GameEffects.Kill.line(state, start, finish, math.max(0.1, width * 1.6), normal, color, environment.Enum.Material.Neon, 0.45)

	for _, part in {inner, outer} do
		GameEffects.Kill.delay(state, hold, function()
			if not part.Parent then return end
			GameEffects.Kill.tween(state, part, 0.22, {Transparency = 1, Size = environment.Vector3.new(0.02, part.Size.Y, part.Size.Z),
				CFrame = part.CFrame + outward * 0.7}, environment.Enum.EasingStyle.Quad)
		end)
		GameEffects.Kill.destroyLater(state, part, hold + 0.3)
	end
end

function GameEffects.Kill.arc(state, position, roll, radius, halfAngle, color, width, drawTime, hold, yaw, white, reverse)
	local environment = state.context.environment
	local camera = state.context.workspace.CurrentCamera
	local frame = camera and camera.CFrame or environment.CFrame.new()
	local yawAngle, rollAngle = math.rad(yaw or 0), math.rad(roll)
	local view = (frame.LookVector * math.cos(yawAngle) + frame.RightVector * math.sin(yawAngle)).Unit
	local direction = frame.RightVector * math.cos(rollAngle) + frame.UpVector * math.sin(rollAngle)
	direction = (direction - view * direction:Dot(view)).Unit
	if reverse then direction = -direction end

	local normal = view:Cross(direction).Unit
	local center = position - normal * radius * math.cos(halfAngle)
	GameEffects.Kill.arcFrom(state, center, normal, direction, radius, -halfAngle, halfAngle, color, width, drawTime, hold, white)
end

function GameEffects.Kill.arcFrom(state, center, normal, direction, radius, startAngle, endAngle, color, width, drawTime, hold, white)
	local points = {}
	for index = 0, 14 do
		local angle = startAngle + (endAngle - startAngle) * index / 14
		points[index] = center + (normal * math.cos(angle) + direction * math.sin(angle)) * radius
	end

	for index = 1, 14 do
		local weight = math.sin(math.pi * (index - 0.5) / 14) ^ 0.7
		local start, finish = points[index - 1], points[index]
		local outward = ((start + finish) / 2 - center).Unit
		GameEffects.Kill.delay(state, drawTime * (index - 1) / 14, function()
			if not state.alive then return end
			GameEffects.Kill.arcSegment(state, start, finish, normal:Cross(direction).Unit, outward, width * weight, color, hold, white)
		end)
	end
end

end

function GameEffects.Kill.ground(state, position, ignore, distance)
	local context, environment = state.context, state.context.environment
	local params = environment.RaycastParams.new()
	params.FilterType, params.RespectCanCollide = environment.Enum.RaycastFilterType.Exclude, true

	local excluded = {}
	for _, object in ignore or {} do
		if environment.typeof(object) == "Instance" then table.insert(excluded, object) end
	end
	local bodies = context.workspace:FindFirstChild("PulseKillBodies")
	if bodies then table.insert(excluded, bodies) end
	if context.workspace.CurrentCamera then table.insert(excluded, context.workspace.CurrentCamera) end
	params.FilterDescendantsInstances = excluded

	local origin = position + environment.Vector3.new(0, 1.5, 0)
	for _ = 1, 8 do
		local result = context.workspace:Raycast(origin, environment.Vector3.new(0, -(distance or 60), 0), params)
		if not result then return nil end

		local part = result.Instance
		local model = part:FindFirstAncestorOfClass("Model")
		local humanoid = model and model:FindFirstChildOfClass("Humanoid")
		if part.Transparency < 0.9 and not humanoid then return result.Position, result.Normal end

		table.insert(excluded, humanoid and model or part)
		params.FilterDescendantsInstances = excluded
	end
	return nil
end

function GameEffects.Kill.groundDecal(state, position, normal, size, color, lifetime, growth, texture, count)
	local environment = state.context.environment
	local reference = math.abs(normal.Z) < 0.9 and environment.Vector3.new(0, 0, 1) or environment.Vector3.new(1, 0, 0)
	local frame = environment.CFrame.fromMatrix(position, normal:Cross(reference).Unit, normal)

	for index = 1, count or 3 do
		local width = size * (1 - (index - 1) * 0.3)
		local offset = index == 1 and environment.Vector3.new(0, 0, 0)
			or environment.Vector3.new((math.random() - 0.5) * size * 0.35, 0, (math.random() - 0.5) * size * 0.35)
		local part = GameEffects.Kill.anchor(state,
			frame * environment.CFrame.new(offset + environment.Vector3.new(0, 0.02 + index * 0.012, 0))
				* environment.CFrame.Angles(0, math.random() * 6.28, 0), environment.Vector3.new(0.2, 0.05, 0.2))
		local decal = environment.Instance.new("Decal")
		decal.Texture = texture and not state.brightTextures[texture] and texture or state.textures.splat
		decal.Face = environment.Enum.NormalId.Top
		decal.Color3 = color:Lerp(environment.Color3.new(0, 0, 0), (index - 1) * 0.15)
		decal.Transparency, decal.Parent = 0.02 + (index - 1) * 0.1, part

		GameEffects.Kill.delay(state, (index - 1) * 0.07, function()
			if part.Parent then GameEffects.Kill.tween(state, part, growth or 0.9, {Size = environment.Vector3.new(width, 0.05, width)}) end
		end)
		GameEffects.Kill.delay(state, lifetime, function()
			if decal.Parent then GameEffects.Kill.tween(state, decal, 1.4, {Transparency = 1}, environment.Enum.EasingStyle.Quad) end
		end)
		GameEffects.Kill.destroyLater(state, part, lifetime + 1.5)
	end
end

function GameEffects.Kill.splat(state, position, size, color, lifetime, ignore, growth, texture, count)
	local ground, normal = GameEffects.Kill.ground(state, position, ignore)
	if ground then GameEffects.Kill.groundDecal(state, ground, normal, size, color, lifetime, growth, texture, count) end
end

function GameEffects.Kill.resetShake(state)
	local context = state.context
	pcall(function() context.runService:UnbindFromRenderStep("PulseKillShakeA") end)
	pcall(function() context.runService:UnbindFromRenderStep("PulseKillShakeB") end)

	if state.shakeRestore then
		local camera = context.workspace.CurrentCamera
		if camera then pcall(function() camera.CFrame = state.shakeRestore end) end
		state.shakeRestore = nil
	end

	table.clear(state.shakes)
	state.shakeBound = false
end

function GameEffects.Kill.shakeCamera(state, position, amplitude, duration)
	if not state.shake or not state.alive then return end

	local context, environment = state.context, state.context.environment
	local camera = context.workspace.CurrentCamera
	if not camera then return end

	local strength = amplitude * math.clamp(1.2 - (camera.CFrame.Position - position).Magnitude / 75, 0, 1)
	if strength < 0.05 then return end

	table.insert(state.shakes, {time = environment.os.clock(), amplitude = strength, duration = duration, seed = math.random() * 50})
	if state.shakeBound then return end
	state.shakeBound = true

	context.runService:BindToRenderStep("PulseKillShakeA", environment.Enum.RenderPriority.First.Value, function()
		if state.shakeRestore then
			local current = context.workspace.CurrentCamera
			if current then current.CFrame = state.shakeRestore end
			state.shakeRestore = nil
		end
		if #state.shakes == 0 then GameEffects.Kill.resetShake(state) end
	end)

	context.runService:BindToRenderStep("PulseKillShakeB", environment.Enum.RenderPriority.Camera.Value + 2, function()
		local current = context.workspace.CurrentCamera
		if not current then return end

		local now, pitch, yaw, roll = environment.os.clock(), 0, 0, 0
		for index = #state.shakes, 1, -1 do
			local shake = state.shakes[index]
			local progress = (now - shake.time) * state.speed / shake.duration
			if progress >= 1 then table.remove(state.shakes, index); continue end

			local strength = shake.amplitude * (1 - progress) ^ 2
			local time = (now - shake.time) * 24
			pitch += math.noise(time, shake.seed) * strength * 2.2
			yaw += math.noise(shake.seed, time) * strength * 2.2
			roll += math.noise(time, shake.seed + 9.3) * strength
		end

		if pitch ~= 0 or yaw ~= 0 or roll ~= 0 then
			state.shakeRestore = current.CFrame
			current.CFrame = current.CFrame * environment.CFrame.Angles(math.rad(pitch), math.rad(yaw), math.rad(roll))
		end
	end)
end

GameEffects.Kill.bodyExcluded = {"LuaSourceContainer", "Tool", "ForceField", "BillboardGui", "SurfaceGui", "Sound", "ParticleEmitter",
	"Trail", "Beam", "Highlight", "Light", "Fire", "Smoke", "Sparkles", "BodyMover", "SelectionBox", "ProximityPrompt", "ClickDetector", "Animator"}

function GameEffects.Kill.canTake(state, character, ghost)
	if not state.originals[character] or state.taken[character] or not character.Parent then return false end
	if ghost and not (state.contexts[character] and state.contexts[character].ghostTakes) then return false end

	local root = character:FindFirstChild("LowerTorso") or character:FindFirstChild("UpperTorso") or character:FindFirstChild("HumanoidRootPart")
	if not root then return false end
	if state.forceTake or root.Anchored then return true end

	local ok, owned = pcall(state.context.environment.isnetworkowner, root)
	if ok and type(owned) == "boolean" then return owned end
	return true
end

function GameEffects.Kill.cleanBody(state, body, taken)
	for _, object in body:GetDescendants() do
		local exclude = false
		for _, class in GameEffects.Kill.bodyExcluded do
			if object:IsA(class) then exclude = true; break end
		end

		if exclude then pcall(function() object:Destroy() end) end
	end

	if taken then
		for _, object in body:GetDescendants() do
			if object:IsA("BasePart") or object:IsA("Decal") then pcall(function() object.LocalTransparencyModifier = 0 end) end
		end
	end

	local humanoid = body:FindFirstChildOfClass("Humanoid")
	if humanoid then
		pcall(function()
			humanoid.RequiresNeck = false
			humanoid.BreakJointsOnDeath = false
			humanoid.EvaluateStateMachine = false
			humanoid.DisplayDistanceType = state.context.environment.Enum.HumanoidDisplayDistanceType.None
			humanoid.HealthDisplayType = state.context.environment.Enum.HumanoidHealthDisplayType.AlwaysOff
			if not taken then humanoid.MaxHealth, humanoid.Health = 100, 100 end
		end)
	end

	body.Name = "PulseKillBody"
	return body
end

function GameEffects.Kill.cloneBody(state, character)
	if GameEffects.Kill.canTake(state, character) then
		state.taken[character] = true
		return GameEffects.Kill.cleanBody(state, character, true)
	end

	local archivable = character.Archivable
	character.Archivable = true
	local ok, body = pcall(function() return character:Clone() end)
	pcall(function() character.Archivable = archivable end)
	if not ok or not body then return nil end

	return GameEffects.Kill.cleanBody(state, body, false)
end

function GameEffects.Kill.link(state, first, second, attachment0, attachment1)
	local environment = state.context.environment
	local joint = environment.Instance.new("BallSocketConstraint")
	joint.Attachment0, joint.Attachment1 = attachment0, attachment1
	joint.LimitsEnabled, joint.UpperAngle = true, 60
	joint.TwistLimitsEnabled, joint.TwistLowerAngle, joint.TwistUpperAngle = true, -45, 45
	joint.Parent = first
	if first and second and first:IsA("BasePart") and second:IsA("BasePart") then
		local collision = environment.Instance.new("NoCollisionConstraint")
		collision.Part0, collision.Part1, collision.Parent = first, second, first
	end

	return joint
end

function GameEffects.Kill.joints(state, body, root)
	local environment = state.context.environment
	local joints, linked = {}, {}
	for _, class in {"BallSocketConstraint", "Motor6D", "AnimationConstraint"} do
		for _, object in body:GetDescendants() do if object.ClassName == class then table.insert(joints, object) end end
	end

	local function seen(first, second)
		return linked[first] and linked[first][second] or linked[second] and linked[second][first]
	end

	local function remember(first, second)
		linked[first] = linked[first] or {}
		linked[first][second] = true
	end

	for _, joint in joints do
		local class = joint.ClassName
		if class == "BallSocketConstraint" then
			if not joint.Attachment0 or not joint.Attachment1 then continue end
			local first, second = joint.Attachment0.Parent, joint.Attachment1.Parent
			if first and second and seen(first, second) then joint:Destroy(); continue end

			if first and second then remember(first, second) end
			joint.Enabled, joint.LimitsEnabled, joint.UpperAngle = true, true, 60
			joint.TwistLimitsEnabled, joint.TwistLowerAngle, joint.TwistUpperAngle = true, -45, 45
			pcall(function() joint.MaxFrictionTorque = 0 end)
			pcall(function() joint.Restitution = 0 end)
			if first and second and first:IsA("BasePart") and second:IsA("BasePart") then
				local collision = environment.Instance.new("NoCollisionConstraint")
				collision.Part0, collision.Part1, collision.Parent = first, second, first
			end
			continue
		end

		local first, second, attachment0, attachment1
		if class == "Motor6D" then
			first, second = joint.Part0, joint.Part1
			if first and second and first ~= root and second ~= root and not seen(first, second) then
				attachment0, attachment1 = environment.Instance.new("Attachment"), environment.Instance.new("Attachment")
				attachment0.CFrame, attachment0.Parent = joint.C0, first
				attachment1.CFrame, attachment1.Parent = joint.C1, second
			end
		end
		if class == "AnimationConstraint" then
			attachment0, attachment1 = joint.Attachment0, joint.Attachment1
			first, second = attachment0 and attachment0.Parent, attachment1 and attachment1.Parent
			if first and second and (not first:IsA("BasePart") or not second:IsA("BasePart") or first == root or second == root or seen(first, second)) then
				attachment0, attachment1 = nil, nil
			end
		end

		if first and second and attachment0 and attachment1 then
			remember(first, second)
			GameEffects.Kill.link(state, first, second, attachment0, attachment1)
		end
		joint:Destroy()
	end
end

function GameEffects.Kill.ghostCandidate(part)
	if not part:IsA("BasePart") or part.Name == "HumanoidRootPart" or part.Transparency >= 0.95 then return false end
	if part:FindFirstChildWhichIsA("WrapLayer") then return false end

	local accessory = part:FindFirstAncestorWhichIsA("Accessory")
	return not accessory or not accessory:FindFirstChildWhichIsA("WrapLayer", true)
end

function GameEffects.Kill.prepareGhost(part, options)
	for _, child in part:GetChildren() do
		if child:IsA("SpecialMesh") then
			if not options.textured then pcall(function() child.TextureId = "" end) end
			continue
		end
		if options.textured and (child:IsA("Decal") or child:IsA("SurfaceAppearance")) then continue end
		pcall(function() child:Destroy() end)
	end

	if part:IsA("MeshPart") and not options.textured then pcall(function() part.TextureID = "" end) end
	part.Anchored, part.CanCollide, part.CanQuery, part.CanTouch, part.CastShadow = true, false, false, false, false
	if options.material then part.Material = options.material end
	if options.color then part.Color = options.color end
	if options.transparency ~= nil then part.Transparency = options.transparency end
	if not options.textured then part.Reflectance = 0 end
	part.Name = "G"
end

function GameEffects.Kill.ghost(state, character, options)
	local taken = GameEffects.Kill.canTake(state, character, true)
	local parts = {}
	if taken then state.taken[character] = true end

	for _, source in character:GetDescendants() do
		if not GameEffects.Kill.ghostCandidate(source) then continue end

		local part = source
		if not taken then
			local ok, clone = pcall(function()
				local archivable = source.Archivable
				source.Archivable = true
				local result = source:Clone()
				source.Archivable = archivable
				return result
			end)
			if not ok or not clone then continue end
			part = clone
		end

		GameEffects.Kill.prepareGhost(part, options)
		if taken then pcall(function() part.LocalTransparencyModifier = 0 end) end
		if not taken then part.Parent = GameEffects.Kill.folder(state) end
		table.insert(parts, part)
	end

	if taken then
		for _, part in parts do part.Parent = GameEffects.Kill.folder(state) end
		GameEffects.Kill.hide(character)
	end
	return parts
end

function GameEffects.Kill.fadeGhost(state, parts, duration, properties)
	local environment = state.context.environment
	for _, part in parts do
		local target = {Transparency = 1}
		if properties then for key, value in properties(part) do target[key] = value end end

		GameEffects.Kill.tween(state, part, duration, target, environment.Enum.EasingStyle.Quad, environment.Enum.EasingDirection.In)
		for _, child in part:GetChildren() do
			if child:IsA("Decal") then GameEffects.Kill.tween(state, child, duration, {Transparency = 1}, environment.Enum.EasingStyle.Quad, environment.Enum.EasingDirection.In) end
		end
		GameEffects.Kill.destroyLater(state, part, duration + 0.1)
	end
end

function GameEffects.Kill.staticBody(state, effect)
	local model = GameEffects.Kill.cloneBody(state, effect.char)
	if not model then return GameEffects.Kill.ghost(state, effect.char, {textured = true}), nil end

	for _, object in model:GetDescendants() do
		if object:IsA("JointInstance") or object:IsA("WeldConstraint") or object:IsA("Constraint") then
			pcall(function() object.Enabled = false end)
		end
	end

	local parts = {}
	for _, part in model:GetDescendants() do
		if not part:IsA("BasePart") then continue end
		part.Anchored, part.CanCollide, part.CanQuery, part.CanTouch, part.CastShadow = true, false, false, false, false
		if part.Name == "HumanoidRootPart" or part.Transparency >= 0.95 then
			part.Transparency = 1
			continue
		end
		table.insert(parts, part)
	end

	model.Parent = GameEffects.Kill.folder(state)
	GameEffects.Kill.destroyLater(state, model, 9)
	return parts, model
end

function GameEffects.Kill.bodyParts(model)
	local parts, body = {}, {}

	for _, part in model:GetDescendants() do
		if not part:IsA("BasePart") then continue end

		local accessory = part:FindFirstAncestorWhichIsA("Accessory")
		part.Anchored, part.CanQuery, part.CanTouch = false, false, false
		part.CanCollide, part.Massless = not accessory, accessory and true or false
		pcall(function() part.CollisionGroup = "Ragdolls" end)

		table.insert(parts, part)
		if not accessory then table.insert(body, part) end
	end

	return parts, body
end

function GameEffects.Kill.launch(state, result, velocity, spin, callback)
	local function applyVelocity()
		for _, part in result.body do pcall(function() part.CanCollide, part.AssemblyLinearVelocity = true, velocity end) end
		if spin and result.torso then pcall(function() result.torso.AssemblyAngularVelocity = spin end) end
		if callback then pcall(callback, result) end
	end

	applyVelocity()
	local steps, connection = 0, nil
	connection = state.context.runService.Stepped:Connect(function()
		steps += 1
		if result.model.Parent then applyVelocity() end
		if steps >= 2 then connection:Disconnect() end
	end)
end

function GameEffects.Kill.fadeBody(state, result, lifetime)
	local context, environment = state.context, state.context.environment
	local duration = lifetime or 2.8

	environment.task.delay(duration / state.speed, function()
		for _, part in result.parts do
			if not part.Parent then continue end

			context.tweenService:Create(part, environment.TweenInfo.new(0.5 / state.speed, environment.Enum.EasingStyle.Quad), {Transparency = 1}):Play()
			for _, child in part:GetChildren() do
				if child:IsA("Decal") then context.tweenService:Create(child, environment.TweenInfo.new(0.5 / state.speed), {Transparency = 1}):Play() end
			end
		end
	end)

	environment.task.delay((duration + 0.6) / state.speed, function() pcall(function() result.model:Destroy() end) end)
end

function GameEffects.Kill.bodiesFolder(state)
	local context = state.context
	if state.bodies and state.bodies.Parent == context.workspace then return state.bodies end

	state.bodies = context.environment.Instance.new("Folder")
	state.bodies.Name, state.bodies.Parent = "PulseKillBodies", context.workspace
	return state.bodies
end

function GameEffects.Kill.body(state, effect, velocity, spin, lifetime, callback, held)
	local context, environment = state.context, state.context.environment
	local model = GameEffects.Kill.cloneBody(state, effect.char)
	if not model then return nil end

	local root = model:FindFirstChild("HumanoidRootPart")
	GameEffects.Kill.joints(state, model, root)
	if root then root:Destroy() end

	local parts, body = GameEffects.Kill.bodyParts(model)
	local original = model ~= effect.char and effect.char or effect.orig
	for _, part in original and original.Parent and original:GetDescendants() or {} do
		if part:IsA("BasePart") then pcall(function() part.CanCollide, part.CollisionGroup = false, "Characters" end) end
	end

	model.Parent = GameEffects.Kill.bodiesFolder(state)
	for _, part in body do part.CanCollide = true end

	local result = {model = model, parts = parts, body = body,
		torso = model:FindFirstChild("UpperTorso") or model:FindFirstChild("Torso") or body[1], head = model:FindFirstChild("Head")}
	local humanoid = model:FindFirstChildOfClass("Humanoid")
	if humanoid then pcall(function() humanoid.PlatformStand = true; humanoid:ChangeState(environment.Enum.HumanoidStateType.Physics) end) end

	result.release = function() end
	if held then
		for _, part in parts do part.Anchored = true end
		result.release = function(nextVelocity, nextCallback)
			if result.released or not model.Parent then return end

			result.released = true
			for _, part in parts do part.Anchored = false end
			GameEffects.Kill.launch(state, result, nextVelocity or velocity, spin, nextCallback or callback)
		end
	end
	if not held then GameEffects.Kill.launch(state, result, velocity, spin, callback) end

	GameEffects.Kill.fadeBody(state, result, lifetime)
	return result
end

function GameEffects.Kill.direction(state, effect)
	local environment = state.context.environment
	local camera = state.context.workspace.CurrentCamera
	local origin = effect.killer and effect.killer.Parent and effect.killer.Position
		or camera and camera.CFrame.Position or effect.root + environment.Vector3.new(0, 0, 5)
	local direction = effect.root - origin
	direction = environment.Vector3.new(direction.X, 0, direction.Z)
	if direction.Magnitude > 0.05 then return direction.Unit, origin end
	return environment.Vector3.new(0, 0, -1), origin
end

function GameEffects.Kill.hideEffectBody(state, character)
	local original = state.originals[character]
	if original and original.Parent then GameEffects.Kill.hide(original) end
	if not state.taken[character] then GameEffects.Kill.hide(character) end
end

end

function GameEffects.Kill.detachHead(body, direction, spin, vector)
	if not body.head then return end

	for _, joint in body.model:GetDescendants() do
		if not joint:IsA("BallSocketConstraint") then continue end
		if joint.Attachment0 and joint.Attachment0.Parent == body.head or joint.Attachment1 and joint.Attachment1.Parent == body.head then
			joint:Destroy()
		end
	end

	body.head.AssemblyLinearVelocity = direction * 36 + vector.new(0, 44, 0)
	body.head.AssemblyAngularVelocity = spin
end

function GameEffects.Kill.impactTrack(state, body)
	if not body or not body.torso then return end

	local velocity, count, last = body.torso.AssemblyLinearVelocity, 0, -1
	GameEffects.Kill.animate(state, 1.8, function(progress)
		local torso = body.torso
		if not torso.Parent then return end

		local current = torso.AssemblyLinearVelocity
		if velocity.Magnitude > 28 and (current - velocity).Magnitude > velocity.Magnitude * 0.55 and count < 3 and progress - last > 0.08 then
			count, last = count + 1, progress
			GameEffects.Kill.impactDust(state, torso.Position, velocity.Magnitude)
		end
		velocity = current
	end)
end

function GameEffects.Kill.impact(state, effect)
	local environment = state.context.environment
	local color, white = environment.Color3.fromRGB(255, 200, 110), environment.Color3.new(1, 1, 1)
	local direction, fallback = GameEffects.Kill.direction(state, effect)
	local source = effect.killer and effect.killer.Parent and effect.killer.Position + environment.Vector3.new(0, 0.6, 0) or fallback
	local delta = effect.chest - source
	local shot = delta.Magnitude > 0.1 and delta.Unit or direction
	local up = math.abs(shot.Y) < 0.95 and environment.Vector3.new(0, 1, 0) or environment.Vector3.new(1, 0, 0)
	local frame = environment.CFrame.fromMatrix(effect.chest, shot:Cross(up).Unit, shot)

	if delta.Magnitude > 1 and delta.Magnitude < 250 then
		local core = GameEffects.Kill.beam(state, source, effect.chest, 0.1, environment.Color3.fromRGB(255, 245, 220))
		local glow = GameEffects.Kill.beam(state, source, effect.chest, 0.4, color, environment.Enum.Material.Neon, 0.6)
		for _, part in {core, glow} do
			GameEffects.Kill.tween(state, part, 0.2, {Transparency = 1, Size = environment.Vector3.new(0.02, 0.02, part.Size.Z)}, environment.Enum.EasingStyle.Quad)
			GameEffects.Kill.destroyLater(state, part, 0.25)
		end
	end

	for index = 1, 2 do
		GameEffects.Kill.delay(state, (index - 1) * 0.05, function()
			local disc = GameEffects.Kill.disc(state, frame + shot * (index - 1) * 1.4, 0.4, white, environment.Enum.Material.Neon, 0.25)
			GameEffects.Kill.tween(state, disc, 0.3, {Size = environment.Vector3.new(0.04, 7 - index, 7 - index), Transparency = 1}, environment.Enum.EasingStyle.Quad)
			GameEffects.Kill.destroyLater(state, disc, 0.35)
		end)
	end

	local anchor = GameEffects.Kill.anchor(state, frame, environment.Vector3.new(0.5, 0.5, 0.5))
	local attachment = environment.Instance.new("Attachment")
	attachment.Parent = anchor
	local sequence = function(...) return GameEffects.Kill.numberSequence(state, ...) end
	local colors = function(first, second) return environment.ColorSequence.new(first, second or first) end
	GameEffects.Kill.particle(state, attachment, {Texture = state.textures.sparkBright, Color = colors(white), Size = sequence(0, 5, 1, 0),
		Lifetime = environment.NumberRange.new(0.12), Brightness = 6, ZOffset = 2}, 1)
	GameEffects.Kill.particle(state, attachment, {Texture = state.textures.fsparks, Color = colors(white, color), Size = sequence(0, 0.5, 1, 0),
		Lifetime = environment.NumberRange.new(0.3, 0.6), Speed = environment.NumberRange.new(30, 55), SpreadAngle = environment.Vector2.new(25, 25),
		Drag = 3, Orientation = environment.Enum.ParticleOrientation.VelocityParallel, Brightness = 5}, 30)
	GameEffects.Kill.particle(state, attachment, {Texture = state.textures.splat,
		Color = colors(environment.Color3.fromRGB(170, 0, 20), environment.Color3.fromRGB(90, 0, 10)), Size = sequence(0, 0.5, 1, 0.2),
		Lifetime = environment.NumberRange.new(0.5, 0.8), Speed = environment.NumberRange.new(14, 26), SpreadAngle = environment.Vector2.new(30, 30),
		Acceleration = environment.Vector3.new(0, -60, 0), Rotation = environment.NumberRange.new(0, 360), LightEmission = 0, Brightness = 1}, 22)

	local horizontal = environment.Vector3.new(shot.X, 0, shot.Z)
	local body = GameEffects.Kill.body(state, effect, (horizontal.Magnitude > 0.05 and horizontal.Unit or direction) * 75 + environment.Vector3.new(0, 16, 0),
		GameEffects.Kill.randomVector(state, 4), 3)
	GameEffects.Kill.hideEffectBody(state, effect.char)
	GameEffects.Kill.impactTrack(state, body)
	GameEffects.Kill.light(state, anchor, color, 6, 14, 0.35)
	GameEffects.Kill.sound(state, GameEffects.Kill.soundIds.sniper, 0.9, 0.85, anchor)
	GameEffects.Kill.sound(state, GameEffects.Kill.soundIds.punch2, 0.7, 1.3, anchor)
	GameEffects.Kill.shakeCamera(state, effect.chest, 0.8, 0.25)
	GameEffects.Kill.destroyLater(state, anchor, 2)
end

function GameEffects.Kill.electricBody(state, body, anchor, color, white)
	if not body then return end

	local environment = state.context.environment
	local highlight = environment.Instance.new("Highlight")
	highlight.FillColor, highlight.OutlineColor = color, white
	highlight.FillTransparency, highlight.OutlineTransparency = 0.3, 0
	highlight.DepthMode, highlight.Adornee, highlight.Parent = environment.Enum.HighlightDepthMode.Occluded, body.model, body.model

	GameEffects.Kill.animate(state, 0.9, function()
		if body.torso and body.torso.Parent then anchor.CFrame = body.torso.CFrame end
	end)
	environment.task.spawn(function()
		for index = 1, 12 do
			if not body.model.Parent or not state.alive then return end

			highlight.FillTransparency = index % 2 == 0 and 0.1 or 0.7
			for _, part in body.body do
				pcall(function()
					local jitter = environment.Vector3.new(math.random() - 0.5, math.random() * 0.7, math.random() - 0.5) * 16
					part.AssemblyLinearVelocity += jitter
				end)
			end
			environment.task.wait(0.06 / state.speed)
		end

		if highlight.Parent then
			GameEffects.Kill.tween(state, highlight, 0.4, {FillColor = environment.Color3.fromRGB(18, 18, 22), FillTransparency = 0.1, OutlineTransparency = 1})
		end
		if body.torso and body.torso.Parent then GameEffects.Kill.electricSmoke(state, body.torso.Position) end
	end)
end

function GameEffects.Kill.soul(state, effect)
	local environment = state.context.environment
	local color, white = environment.Color3.fromRGB(170, 225, 255), environment.Color3.new(1, 1, 1)
	local glow = GameEffects.Kill.ghost(state, effect.char, {material = environment.Enum.Material.Neon, color = color, transparency = 0.6})
	local shell = GameEffects.Kill.ghost(state, effect.char, {material = environment.Enum.Material.ForceField, color = color, transparency = 0})
	local direction = GameEffects.Kill.direction(state, effect)
	GameEffects.Kill.body(state, effect, direction * 3 + environment.Vector3.new(0, 2, 0), GameEffects.Kill.randomVector(state, 1), 3.2)
	GameEffects.Kill.hideEffectBody(state, effect.char)

	local parts, frames, transparency = {}, {}, {}
	for _, part in glow do table.insert(parts, part) end
	for _, part in shell do table.insert(parts, part) end
	for index, part in parts do frames[index], transparency[index] = part.CFrame, part.Transparency end

	local anchor = GameEffects.Kill.anchor(state, environment.CFrame.new(effect.chest), environment.Vector3.new(2, 4, 1.4))
	local sequence = function(...) return GameEffects.Kill.numberSequence(state, ...) end
	local colors = function(first, second) return environment.ColorSequence.new(first, second or first) end
	local mist = GameEffects.Kill.particle(state, anchor, {Texture = state.textures.soft, Color = colors(color, white),
		Size = sequence(0, 1, 1, 0), Transparency = sequence(0, 0.45, 1, 1), Lifetime = environment.NumberRange.new(0.6, 1),
		Speed = environment.NumberRange.new(0.5, 1.5), SpreadAngle = environment.Vector2.new(180, 180),
		Shape = environment.Enum.ParticleEmitterShape.Box, Brightness = 2.5}, 0)
	mist.Rate, mist.Enabled = 45, true
	local stars = GameEffects.Kill.particle(state, anchor, {Texture = state.textures.star, Color = colors(white, color),
		Size = sequence(0, 0.35, 1, 0), Lifetime = environment.NumberRange.new(0.5, 0.9), Speed = environment.NumberRange.new(1, 2),
		SpreadAngle = environment.Vector2.new(180, 180), Rotation = environment.NumberRange.new(0, 360), RotSpeed = environment.NumberRange.new(-120, 120),
		Shape = environment.Enum.ParticleEmitterShape.Box, Brightness = 4}, 0)
	stars.Rate, stars.Enabled = 14, true
	local attachment = environment.Instance.new("Attachment")
	attachment.Parent = anchor
	GameEffects.Kill.particle(state, attachment, {Texture = state.textures.soft, Color = colors(white, color), Size = sequence(0, 2, 1, 6),
		Transparency = sequence(0, 0.3, 1, 1), Lifetime = environment.NumberRange.new(0.5), Brightness = 3}, 1)

	GameEffects.Kill.animate(state, 2, function(progress)
		local height = progress * progress * 10 + progress * 1.2
		local offset = environment.Vector3.new(math.sin(progress * 7) * 0.35 * (1 - progress), height, math.cos(progress * 5) * 0.25 * (1 - progress))
		for index, part in parts do
			part.CFrame = frames[index] + offset
			part.Transparency = transparency[index] + (1 - transparency[index]) * progress ^ 2.2
		end
		anchor.CFrame = environment.CFrame.new(effect.chest + offset)
	end, function()
		mist.Enabled, stars.Enabled = false, false
		for _, part in parts do pcall(function() part:Destroy() end) end
		GameEffects.Kill.particle(state, attachment, {Texture = state.textures.star, Color = colors(white, color), Size = sequence(0, 0.6, 1, 0),
			Lifetime = environment.NumberRange.new(0.5, 0.9), Speed = environment.NumberRange.new(4, 9), SpreadAngle = environment.Vector2.new(180, 180),
			Drag = 3, Brightness = 5}, 18)
		GameEffects.Kill.particle(state, attachment, {Texture = state.textures.soft, Color = colors(color), Size = sequence(0, 3, 1, 7),
			Transparency = sequence(0, 0.3, 1, 1), Lifetime = environment.NumberRange.new(0.6), Brightness = 3}, 1)
		GameEffects.Kill.sound(state, GameEffects.Kill.soundIds.chimes, 0.3, 1.8, anchor, 2)
	end)
	GameEffects.Kill.light(state, anchor, color, 4, 14, 2.2)
	GameEffects.Kill.sound(state, GameEffects.Kill.soundIds.ghost, 0.8, 1.05, anchor)
	GameEffects.Kill.destroyLater(state, anchor, 3.8)
end

			GameEffects.Kill.sound(state, id, 0.8, 0.6 + math.random() * 1.6, anchor)
		end)
	end

function GameEffects.Kill.glitchLayers(state, layers, right)
	local environment = state.context.environment
	local frames = {}
	for index, layer in layers do
		frames[index] = {}
		for partIndex, part in layer do frames[index][partIndex] = part.CFrame end
	end

	environment.task.spawn(function()
		for index = 1, 12 do
			for layerIndex, layer in layers do
				local jitter = (math.random() - 0.5) * 0.9
				local offset = layerIndex == 1 and -0.25 or (layerIndex == 2 and 0.25 or 0)
				local shift = jitter + offset
				local vertical = math.random() < 0.3 and (math.random() - 0.5) * 0.6 or 0
				for partIndex, part in layer do
					if not part.Parent then continue end
					part.CFrame = frames[layerIndex][partIndex] + right * shift + environment.Vector3.new(0, vertical, 0)
					part.Transparency = math.clamp(0.3 + index * 0.05 + math.random() * 0.3, 0, 1)
				end
			end
			environment.task.wait(0.05 / state.speed)
		end
		for _, layer in layers do GameEffects.Kill.fadeGhost(state, layer, 0.2) end
	end)
end

			if anchor.Parent then anchor:FindFirstChildWhichIsA("ParticleEmitter"):Emit(6) end

function GameEffects.Kill.poisonHighlight(state, body, anchor, color)
	if not body then return end

	local environment = state.context.environment
	local highlight = environment.Instance.new("Highlight")
	highlight.FillColor, highlight.OutlineColor = color, environment.Color3.fromRGB(40, 120, 20)
	highlight.FillTransparency, highlight.OutlineTransparency = 0.6, 0.3
	highlight.DepthMode, highlight.Adornee, highlight.Parent = environment.Enum.HighlightDepthMode.Occluded, body.model, body.model
	GameEffects.Kill.animate(state, 2.4, function(progress)
		highlight.FillTransparency = 0.4 + 0.2 * math.sin(progress * 22)
		if body.torso and body.torso.Parent then anchor.CFrame = body.torso.CFrame end
	end, function()
		if highlight.Parent then
			GameEffects.Kill.tween(state, highlight, 0.5, {FillColor = environment.Color3.fromRGB(30, 60, 20), FillTransparency = 0.15, OutlineTransparency = 1})
		end
	end)
end
