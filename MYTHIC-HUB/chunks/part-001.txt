local FovSettings = {}

function FovSettings.initialize(configuration)
	local settings = configuration.sheriff.fov
	if type(settings) ~= "table" then
		settings = {}
		configuration.sheriff.fov = settings
	end

	settings.on = settings.on or false
	settings.size = settings.size or 220
	settings.r, settings.g, settings.b = settings.r or 120, settings.g or 220, settings.b or 255
	settings.lr, settings.lg, settings.lb = settings.lr or 255, settings.lg or 255, settings.lb or 255
	settings.style, settings.tr, settings.fa = settings.style or "Glow", settings.tr or 20, settings.fa or 12
	if settings.fill == nil then settings.fill = false end
	if settings.anim == nil then settings.anim = true end

	return settings
end

function FovSettings.changeSize(context, value)
	context.settings.size = context.environment.math.floor(value)
end

function FovSettings.changeFillStrength(context, value)
	context.settings.fa = context.environment.math.floor(value)
	context.redraw()
end

function FovSettings.changeTransparency(context, value)
	context.settings.tr = context.environment.math.floor(value)
	context.redraw()

	if not context.picker or not context.picker.SetAlpha then
		return
	end

	context.picker.SetAlpha(1 - context.settings.tr / 100)
end

function FovSettings.changeAnimation(context, isEnabled)
	context.settings.anim = not not isEnabled
end

function FovSettings.changeLockColor(context, color)
	local floor = context.environment.math.floor
	context.settings.lr = floor(color.R * 255 + 0.5)
	context.settings.lg = floor(color.G * 255 + 0.5)
	context.settings.lb = floor(color.B * 255 + 0.5)

	context.redraw()
end

function FovSettings.changeFilled(context, isEnabled)
	local isFilled = true

	if not isEnabled then
		isFilled = context.disabledValue()
	end

	context.settings.fill = isFilled
	context.redraw()
end

function FovSettings.formatSize(environment, value)
	local numeric = environment.tonumber(value)

	if numeric and numeric >= 1500 then
		return "в€ћ"
	end

	numeric = environment.tonumber(value) or 0
	return environment.tostring(environment.math.floor(numeric))
end

function FovSettings.changeColor(context, selection)
	local color = selection.color
	local floor = context.environment.math.floor
	context.settings.r = floor(color.R * 255 + 0.5)
	context.settings.g = floor(color.G * 255 + 0.5)
	context.settings.b = floor(color.B * 255 + 0.5)

	FovSettings._syncTransparency(context, selection.alpha)
	context.redraw()
end

function FovSettings._syncTransparency(context, alpha)
	if not context.environment.tonumber(alpha) then
		return
	end

	local transparency = context.environment.math.floor((1 - alpha) * 100 + 0.5)

	if transparency == context.settings.tr then
		return
	end

	context.settings.tr = transparency

	if not context.slider or not context.slider.Set then
		return
	end

	context.slider.Set(transparency)
end

local Layout = {}

function Layout.resolve(routes, section, subsection, item)
	local route = routes[section .. "/" .. tostring(item)]

	if not route and subsection then
		route = routes[section .. "/" .. tostring(subsection) .. ":*"]
	end

	if not route then
		route = routes[section .. "/*"]
	end

	if not route then
		return section, nil
	end

	if route[2] ~= "@sub" then
		subsection = route[2]
	end

	return route[1], subsection, route[3]
end

function Layout.createTabs(window, layout)
	local tabs = {}
	local currentGroup

	for _, definition in ipairs(layout.NovaLayout.tabs) do
		if definition.group ~= currentGroup then
			currentGroup = definition.group
			window:TabGroup(currentGroup)
		end

		local tab = window:Tab({Name = definition.name, Icon = definition.icon, Desc = definition.desc, Badge = definition.badge})
		tabs[definition.name] = tab

		if definition.subs then
			tab:SubTabs(definition.subs)
		end
	end

	return tabs
end

local PulseTab = {}

local function emptyPulseTab()
	local tab = {_curSub = nil, _subSelectFn = nil}
		local function noop() end
		local handle = {Set = noop, Get = noop, Refresh = noop}
		tab.SubTabs, tab.Sub, tab.Group, tab.Header, tab.Container, tab.Button, tab.Grid, tab.Invite = noop, noop, noop, noop, noop, noop, noop, noop
		tab.Columns = function() return {Left = nil, Right = nil} end
		tab.Section = function() return {Capture = noop, SetOpen = noop, IsOpen = noop, SetToggle = noop} end
		tab.Note = function() return {Text = ""} end
		tab.Toggle, tab.Keybind, tab.ToggleKey, tab.Slider, tab.Dropdown, tab.Colorpicker = function() return handle end, function() return handle end,
			function() return handle end, function() return handle end, function() return handle end, function() return handle end
		tab.IconList = function() return {Rebuild = noop, Selected = {}} end
		tab.ShopList = function() return {Selected = {}, RefreshSelection = noop, Refresh = noop} end
		tab.PlayerList = function() return {GetSelectedId = noop, GetSelected = noop, SetSelected = noop, Refresh = noop} end
	return tab
end

function PulseTab.create(environment, hub, name, icon, context)
	local tab = {_curSub = nil, _subSelectFn = nil}
	local window, nova, translator, layout = hub.W, hub.Nova, context.translator, context.layout
	if hub.dead or not window then return emptyPulseTab() end

	local function findTab(target)
		if hub.NovaTabs[target] then return hub.NovaTabs[target] end
		local ok, native = environment.pcall(function() return window:Tab({Name = target, Icon = context.icons[icon] or icon or "circle"}) end)
		if not ok or not native then return nil end
		environment.pcall(function() translator.bind(native.Label, target) end)
		hub.NovaTabs[target] = native
		return native
	end
	local native = findTab(layout.novaResolve(name, nil, "*"))
	if not native then return emptyPulseTab() end
	tab.Native = native
	hub.tabs[name] = {name = name, tab = native}
	table.insert(hub._tabList, hub.tabs[name])
	local section, capture, columns, side, subDefinitions = nil, nil, false, 1, {}
	local columnsBySub = {}
	local function translate(text) return translator.tr(text) end
	local function bind(object, text, extra)
		if environment.typeof(object) ~= "Instance" or text == nil then return end
		environment.pcall(function() translator.bind(object, text, extra) end)
	end
	local function configKey(text)
		local key = tostring(name) .. "/" .. tostring(tab._curSub or "-") .. "/" .. tostring(text)
		local count = (hub._ckeys[key] or 0) + 1
		hub._ckeys[key] = count
		return count > 1 and key .. "#" .. count or key
	end
	local function saved(key, kind)
		if not key then return nil end
		local value = context.config[key]
		if type(value) == kind or kind == "string" and type(value) == "number" then return value end
		return nil
	end
	local function store(key, value) if key then context.config[key] = value end end
	local function sideFor(requested)
		if columns then
			if requested == "Right" then side = 2 end
			if requested == "Left" then side = 1 end
			return side == 2 and "Right" or "Left"
		end
		if requested == "Right" or requested == "Left" then return requested end
		return nil
	end
	local function targetFor(title)
		local sub = #subDefinitions > 0 and (tab._navSub or subDefinitions[1].key) or tab._navSub
		local target, targetSub, targetTitle = layout.novaResolve(name, sub, tostring(title or "General"))
		local destination = findTab(target) or native
		if targetSub and (not destination.SubButtons or not destination.SubButtons[targetSub]) then targetSub = nil end
		targetSub = targetSub or (destination.SubDefs and destination.SubDefs[1].key)
		return target, targetSub, targetTitle or tostring(title or "General"), destination
	end
	local function createSection(title, requestedSide, pinned)
		if capture then capture.detached = true end
		local target, sub, mapped, destination = targetFor(title)
		local result
		environment.pcall(function()
			result = destination:Section({Title = translate(mapped), Icon = layout.NovaSectionIcons[mapped], Sub = sub, Side = pinned and sideFor(requestedSide) or nil})
		end)
		if not result then return nil end
		bind(result.TitleLabel, mapped)
		result._pulseCount, result._pulseText, result._pulseTarget = 0, mapped, target .. "/" .. tostring(sub)
		section = result
		return result
	end
	local function currentSection()
		if not section then createSection("General") end
		return section
	end
	local function indent()
		if capture and capture.drawer then return 0 end
		return capture and not capture.detached and (capture.depth or 1) or 0
	end
	local function track(handle)
		if handle == nil then return nil end
		if section then section._pulseCount = (section._pulseCount or 0) + 1 end
		if capture and not capture.detached then table.insert(capture, handle) end
		return handle
	end
	local function setVisible(item, visible)
		if type(item) == "table" then
			if item.SetParentHidden then environment.pcall(item.SetParentHidden, item, not visible) end
			return
		end
		if environment.typeof(item) == "Instance" and item:IsA("GuiObject") then item.Visible = visible end
	end
	local function setChildrenVisible(children, visible)
		if type(children) ~= "table" then return end
		for _, child in ipairs(children) do
			setVisible(child, visible)
			if type(child) == "table" and child._pulseKids then
				local childVisible = true
				if child._pulseState then
					local ok, value = environment.pcall(child._pulseState)
					childVisible = ok and value == true
				end
				setChildrenVisible(child._pulseKids, visible and childVisible)
			end
		end
	end
	local function collect(callback, drawer)
		if not callback then return nil end
		local previous, oldSection, oldSide = capture, section, side
		if drawer then
			capture, section = {depth = 0, drawer = true}, drawer
			section._pulseCount = 0
		else
			capture = {depth = previous and not previous.detached and (previous.depth or 1) + 1 or 1}
		end
		local ok, message = environment.pcall(callback)
		if not ok then environment.warn("[MYTHIC HUB] " .. tostring(message)) end
		local children = capture
		capture, section, side = previous, oldSection, oldSide
		if not drawer then children.detached = nil end
		return children
	end
	local function bindHandle(handle, text)
		if type(handle) == "table" and handle.Title then bind(handle.Title, text) end
	end
	local function matchesTarget(candidate, title)
		local currentSub = #subDefinitions > 0 and (tab._navSub or subDefinitions[1].key) or tab._navSub
		local target, sub = layout.novaResolve(name, currentSub, title)
		local destination = hub.NovaTabs[target] or native
		if sub and (not destination.SubButtons or not destination.SubButtons[sub]) then sub = nil end
		sub = sub or (destination.SubDefs and destination.SubDefs[1].key)
		return candidate._pulseTarget == target .. "/" .. tostring(sub)
	end
	local function reuse(candidate, title)
		if type(candidate) ~= "table" or not candidate.TitleLabel then return false end
		if not matchesTarget(candidate, title) then
			environment.pcall(function()
				local sections = candidate.Tab.Sections
				for index = #sections, 1, -1 do if sections[index] == candidate then table.remove(sections, index) end end
				candidate.Page.Load[candidate.Side] = math.max(0, candidate.Page.Load[candidate.Side] - 1)
				local object = candidate.Wrap or candidate.Card
				object:Destroy()
			end)
			if section == candidate then section = nil end
			return false
		end
		environment.pcall(function()
			local image = layout.NovaSectionIcons[title]
			if image and candidate.HeadIcon then nova.Icons.apply(candidate.HeadIcon, image) end
		end)
		candidate._pulseText = title
		environment.pcall(function()
			for index = #translator.reg, 1, -1 do if translator.reg[index].inst == candidate.TitleLabel then table.remove(translator.reg, index) end end
		end)
		bind(candidate.TitleLabel, title)
		return true
	end

	function tab.SubTabs(_, definitions)
		subDefinitions = {}
		for _, definition in ipairs(definitions or {}) do

		end
	end

	function tab.Toggle(_, options)
		local key = not options.NoSave and configKey(options.Text) or nil
		if key then context.defaults[key] = options.Default and true or false end
		local previous = saved(key, "boolean")
		if previous ~= nil then options.Default = previous end
		local value = options.Default and true or false
		local current, nativeHandle = currentSection(), nil
		environment.pcall(function()
			nativeHandle = current:Toggle({Text = translate(options.Text), Desc = options.Desc and translate(options.Desc) or nil, Default = value,
				Tooltip = options.Tooltip and translate(options.Tooltip) or nil, Indent = indent(), Locked = options.Locked, Drawer = options.Settings ~= nil,
				Callback = function(nextValue)
					value = nextValue and true or false
					store(key, value)
					if options.Callback then options.Callback(value) end
				end})
		end)
		bindHandle(nativeHandle, options.Text)
		if nativeHandle and nativeHandle.DescLabel and options.Desc then bind(nativeHandle.DescLabel, options.Desc) end
		track(nativeHandle)
		local handle = {Row = nativeHandle and nativeHandle.Row or nil}
		handle.Get = function() return value end
		handle.Set = function(nextValue)
			value = nextValue and true or false
			store(key, value)
			if nativeHandle then environment.pcall(nativeHandle.SetSilent, nativeHandle, value) end
			if options.Callback then options.Callback(value) end
		end
		handle.SetSilent = function(nextValue)
			value = nextValue and true or false
			store(key, value)
			if nativeHandle then environment.pcall(nativeHandle.SetSilent, nativeHandle, value) end
		end
		if key then context.setters[key] = function(nextValue) handle.Set(nextValue and true or false) end end
		if options.Settings then
			if nativeHandle and nativeHandle.Drawer then
				collect(options.Settings, nativeHandle.Drawer)
			else
				local children = collect(options.Settings)
				setChildrenVisible(children, value)
			end
		end
		environment.pcall(function() if options.Callback then options.Callback(value) end end)
		return handle
	end
	function tab.Slider(_, options)
		local minimum, maximum, decimals = options.Min or 0, options.Max or 1, options.Decimals or 0
		local key = not options.NoSave and configKey(options.Text) or nil
		if key then context.defaults[key] = tonumber(options.Default) end
		local previous = saved(key, "number")
		if previous ~= nil then options.Default = math.clamp(previous, minimum, maximum) end
		local factor = 10 ^ math.max(0, math.floor(tonumber(decimals) or 0))
		local value = math.floor(math.clamp(tonumber(options.Default) or minimum, minimum, maximum) * factor + 0.5) / factor
		local current, nativeHandle = currentSection(), nil
		environment.pcall(function()
			nativeHandle = current:Slider({Text = translate(options.Text), Min = minimum, Max = maximum, Default = value, Decimals = decimals,
				Suffix = options.Suffix or "", Tooltip = options.Tooltip and translate(options.Tooltip) or nil, Indent = indent(), Display = options.Display,
				Callback = function(nextValue) value = nextValue; store(key, value); if options.Callback then options.Callback(value) end end})
		end)
		bindHandle(nativeHandle, options.Text)
		track(nativeHandle)
		local handle = {Row = nativeHandle and nativeHandle.Row or nil}
		handle.Get = function() return value end
		handle.SetSilent = function(nextValue)
			value = math.clamp(tonumber(nextValue) or minimum, minimum, maximum)
			store(key, value)
			if nativeHandle then environment.pcall(nativeHandle.SetSilent, nativeHandle, value) end
		end
		handle.Set = function(nextValue)
			handle.SetSilent(nextValue)
			if options.Callback then options.Callback(value) end
		end
		handle.Refresh = function() if nativeHandle then environment.pcall(nativeHandle.SetSilent, nativeHandle, value) end end
		if key then context.setters[key] = function(nextValue) if tonumber(nextValue) then handle.Set(nextValue) end end end
		if options.Settings then collect(options.Settings) end
		if options.Callback then options.Callback(value) end
		return handle
	end
	function tab.Dropdown(_, options)
		local function mappedOptions(values)
			local result = {}
			for _, item in ipairs(values or {}) do
				local label = tostring(item.label or item.code)
				if options.TranslateOptions ~= false then label = translate(label) end
				table.insert(result, {code = item.code, label = label, image = item.image})
			end
			return result
		end
		if options.Multi then
			local value = type(options.Default) == "table" and options.Default or {}
			local current, nativeHandle = currentSection(), nil
			environment.pcall(function()
				nativeHandle = current:Dropdown({Text = translate(options.Text), Options = mappedOptions(options.Options), Multi = true, Default = value,
					Placeholder = options.Placeholder and translate(options.Placeholder) or nil, Tooltip = options.Tooltip and translate(options.Tooltip) or nil,
					Indent = indent(), Callback = function(nextValue)
						value = type(nextValue) == "table" and nextValue or {}
						if options.Callback then options.Callback(value) end
					end})
			end)
			bindHandle(nativeHandle, options.Text)
			track(nativeHandle)
			return {Row = nativeHandle and nativeHandle.Row or nil, Get = function() return value end,
				Set = function(nextValue)
					value = type(nextValue) == "table" and nextValue or {}
					if nativeHandle then environment.pcall(nativeHandle.Set, nativeHandle, value, true) end
					if options.Callback then environment.pcall(options.Callback, value) end
				end,
				SetOptions = function(values)
					options.Options = values
					if nativeHandle then environment.pcall(nativeHandle.SetOptions, nativeHandle, mappedOptions(values)) end
				end}
		end
		local key = not options.NoSave and configKey(options.Text) or nil
		if key then context.defaults[key] = options.Default end
		local labels = {}
		for _, item in ipairs(options.Options or {}) do labels[item.code] = tostring(item.label or item.code) end
		local previous = saved(key, "string")
		if previous ~= nil and labels[previous] then options.Default = previous end
		local value, current, nativeHandle = options.Default, currentSection(), nil
		environment.pcall(function()
			nativeHandle = current:Dropdown({Text = translate(options.Text), Options = mappedOptions(options.Options), Default = value,
				Tooltip = options.Tooltip and translate(options.Tooltip) or nil, Indent = indent(), Callback = function(nextValue)
					if nextValue == nil then return end
					value = nextValue
					store(key, nextValue)
					if options.Callback then options.Callback(nextValue) end
				end})
		end)
		bindHandle(nativeHandle, options.Text)
		track(nativeHandle)
		environment.pcall(function()
			table.insert(environment.getgenv().__pulseNovaLangHooks, function() nativeHandle:SetOptions(mappedOptions(options.Options)) end)
		end)
		local handle = {Row = nativeHandle and nativeHandle.Row or nil, Get = function() return value end}
		handle.SetSilent = function(nextValue)
			if labels[nextValue] == nil then return false end
			value = nextValue
			store(key, nextValue)
			if nativeHandle then environment.pcall(nativeHandle.Set, nativeHandle, nextValue, true) end
			return true
		end
		handle.Set = function(nextValue)
			if not handle.SetSilent(nextValue) then return end
			if options.Callback then environment.pcall(options.Callback, nextValue) end
		end
		handle.SetOptions = function(values)
			options.Options, labels = values, {}
			for _, item in ipairs(values or {}) do labels[item.code] = tostring(item.label or item.code) end
			if nativeHandle then environment.pcall(nativeHandle.SetOptions, nativeHandle, mappedOptions(values)) end
		end
		if key then context.setters[key] = function(nextValue) handle.Set(nextValue) end end
		if options.Settings then collect(options.Settings) end
		environment.pcall(function() if options.Callback then options.Callback(value) end end)
		return handle
	end

	function tab.Colorpicker(_, options)
		local key = not options.NoSave and configKey(options.Text) or nil
		local value = environment.typeof(options.Default) == "Color3" and options.Default or environment.Color3.fromRGB(255, 255, 255)
		if key then
			local _, hex = environment.pcall(function() return value:ToHex() end)
			context.defaults[key] = hex
		end
		local previous = saved(key, "string")
		if type(previous) == "string" then
			local ok, color = environment.pcall(environment.Color3.fromHex, previous)
			if ok and environment.typeof(color) == "Color3" then value = color end
		end
		local function saveColor(color)
			if key then environment.pcall(function() context.config[key] = color:ToHex() end) end
		end
		local current, nativeHandle = currentSection(), nil
		environment.pcall(function()
			nativeHandle = current:Colorpicker({Text = translate(options.Text), Default = value, Alpha = options.Alpha == true, DefaultAlpha = options.DefaultAlpha,
				Indent = indent(), Callback = function(color, alpha)
					if environment.typeof(color) ~= "Color3" then return end
					value = color
					saveColor(color)
					if options.Callback then options.Callback(color, alpha) end
				end})
		end)
		bindHandle(nativeHandle, options.Text)
		track(nativeHandle)
		local handle = {Row = nativeHandle and nativeHandle.Row or nil, Get = function() return value end}
		handle.SetSilent = function(color)
			if environment.typeof(color) ~= "Color3" then return false end
			value = color
			saveColor(color)
			if nativeHandle then environment.pcall(nativeHandle.SetSilent, nativeHandle, color) end
			return true
		end
		handle.Set = function(color)
			if not handle.SetSilent(color) then return end
			if options.Callback then options.Callback(color) end
		end
		handle.SetAlpha = function(alpha) if nativeHandle then environment.pcall(nativeHandle.SetSilent, nativeHandle, value, alpha) end end
		if key then
			context.setters[key] = function(hex)
				if type(hex) ~= "string" then return end
				local ok, color = environment.pcall(environment.Color3.fromHex, hex)
				if ok then handle.Set(color) end
			end
		end
		if options.Settings then collect(options.Settings) end
		environment.pcall(function() if options.Callback then options.Callback(value, options.DefaultAlpha) end end)
		return handle
	end

		corner(field, 9)
		theme.bind(field, "BackgroundColor3", "Surface2")
		local fieldStroke = create("UIStroke", {Thickness = 1, Transparency = 0.86, ApplyStrokeMode = environment.Enum.ApplyStrokeMode.Border}, field)
		theme.bind(fieldStroke, "Color", "Glass")

		local search = create("TextBox", {Position = environment.UDim2.fromOffset(30, 0), Size = environment.UDim2.new(1, -36, 1, 0), BackgroundTransparency = 1,
			BorderSizePixel = 0, ClearTextOnFocus = false, Text = "", PlaceholderText = translate("Search player"), TextSize = 12,
			TextXAlignment = environment.Enum.TextXAlignment.Left, TextTruncate = environment.Enum.TextTruncate.AtEnd, ZIndex = 2}, field)
		search.FontFace = nova.Font.Medium()
		theme.bind(search, "TextColor3", "Text")
		theme.bind(search, "PlaceholderColor3", "Faint")
		environment.pcall(function() translator.bind(search, "Search player", "PlaceholderText") end)

			nova.paint(fieldStroke, "Color", "Accent", "Quick")
			motion.tween(fieldStroke, "Quick", {Transparency = 0.4})
		end)

			Auto = environment.Enum.AutomaticSize.X, Color = "Muted", Z = 2})

			if query == "" then return true end
			return string.find(string.lower(player.DisplayName), query, 1, true) ~= nil or string.find(string.lower(player.Name), query, 1, true) ~= nil

			corner(avatar, 15)

			theme.bind(ring, "Color", "Glass")
			nova.label({Parent = row, Text = player.DisplayName, Font = nova.Font.Semi(), Size = 13, Pos = environment.UDim2.fromOffset(46, 6),
				Box = environment.UDim2.new(1, -94, 0, 17), Truncate = true, Color = "Text"})
			local subtitle = nova.label({Parent = row, Text = "@" .. player.Name, Font = nova.Font.Medium(), Size = 11, Pos = environment.UDim2.fromOffset(46, 24),
				Box = environment.UDim2.new(1, -94, 0, 14), Truncate = true, Color = "Faint"})
			local button = create("TextButton", {AnchorPoint = environment.Vector2.new(1, 0.5), Position = environment.UDim2.new(1, -7, 0.5, 0),
				Size = environment.UDim2.fromOffset(34, 30), Text = "", AutoButtonColor = false, BorderSizePixel = 0, BackgroundTransparency = 0.88}, row)
			corner(button, 9)
			theme.bind(button, "BackgroundColor3", "Accent")
			local icon = nova.icon({Parent = button, Icon = actionIcon, Size = 15, Pos = environment.UDim2.fromScale(0.5, 0.5),
				Anchor = environment.Vector2.new(0.5, 0.5), Color = "Accent", Z = 2})
			local entry = {row = row, button = button, buttonIcon = icon, subtitle = subtitle, ring = ring, userId = player.UserId, player = player, hover = false}
			local function selectPlayer()
				environment.pcall(function() nova.Sound.play("click") end)
				if options.Momentary then
					entry.flash = true
					environment.pcall(nova.Icons.apply, icon, "check")
					style(entry)
					environment.task.delay(0.7, function()
						entry.flash = false
						if icon.Parent then environment.pcall(nova.Icons.apply, icon, actionIcon); style(entry) end
					end)
					if options.OnSelect then environment.task.spawn(environment.pcall, options.OnSelect, player, player.UserId) end
					return
				end
				selectedId = player.UserId
				styleAll()
				if options.OnSelect then
					if selectedId then environment.pcall(options.OnSelect, player, player.UserId) else environment.pcall(options.OnSelect, nil, nil) end
				end
			end
			connect(row.MouseButton1Click, selectPlayer)
			connect(button.MouseButton1Click, selectPlayer)
			connect(row.MouseEnter, function() entry.hover = true; style(entry) end)
			utility.leave(row, function() entry.hover = false; style(entry) end)
			connect(button.MouseEnter, function() entry.buttonHover = true; style(entry) end)
			utility.leave(button, function() entry.buttonHover = false; style(entry) end)
			entry.sync = function()
				if options.SubText then
					local ok, text = environment.pcall(options.SubText, player)
					if ok and text ~= nil and subtitle.Text ~= tostring(text) then subtitle.Text = tostring(text) end
				end
				if options.SubColor then
					local ok, color = environment.pcall(options.SubColor, player)
					if ok and environment.typeof(color) == "Color3" then
						if subtitle.TextColor3 ~= color then override(subtitle, "TextColor3", color) end
						if ring.Color ~= color then override(ring, "Color", color) end
					end
				end
			end
			entry.sync()
			style(entry)
			return entry
		end
		local function refresh()
			if not context.alive() or not list.Parent then return end
			local visiblePlayers = {}
			for _, player in ipairs(players:GetPlayers()) do
				if options.ExcludeLocal and player == context.localPlayer then continue end
				if options.Filter then
					local ok, allowed = environment.pcall(options.Filter, player)
					if not ok or allowed == false then continue end
				end
				table.insert(visiblePlayers, player)
			end
			table.sort(visiblePlayers, function(left, right)
				if options.SortKey then
					local _, a = environment.pcall(options.SortKey, left)
					local _, b = environment.pcall(options.SortKey, right)
					if a ~= b and type(a) == type(b) then return a < b end
				end
				return string.lower(left.DisplayName) < string.lower(right.DisplayName)
			end)
			local present = {}
			for _, player in ipairs(visiblePlayers) do present[player.UserId] = true end
			for id, entry in pairs(entries) do
				if not present[id] then
					environment.pcall(function() entry.row:Destroy() end)
					entries[id] = nil
					if selectedId == id then
						selectedId = nil
						if options.OnSelect then environment.pcall(options.OnSelect, nil, nil) end
					end
				end
			end
			local matchesCount = 0
			for index, player in ipairs(visiblePlayers) do
				local entry = entries[player.UserId]
				if entry then
					entry.row.LayoutOrder = index
					entry.sync()
				else
					entry = makeRow(player, index)
					entries[player.UserId] = entry
				end
				entry.row.Visible = matches(player)
				if entry.row.Visible then matchesCount += 1 end
			end
			count.Text = tostring(#visiblePlayers)
			empty.Visible = matchesCount == 0
			emptyLabel.Text = translate(#visiblePlayers == 0 and "No players" or "Nothing found")
		end
		connect(search:GetPropertyChangedSignal("Text"), function()
			query = string.lower((search.Text:gsub("^%s+", ""):gsub("%s+$", "")))
			environment.pcall(refresh)
		end)
		refresh()
		connect(players.PlayerAdded, function() environment.task.delay(0.3, refresh) end)
		connect(players.PlayerRemoving, function() environment.task.delay(0.3, refresh) end)
		environment.task.spawn(function()
			while context.alive() and list.Parent do
				environment.task.wait(refreshInterval)
				if list.Parent then environment.pcall(refresh) end
			end
		end)
		return {GetSelectedId = function() return selectedId end,
			GetSelected = function() return selectedId and players:GetPlayerByUserId(selectedId) or nil end,
			SetSelected = function(id) selectedId = id; styleAll() end, Refresh = refresh}
	end

	return tab
end

local FovRenderer = {}

function FovRenderer.create(environment, create)
	local context = {environment = environment, create = create}

	return {
		make = function(parent) return FovRenderer.make(environment, create, parent) end,
		render = function(...) return FovRenderer.render(context, ...) end,
	}
end

function FovRenderer.render(context, item, radius, color, lockColor, locked, filled, fillStrength, style, animated, delta, opacity, rotation)
	local environment = context.environment
	local math = environment.math
	radius = math.max(2, radius)
	local smoothing = 1 - math.exp(-delta * 16)
	item.radius = item.radius or radius
	item.radius += (radius - item.radius) * smoothing
	radius = item.radius
	local target = locked and 1 or 0
	item.lock += (target - item.lock) * smoothing

	if rotation ~= nil then
		item.spin = rotation
	end

	if rotation == nil and animated then
		item.spin = (item.spin + delta * 75) % 360
	end

	local currentColor = color:Lerp(lockColor, item.lock * 0.42)
	local diameter = radius * 2
	local coreThickness = 1.6 + item.lock * 0.77
	local gradient = style == "Gradient"
	local glow = style == "Glow" or gradient
	local dashed = style == "Dashed"
	local classic = style == "Classic"
	local white = environment.Color3.new(1, 1, 1)
	local black = environment.Color3.new(0, 0, 0)

	item.fill.Visible = filled and true or false
	if filled then
		local strength = math.clamp((environment.tonumber(fillStrength) or 10) / 100 * (1 + item.lock * 0.35), 0, 0.9)
		local keypoint = environment.NumberSequenceKeypoint.new
		item.fill.Size = environment.UDim2.fromOffset(diameter, diameter)
		item.fill.BackgroundColor3 = currentColor
		item.fill.BackgroundTransparency = 0
		item.fillGradient.Rotation = 90
		item.fillGradient.Transparency = environment.NumberSequence.new({
			keypoint(0, 1 - strength * 0.55), keypoint(0.5, 1 - strength), keypoint(1, 1 - strength * 0.55),
		})
	end

	item.base.Visible = style ~= "Glow"
	item.base.Size = environment.UDim2.fromOffset(diameter, diameter)
	item.baseStroke.Color = currentColor
	item.baseStroke.Thickness = classic and 1.5 or 1
	item.baseStroke.Transparency = 1 - opacity * (classic and 1 or (dashed and 0.18 or 0.3))

	item.shade.Visible = not (classic or dashed)
	item.shade.Size = environment.UDim2.fromOffset(math.max(2, diameter - coreThickness - 2), math.max(2, diameter - coreThickness - 2))
	item.shadeStroke.Color = black
	item.shadeStroke.Thickness = coreThickness + 2
	item.shadeStroke.Transparency = 1 - opacity * (gradient and 0.14 or 0.26)

	item.core.Visible = not (classic or dashed)
	item.core.Size = environment.UDim2.fromOffset(math.max(2, diameter - coreThickness), math.max(2, diameter - coreThickness))
	item.coreStroke.Color = white
	item.coreStroke.Thickness = coreThickness
	item.coreStroke.Transparency = 1 - opacity
	local colorPoint = environment.ColorSequenceKeypoint.new
	local middleColor = gradient and environment.Color3.new(0.12, 0.55, 1) or currentColor:Lerp(white, 0.55)
	item.coreGradient.Color = environment.ColorSequence.new({colorPoint(0, currentColor), colorPoint(0.5, middleColor), colorPoint(1, currentColor)})
	item.coreGradient.Rotation = item.spin
	if gradient then
		local numberPoint = environment.NumberSequenceKeypoint.new
		item.coreGradient.Transparency = environment.NumberSequence.new({numberPoint(0, 0), numberPoint(0.5, 0.92), numberPoint(1, 0)})
	else
		item.coreGradient.Transparency = environment.NumberSequence.new(0)
	end

	FovRenderer._renderGlow(context, item, diameter, currentColor, opacity, glow, gradient)
	FovRenderer._renderDashes(context, item, radius, currentColor, opacity, dashed)
end

function FovRenderer._renderGlow(context, item, diameter, color, opacity, visible, gradient)
	local environment = context.environment
	local intensity = opacity * (gradient and 0.55 or 1) * (1 + item.lock * 0.63)

	for _, layer in ipairs(item.glow) do
		layer.frame.Visible = visible

		if not visible then
			continue
		end

		local size = environment.math.max(2, diameter + layer.inset * 2)
		layer.frame.Size = environment.UDim2.fromOffset(size, size)
		layer.stroke.Color = color
		layer.stroke.Thickness = layer.thickness * (1 + item.lock * 0.42)
		layer.stroke.Transparency = environment.math.clamp(1 - (1 - layer.alpha) * intensity, 0, 1)
	end
end

function FovRenderer._renderDashes(context, item, radius, color, opacity, visible)
	local environment = context.environment
	local math = environment.math
	local count = visible and math.clamp(math.floor(2 * math.pi * radius / 16), 12, 72) or 0

	while #item.dashes < count do
		local dash = context.create("Frame", {
			AnchorPoint = environment.Vector2.new(0.5, 0.5), BorderSizePixel = 0, ZIndex = 4,
		}, item.holder)
		context.create("UICorner", {CornerRadius = environment.UDim.new(1, 0)}, dash)
		table.insert(item.dashes, dash)
	end

	for index, dash in ipairs(item.dashes) do
		dash.Visible = index <= count

		if index > count then
			continue
		end

		local angle = (index - 1) * math.pi * 2 / count + math.rad(item.spin * 0.35)
		dash.Size = environment.UDim2.fromOffset(math.max(3, math.pi * radius / count), 2 + item.lock * 0.77)
		dash.Position = environment.UDim2.fromOffset(math.cos(angle) * radius, math.sin(angle) * radius)
		dash.Rotation = math.deg(angle) + 90
		dash.BackgroundColor3 = color
		dash.BackgroundTransparency = 1 - opacity * 0.96
	end
end

function FovRenderer.make(environment, create, parent)
	local center = environment.Vector2.new(0.5, 0.5)
	local position = environment.UDim2.fromScale(0.5, 0.5)
	local holder = create("Frame", {
		AnchorPoint = center, Position = position, Size = environment.UDim2.fromOffset(0, 0),
		BackgroundTransparency = 1, BorderSizePixel = 0,
	}, parent)
	local fill = create("Frame", {AnchorPoint = center, Position = position, BorderSizePixel = 0, ZIndex = 1}, holder)
	create("UICorner", {CornerRadius = environment.UDim.new(1, 0)}, fill)
	local fillGradient = create("UIGradient", {}, fill)
	local glow = {}

	for _, settings in ipairs({{3, 0.72, 0}, {7, 0.87, 0}, {13, 0.945, 0}, {3, 0.8, -3}, {8, 0.925, -8}}) do
		local frame, stroke = FovRenderer._circle(environment, create, holder, 2)
		table.insert(glow, {frame = frame, stroke = stroke, thickness = settings[1], alpha = settings[2], inset = settings[3]})
	end

	local shade, shadeStroke = FovRenderer._circle(environment, create, holder, 3)
	local base, baseStroke = FovRenderer._circle(environment, create, holder, 3)
	local core, coreStroke = FovRenderer._circle(environment, create, holder, 4)
	local coreGradient = create("UIGradient", {}, coreStroke)

	return {holder = holder, fill = fill, fillGradient = fillGradient, glow = glow,
		base = base, baseStroke = baseStroke, shade = shade, shadeStroke = shadeStroke,
		core = core, coreStroke = coreStroke, coreGradient = coreGradient, dashes = {}, spin = 0, lock = 0}
end

function FovRenderer._circle(environment, create, parent, zIndex)
	local frame = create("Frame", {
		AnchorPoint = environment.Vector2.new(0.5, 0.5), Position = environment.UDim2.fromScale(0.5, 0.5),
		BackgroundTransparency = 1, BorderSizePixel = 0, ZIndex = zIndex,
	}, parent)
	create("UICorner", {CornerRadius = environment.UDim.new(1, 0)}, frame)
	local stroke = create("UIStroke", {
		ApplyStrokeMode = environment.Enum.ApplyStrokeMode.Border, LineJoinMode = environment.Enum.LineJoinMode.Round,
	}, frame)

	return frame, stroke
end

local InputBindings = {}

function InputBindings.install(environment, inputService)
	local globals = environment.getgenv()
	local enum = environment.Enum
	local mouse = {
		MouseButton1 = enum.UserInputType.MouseButton1,
		MouseButton2 = enum.UserInputType.MouseButton2,
		MouseButton3 = enum.UserInputType.MouseButton3,
	}
	globals.__pulseBindMouse = mouse
	globals.__pulseBindCapture = false

	globals.__pulseBindFromName = function(name)
		if type(name) ~= "string" or name == "" then return nil end
		if mouse[name] then return mouse[name] end

		local ok, key = pcall(function() return enum.KeyCode[name] end)
		if ok and environment.typeof(key) == "EnumItem" then return key end
		return nil
	end

	globals.__pulseBindReady = function()
		return not globals.__pulseBindCapture and not inputService:GetFocusedTextBox()
	end

	globals.__pulseBindHit = function(input, binding)
		if environment.typeof(binding) ~= "EnumItem" then return false end
		if binding.EnumType == enum.UserInputType then return input.UserInputType == binding end
		return binding.EnumType == enum.KeyCode and input.UserInputType == enum.UserInputType.Keyboard
			and input.KeyCode == binding
	end
end

local Movement = {}

function Movement.install(context)
	local environment = context.environment
	local settings = context.configuration
	local section = context.section

	context.connect(context.inputService.JumpRequest, function()
		if not settings.infJump then return end

		local character = context.player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		if not humanoid or humanoid.Health <= 0 then return end

		pcall(function() humanoid:ChangeState(environment.Enum.HumanoidStateType.Jumping) end)
	end)

	section:ToggleKey({
		Text = "Infinite Jumps", Default = context.defaults.infJump,
		Bind = environment.getgenv().__pulseBindFromName(settings.infJumpKey or ""),
		Callback = function(enabled) settings.infJump = not not enabled end,
		OnBind = function(key) settings.infJumpKey = key and key.Name or "" end,
	})
	section:Toggle({
		Text = "Anti-Fling", Default = context.defaults.antiFling,
		Callback = function(enabled) settings.antiFling = not not enabled end,
	})
end

function Movement.installNoclip(context)
	local environment = context.environment
	local settings = context.configuration.noclip
	local saved, parts = {}, {}
	local character, changed, refreshedAt = nil, false, 0
	local listeners = {}

	local function collect(model)
		local result, seen = {}, {}
		local function add(part)
			if not part or seen[part] then return end
			seen[part] = true
			table.insert(result, part)
		end
		for _, part in ipairs(model:GetDescendants()) do
			if part:IsA("BasePart") then add(part) end
		end
		local root = model:FindFirstChild("HumanoidRootPart")
		if root then
			local ok, connected = pcall(function() return root:GetConnectedParts(true) end)
			if ok and type(connected) == "table" then
				for _, part in ipairs(connected) do add(part) end
			end
		end
		for _, name in ipairs({"DisplayRefKnife", "DisplayRefGun"}) do
			local reference = model:FindFirstChild(name)
			local item = reference and reference:IsA("ObjectValue") and reference.Value
			if item and item.Parent then
				if item:IsA("BasePart") then add(item) end
				for _, part in ipairs(item:GetDescendants()) do
					if part:IsA("BasePart") then add(part) end
				end
			end
		end
		return result
	end
	local function restore()
		for part, value in pairs(saved) do
			if part.Parent then pcall(function() part.CanCollide = value end) end
			saved[part] = nil
		end
	end
	local globals = environment.getgenv()
	globals.__pulseNoclipParts = collect
	context.connect(context.runService.Stepped, function()
		if not settings.on then return end
		local current = context.player.Character
		if not current then return end

		local now = environment.os.clock()
		if current ~= character or changed or now - refreshedAt > 1 then
			if current ~= character then
				character = current
				for _, listener in ipairs(listeners) do pcall(function() listener:Disconnect() end) end
				listeners = {
					context.connect(current.DescendantAdded, function() changed = true end),
					context.connect(current.DescendantRemoving, function() changed = true end),
				}
			end
			changed, refreshedAt = false, now
			parts = collect(current)
		end

		for _, part in ipairs(parts) do
			if part.CanCollide then
				if saved[part] == nil then saved[part] = true end
				part.CanCollide = false
			end
		end
	end)
	local toggle = context.section:ToggleKey({Text = "No Clip", Default = context.defaults.noclip.on,
		Bind = globals.__pulseBindFromName(settings.key or ""), Callback = function(enabled)
			settings.on = not not enabled
			if not enabled then restore() end
		end, OnBind = function(key) settings.key = key and key.Name or "" end})
	globals.__pulseSetNoclip = function(enabled)
		enabled = not not enabled
		if toggle and toggle.Set then
			pcall(toggle.Set, enabled)
			return
		end
		settings.on = enabled
		if not enabled then restore() end
	end
	globals.__pulseGetNoclip = function() return not not settings.on end

	return function()
		settings.on = false
		restore()
	end
end

function Movement.installCameraNoclip(context)
	local environment = context.environment
	local globals = environment.getgenv()
	local configuration = context.configuration
	if type(configuration.noclipCam) ~= "table" then configuration.noclipCam = {on = false, key = ""} end
	local settings = configuration.noclipCam
	local state = rawget(globals, "__pulseNoclipCam")
	if type(state) ~= "table" then
		state = {on = false, hooked = false}
		globals.__pulseNoclipCam = state
	end
	local previousMode

	local function findPopper()
		local scripts = context.player:FindFirstChild("PlayerScripts")
		local playerModule = scripts and scripts:FindFirstChild("PlayerModule")
		local camera = playerModule and playerModule:FindFirstChild("CameraModule")
		local zoom = camera and camera:FindFirstChild("ZoomController")
		local module = zoom and zoom:FindFirstChild("Popper")
		if not module then return nil end

		if type(environment.getgc) == "function" then
			local source = module:GetFullName()
			for _, callback in ipairs(environment.getgc(false)) do
				if type(callback) ~= "function" or type(environment.islclosure) == "function" and not environment.islclosure(callback) then
					continue
				end

				local ok, path, arguments = pcall(environment.debug.info, callback, "sa")
				if ok and type(path) == "string" and (path == source or path:sub(-#module.Name - 1) == "." .. module.Name)
					and arguments == 3 then
					local name = environment.debug.info(callback, "n")
					if not name or name == "" or string.lower(name) == "popper" then return callback end
				end
			end
		end

		local ok, callback = pcall(environment.require, module)
		if ok and type(callback) == "function" then return callback end
		return nil
	end
	local function installHook()
		if state.hooked then return true end
		if type(environment.hookfunction) ~= "function" then return false end
		local popper = findPopper()
		if not popper then return false end

		local original
		local ok = pcall(function()
			original = environment.hookfunction(popper, function(origin, distance, focus)
				local current = rawget(globals, "__pulseNoclipCam")
				if current and current.on and current.mode == "hook" and type(distance) == "number" then
					return distance
				end
				return original(origin, distance, focus)
			end)
		end)
		if not ok or not original then return false end
		state.hooked = true
		return true
	end
	local function setEnabled(enabled)
		state.on = not not enabled
		if state.on then
			if installHook() then
				state.mode = "hook"
				return
			end
			state.mode = "invisicam"
			pcall(function()
				if previousMode == nil then previousMode = context.player.DevCameraOcclusionMode end
				context.player.DevCameraOcclusionMode = environment.Enum.DevCameraOcclusionMode.Invisicam
			end)
			return
		end

		if state.mode == "invisicam" and previousMode ~= nil then
			pcall(function() context.player.DevCameraOcclusionMode = previousMode end)
			previousMode = nil
		end
		state.mode = nil
	end

	context.section:ToggleKey({Text = "No Clip Camera", Default = context.defaults.noclipCam and context.defaults.noclipCam.on,
		Bind = globals.__pulseBindFromName(settings.key or ""), Callback = function(enabled)
			settings.on = not not enabled
			setEnabled(settings.on)
		end, OnBind = function(key) settings.key = key and key.Name or "" end})
	return function() setEnabled(false) end
end

function Movement.installAntiCoin(context)
	local environment = context.environment
	local globals, workspace = environment.getgenv(), context.workspace
	local info = {n = 0, leaks = 0, strict = false}
	local coins = setmetatable({}, {__mode = "k"})
	local characterParts = setmetatable({}, {__mode = "k"})
	local enabled = false
	globals.__pulseAntiCoinInfo, globals.__pulseAntiCoin = info, false

	local function blockCoin(part)
		if not part:IsA("BasePart") then return end
		if coins[part] == nil then coins[part] = part.CanTouch end
		if part.CanTouch then
			part.CanTouch = false
			info.n += 1
		end
	end
	local function scanCoins()
		for _, map in ipairs(workspace:GetChildren()) do
			local container = map:FindFirstChild("CoinContainer")
			if container then
				for _, part in ipairs(container:GetChildren()) do pcall(blockCoin, part) end
			end
		end
	end
	local function blockCharacter()
		local character = context.player.Character
		if not character then return end
		for _, part in ipairs(character:GetChildren()) do
			if part:IsA("BasePart") then
				if characterParts[part] == nil then characterParts[part] = part.CanTouch end
				if part.CanTouch then pcall(function() part.CanTouch = false end) end
			end
		end
	end
	local function restore()
		for part, original in pairs(coins) do
			if part.Parent then pcall(function() part.CanTouch = original end) end
			coins[part] = nil
		end
		for part, original in pairs(characterParts) do
			if part.Parent then pcall(function() part.CanTouch = original end) end
			characterParts[part] = nil
		end
		info.strict = false
	end
	local function setEnabled(value)
		enabled = not not value
		globals.__pulseAntiCoin = enabled
		if enabled then pcall(scanCoins) else restore() end
	end
	info.set = setEnabled
	context.connect(workspace.DescendantAdded, function(part)
		if enabled and part.Parent and part.Parent.Name == "CoinContainer" then pcall(blockCoin, part) end
	end)
	context.connect(context.player.CharacterAdded, function()
		if not enabled or not info.strict then return end
		environment.task.wait(0.5)
		pcall(blockCharacter)
	end)
	environment.task.spawn(function()
		local gameplay
		while context.isAlive() and not gameplay do
			local remotes = context.storage:FindFirstChild("Remotes")
			gameplay = remotes and remotes:FindFirstChild("Gameplay")
			if not gameplay then environment.task.wait(1) end
		end
		local collected = gameplay and gameplay:FindFirstChild("CoinCollected")
		if not collected then return end
		context.connect(collected.OnClientEvent, function()
			if not enabled then return end
			info.leaks += 1
			if info.strict then return end
			info.strict = true
			pcall(blockCharacter)
			pcall(function()
				context.window:Notify({Title = context.translation.tr("Anti Coin"),
					Text = context.translation.tr("Coin leaked, strict mode on"), Kind = "error", Duration = 3})
			end)
		end)
	end)
	environment.task.spawn(function()
		while context.isAlive() do
			environment.task.wait(0.5)
			if enabled then
				pcall(scanCoins)
				if info.strict then pcall(blockCharacter) end
			end
		end
	end)
	context.section:Toggle({Text = "Anti Coin", Default = false, Callback = setEnabled})
	return function() setEnabled(false) end
end

function Movement.installBombJump(context)
	local environment = context.environment
	local globals = environment.getgenv()
	local settings, player = context.configuration, context.player
	local busy = false

	local function jump()
		local character = player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		local root = character and character:FindFirstChild("HumanoidRootPart")
		if not humanoid or not root or humanoid.Health <= 0 then return end

		local power = humanoid.UseJumpPower == false
			and math.sqrt(2 * (context.workspace.Gravity or 196.2) * math.max(humanoid.JumpHeight or 7.2, 1))
			or math.max(humanoid.JumpPower or 50, 50)
		globals.__pulseBombJumpUntil = math.max(tonumber(globals.__pulseBombJumpUntil) or 0, environment.os.clock() + 1.5)
		local velocity = root.AssemblyLinearVelocity
		root.AssemblyLinearVelocity = environment.Vector3.new(velocity.X, power, velocity.Z)
		pcall(function() humanoid:ChangeState(environment.Enum.HumanoidStateType.Jumping) end)
	end
	local function findBomb()
		local character = player.Character
		local bomb = character and character:FindFirstChild("FakeBomb")
		if bomb then return bomb end
		local backpack = player:FindFirstChild("Backpack")
		if not backpack then return nil end
		local toys = backpack:FindFirstChild("Toys")
		return toys and toys:FindFirstChild("FakeBomb") or backpack:FindFirstChild("FakeBomb")
	end
	local function equip()
		local character = player.Character
		if not character or not character:FindFirstChildOfClass("Humanoid") then return nil end
		local bomb = character:FindFirstChild("FakeBomb") or findBomb()
		if not bomb then return nil end
		pcall(function() bomb.Parent = character end)
		environment.task.wait(0.08)
		bomb = character:FindFirstChild("FakeBomb") or bomb
		pcall(function()
			local handle = bomb:FindFirstChild("Handle")
			local hand = character:FindFirstChild("RightHand") or character:FindFirstChild("Right Arm")
			if not handle or not hand or hand:FindFirstChild("RightGrip") then return end
			handle.Anchored = false
			local weld = environment.Instance.new("Weld")
			weld.Name, weld.Part0, weld.Part1 = "RightGrip", hand, handle
			weld.C0 = hand.Name == "RightHand" and environment.CFrame.new(0, -0.1, 0)
				or environment.CFrame.new(0, -1, 0) * environment.CFrame.Angles(-math.pi / 2, 0, 0)
			weld.C1, weld.Parent = bomb.Grip, hand
		end)
		return bomb.Parent == character and bomb or nil
	end
	local function throw(position)
		local bomb = equip()
		local remote = bomb and bomb:FindFirstChild("Remote")
		if not remote then return false end
		globals.__pulseBombJumpUntil = math.max(tonumber(globals.__pulseBombJumpUntil) or 0, environment.os.clock() + 2)
		pcall(function() remote:FireServer(environment.CFrame.new(position), 50) end)
		return true
	end
	local function watch(character)
		local function watchTool(tool)
			if tool.Name == "FakeBomb" and tool:IsA("Tool") then
				context.connect(tool.Activated, function() if settings.bombJump then jump() end end)
			end
		end
		for _, child in ipairs(character:GetChildren()) do watchTool(child) end
		context.connect(character.ChildAdded, watchTool)
	end
	if player.Character then watch(player.Character) end
	context.connect(player.CharacterAdded, function(character)
		environment.task.wait(0.3)
		watch(character)
	end)
	globals.__pulseAutoBombJump = function()
		if busy then return end
		busy = true
		environment.task.spawn(function()
			local character = player.Character
			local root = character and character:FindFirstChild("HumanoidRootPart")
			if root then
				equip()
				environment.task.wait(0.1)
				jump()
				environment.task.wait(0.12)
				throw(root.Position - environment.Vector3.new(0, 2.5, 0))
				environment.task.wait(0.22)
				jump()
			end
			busy = false
		end)
	end
	context.section:Toggle({Text = "Bomb Jump", Default = context.defaults.bombJump,
		Callback = function(enabled) settings.bombJump = not not enabled end, Settings = function()
			local key = globals.__pulseBindFromName(settings.bombJumpKey)
			context.section:Keybind({Text = "Bomb Jump", Default = key, Active = function() return busy end,
				Callback = function(value)
					key = value
					settings.bombJumpKey = value and value.Name or ""
				end})
			context.connect(context.inputService.InputBegan, function(input)
				if not globals.__pulseBindReady() or not globals.__pulseBindHit(input, key) then return end
				if globals.__pulseAutoBombJump then pcall(globals.__pulseAutoBombJump) end
			end)
		end})
end

function Movement.installInvisibility(context)
	local environment = context.environment
	local globals, player = environment.getgenv(), context.player
	local runService, workspace = context.runService, context.workspace
	local active, pending, humanoid, root = false, false, nil, nil
	local parts, heartbeat, lastReal, renderReal, ghostPosition = {}, nil, nil, nil, nil
	local groundedOffset, nudging, nudgeUntil, nudgeConnection = false, false, 0, nil
	local raycastParams, raycastCharacter

	local function nudge(duration)
		nudgeUntil = math.max(nudgeUntil, environment.os.clock() + (duration or 0.5))
		if nudgeConnection then return end
		nudgeConnection = context.connect(runService.Stepped, function()
			if environment.os.clock() > nudgeUntil or not humanoid or not root or not root.Parent or humanoid.Health <= 0 then
				if nudgeConnection then pcall(function() nudgeConnection:Disconnect() end) end
				nudgeConnection = nil
				return
			end
			if humanoid.MoveDirection.Magnitude > 0.01 then return end
			nudging = not nudging
			pcall(function()
				root.CFrame = root.CFrame * environment.CFrame.new(0, nudging and 0.03 or -0.03, 0)
					* environment.CFrame.Angles(0, nudging and 0.01 or -0.01, 0)
			end)
		end)
	end
	local function refreshCharacter()
		local character = player.Character
		if not character then
			humanoid, root, parts = nil, nil, {}
			return
		end
		humanoid, root, parts = character:FindFirstChildOfClass("Humanoid"), character:FindFirstChild("HumanoidRootPart"), {}
		for _, part in ipairs(character:GetDescendants()) do
			if part:IsA("BasePart") and part.Transparency == 0 then
				local near = part.Name == "Head" or part.Parent ~= nil and part.Parent:IsA("Accessory") or false
				table.insert(parts, {part = part, near = near})
			end
		end
	end
	local function transparency(enabled)
		pcall(function()
			for _, item in ipairs(parts) do
				if item.part and item.part.Parent then item.part.LocalTransparencyModifier = enabled and 0.5 or 0 end
			end
		end)
	end
	local function firstPerson()
		local camera = workspace.CurrentCamera
		if not camera then return false, nil end
		local position = camera.CFrame.Position
		if player.CameraMode == environment.Enum.CameraMode.LockFirstPerson then return true, position end
		local character = player.Character
		local head = character and character:FindFirstChild("Head")
		return head ~= nil and (head.Position - position).Magnitude < 1.5 or false, position
	end
	local function unbind()
		pcall(function() runService:UnbindFromRenderStep("PulseInvisRestore") end)
		pcall(function() runService:UnbindFromRenderStep("PulseInvisGhost") end)
	end
	local function stop()
		if not active then return end
		active, pending = false, false
		pcall(function() if globals.__pulseInvis then globals.__pulseInvis.active, globals.__pulseInvis.real = false, nil end end)
		if heartbeat then
			pcall(function() heartbeat:Disconnect() end)
			heartbeat = nil
		end
		unbind()
		transparency(false)
		if root and root.Parent then
			local target = lastReal or root.CFrame
			pcall(function() root.CFrame = target end)
			pcall(function() root.AssemblyLinearVelocity = environment.Vector3.new(0, 0, 0) end)
			pcall(function() root.AssemblyAngularVelocity = environment.Vector3.new(0, 0, 0) end)
			nudge(0.5)
		end
	end
	local function start()
		if active then return end
		refreshCharacter()
		if not humanoid or not root then return end
		active, pending = true, false
		transparency(true)
		heartbeat = context.connect(runService.Heartbeat, function()
			if not active then return end
			local valid = humanoid and root and root.Parent
			if not valid or root.Parent ~= player.Character or humanoid.Health <= 0 then
				if globals.__pulseInvis then globals.__pulseInvis.active, globals.__pulseInvis.real = false, nil end
				return
			end
			pcall(function()
				local revert = rawget(globals, "__pulseAfRevert")
				if revert then pcall(revert) end
				local moving = humanoid.MoveDirection.Magnitude > 0.01
				if not raycastParams then
					raycastParams = environment.RaycastParams.new()
					raycastParams.FilterType = environment.Enum.RaycastFilterType.Exclude
				end
				local character = player.Character
				if raycastCharacter ~= character then
					raycastCharacter = character
					raycastParams.FilterDescendantsInstances = {character}
				end
				local grounded = workspace:Raycast(root.Position, environment.Vector3.new(0, -5, 0), raycastParams) ~= nil
				local info = globals.__pulseInvis or {}
				info.active, info.grounded = true, grounded
				globals.__pulseInvis = info
				local real, offset = root.CFrame, -200000
				if grounded and not moving then
					groundedOffset = not groundedOffset
					if groundedOffset then offset = -199999.5 end
				end
				renderReal, lastReal = real, real
				info.real, info.realAt = real, environment.os.clock()
				if (tonumber(globals.__pulseInvisPause) or 0) > environment.os.clock() then return end
				root.CFrame = real * environment.CFrame.new(0, offset, 0)
				ghostPosition, pending = root.Position, true
			end)
		end)
		pcall(function()
			context.bindRender("PulseInvisRestore", environment.Enum.RenderPriority.Camera.Value - 1, function()
				if pending and root and root.Parent and renderReal then
					if not ghostPosition or (root.Position - ghostPosition).Magnitude < 2 then
						pcall(function() root.CFrame = renderReal end)
					end
					pending = false
				end
			end)
		end)
		nudge(0.5)
		pcall(function()
			context.bindRender("PulseInvisGhost", environment.Enum.RenderPriority.Camera.Value + 1, function()
				if not active then return end
				local first, cameraPosition = firstPerson()
				pcall(function()
					for _, item in ipairs(parts) do
						local part = item.part
						if part and part.Parent then
							local alpha = 0.5
							if first then
								if item.near then
									alpha = 1
								else
									local distance = cameraPosition and (part.Position - cameraPosition).Magnitude or 99
									if distance < 1.2 then alpha = 1 end
									if distance >= 1.2 and distance < 2.4 then alpha = math.max(0.5, 1 - (distance - 1.2) / 1.2) end
								end
							end
							part.LocalTransparencyModifier = alpha
						end
					end
				end)
			end)
		end)
	end
	context.connect(player.CharacterAdded, function()
		local enabled = not not context.configuration.invis.on
		active, pending = false, false
		if heartbeat then
			pcall(function() heartbeat:Disconnect() end)
			heartbeat = nil
		end
		unbind()
		environment.task.spawn(function()
			local character = player.Character
			if character then
				pcall(function() character:WaitForChild("HumanoidRootPart", 8) end)
				pcall(function() character:WaitForChild("Humanoid", 8) end)
			end
			environment.task.wait(0.4)
			refreshCharacter()
			if enabled and context.configuration.invis.on and player.Character then start() end
		end)
	end)
	context.section:ToggleKey({Text = "Invisible", Default = false,
		Bind = globals.__pulseBindFromName(context.configuration.invis.key), Callback = function(enabled)
			context.configuration.invis.on = not not enabled
			if enabled then start() else stop() end
		end, OnBind = function(key) context.configuration.invis.key = key and key.Name or "" end})
	globals.__pulseSetInvis = function(enabled)
		enabled = not not enabled
		if enabled == not not context.configuration.invis.on then return end
		context.configuration.invis.on = enabled
		if enabled then start() else stop() end
	end
	globals.__pulseGetInvis = function() return not not context.configuration.invis.on end
	return stop
end

function Movement.installSpinAndBhop(context)
	local environment = context.environment
	local globals, settings = environment.getgenv(), context.configuration
	local modes = {"Smooth", "Jitter", "Backwards", "Helicopter", "Break Dance", "Top Rock", "Air Dance",
		"Hero Flip", "Twirl", "Jacks", "Floss", "Tantrum", "Robot"}
	local animations = {
		Helicopter = {yaw = 2.4, emote = {10714347453, 1}}, ["Break Dance"] = {yaw = 1.8, emote = {10714003221, 1.3}},
		["Top Rock"] = {yaw = 1, emote = {10714336041, 1.4}}, ["Air Dance"] = {yaw = 1.4, emote = {10713957138, 1.5}},
		["Hero Flip"] = {yaw = 1.2, emote = {10714360164, 1.2}}, Twirl = {yaw = 2.2, emote = {10714293450, 1.5}},
		Jacks = {yaw = 1.6, emote = {10714375667, 1.6}}, Floss = {yaw = 1.2, emote = {10714340543, 1.4}},
		Tantrum = {yaw = 1.5, emote = {10714340558, 1.5}}, Robot = {yaw = 0.8, emote = {10714392151, 1.3}},
	}
	local animationState = {tracks = {}}
	globals.__pulseSpinAnim = {anim = animations, sa = animationState}
	local spinning, jitterTimer, jitter = false, 0, false
	local speed, landedAt, wasGrounded, reserve, chainUntil = 0, 0, true, 0, 0
	local sounds

	local function stopAnimation()
		local track = animationState.track
		animationState.track = nil
		if track then pcall(function() track:Stop(0.25) end) end
	end
	local function animate(character, humanoid, animation, spinSpeed)
		if animationState.char ~= character then
			stopAnimation()
			animationState.char, animationState.tracks = character, {}
		end
		local track = animationState.tracks[animation[1]]
		if not track then
			local ok, result = pcall(function()
				local animator = humanoid:FindFirstChildOfClass("Animator") or humanoid
				local asset = environment.Instance.new("Animation")
				asset.AnimationId = "rbxassetid://" .. animation[1]
				local loaded = animator:LoadAnimation(asset)
				loaded.Priority, loaded.Looped = environment.Enum.AnimationPriority.Action4, true
				return loaded
			end)
			if not ok or not result then return end
			track = result
			animationState.tracks[animation[1]] = track
		end
		if animationState.track ~= track then
			stopAnimation()
			animationState.track = track
			pcall(function() track:Play(0.25) end)
		end
		local playback = math.clamp(animation[2] * (0.8 + spinSpeed / 50), 0.5, 4)
		pcall(function() if math.abs(track.Speed - playback) > 0.01 then track:AdjustSpeed(playback) end end)
	end
	local function jumpSound()
		if not settings.bhopSound then return end
		if not sounds then
			sounds = {}
			for _, id in ipairs({7338942417, 7338942345, 7338942259, 7338942180}) do
				local ok, sound = pcall(function()
					local object = environment.Instance.new("Sound")
					object.SoundId, object.Volume, object.Parent = "rbxassetid://" .. id, 2, context.workspace
					return object
				end)
				if ok and sound then table.insert(sounds, sound) end
			end
		end
		if #sounds > 0 then
			local sound = sounds[math.random(1, #sounds)]
			pcall(function() sound:Play() end)
		end
	end
	local function yaw(frame)
		local look = frame.LookVector
		return math.atan2(-look.X, -look.Z)
	end
	local function spinAngle(root, delta)
		local mode = settings.spin.mode or "Smooth"
		local spinSpeed = math.clamp(tonumber(settings.spin.speed) or 10, 1, 100)
		local current, animation = yaw(root.CFrame), animations[mode]
		if animation then return current + spinSpeed * animation.yaw * delta end
		local camera = context.workspace.CurrentCamera
		local cameraYaw = camera and yaw(camera.CFrame) or 0
		if mode == "Jitter" then
			jitterTimer += delta
			if jitterTimer >= 1 / math.max(spinSpeed * 0.6, 0.5) then jitterTimer, jitter = 0, not jitter end
			return cameraYaw + (jitter and math.rad(75) or -math.rad(75))
		end
		if mode == "Backwards" then return cameraYaw + math.pi end
		return current + spinSpeed * delta
	end
	context.connect(context.runService.Stepped, function()
		local animation = settings.spin and settings.spin.on and animations[settings.spin.mode or ""]
		if not animation then
			stopAnimation()
			return
		end
		local character = context.player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		if not humanoid or humanoid.Health <= 0 or character:FindFirstChild("Gun") then
			stopAnimation()
			return
		end
		animate(character, humanoid, animation.emote, math.clamp(tonumber(settings.spin.speed) or 10, 1, 100))
	end)
	context.connect(context.runService.Heartbeat, function(delta)
		if not (settings.spin and settings.spin.on) and not settings.bhop and not spinning then
			if globals.__pulseBhopChain then
				speed, reserve, chainUntil = 0, 0, 0
				globals.__pulseBhopSpeed, globals.__pulseBhopChain = 0, false
			end
			return
		end
		local character = context.player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		local root = character and character:FindFirstChild("HumanoidRootPart")
		if settings.spin and settings.spin.on then
			if humanoid and root and humanoid.Health > 0 then
				spinning = true
				pcall(function()
					humanoid.AutoRotate = false
					if environment.os.clock() < (tonumber(globals.__pulseShotHold) or 0) then return end
					root.CFrame = environment.CFrame.new(root.Position) * environment.CFrame.Angles(0, spinAngle(root, delta), 0)
				end)
			end
		end
		if not (settings.spin and settings.spin.on) and spinning then
			spinning = false
			if humanoid and root then
				pcall(function()
					humanoid.AutoRotate = true
					root.AssemblyAngularVelocity = environment.Vector3.new(0, 0, 0)
				end)
			end
		end

		local status
		pcall(function() status = humanoid:GetState() end)
		local stateTypes = environment.Enum.HumanoidStateType
		if settings.bhop and humanoid and root and humanoid.Health > 0 and not globals.__pulseTouchFlinging and status ~= stateTypes.Climbing then
			local invisible = globals.__pulseInvis
			local grounded = invisible and invisible.active and invisible.grounded
				or (not (invisible and invisible.active) and humanoid.FloorMaterial ~= environment.Enum.Material.Air
					and status ~= stateTypes.Freefall and status ~= stateTypes.Jumping)
			local now, velocity = environment.os.clock(), root.AssemblyLinearVelocity
			local power, walkSpeed = settings.bhopPower or 5, humanoid.WalkSpeed
			if grounded and not wasGrounded then
				landedAt = now
			end
			if not grounded and wasGrounded and velocity.Y > 2 and humanoid.MoveDirection.Magnitude > 0.1 then
				speed = speed > 0 and now - landedAt <= 0.25 and speed + 2 + power * 1.2 or math.max(speed, walkSpeed + 2 + power)
				jumpSound()
			end
			if grounded and now - landedAt > 0.3 then speed = 0 end
			local movement = humanoid.MoveDirection
			if speed > 0 and movement.Magnitude > 0.1 then
				speed = math.min(speed, math.max(walkSpeed * (2 + 0.4 * power), 100))
				if not grounded or now - landedAt <= 0.25 then
					local direction = environment.Vector3.new(movement.X, 0, movement.Z).Unit
					local desired = math.max(speed, tonumber(globals.__pulseGlitchCur) or 0)
					root.AssemblyLinearVelocity = environment.Vector3.new(direction.X * desired, velocity.Y, direction.Z * desired)
				end
			end
			wasGrounded = grounded
			if speed > 0 then
				reserve, chainUntil = math.max(reserve, speed + 2 + power * 1.2), now + 0.5
			end
			if not (speed > 0) and now > chainUntil then
				reserve = 0
			end
			globals.__pulseBhopSpeed = math.max(speed, reserve, walkSpeed + 2 + power)
			globals.__pulseBhopChain = speed > 0 or now <= chainUntil
		else
			wasGrounded, speed = true, 0
			if reserve > 0 and environment.os.clock() <= chainUntil then
				globals.__pulseBhopSpeed, globals.__pulseBhopChain = reserve, true
			else
				reserve, chainUntil = 0, 0
				globals.__pulseBhopSpeed, globals.__pulseBhopChain = 0, false
			end
		end
	end)
	if not table.find(modes, settings.spin.mode) then settings.spin.mode = "Smooth" end
	if not table.find(modes, context.defaults.spin.mode) then context.defaults.spin.mode = "Smooth" end
	context.section:ToggleKey({Text = "Spin", Default = context.defaults.spin.on,
		Bind = globals.__pulseBindFromName(settings.spin.key), Callback = function(enabled) settings.spin.on = not not enabled end,
		OnBind = function(key) settings.spin.key = key and key.Name or "" end, Settings = function()
			local options = {}
			for _, mode in ipairs(modes) do table.insert(options, {code = mode, label = mode == "Smooth" and "Classic" or mode}) end
			context.section:Dropdown({Text = "Spin Mode", Options = options, Default = context.defaults.spin.mode, Callback = function(value)
				settings.spin.mode, jitterTimer, jitter = value, 0, false
			end})
			context.section:Slider({Text = "Spin Speed", Min = 1, Max = 100, Default = context.defaults.spin.speed or 10, Decimals = 0,
				Callback = function(value) settings.spin.speed = value end})
		end})
	context.section:ToggleKey({Text = "Bhop", Default = context.defaults.bhop,
		Bind = globals.__pulseBindFromName(settings.bhopKey), Callback = function(enabled) settings.bhop = not not enabled end,
		OnBind = function(key) settings.bhopKey = key and key.Name or "" end, Settings = function()
			context.section:Slider({Text = "Bhop Power", Min = 1, Max = 10, Default = context.defaults.bhopPower or 5, Decimals = 0,
				Callback = function(value) settings.bhopPower = value end})
			context.section:Toggle({Text = "Bhop Sound", Default = context.defaults.bhopSound,
				Callback = function(value) settings.bhopSound = not not value end})
		end})

	return function()
		stopAnimation()
		local character = context.player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		local root = character and character:FindFirstChild("HumanoidRootPart")
		if spinning and humanoid and root then
			pcall(function()
				humanoid.AutoRotate = true
				root.AssemblyAngularVelocity = environment.Vector3.new(0, 0, 0)
			end)
		end
		globals.__pulseBhopSpeed, globals.__pulseBhopChain = 0, false
	end
end

function Movement.installFly(context)
	local environment = context.environment
	local enum, vector, cframe = environment.Enum, environment.Vector3, environment.CFrame
	local states = enum.HumanoidStateType
	local zero, up = vector.new(0, 0, 0), vector.new(0, 1, 0)
	local settings, globals = context.configuration.fly, environment.getgenv()
	local state = {velocity = zero, bank = 0, lastUp = 0}
	local upHeld, downHeld, lastJump = false, false, 0
	local methods = {}

	local function destroy(name)
		local object = state[name]
		if object then pcall(function() object:Destroy() end) end
		state[name] = nil
	end
	local function motors(root)
		if not state.velocityMotor or state.velocityMotor.Parent ~= root then
			destroy("velocityMotor")
			local motor = environment.Instance.new("BodyVelocity")
			motor.MaxForce = vector.new(9e9, 9e9, 9e9)
			motor.Velocity = vector.new(0, 0.1, 0)
			motor:SetAttribute("PulseMine", true)
			motor.Parent = root
			state.velocityMotor = motor
		end
		if not state.gyro or state.gyro.Parent ~= root then
			destroy("gyro")
			local gyro = environment.Instance.new("BodyGyro")
			gyro.P = 90000
			gyro.MaxTorque = vector.new(9e9, 9e9, 9e9)
			gyro.CFrame = root.CFrame
			gyro:SetAttribute("PulseMine", true)
			gyro.Parent = root
			state.gyro, state.gyroFrame = gyro, root.CFrame
		end
	end
	local function changeState(humanoid, target)
		local ok, current = pcall(function() return humanoid:GetState() end)
		if ok and current ~= target then pcall(function() humanoid:ChangeState(target) end) end
	end
	local function setStates(humanoid, disabled, enabled)
		state.savedStates = state.savedStates or {}
		local function remember(value)
			if state.savedStates[value] ~= nil then return end
			local ok, current = pcall(function() return humanoid:GetStateEnabled(value) end)
			state.savedStates[value] = not (ok and current == false)
		end
		for _, value in ipairs(disabled) do
			remember(value)
			pcall(function() humanoid:SetStateEnabled(value, false) end)
		end
		if enabled then
			remember(enabled)
			pcall(function() humanoid:SetStateEnabled(enabled, true) end)
		end
	end
	local function cleanup()
		local humanoid = state.humanoid
		local character = humanoid and humanoid.Parent
		local savedStates = state.savedStates
		state.savedStates = nil
		if humanoid and savedStates then
			for value, enabled in pairs(savedStates) do pcall(function() humanoid:SetStateEnabled(value, enabled) end) end
			if humanoid.Parent and humanoid.Health > 0 then pcall(function() humanoid:ChangeState(states.Freefall) end) end
		end
		destroy("velocityMotor")
		destroy("gyro")
		destroy("platform")
		if state.stand and humanoid then pcall(function() humanoid.PlatformStand = false end) end
		if state.animationDisabled and character then
			local animate = character:FindFirstChild("Animate")
			if animate then pcall(function() animate.Disabled = false end) end
		end
		state.stand, state.animationDisabled = false, false
		state.method, state.humanoid, state.root, state.gyroFrame = nil, nil, nil, nil
		state.level, state.position, state.yaw, state.face = nil, nil, nil, nil
		state.velocity, state.bank, state.lastUp = zero, 0, 0
	end
	local function stand(character, humanoid)
		if not humanoid.PlatformStand then pcall(function() humanoid.PlatformStand = true end) end
		state.stand = true
		local animate = character:FindFirstChild("Animate")
		if animate and not animate.Disabled then
			pcall(function() animate.Disabled = true end)
			state.animationDisabled = true
		end
		local animator = humanoid:FindFirstChildOfClass("Animator")
		if animator then
			for _, track in ipairs(animator:GetPlayingAnimationTracks()) do pcall(function() track:Stop(0) end) end
		end
	end
	local function facing(root, movement, velocity, fallback, delta)
		local horizontal = vector.new(movement.X, 0, movement.Z)
		local drift = vector.new(velocity.X, 0, velocity.Z)
		local direction = horizontal.Magnitude > 0.01 and horizontal.Unit or (drift.Magnitude > 0.5 and drift.Unit or fallback)
		local target = cframe.lookAt(root.Position, root.Position + direction)
		state.gyroFrame = state.gyroFrame and state.gyroFrame:Lerp(target, math.min(1, 8 * delta)) or target
		state.gyro.CFrame = state.gyroFrame
	end
	local function height(humanoid, root)
		if humanoid.RigType == enum.HumanoidRigType.R6 then return root.Size.Y / 2 + 2 end
		return humanoid.HipHeight + root.Size.Y / 2
	end
	local function direction(humanoid)
		local camera, move = context.workspace.CurrentCamera, humanoid.MoveDirection
		local forward, result = vector.new(0, 0, -1), zero
		if camera then
			local cameraFrame = camera.CFrame
			local look = cameraFrame.LookVector
			local flat = vector.new(look.X, 0, look.Z)
			forward = flat.Magnitude > 0.01 and flat.Unit or forward
			if move.Magnitude > 0 then
				local right = cameraFrame.RightVector
				local flatRight = vector.new(right.X, 0, right.Z)
				flatRight = flatRight.Magnitude > 0.01 and flatRight.Unit or vector.new(1, 0, 0)
				result = look * move:Dot(forward) + flatRight * move:Dot(flatRight)
			end
		end
		local vertical = (upHeld or environment.os.clock() - lastJump < 0.15) and 1 or 0
		if downHeld then vertical -= 1 end
		return result, vertical, forward
	end

	methods.Classic = {
		start = function(humanoid, root)
			motors(root)
			setStates(humanoid, {states.Freefall})
			pcall(function() humanoid:ChangeState(states.Running) end)
		end,
		step = function(humanoid, root, character, delta, movement, vertical, forward, speed)
			motors(root)
			stand(character, humanoid)
			local target = movement + up * vertical
			local velocity = target.Magnitude > 0.01 and target.Unit * speed or zero
			state.velocityMotor.Velocity = velocity
			facing(root, movement, velocity, forward, delta)
		end,
	}
	methods.CFrame = {
		start = function(humanoid, root)
			setStates(humanoid, {states.Freefall, states.FallingDown, states.Ragdoll, states.Jumping})
			state.position = root.Position
		end,
		step = function(humanoid, root, _, delta, movement, vertical, forward, speed)
			changeState(humanoid, states.Running)
			local position = root.Position
			if not state.position or (position - state.position).Magnitude > 8 then state.position = position end
			local target = movement + up * vertical
			if target.Magnitude > 0.01 then state.position += target.Unit * speed * delta end
			local look = root.CFrame.LookVector
			local facingNow = vector.new(look.X, 0, look.Z)
			facingNow = facingNow.Magnitude > 0.01 and facingNow.Unit or forward
			local desired = vector.new(movement.X, 0, movement.Z)
			desired = desired.Magnitude > 0.01 and desired.Unit or facingNow
			local smoothed = facingNow:Lerp(desired, math.min(1, 10 * delta))
			if smoothed.Magnitude < 0.01 then smoothed = desired end
			root.CFrame = cframe.lookAt(state.position, state.position + smoothed.Unit)
			root.AssemblyLinearVelocity, root.AssemblyAngularVelocity = zero, zero
		end,
	}
	methods.Swim = {
		start = function(humanoid, root)
			motors(root)
			setStates(humanoid, {states.Freefall, states.Running, states.RunningNoPhysics, states.Landed, states.Jumping,
				states.Climbing, states.FallingDown, states.GettingUp, states.Ragdoll}, states.Swimming)
			changeState(humanoid, states.Swimming)
		end,
		step = function(humanoid, root, _, delta, movement, vertical, forward, speed)
			motors(root)
			changeState(humanoid, states.Swimming)
			local target = movement + up * vertical
			local velocity = target.Magnitude > 0.01 and target.Unit * speed or zero
			state.velocity = state.velocity:Lerp(velocity, math.min(1, 9 * delta))
			state.velocityMotor.Velocity = state.velocity
			facing(root, movement, state.velocity, forward, delta)
		end,
	}
	methods["Air Walk"] = {
		start = function(humanoid, root)
			setStates(humanoid, {states.Jumping, states.Freefall, states.FallingDown, states.Ragdoll})
			state.level = root.Position.Y - height(humanoid, root)
		end,
		step = function(humanoid, root, _, delta, _, vertical, _, speed)
			if not state.platform or not state.platform.Parent then
				local platform = environment.Instance.new("Part")
				platform.Name, platform.Size = "PulseAirWalk", vector.new(10, 1, 10)
				platform.Anchored, platform.CanTouch, platform.CastShadow, platform.Transparency = true, false, false, 1
				platform:SetAttribute("PulseMine", true)
				platform.CFrame = cframe.new(root.Position.X, state.level - 0.5, root.Position.Z)
				platform.Parent = context.workspace
				state.platform = platform
			end
			changeState(humanoid, states.Running)
			local ascent, position = math.clamp(speed, 10, 120), root.Position
			local feet = position.Y - height(humanoid, root)
			if vertical ~= 0 then
				state.level += vertical * ascent * delta
				local velocity = root.AssemblyLinearVelocity
				root.AssemblyLinearVelocity = vector.new(velocity.X, vertical * ascent, velocity.Z)
			else
				if state.lastUp ~= 0 then
					local velocity = root.AssemblyLinearVelocity
					root.AssemblyLinearVelocity = vector.new(velocity.X, 0, velocity.Z)
				end
				if humanoid.FloorMaterial ~= enum.Material.Air and feet > state.level + 0.2 then state.level = feet end
			end
			state.lastUp = vertical
			state.platform.CFrame = cframe.new(position.X, state.level - 0.5, position.Z)
		end,
	}
	methods.Glide = {
		start = function(humanoid, root)
			motors(root)
			setStates(humanoid, {states.Freefall})
			pcall(function() humanoid:ChangeState(states.Running) end)
		end,
		step = function(humanoid, root, character, delta, movement, vertical, forward, speed)
			motors(root)
			stand(character, humanoid)
			local target = movement + up * vertical
			local velocity = target.Magnitude > 0.01 and target.Unit * speed or zero
			local rate = velocity.Magnitude > 0.01 and 2.2 or 1.6
			state.velocity = state.velocity:Lerp(velocity, 1 - math.exp(-rate * delta))
			if velocity.Magnitude < 0.01 and state.velocity.Magnitude < 0.3 then state.velocity = zero end
			state.velocityMotor.Velocity = state.velocity
			local horizontal = vector.new(state.velocity.X, 0, state.velocity.Z)
			local horizontalSpeed = horizontal.Magnitude
			if horizontalSpeed > 1 then state.face = horizontal.Unit end
			local face = state.face or forward
			local ratio = math.clamp(state.velocity.Magnitude / math.max(speed, 1), 0, 1)
			local pitch = math.atan2(state.velocity.Y, math.max(horizontalSpeed, 0.001))
			local angle = math.clamp(ratio * (math.rad(65) - pitch * 0.6), 0, math.rad(115))
			local yaw = math.atan2(-face.X, -face.Z)
			local turn = state.yaw and ((yaw - state.yaw + math.pi) % (2 * math.pi) - math.pi) / math.max(delta, 1 / 240) or 0
			state.yaw = yaw
			local bank = math.clamp(turn * 0.3, -0.75, 0.75) * ratio
			state.bank += (bank - state.bank) * (1 - math.exp(-6 * delta))
			local targetFrame = cframe.lookAt(root.Position, root.Position + face)
				* cframe.Angles(0, 0, state.bank) * cframe.Angles(-angle, 0, 0)
			state.gyroFrame = state.gyroFrame and state.gyroFrame:Lerp(targetFrame, math.min(1, 8 * delta)) or targetFrame
			state.gyro.CFrame = state.gyroFrame
		end,
	}
	context.connect(context.inputService.JumpRequest, function() lastJump = environment.os.clock() end)
	context.connect(context.inputService.InputBegan, function(input, processed)
		if processed or input.UserInputType ~= enum.UserInputType.Keyboard then return end
		if input.KeyCode == enum.KeyCode.Space then upHeld = true end
		if input.KeyCode == enum.KeyCode.LeftControl then downHeld = true end
	end)
	context.connect(context.inputService.InputEnded, function(input)
		if input.KeyCode == enum.KeyCode.Space then upHeld = false end
		if input.KeyCode == enum.KeyCode.LeftControl then downHeld = false end
	end)
	context.connect(context.runService.Stepped, function(_, delta)
		if not context.configuration.fly or not settings.on then
			globals.__pulseFlyActive = false
			if state.method then cleanup() end
			return
		end

		local speed = tonumber(settings.speed) or 60
		globals.__pulseFlyActive, globals.__pulseFlySpeed = true, speed
		local character = context.player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		local root = character and character:FindFirstChild("HumanoidRootPart")
		if not humanoid or not root or humanoid.Health <= 0 then
			if state.method then cleanup() end
			return
		end
		local method = methods[settings.method] and settings.method or "Classic"
		if state.method ~= method or state.humanoid ~= humanoid or state.root ~= root then
			if state.method then cleanup() end
			state.method, state.humanoid, state.root = method, humanoid, root
			methods[method].start(humanoid, root)
		end
		delta = math.clamp(tonumber(delta) or 1 / 60, 0, 0.1)
		local movement, vertical, forward = direction(humanoid)
		methods[method].step(humanoid, root, character, delta, movement, vertical, forward, speed)
	end)
	context.section:ToggleKey({Text = "Fly", Default = context.defaults.fly.on,
		Bind = globals.__pulseBindFromName(settings.key), Callback = function(enabled) settings.on = not not enabled end,
		OnBind = function(key) settings.key = key and key.Name or "" end, Settings = function()
			context.section:Slider({Text = "Fly Speed", Min = 10, Max = 300, Default = context.defaults.fly.speed or 60,
				Decimals = 0, Callback = function(value) settings.speed = value end})
			local options = {}
			for _, method in ipairs({"Classic", "CFrame", "Swim", "Air Walk", "Glide"}) do
				table.insert(options, {code = method, label = method})
			end
			context.section:Dropdown({Text = "Fly Method", Options = options, Default = context.defaults.fly.method or "Classic",
				Callback = function(value) if methods[value] then settings.method = value end end})
		end})

	return function()
		globals.__pulseFlyActive = false
		cleanup()
	end
end

function Movement.installSpeed(context)
	local environment = context.environment
	local section = context.section
	local state = {on = false, value = 16, method = "WalkSpeed", base = 16, applied = false, elapsed = 0}
	local methods = {"WalkSpeed", "CFrame", "TP Walk", "Velocity", "Mover"}
	local jumpEnabled, jumpPower = false, 50

	local function humanoid()
		local character = context.player.Character
		return character and character:FindFirstChildOfClass("Humanoid")
	end
	local function dropMover()
		if not state.mover then return end
		pcall(function() state.mover:Destroy() end)
		state.mover = nil
	end
	local function apply()
		local characterHumanoid = humanoid()
		if not characterHumanoid then return end

		if state.on and state.method == "WalkSpeed" then
			if characterHumanoid.WalkSpeed ~= state.value then characterHumanoid.WalkSpeed = state.value end
			state.applied = true
			return
		end
		if state.applied then
			state.applied = false
			if characterHumanoid.WalkSpeed ~= state.base then characterHumanoid.WalkSpeed = state.base end
		end
	end
	local function step(_, delta)
		if not state.on or state.method == "WalkSpeed" or context.configuration.fly and context.configuration.fly.on then
			dropMover()
			return
		end

		local characterHumanoid = humanoid()
		local character = context.player.Character
		local root = character and character:FindFirstChild("HumanoidRootPart")
		if not characterHumanoid or not root or characterHumanoid.Health <= 0 or characterHumanoid.Sit then
			dropMover()
			return
		end

		delta = math.clamp(tonumber(delta) or 1 / 60, 0, 0.1)
		local direction = characterHumanoid.MoveDirection
		local movement = environment.Vector3.new(direction.X, 0, direction.Z)
		if movement.Magnitude > 1 then movement = movement.Unit end
		local moving = movement.Magnitude > 0.01
		local extra = state.value - characterHumanoid.WalkSpeed

		if state.method == "CFrame" then
			dropMover()
			if moving and extra > 0 then root.CFrame = root.CFrame + movement * extra * delta end
			return
		end
		if state.method == "TP Walk" then
			dropMover()
			state.elapsed += delta
			if state.elapsed >= 0.1 then
				local interval = math.min(state.elapsed, 0.2)
				state.elapsed = 0
				if moving and extra > 0 then root.CFrame = root.CFrame + movement * extra * interval end
			end
			return
		end
		if state.method == "Velocity" then
			dropMover()
			if moving then
				root.AssemblyLinearVelocity = environment.Vector3.new(movement.X * state.value,
					root.AssemblyLinearVelocity.Y, movement.Z * state.value)
			end
			return
		end
		if state.method == "Mover" then
			if not state.mover or state.mover.Parent ~= root then
				dropMover()
				local mover = environment.Instance.new("BodyVelocity")
				mover.Name = "PulseSpeedMover"
				mover.MaxForce = environment.Vector3.new(100000, 0, 100000)
				mover.P = 10000
				mover:SetAttribute("PulseMine", true)
				mover.Velocity = environment.Vector3.zero
				mover.Parent = root
				state.mover = mover
			end
			state.mover.Velocity = movement * state.value
		end
	end
	local function applyJump()
		local characterHumanoid = humanoid()
		if characterHumanoid and jumpEnabled then
			characterHumanoid.UseJumpPower = true
			characterHumanoid.JumpPower = jumpPower
		end
	end

	context.connect(context.runService.Stepped, step)
	local speed
	local globals = environment.getgenv()
	globals.__pulseSetSpeed = function(enabled)
		enabled = not not enabled
		if speed and speed.Set then
			pcall(speed.Set, enabled)
			return
		end

		state.on = enabled
		if not enabled then dropMover() end
		apply()
	end
	globals.__pulseGetSpeed = function() return not not state.on end
	speed = section:ToggleKey({Text = "Walk Speed", Default = false, Callback = function(enabled)
		state.on = not not enabled
		if not state.on then dropMover() end
		apply()
	end, Settings = function()
		section:Slider({Text = "Walk Speed Value", Min = 16, Max = 100, Default = 16, Decimals = 0,
			Callback = function(value)
				state.value = tonumber(value) or 16
				apply()
			end})
		local options = {}
		for _, method in ipairs(methods) do table.insert(options, {code = method, label = method}) end
		section:Dropdown({Text = "Speed Method", Options = options, Default = "WalkSpeed", Callback = function(value)
			if not table.find(methods, value) then return end
			state.method, state.elapsed = value, 0
			dropMover()
			apply()
		end})
	end})
	section:Toggle({Text = "Jump Power", Default = false, Callback = function(enabled)
		jumpEnabled = enabled
		if enabled then
			applyJump()
			return
		end

		local characterHumanoid = humanoid()
		if characterHumanoid then
			characterHumanoid.UseJumpPower = true
			characterHumanoid.JumpPower = 50
		end
	end, Settings = function()
		section:Slider({Text = "Jump Power Value", Min = 50, Max = 200, Default = 50, Decimals = 0,
			Callback = function(value)
				jumpPower = value
				applyJump()
			end})
	end})
	if context.isAlive then
		environment.task.spawn(function()
			while context.isAlive() do
				environment.task.wait(0.5)
				local characterHumanoid = humanoid()
				if characterHumanoid then
					apply()
					if jumpEnabled then
						if not characterHumanoid.UseJumpPower then characterHumanoid.UseJumpPower = true end
						if characterHumanoid.JumpPower ~= jumpPower then characterHumanoid.JumpPower = jumpPower end
					end
				end
			end
		end)
	end

	return function()
		state.on = false
		dropMover()
		apply()
	end
end

function Movement.installGlitch(context)
	local environment = context.environment
	local globals = environment.getgenv()
	local state = {on = false, max = 120, acceleration = 60, speed = 0, landedAt = 0, airborne = false}
	local humanoidStates = environment.Enum.HumanoidStateType

	local function publish(speed)
		globals.__pulseGlitchCur = speed
		globals.__pulseGlitchSpeed = speed > 0 and speed + 6 or 0
	end
	local function reset()
		if state.speed == 0 and not state.airborne then return end
		state.speed, state.airborne = 0, false
		publish(0)
	end
	local function step(delta)
		if not state.on then
			reset()
			return
		end

		local character = context.player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		local root = humanoid and humanoid.RootPart
		if not root or humanoid.Health <= 0 or globals.__pulseFlyActive or globals.__pulseTouchFlinging then
			reset()
			return
		end

		local status = humanoid:GetState()
		if status == humanoidStates.Climbing or status == humanoidStates.Swimming
			or status == humanoidStates.Seated or status == humanoidStates.Dead then
			reset()
			return
		end

		delta = math.min(delta or 1 / 60, 0.1)
		local now = environment.os.clock()
		local inAir = status == humanoidStates.Freefall or status == humanoidStates.Jumping
		local direction = humanoid.MoveDirection
		local movement = environment.Vector3.new(direction.X, 0, direction.Z)
		local moving = movement.Magnitude > 0.1
		local velocity = root.AssemblyLinearVelocity
		local horizontal = environment.Vector3.new(velocity.X, 0, velocity.Z)
		local current = horizontal.Magnitude

		if inAir then
			state.airborne = true
			if moving then
				state.speed = math.max(state.speed, humanoid.WalkSpeed)
				if current > state.speed and current <= state.max then state.speed = current end
				if state.speed < state.max then state.speed = math.min(state.max, state.speed + state.acceleration * delta) end
			end
		else
			if state.airborne then state.airborne, state.landedAt = false, now end
			if now - state.landedAt > 0.3 then state.speed = 0 end
		end

		state.speed = math.min(state.speed, state.max)
		publish(state.speed)
		if state.speed > 0 and moving and (inAir or now - state.landedAt <= 0.25) and current < state.speed + 1 then
			local desired = movement.Unit * state.speed
			if (horizontal - desired).Magnitude > 0.5 then
				root.AssemblyLinearVelocity = environment.Vector3.new(desired.X, velocity.Y, desired.Z)
			end
		end
	end

	context.connect(context.runService.Heartbeat, step)
	local toggle = context.section:ToggleKey({Text = "Speed Glitch", Default = false, Callback = function(enabled)
		state.on, state.speed, state.airborne = not not enabled, 0, false
		publish(0)
	end, Settings = function()
		context.section:Slider({Text = "Glitch Speed", Min = 30, Max = 300, Default = 120, Decimals = 0,
			Callback = function(value) state.max = tonumber(value) or 120 end})
		context.section:Slider({Text = "Glitch Acceleration", Min = 10, Max = 300, Default = 60, Decimals = 0,
			Callback = function(value) state.acceleration = tonumber(value) or 60 end})
	end})
	globals.__pulseSetSGlitch = function(enabled)
		enabled = not not enabled
		if toggle and toggle.Set then
			pcall(toggle.Set, enabled)
			return
		end
		state.on = enabled
	end
	globals.__pulseGetSGlitch = function() return not not state.on end

	return function()
		state.on = false
		publish(0)
	end
end

function Movement.installWallHop(context)
	local environment = context.environment
	local vector, cframe, enum = environment.Vector3, environment.CFrame, environment.Enum
	local workspace, globals = context.workspace, environment.getgenv()
	local state = {on = false, mode = "Press", angle = 90, camFlick = false, last = 0, press = 0,
		edgeAt = -1, edgeAir = false, used = -2, held = false, filterT = 0}
	globals.__pulseWallHop = state
	local rays = environment.RaycastParams.new()
	rays.FilterType = enum.RaycastFilterType.Exclude
	pcall(function() rays.RespectCanCollide = true end)
	local overlaps = environment.OverlapParams.new()
	overlaps.FilterType = enum.RaycastFilterType.Exclude
	pcall(function() overlaps.RespectCanCollide = true end)
	pcall(function() overlaps.MaxParts = 1 end)
	local directions = {}
	for index = 0, 7 do
		local angle = index * math.pi / 4
		table.insert(directions, vector.new(math.sin(angle), 0, -math.cos(angle)))
	end
	local function filter(character)
		local now = environment.os.clock()
		if state.fchar == character and now - state.filterT < 1 then return end
		state.fchar, state.filterT = character, now
		local ignored = {character}
		for _, player in ipairs(context.players:GetPlayers()) do
			if player.Character and player.Character ~= character then table.insert(ignored, player.Character) end
		end
		local platform = workspace:FindFirstChild("PulseAirWalk")
		if platform then table.insert(ignored, platform) end
		rays.FilterDescendantsInstances, overlaps.FilterDescendantsInstances = ignored, ignored
	end
	local function jumpPower(humanoid)
		if humanoid.UseJumpPower then return math.max(humanoid.JumpPower, 20) end
		return math.sqrt(2 * workspace.Gravity * math.max(humanoid.JumpHeight, 2))
	end
	local function findWall(character, humanoid, root)
		filter(character)
		local look = root.CFrame.LookVector
		local horizontal = vector.new(look.X, 0, look.Z)
		local rotation = cframe.Angles(0, horizontal.Magnitude > 0.01 and math.atan2(-horizontal.X, -horizontal.Z) or 0, 0)
		local distance = root.Size.X / 2 + 0.85
		local height = humanoid.RigType == enum.HumanoidRigType.R6 and root.Size.Y / 2 + 2 or humanoid.HipHeight + root.Size.Y / 2
		local position, now = root.Position, environment.os.clock()
		if not state.terrAt or now - state.terrAt > 15 then
			state.terrAt = now
			local ok, cells = pcall(function() return workspace.Terrain:CountCells() end)
			state.noTerrain = ok and cells == 0
		end
		if state.noTerrain then
			local vertical = math.max(height * 0.6, root.Size.Y * 0.5)
			local ok, nearby = pcall(function()
				return workspace:GetPartBoundsInRadius(position, math.sqrt(distance * distance + vertical * vertical) + 0.25, overlaps)
			end)
			if ok and type(nearby) == "table" and #nearby == 0 then return nil end
		end
		local nearest, nearestDirection, nearestDistance
		local function cast(direction)
			for _, y in ipairs({-height * 0.6, 0, root.Size.Y * 0.5}) do
				local hit = workspace:Raycast(position + vector.new(0, y, 0), direction * distance, rays)
				if hit and math.abs(hit.Normal.Y) < 0.4 and (not nearestDistance or hit.Distance < nearestDistance) then
					nearest, nearestDirection, nearestDistance = hit, direction, hit.Distance
				end
			end
		end
		local movement = humanoid.MoveDirection
		movement = vector.new(movement.X, 0, movement.Z)
		if movement.Magnitude > 0.1 then cast(movement.Unit) end
		for _, direction in ipairs(directions) do cast(rotation:VectorToWorldSpace(direction)) end
		return nearest, nearestDirection
	end
	local function rotateCamera(angle, position)
		local camera = workspace.CurrentCamera
		if camera then
			pcall(function() camera.CFrame = cframe.new(position) * cframe.Angles(0, angle, 0) * cframe.new(-position) * camera.CFrame end)
		end
	end
	context.connect(context.inputService.JumpRequest, function() state.press = environment.os.clock() end)
	context.connect(context.runService.Stepped, function()
		local character = context.player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		local root = character and character:FindFirstChild("HumanoidRootPart")
		if not humanoid or not root or humanoid.Health <= 0 then return end
		local now, back = environment.os.clock(), state.back
		if back and now >= back.t then
			state.back = nil
			if back.hrp == root and root.Parent then pcall(function() root.CFrame = cframe.new(root.Position) * back.rot end) end
			if back.cam then rotateCamera(-back.cam, root.Position) end
			if back.ar then pcall(function() if back.ar.AutoRotate == false then back.ar.AutoRotate = true end end) end
		end
		local airborne = humanoid.FloorMaterial == enum.Material.Air
		local pressed = humanoid.Jump or context.inputService:IsKeyDown(enum.KeyCode.Space) or now - state.press < 0.12
		if pressed and not state.held then state.edgeAt, state.edgeAir = now, airborne end
		state.held = pressed
		if not state.on or state.back or globals.__pulseFlyActive or not airborne or now - state.last < 0.25 then return end
		local current = humanoid:GetState()
		if current ~= enum.HumanoidStateType.Freefall and current ~= enum.HumanoidStateType.Jumping then return end
		local vertical, power = root.AssemblyLinearVelocity.Y, jumpPower(humanoid)
		if state.mode == "Hold" and (not pressed or vertical > power * 0.1) then return end
		if state.mode ~= "Hold" and (not state.edgeAir or state.used == state.edgeAt or now - state.edgeAt > 0.15 or vertical > power * 0.7) then return end
		local hit, direction = findWall(character, humanoid, root)
		if not hit then return end
		state.last, state.used = now, state.edgeAt
		local normal = vector.new(hit.Normal.X, 0, hit.Normal.Z)
		normal = normal.Magnitude > 0.01 and normal.Unit or -direction
		local look = root.CFrame.LookVector
		look = vector.new(look.X, 0, look.Z)
		local sign = look.Magnitude > 0.01 and look:Cross(-normal).Y < 0 and -1 or 1
		local angle, rotation = math.rad(state.angle) * sign, root.CFrame - root.CFrame.Position
		local autoRotate = humanoid.AutoRotate and humanoid or nil
		if autoRotate then pcall(function() humanoid.AutoRotate = false end) end
		pcall(function() root.CFrame = cframe.new(root.Position) * cframe.Angles(0, angle, 0) * rotation end)
		state.back = {t = now + 0.05, rot = rotation, hrp = root, cam = state.camFlick and angle or nil, ar = autoRotate}
		if state.camFlick then rotateCamera(angle, root.Position) end
		pcall(function() humanoid:ChangeState(enum.HumanoidStateType.Jumping) end)
		state.hops = (state.hops or 0) + 1
	end)
	context.section:Header("Wall Hop")
	context.section:ToggleKey({Text = "Wall Hop", Default = false, Callback = function(enabled) state.on = not not enabled end,
		Settings = function()
			context.section:Dropdown({Text = "Hop Mode", Options = {{code = "Press", label = "Press"}, {code = "Hold", label = "Hold"}},
				Default = "Press", Callback = function(value) if value == "Press" or value == "Hold" then state.mode = value end end})
			context.section:Slider({Text = "Flick Angle", Min = 30, Max = 180, Default = 90, Decimals = 0, Suffix = "В°",
				Callback = function(value) state.angle = tonumber(value) or 90 end})
			context.section:Toggle({Text = "Camera Flick", Default = false, Callback = function(value) state.camFlick = not not value end})
		end})
	return function()
		state.on = false
		if state.back and state.back.ar then pcall(function() state.back.ar.AutoRotate = true end) end
	end
end

function Movement.installGravity(context)
	local environment = context.environment
	local section = context.section
	local state = {on = false, value = 196, applied = false, hipOn = false, hip = 4}

	local function applyGravity()
		if state.on then
			if math.abs(context.workspace.Gravity - state.value) > 0.01 then
				context.workspace.Gravity = state.value
			end
			state.applied = true
			return
		end

		if state.applied then
			state.applied = false
			local base = rawget(environment.getgenv(), "__pulseWorldBase")
			context.workspace.Gravity = type(base) == "table" and tonumber(base.Gravity) or 196.2
		end
	end
	local function applyHip()
		local character = context.player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")

		if state.hipOn and humanoid then
			if state.humanoid ~= humanoid then
				state.humanoid, state.hipBase = humanoid, humanoid.HipHeight
			end
			if math.abs(humanoid.HipHeight - state.hip) > 0.01 then humanoid.HipHeight = state.hip end
			return
		end
		if not state.hipOn and state.humanoid then
			if state.humanoid == humanoid and state.hipBase then
				pcall(function() humanoid.HipHeight = state.hipBase end)
			end
			state.humanoid, state.hipBase = nil, nil
		end
	end

	section:ToggleKey({Text = "Gravity", Default = false, Callback = function(enabled)
		state.on = not not enabled
		applyGravity()
	end, Settings = function()
		section:Slider({Text = "Gravity Value", Min = 5, Max = 400, Default = 196, Decimals = 0,
			Callback = function(value)
				state.value = tonumber(value) or 196
				applyGravity()
			end})
	end})
	section:Toggle({Text = "Hip Height", Default = false, Callback = function(enabled)
		state.hipOn = not not enabled
		applyHip()
	end, Settings = function()
		section:Slider({Text = "Hip Height Value", Min = 0, Max = 30, Default = 4, Decimals = 1,
			Callback = function(value)
				state.hip = tonumber(value) or 4
				applyHip()
			end})
	end})
	if context.isAlive then
		environment.task.spawn(function()
			while context.isAlive() do
				environment.task.wait(0.5)
				pcall(applyGravity)
				pcall(applyHip)
			end
		end)
	end

	return function()
		state.on, state.hipOn = false, false
		pcall(applyGravity)
		pcall(applyHip)
	end
end

function FovSection.create(context)
	local environment = context.environment
	local parent = context.section:Container()
	local container = context.create("Frame", {
		Size = environment.UDim2.new(1, 0, 0, 84),
		BackgroundTransparency = 1,

		FillDirection = environment.Enum.FillDirection.Horizontal,

		SortOrder = environment.Enum.SortOrder.LayoutOrder,
	}, container)

	local cards = {}
	local function repaint()
		for _, card in environment.ipairs(cards) do

		end
	end

	local function redraw()
		local settings = context.settings
		local color = environment.Color3.fromRGB(settings.r, settings.g, settings.b)
		local lockColor = environment.Color3.fromRGB(settings.lr, settings.lg, settings.lb)

		for _, card in environment.ipairs(cards) do
			environment.pcall(context.renderer.render, card.item, 11, color, lockColor, false,
				settings.fill == true, settings.fa, card.name, context.disabledValue(), 0, alpha, 40)
		end
	end

	context.setRedraw(redraw)

	for index, styleName in environment.ipairs(context.styles) do
		FovSection._createCard(context, {
			index = index, name = styleName, parent = container, cards = cards, repaint = repaint,
		})
	end

	repaint()
	redraw()

	environment.getgenv().__pulseFovPaint = function()
		repaint()
		context.redraw()
	end

	local settingsContext = {
		environment = environment, settings = context.settings, section = context.section,
		configuration = context.configuration, redraw = context.redraw,
		disabledValue = context.disabledValue, alphaEnabled = context.alphaEnabled,
	}

	FovSection._createControls(settingsContext)
end

function Configuration.merge(defaults, saved)
	local result = Configuration._copy(defaults)

	if type(saved) == "table" then
		Configuration._overlay(result, saved)
	end

	return result
end

function Configuration.defaultView(settings, versions)
	return setmetatable(Configuration.defaults(versions), {__index = function(_, key) return settings[key] end})
end

function Configuration._copy(value)
	if type(value) ~= "table" then return value end

	local result = {}
	for key, child in pairs(value) do
		result[key] = Configuration._copy(child)
	end

	return result
end

function Configuration._overlay(target, source)
	for key, value in pairs(source) do
		if type(value) == "table" and type(target[key]) == "table" then
			Configuration._overlay(target[key], value)
			continue
		end

		target[key] = Configuration._copy(value)
	end
end

function Configuration.migrate(settings)
	if type(settings.invis) == "table" then
		settings.invis.on = false
	end

	local sheriff = settings.sheriff
	if type(sheriff) == "table" then
		if not sheriff.aimV3Mig then
			sheriff.aimVer = "V3"
			sheriff.aimV3Mig = true
		end

		sheriff.aimVer = "V4"
		sheriff.aimV4Mig = true
	end

	local reserved = {W = true, A = true, S = true, D = true, Up = true, Down = true,
		Left = true, Right = true, Space = true}
	local function clearBinding(parent, key)
		if type(parent) == "table" and reserved[parent[key]] then
			parent[key] = ""
		end
	end

	clearBinding(settings, "bhopKey")
	clearBinding(settings, "bombJumpKey")
	clearBinding(settings, "infJumpKey")
	clearBinding(settings.invis, "key")
	clearBinding(settings.fly, "key")
	clearBinding(settings.spin, "key")
	clearBinding(sheriff, "shootKey")
	clearBinding(settings.ui, "menuKey")
end

function Configuration.load(environment, httpService, versions)
	local settings = Configuration.defaults(versions)
	settings.ctrl = {}

	if type(environment.isfile) ~= "function" or type(environment.readfile) ~= "function" then
		return settings
	end

	pcall(function()
		local path = "PulseHubMM2_config.json"
		local saved

		for attempt = 1, 5 do
			if not environment.isfile(path) then return end

			local contents = environment.readfile(path)
			if type(contents) ~= "string" or contents == "" then return end

			local ok, decoded = pcall(httpService.JSONDecode, httpService, contents)
			if ok and type(decoded) == "table" then
				saved = decoded
				break
			end

			if attempt < 5 then environment.task.wait(0.3) end
		end

		if not saved then
			pcall(function()
				local backup = "PulseHubMM2_config.bak.json"
				if not environment.isfile(backup) then return end

				local contents = environment.readfile(backup)
				if type(contents) == "string" and contents ~= "" then
					local ok, decoded = pcall(httpService.JSONDecode, httpService, contents)
					if ok and type(decoded) == "table" then saved = decoded end
				end
			end)
		end

		if saved then settings = Configuration.merge(settings, saved) end
		Configuration.migrate(settings)
	end)

	return settings
end

function Configuration.snapshot(context)
	local environment = context.environment
	local settings = context.sanitize(context.settings) or {}

	environment.pcall(function()
		if environment.type(settings.invis) == "table" then
			settings.invis.on = false
		end
	end)

	local animations
	environment.pcall(function()
		local controller = environment.rawget(environment.getgenv(), "__pulseFA")

		if controller and environment.type(controller.config) == "table" then
			animations = context.json:JSONEncode(controller.config)
		end
	end)

	if not animations then
		environment.pcall(function()
			if environment.isfile and environment.isfile("BundleAnimations/config.json") then
				animations = environment.readfile("BundleAnimations/config.json")
			end
		end)
	end

	local visuals, skins, nova
	local components = context.components

	environment.pcall(function()
		if components.visSnap then
			visuals = components.visSnap()
		end
	end)
	environment.pcall(function()
		if components.skinsSnap then
			skins = components.skinsSnap()
			return
		end

		if environment.isfile and environment.isfile("PulseHub/mm2_skins.json") then
			skins = environment.readfile("PulseHub/mm2_skins.json")
		end
	end)
	environment.pcall(function()
		nova = components.novaSnap()
	end)

	return {
		v = 1, game = "mm2", s = settings, fa = animations,
		vis = visuals, skins = skins, nova = nova, dev = components.devKind(),
	}
end

function Configuration.applyAppearance(environment, window, appearance)
	local methods = {
		base = "SetBase", radius = "SetCornerRadius", font = "SetFont", textScale = "SetTextScale", scale = "SetScale",
		blur = "SetBlur", bgStyle = "SetBackground", bgIntensity = "SetBackgroundIntensity", cardStyle = "SetCardStyle",
		density = "SetDensity", sidebarWidth = "SetSidebarWidth", opacity = "SetOpacity", tooltips = "SetTooltips",
		sounds = "SetSounds", volume = "SetVolume", motion = "SetMotion", autoFit = "SetAutoFit",
	}
	for key, value in appearance do window.State.data[key] = Configuration._copy(value) end
	pcall(window.State.save)

	if appearance.base then pcall(window.SetBase, window, appearance.base) end
	for _, key in {"panel", "accent"} do
		if type(appearance[key]) ~= "string" then continue end
		local method = key == "panel" and window.SetPanelColor or window.SetAccent
		pcall(function() method(window, environment.Color3.fromHex(appearance[key])) end)
	end
	for key, method in methods do
		if key == "base" or appearance[key] == nil or not window[method] then continue end
		pcall(window[method], window, appearance[key])
	end
	if window.SyncAppearance then pcall(window.SyncAppearance, window) end
end

function Configuration.applyAnimations(context, encoded)
	local environment = context.environment
	local globals = type(environment.getgenv) == "function" and environment.getgenv() or {}
	local controller = globals.__pulseFA
	if not controller or type(controller.applyConfig) ~= "function" then
		if type(encoded) == "string" then pcall(environment.writefile, "BundleAnimations/config.json", encoded) end
		return
	end

	local animations = {}
	if type(encoded) == "string" then
		local ok, decoded = pcall(context.http.JSONDecode, context.http, encoded)
		if ok and type(decoded) == "table" then animations = decoded end
	end
	pcall(controller.applyConfig, animations)
end

function Configuration.apply(context, data)
	if type(data) ~= "table" or type(data.s) ~= "table" then return false, "Bad config data" end

	local settings, saved = context.settings, data.s
	for key, value in saved do
		if key == "ui" or key == "ctrl" then continue end
		if type(value) == "table" and type(settings[key]) == "table" then
			Configuration._overlay(settings[key], value)
			continue
		end
		settings[key] = Configuration._copy(value)
	end
	if type(settings.invis) == "table" then settings.invis.on = false end

	if context.applyState then context.applyState.applying = true end
	if type(saved.ctrl) == "table" then
		for key, value in saved.ctrl do
			if key == "Settings/-/Language" then continue end
			settings.ctrl[key] = Configuration._copy(value)
			local setter = context.setters[key]
			if setter then pcall(setter, value) end
		end
	end
	if context.applyState then context.applyState.applying = false end
	for _, hook in context.applyHooks or {} do pcall(hook) end

	Configuration.applyAnimations(context, data.fa)
	if type(data.vis) == "table" and context.visApply then pcall(context.visApply, data.vis) end
	if type(data.skins) == "string" then
		if context.weaponSkins and context.weaponSkins.ready then
			pcall(context.weaponSkins.applySnapshot, data.skins)
		else
			pcall(context.environment.writefile, "PulseHub/mm2_skins.json", data.skins)
		end
	end
	if type(data.nova) == "table" and context.window then
		pcall(Configuration.applyAppearance, context.environment, context.window, data.nova)
	end
	return true
end

function Configuration.prepareList(context)
	local environment, http = context.environment, context.http
	local list = Configuration.loadList(environment, http)
	local preset = context.preset
	if type(preset) ~= "table" then return list end

	local builtin
	for index = #list, 1, -1 do
		if list[index].name ~= "MYTHIC CFG" then continue end
		if builtin then table.remove(list, index) else builtin = list[index] end
	end
	if builtin then
		builtin.data, builtin.builtin = {v = 1, game = "mm2", s = preset, reset = true}, true
	else
		table.insert(list, 1, {name = "MYTHIC CFG", data = {v = 1, game = "mm2", s = preset, reset = true}, builtin = true})
	end
	Configuration.saveList(environment, http, list)
	return list
end

function Configuration.capture(context)

end

function Configuration.sanitize(environment, value, depth, seen)
	depth = depth or 0

	if depth > 12 then
		return nil
	end

	local kind = environment.type(value)

	if kind == "string" or kind == "boolean" then
		return value
	end

	if kind == "number" then
		if value ~= value or value == environment.math.huge or value == -environment.math.huge then
			return nil
		end

		return value
	end

	if kind == "table" then
		return Configuration._copyTable(environment, value, depth, seen)
	end

	if environment.typeof and environment.typeof(value) == "EnumItem" then
		return environment.tostring(value)
	end

	return nil
end

function Configuration._copyTable(environment, value, depth, seen)
	seen = seen or {}

	if seen[value] then
		return nil
	end

	seen[value] = true
	local cleaned = {}

	for key, child in environment.pairs(value) do
		if environment.type(key) == "string" or environment.type(key) == "number" then
			local result = Configuration.sanitize(environment, child, depth + 1, seen)

			if result ~= nil then
				cleaned[key] = result
			end
		end
	end

	seen[value] = nil
	return cleaned
end

function Configuration.saveWhileCurrent(context)
	local environment = context.environment
	local lastWritten

	while environment.getgenv().__pulseMM2SaveGen == context.generation do
		environment.task.wait(2)

		if environment.getgenv().__pulseMM2SaveGen ~= context.generation then
			return
		end

		local encoded, serialized = Configuration._encode(context)

		if encoded and environment.type(serialized) == "string" and serialized ~= lastWritten then
			if Configuration._write(environment, serialized) then
				lastWritten = serialized
			end
		end
	end
end

function Configuration._encode(context)
	local environment = context.environment
	local encoded, serialized = environment.pcall(function()
		return context.json:JSONEncode(context.settings)
	end)

	if encoded then
		environment.getgenv().__pulseCfgErr = nil
		return encoded, serialized
	end

	environment.getgenv().__pulseCfgErr = environment.tostring(serialized)
	encoded, serialized = environment.pcall(function()
		return context.json:JSONEncode(Configuration.sanitize(environment, context.settings))
	end)

	if not encoded then
		environment.getgenv().__pulseCfgErr = "С‡РёСЃС‚РєР° РЅРµ РїРѕРјРѕРіР»Р°: " .. environment.tostring(serialized)
	end

	return encoded, serialized
end

function Configuration._write(environment, serialized)
	if not environment.writefile then
		environment.getgenv().__pulseCfgErr = "РЅРµС‚ writefile"
		return false
	end

	local written, failure = environment.pcall(environment.writefile, "PulseHubMM2_config.json", serialized)

	if not written then
		environment.getgenv().__pulseCfgErr = "writefile: " .. environment.tostring(failure)
	end

	return written
end

local FovLifecycle = {}

function FovLifecycle.ensure(context)
	local state = context.state
	local screen = state.screen
	local renderer = state.renderer

	if screen and screen.Parent and renderer and renderer.holder.Parent then
		return
	end

	if screen then
		context.environment.pcall(function()
			screen:Destroy()
		end)
	end

	state.screen = context.create("ScreenGui", {
		Name = "PulseAimFOV", ResetOnSpawn = false, IgnoreGuiInset = true,

		context.environment.pcall(function()
			state.screen.Parent = context.player:FindFirstChildOfClass("PlayerGui")
		end)
	end

	state.renderer = context.renderers.make(state.screen)
end

local RoleData = {}

function RoleData.infer(player)
	local character = player.Character
	if not character then return end

	for _, tool in ipairs(character:GetChildren()) do
		if not tool:IsA("Tool") then
			continue
		end

		local name = string.lower(tool.Name)
		if string.find(name, "knife", 1, true) then return "Murderer" end
		if string.find(name, "gun", 1, true) then return "Sheriff" end
	end
end

function RoleData.source(context)
	local state = context.state
	if state.source then return state.source end

	local storage = context.storage
	local function fromFolder(folder)
		return RoleData._load(context.environment, folder and folder:FindFirstChild("CurrentRoundClient"))
	end

	state.source = fromFolder(storage:FindFirstChild("Modules"))
		or RoleData._load(context.environment, storage:FindFirstChild("CurrentRoundClient"))
		or fromFolder(storage:FindFirstChild("Client"))

	if state.source then return state.source end

	local modules = storage:FindFirstChild("Modules")
	local client = modules and modules:FindFirstChild("Client")
	state.source = fromFolder(client)

	return state.source
end

function RoleData.refresh(context)
	local state = context.state
	local source = RoleData.source(context)

	if source then
		local ok, roles = context.environment.pcall(function() return source.GetLatestPlayerData() end)
		if ok and type(roles) == "table" then state.roles = roles end
		return
	end

	local roles = {}
	for _, player in ipairs(context.players:GetPlayers()) do
		local role = RoleData.infer(player)
		if not role then
			continue
		end

		local character = player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		local dead = humanoid and humanoid.Health <= 0 or false
		roles[player.Name] = {Role = role, Dead = dead, Killed = false}
	end

	state.roles = roles
end

function RoleData._load(environment, module)
	if not module or not module:IsA("ModuleScript") then return end

	local ok, source = environment.pcall(environment.require, module)
	if not ok or type(source) ~= "table" or type(source.GetLatestPlayerData) ~= "function" then return end

	return source
end

local RoleCache = {}

function RoleCache.create(context)
	local environment = context.environment
	local state = context.state
	state.refreshing = false
	state.lastRefresh = 0

	return function()
		if not state.refreshing and environment.os.clock() - state.lastRefresh >= 0.25 then
			state.refreshing = true
			environment.task.spawn(function()
				environment.pcall(context.refresh)
				state.lastRefresh = environment.os.clock()
				state.refreshing = false
			end)
		end

		return state.roles
	end
end

function RoleData.create(context)
	local state = {roles = {}}
	local data = {environment = context.environment, storage = context.storage, players = context.players, state = state}
	local getRoles = RoleCache.create({
		environment = context.environment, state = state,
		refresh = function() RoleData.refresh(data) end,
	})

	return getRoles, state
end

local FovTarget = {}

function FovTarget.isInside(context, character)
	local settings = context.configuration.sheriff.fov
	if not settings or not settings.on then return true end
	if tonumber(settings.size) and tonumber(settings.size) >= 1500 then return character ~= nil end

	local part = character and (character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Head"))
	if not part then return false end

	local camera = context.workspace.CurrentCamera
	if not camera then return true end

	local ok, inside = pcall(function()
		local point, visible = camera:WorldToViewportPoint(part.Position)
		if not visible or point.Z <= 0 then return false end

		local viewport = camera.ViewportSize
		local x, y = point.X - viewport.X * 0.5, point.Y - viewport.Y * 0.5
		local radius = tonumber(settings.size) or 220
		return x * x + y * y <= radius * radius
	end)

	return ok and inside or false
end

function FovTarget.hasMurderer(context)
	local roles = context.roles()

	for _, player in ipairs(context.players:GetPlayers()) do
		if player == context.localPlayer then
			continue
		end

		local role = roles[player.Name]
		if not role or role.Role ~= "Murderer" or role.Dead == true or role.Killed == true then
			continue
		end

		local character = player.Character
		if not character then
			continue
		end

		local humanoid = character:FindFirstChildOfClass("Humanoid")
		if not humanoid or humanoid.Health <= 0 then
			continue
		end

		if character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Head") then
			return true
		end
	end

	return false
end

local FovController = {}
FovController.__index = FovController

function FovController.create(context)
	return setmetatable({
		context = context, environment = context.environment, state = context.state, settings = context.settings,
		renderers = context.renderers, isLocked = context.isLocked,
		lastFrame = context.environment.os.clock(), nextLockCheck = 0, locked = false,
	}, FovController)
end

function FovController.setEnabled(controller, enabled)
	controller.settings.on = not not enabled
	controller:refresh()
end

function FovController.refresh(controller)
	local state = controller.state

	if not controller.settings.on then
		if state.screen then state.screen.Enabled = false end
		return
	end

	controller.context.ensure()
	if state.screen then state.screen.Enabled = true end
	if controller.context.redraw then controller.context.redraw() end
end

function FovController.step(controller)
	local environment = controller.environment
	local settings = controller.settings
	local state = controller.state
	local now = environment.os.clock()
	local delta = environment.math.clamp(now - controller.lastFrame, 0, 0.1)
	controller.lastFrame = now

	if not settings.on or not state.screen or not state.renderer or not state.renderer.holder.Parent then
		return
	end

	if now >= controller.nextLockCheck then
		controller.nextLockCheck = now + 0.1
		local ok, locked = environment.pcall(controller.isLocked)
		controller.locked = ok and locked or false
	end

	local color = environment.Color3.fromRGB(settings.r, settings.g, settings.b)
	local lockColor = environment.Color3.fromRGB(settings.lr, settings.lg, settings.lb)
	local opacity = 1 - environment.math.clamp((environment.tonumber(settings.tr) or 20) / 100, 0, 1)
	controller.renderers.render(state.renderer, environment.tonumber(settings.size) or 220, color, lockColor,
		controller.locked, settings.fill == true, settings.fa, settings.style, settings.anim ~= false, delta, opacity)
end

local FovFeature = {}

function FovFeature.install(context)
	local state = {}
	local renderers = FovRenderer.create(context.environment, context.create)
	local lifecycle = {
		environment = context.environment, create = context.create, coreGui = context.coreGui,
		player = context.player, renderers = renderers, state = state,
	}
	local target = {roles = context.roles, players = context.players, localPlayer = context.player}
	local redraw
	local controllerContext = {
		environment = context.environment, settings = context.settings, state = state,
		renderers = renderers, isLocked = function() return FovTarget.hasMurderer(target) end,
		ensure = function() FovLifecycle.ensure(lifecycle) end,
		redraw = function() if redraw then redraw() end end,
	}
	local controller = FovController.create(controllerContext)
	local sectionContext = {
		environment = context.environment, section = context.section, settings = context.settings,
		configuration = context.configuration, nova = context.nova, translation = context.translation,
		create = context.create, connect = context.connect, round = context.round, renderer = renderers,
		styles = {"Glow", "Gradient", "Dashed", "Classic"}, defaultOn = context.defaultOn, controller = controller,
		disabledValue = function() return false end, alphaEnabled = function() return true end,
		setRedraw = function(callback) redraw = callback end,
		redraw = function() redraw() end,
	}
	context.connect(context.runService.RenderStepped, function() controller:step() end)
	local toggle = FovSection.install(sectionContext)

	return {controller = controller, state = state, toggle = toggle}
end

local RenderBindings = {}

	local environment = context.environment
	local runService = environment.game:GetService("RunService")

	environment.pcall(function()
		runService:UnbindFromRenderStep(name)

			end)
		end)
	end)
end

local ThemeBindings = {}

local THEME_ROLES = {
	Accent = "Accent", Text = "Text", Surface = "Surface", SurfaceAlt = "Surface2",
	Background = "Background", Outline = "Line", Divider = "Line", Border = "Line",
	Stroke = "Line", Faint = "Faint", Muted = "Muted", Elevated = "Elevated",
	Warning = "Warning", Success = "Success", Danger = "Danger",
}

function ThemeBindings.create(environment, library)
	local bindings = {}

	function bindings.key(role)
		if role == "Info" then
			return function()
				return environment.Color3.fromRGB(90, 169, 255)
			end
		end

		return THEME_ROLES[role] or "Text"
	end

	function bindings.color(role)
		return library.Theme.resolve(bindings.key(role))
	end

	function bindings.bind(object, property, role)
		if environment.typeof(object) ~= "Instance" then
			return object
		end

		library.Theme.bind(object, property, bindings.key(role))
		return object
	end

	return bindings
end

function ThemeBindings.addAccent(context, object, property)
	property = property or "BackgroundColor3"
	local registry = context.registry
	local items = registry.items
	registry._adds = (registry._adds or 0) + 1

	if registry._adds >= 64 then
		registry._adds = 0
		ThemeBindings._removeDetached(items)
	end

	items[#items + 1] = {inst = object, prop = property}

	context.environment.pcall(function()
		object[property] = context.palette.A1
	end)
	context.environment.pcall(function()
		object:SetAttribute("_skinSkip", context.skinSkip())
	end)
end

function ThemeBindings._removeDetached(items)
	local retained = 0

	for index = 1, #items do
		local item = items[index]

		if item.inst and item.inst.Parent then
			retained += 1
			items[retained] = item
		end
	end

	for index = #items, retained + 1, -1 do
		items[index] = nil
	end
end

local ThemeRefresh = {}

function ThemeRefresh.apply(environment, theme, isNeutral)
	local root = theme.root

	if not root then
		return
	end

	local transparency = theme.transp
	local light = theme.light
	local surface = theme.surf

	local function refresh(object)
		if not object:IsA("GuiObject") or object:GetAttribute("_skinSkip") then
			return
		end

		environment.pcall(function()
			ThemeRefresh._background(environment, theme, isNeutral, surface, object)

			if object.Name == "Sect" then
				ThemeRefresh._section(environment, theme, object)
			end

			ThemeRefresh._text(environment, theme, isNeutral, light, object)
			ThemeRefresh._transparency(environment, theme, transparency, object)
		end)
	end

	refresh(root)

	for _, object in environment.ipairs(root:GetDescendants()) do
		refresh(object)
	end

	for _, hook in environment.ipairs(theme.refreshHooks) do
		environment.pcall(hook)
	end
end

function ThemeRefresh._brightness(object, attribute, property, isNeutral)
	local brightness = object:GetAttribute(attribute)

	if brightness ~= nil then
		return brightness
	end

	local color = object[property]
	brightness = isNeutral(color) and color.R or -1
	object:SetAttribute(attribute, brightness)

	return brightness
end

function ThemeRefresh._background(environment, theme, isNeutral, surface, object)
	local brightness = ThemeRefresh._brightness(object, "_surfL", "BackgroundColor3", isNeutral)

	if brightness < 0 then
		return
	end

	object.BackgroundColor3 = surface(brightness)
	local gradient = object:FindFirstChild("PulseSurfGrad")
	local isFrame = object.ClassName == "Frame"

	if theme.surfGradSeq and isFrame then
		if not gradient then
			gradient = environment.Instance.new("UIGradient")
			gradient.Name = "PulseSurfGrad"
			gradient.Parent = object
		end

		gradient.Color = theme.surfGradSeq
		gradient.Rotation = theme.surfGradRot or 0
		return
	end

	if gradient then
		gradient:Destroy()
	end

	if theme.surfGradSeq and not isFrame and theme.surfGradMid then
		local tint = theme.surfGradMid
		local shade = environment.math.clamp(brightness * (theme.lum or 1) * 2.4, 0, 1)
		object.BackgroundColor3 = environment.Color3.new(
			environment.math.clamp(tint.R * shade, 0, 1),
			environment.math.clamp(tint.G * shade, 0, 1),
			environment.math.clamp(tint.B * shade, 0, 1)
		)
	end
end

function ThemeRefresh._section(environment, theme, object)
	local baseTransparency = object:GetAttribute("_bgTr0")

	if baseTransparency == nil then
		baseTransparency = object.BackgroundTransparency
		object:SetAttribute("_bgTr0", baseTransparency)
	end

	local cardTransparency = baseTransparency

	if theme.cardTr then
		cardTransparency = environment.math.clamp(baseTransparency + theme.cardTr, 0, 0.95)
	end

	object.BackgroundTransparency = cardTransparency
	local isGlass = theme.glassWant and theme.cardTr and true or false
	local stroke = object:FindFirstChildOfClass("UIStroke")

	if stroke then
		stroke.Color = isGlass and environment.Color3.fromRGB(255, 255, 255)
			or environment.Color3.fromRGB(23, 24, 27)
		stroke.Transparency = isGlass and 0.88 or 0
	end

	local corner = object:FindFirstChildOfClass("UICorner")

	if corner then
		corner.CornerRadius = environment.UDim.new(0, isGlass and 10 or 6)
	end

	ThemeRefresh._glass(environment, object, isGlass)
end

function ThemeRefresh._glass(environment, object, isEnabled)
	local gradient = object:FindFirstChild("__PulseGlass")

	if not isEnabled then
		if gradient then
			gradient:Destroy()
		end

		return
	end

	if not gradient then
		gradient = environment.Instance.new("UIGradient")
		gradient.Name = "__PulseGlass"
		gradient.Rotation = 90
		gradient.Parent = object
	end

	gradient.Transparency = environment.NumberSequence.new({
		environment.NumberSequenceKeypoint.new(0, 0),
		environment.NumberSequenceKeypoint.new(1, 0.3),
	})
end

function ThemeRefresh._text(environment, theme, isNeutral, light, object)
	local function color(brightness)
		local tint = theme.textTintC

		if tint then
			return environment.Color3.new(
				environment.math.clamp(tint.R * brightness, 0, 1),
				environment.math.clamp(tint.G * brightness, 0, 1),
				environment.math.clamp(tint.B * brightness, 0, 1)
			)
		end

		if light then
			local shade = environment.math.clamp(0.3 - brightness * 0.22, 0, 1)
			return environment.Color3.new(shade, shade, shade)
		end

		return environment.Color3.new(brightness, brightness, brightness)
	end

	if object:IsA("TextLabel") or object:IsA("TextButton") or object:IsA("TextBox") then
		local brightness = ThemeRefresh._brightness(object, "_txtL", "TextColor3", isNeutral)

		if brightness >= 0 then
			object.TextColor3 = color(brightness)
		end
	end

	if object:IsA("TextBox") then
		local brightness = ThemeRefresh._brightness(object, "_phL", "PlaceholderColor3", isNeutral)

		if brightness >= 0 then
			object.PlaceholderColor3 = color(brightness)
		end
	end
end

function ThemeRefresh._transparency(environment, theme, transparency, object)
	local selected = object:GetAttribute("_tabSel")

	if selected ~= nil then
		object.BackgroundTransparency = selected and (transparency <= 0 and 0 or transparency) or 1
		return
	end

	local original = object:GetAttribute("_btr")

	if original == nil then
		original = object.BackgroundTransparency
		object:SetAttribute("_btr", original)
	end

	if original < 1 then
		object.BackgroundTransparency = transparency <= 0 and 0 or environment.math.min(original + transparency, theme.cap)
	end
end

local Appearance = {}

Appearance.FunTools = {}

function Appearance.FunTools.backpack(state)
	local backpack = state.context.player:FindFirstChildOfClass("Backpack")
	return backpack, backpack and backpack:FindFirstChild("Toys")
end

function Appearance.FunTools.toyDatabase(state)
	if state.toyDatabase ~= nil then return state.toyDatabase or nil end
	local context = state.context
	local ok, database = pcall(function()
		local folder = context.storage:WaitForChild("Database", 5)
		return context.environment.require(folder:WaitForChild("Sync", 5))
	end)
	state.toyDatabase = ok and type(database) == "table" and type(database.Toys) == "table" and database.Toys or false
	return state.toyDatabase or nil
end

function Appearance.FunTools.toyOptions(state)
	local _, folder = Appearance.FunTools.backpack(state)
	local database = Appearance.FunTools.toyDatabase(state)
	local options, seen = {}, {}
	if not folder then return options end
	for _, toy in folder:GetChildren() do
		if not toy:IsA("Tool") or seen[toy.Name] then continue end
		seen[toy.Name] = true
		local info = database and database[toy.Name]
		local image = toy.TextureId ~= "" and toy.TextureId or nil
		if not image and type(info) == "table" and info.Image then
			image = "rbxthumb://type=Asset&w=150&h=150&id=" .. tostring(info.Image)
		end
		local label = type(info) == "table" and type(info.Name) == "string" and info.Name ~= "" and info.Name or toy.Name
		table.insert(options, {code = toy.Name, label = label, image = image})
	end
	table.sort(options, function(left, right) return left.label < right.label end)
	return options
end

function Appearance.FunTools.toySignature(options)
	local names = {}
	for _, option in options do table.insert(names, option.code) end
	return table.concat(names, "|")
end

function Appearance.FunTools.refreshToys(state)
	local options = Appearance.FunTools.toyOptions(state)
	local signature = Appearance.FunTools.toySignature(options)
	if signature == state.toySignature or #options == 0 then return end
	state.toySignature = signature
	state.toyDropdown.SetOptions(options)
	state.toyDropdown.Set(state.selectedToys)
end

function Appearance.FunTools.equipToy(state, name)
	local context = state.context
	local character = context.player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not humanoid or humanoid.Health <= 0 then return end
	local backpack, toys = Appearance.FunTools.backpack(state)
	if not backpack then return end
	local existing = backpack:FindFirstChild(name) or character:FindFirstChild(name)
	if existing and existing:IsA("Tool") then return end
	if not toys or not toys:FindFirstChild(name) then return end
	local now = os.clock()
	if now - (state.replicatedAt[name] or 0) < 3 then return end
	state.replicatedAt[name] = now
	local remotes = context.storage:FindFirstChild("Remotes")
	local extras = remotes and remotes:FindFirstChild("Extras")
	local remote = extras and extras:FindFirstChild("ReplicateToy")
	if not remote then return end
	local ok, toy = pcall(remote.InvokeServer, remote, name)
	if not ok or context.environment.typeof(toy) ~= "Instance" or not toy:IsA("Tool") then return end
	if toy.Parent == backpack or toy.Parent == character then return end
	pcall(function() toy.Parent = backpack end)
end

function Appearance.FunTools.playEmote(state, name)
	if state.emoteRemote then pcall(state.emoteRemote.Fire, state.emoteRemote, name) end
end

function Appearance.FunTools.install(context)
	local configuration = context.configuration
	if type(configuration.toyAuto) ~= "table" then configuration.toyAuto = {} end
	local toySettings = configuration.toyAuto
	if type(toySettings.toys) ~= "table" then
		toySettings.toys = type(toySettings.toy) == "string" and toySettings.toy ~= "" and {toySettings.toy} or {}
	end
	local state = {context = context, alive = true, emote = "ninja", emoteOn = false, emoteReady = false,
		selectedToys = toySettings.toys, toysOn = toySettings.on == true, replicatedAt = {}, spawnAt = os.clock(), ticks = 0}
	pcall(function() state.emoteRemote = context.storage.Remotes.Misc.PlayEmote end)
	local section = context.section
	section:Header("Emotes")
	local options = {}
	for _, name in {"ninja", "dab", "floss", "headless", "zen", "zombie", "sit"} do
		table.insert(options, {code = name, label = name})
	end
	section:Dropdown({Text = "Emote", Default = "ninja", TranslateOptions = false, Options = options, Callback = function(name)
		state.emote = name
		if state.emoteReady and not (context.applyState and context.applyState.applying) then Appearance.FunTools.playEmote(state, name) end
	end})
	state.emoteReady = true
	section:Toggle({Text = "Auto Emote", Default = false, Callback = function(enabled) state.emoteOn = enabled end})
	section:Header("Equip Toy")
	local toys = Appearance.FunTools.toyOptions(state)
	state.toySignature = Appearance.FunTools.toySignature(toys)
	state.toyDropdown = section:Dropdown({Text = "Toys", Multi = true, Default = state.selectedToys,
		TranslateOptions = false, NoSave = true, Options = toys, Callback = function(values)
			local selected = {}
			for _, value in type(values) == "table" and values or {} do
				if type(value) == "string" and value ~= "" then table.insert(selected, value) end
			end
			state.selectedToys, toySettings.toys = selected, selected
		end})
	section:Toggle({Text = "Auto Equip Toy", Default = toySettings.on == true, Callback = function(enabled)
		state.toysOn = not not enabled
		toySettings.on = state.toysOn
	end})
	context.connect(context.player.CharacterAdded, function() state.spawnAt = os.clock() end)
	context.environment.task.spawn(function()
		while state.alive and context.isAlive() do
			context.environment.task.wait(1.4)
			if state.emoteOn then Appearance.FunTools.playEmote(state, state.emote) end
		end
	end)
	context.environment.task.spawn(function()
		while state.alive and context.isAlive() do
			context.environment.task.wait(0.5)
			state.ticks += 1
			if state.ticks % 6 == 0 then pcall(Appearance.FunTools.refreshToys, state) end
			if not state.toysOn or os.clock() - state.spawnAt <= 1.2 then continue end
			for _, name in state.selectedToys do
				if not state.toysOn then break end
				pcall(Appearance.FunTools.equipToy, state, name)
			end
		end
	end)
	return state, function() state.alive = false end
end


function Appearance.installXray(context)
	local environment = context.environment
	local globals = environment.getgenv()
	local settings = context.configuration.xray
	local original = {}
	local queue, index = {}, 1

	pcall(function()
		if type(globals.__pulseXrayOrig) ~= "table" then return end
		for part, transparency in pairs(globals.__pulseXrayOrig) do
			pcall(function() if part.Parent then part.Transparency = transparency end end)
		end
	end)
	globals.__pulseXrayOrig = original
	local function eligible(part)
		if not part:IsA("BasePart") or part:IsA("Terrain") or part.Transparency >= 1 then return false end
		local model = part:FindFirstAncestorWhichIsA("Model")
		while model do
			if context.players:GetPlayerFromCharacter(model) then return false end
			model = model.Parent and model.Parent:FindFirstAncestorWhichIsA("Model") or nil
		end
		return true
	end
	local function opacity(value)
		return value + (1 - value) * settings.strength
	end
	local function refresh()
		for part, transparency in pairs(original) do
			if part.Parent then
				part.Transparency = opacity(transparency)
			else
				original[part] = nil
			end
		end
	end
	local function restore()
		for part, transparency in pairs(original) do
			if part.Parent then part.Transparency = transparency end
			original[part] = nil
		end
		queue, index = {}, 1
	end

	context.connect(context.runService.Heartbeat, function()
		if index > #queue then return end
		if not settings.on then
			queue, index = {}, 1
			return
		end

		local processed = 0
		while index <= #queue and processed < 180 do
			local part = queue[index]
			index += 1
			processed += 1
			if part and part.Parent and original[part] == nil and eligible(part) then
				original[part] = part.Transparency
				part.Transparency = opacity(part.Transparency)
			end
		end
		if index > #queue then queue, index = {}, 1 end
	end)
	context.connect(context.workspace.DescendantAdded, function(part)
		if settings.on and part:IsA("BasePart") then table.insert(queue, part) end
	end)
	context.section:Toggle({Text = "X-Ray", Default = context.defaults.xray.on, Callback = function(enabled)
		settings.on = enabled
		if enabled then queue, index = context.workspace:GetDescendants(), 1 else restore() end
	end, Settings = function()
		context.section:Slider({Text = "X-Ray Strength", Min = 10, Max = 100,
			Default = math.floor(settings.strength * 100 + 0.5), Decimals = 0, Suffix = "%", Callback = function(value)
				settings.strength = value / 100
				if settings.on then refresh() end
			end})
	end})

	return restore
end

function Appearance.installFakeItems(context)
	local environment = context.environment
	local globals = environment.getgenv()
	local legEnabled, headEnabled = false, false
	local legSaved, headSaved, extras, legacyMesh = {}, {}, {}, nil
	local meshId, textureId = "rbxassetid://9598310133", "rbxassetid://902843398"

	local function restoreLeg()
		for _, object in ipairs(extras) do pcall(function() object:Destroy() end) end
		extras = {}
		for part, original in pairs(legSaved) do
			pcall(function()
				if original.MeshId ~= nil then part.MeshId = original.MeshId end
				if original.TextureID ~= nil then part.TextureID = original.TextureID end
				if original.Transparency ~= nil then part.Transparency = original.Transparency end
			end)
		end
		legSaved = {}
		if legacyMesh then
			pcall(function() legacyMesh:Destroy() end)
			legacyMesh = nil
		end
	end
	local function hideLeg(part)
		if not part then return end
		pcall(function()
			legSaved[part] = legSaved[part] or {}
			if legSaved[part].Transparency == nil then legSaved[part].Transparency = part.Transparency end
			part.Transparency = 1
		end)
	end
	local function replaceLeg(part)
		if not part then return end
		local replaced = false
		pcall(function()
			legSaved[part] = {MeshId = part.MeshId, TextureID = part.TextureID, Transparency = part.Transparency}
			part.MeshId, part.TextureID = meshId, textureId
			replaced = part.MeshId == meshId
		end)
		if replaced then return end

		pcall(function()
			local object = environment.Instance.new("Part")
			object.Name = "PulseKorblox"
			object.Size = environment.Vector3.new(1, 1, 1)
			object.CanCollide, object.CanQuery, object.CanTouch = false, false, false
			object.Massless, object.Anchored, object.Transparency = true, false, 0
			local mesh = environment.Instance.new("SpecialMesh")
			mesh.MeshType = environment.Enum.MeshType.FileMesh
			mesh.MeshId, mesh.TextureId = meshId, textureId
			mesh.Scale, mesh.Parent = environment.Vector3.new(1, 1, 1), object
			object.Parent, object.CFrame = part, part.CFrame
			local weld = environment.Instance.new("Weld")
			weld.Part0, weld.Part1, weld.C0, weld.Parent = part, object, environment.CFrame.new(), object
			table.insert(extras, object)
			part.Transparency = 1
		end)
	end
	local function applyLeg()
		local character = context.player.Character
		if not legEnabled or not character then return end
		restoreLeg()
		local lower = character:FindFirstChild("RightLowerLeg")
		if lower then
			replaceLeg(character:FindFirstChild("RightUpperLeg"))
			hideLeg(lower)
			hideLeg(character:FindFirstChild("RightFoot"))
			return
		end

		local leg = character:FindFirstChild("Right Leg")
		if leg then
			legacyMesh = environment.Instance.new("SpecialMesh")
			legacyMesh.MeshType = environment.Enum.MeshType.FileMesh
			legacyMesh.MeshId, legacyMesh.TextureId, legacyMesh.Parent = meshId, textureId, leg
		end
	end
	local function restoreHead()
		for object, transparency in pairs(headSaved) do
			pcall(function() object.Transparency = transparency end)
		end
		headSaved = {}
	end
	local function applyHead()
		local character = context.player.Character
		local head = character and character:FindFirstChild("Head")
		if not headEnabled or not head then return end
		restoreHead()
		pcall(function()
			headSaved[head] = head.Transparency
			head.Transparency = 1
		end)
		for _, object in ipairs(character:GetDescendants()) do
			if object:IsA("Decal") and object.Name == "face" then
				pcall(function()
					headSaved[object] = object.Transparency
					object.Transparency = 1
				end)
			end
		end
	end
	local function refreshCosmetics()
		local callback = rawget(globals, "__pulseSkinCos")
		if type(callback) == "function" then pcall(callback) end
	end
	context.section:Toggle({Text = "Fake Korblox", Default = false, Callback = function(enabled)
		legEnabled = not not enabled
		if legEnabled then applyLeg() else restoreLeg() end
		refreshCosmetics()
	end})
	context.section:Toggle({Text = "Fake Headless", Default = false, Callback = function(enabled)
		headEnabled = not not enabled
		if headEnabled then applyHead() else restoreHead() end
		refreshCosmetics()
	end})
	globals.__pulseCosMine = function() return legEnabled, headEnabled end
	context.connect(context.player.CharacterAdded, function()
		environment.task.wait(0.6)
		legSaved, legacyMesh, headSaved, extras = {}, nil, {}, {}
		applyLeg()
		applyHead()
	end)

	return function()
		restoreLeg()
		restoreHead()
	end
end

function Appearance.setHeadless(environment, character, state, isEnabled)
	if isEnabled then
