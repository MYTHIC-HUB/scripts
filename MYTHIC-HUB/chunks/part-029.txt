	if state.dWide ~= false and state.modern ~= false and not untrusted then
		add("mdL", delay + 0.12, errorFor("mdL"), state.dWideW1 or 0.3, true)
		add("mdL", delay + 0.25, errorFor("mdL"), state.dWideW2 or 0.15, true)
	end
	if state.dEarly ~= false and state.modern ~= false and delay > 0.11 and not untrusted then
		add("mdE", delay - 0.1, errorFor("mdE"), state.dEarlyW or 0.15, true)
	end
	if state.rehopW then
		local jumps, jumping = 0, environment.Enum.HumanoidStateType.Jumping
		for index = #samples, 2, -1 do
			if time - samples[index].t > (state.rehopT or 2.5) then break end
			if samples[index].st == jumping and samples[index - 1].st ~= jumping then jumps += 1 end
		end
		state.lastJumps = jumps
		if jumps >= (state.rehopN or 2) then
			state.rehopTmp = true
			pcall(add, "mdH", delay, errorFor("md"), state.rehopW, true)
			state.rehopTmp = nil
		end
	end
	local airDuration = 0
	for index = #samples, 1, -1 do
		if not samples[index].air then break end
		airDuration = time - samples[index].t
	end
	local scale = state.hpScale2 or 0.8
	local function addModel(kind, sampleDelay, weight, error, better)
		local trajectory = AimMotion.simulate(context, state, samples, {time = time, delay = sampleDelay, kind = kind,
			duration = duration, factor = 1, parameters = parameters, jumps = false})
		table.insert(trajectories, {kind = kind, w = weight * math.exp(-(error * error) / scale), traj = trajectory, better = better})
	end
	addModel("fly", 0, airDuration > 1.2 and 1.4 or 0.35, errorFor("fly", 1.4))
	local velocity = AimGeometry.kinVelocity(environment, samples, time, 0.08) or vector.zero
	local fast = vector.new(velocity.X, 0, velocity.Z).Magnitude > 22
	addModel("mdv", delay, fast and 1 or 0.3, errorFor("mdv", 1.4))
	addModel("arc", 0, fast and 1 or 0.3, errorFor("arc", 1.4))
	if state.velOn ~= false then
		local sample = AimGeometry.at(samples, time) or samples[#samples]
		local current = environment.typeof(sample.v) == "Vector3" and sample.v or velocity
		local speed = vector.new(current.X, 0, current.Z).Magnitude
		if state.fastWin then speed = math.max(speed, AimMotion.recentSpeed(environment, samples, time, state.fastWin)) end
		if speed > (state.velMin or 30) then
			addModel("vel", 0, state.hpVel or 1.2, errorFor("vel", 1.4))
			if state.velF ~= false then
				addModel("velf", 0, state.hpVelF or 1.2, errorFor("velf", 1.4), errorFor("velf", 9) < errorFor("vel", 9))
			end
		end
	end
	local position = AimGeometry.posAt(samples, time) or samples[#samples].p
	if fast or state.stopAll == true and vector.new(velocity.X, 0, velocity.Z).Magnitude > 6 and not samples[#samples].air then
		local stopWeight, halfWeight
		if fast then
			stopWeight = 0.8 * math.exp(-errorFor("stop", 2) ^ 2 / scale)
			halfWeight = 0.8 * math.exp(-errorFor("half", 1.5) ^ 2 / scale)
		end
		if not fast then
			stopWeight = 1.2 * math.exp(-(errorFor("md") ^ 2) / 0.8) * (state.stopW or 0.1)
			halfWeight = stopWeight
		end
		table.insert(trajectories, {kind = "stop", w = stopWeight, traj = {{tau = 0, p = position}, {tau = duration, p = position}}})
		local half = {}
		for index = 0, math.ceil(duration * 60) do
			local elapsed = index / 60
			table.insert(half, {tau = elapsed, p = position + velocity * (elapsed * 0.5)})
		end
		table.insert(trajectories, {kind = "half", w = halfWeight, traj = half})
	end
	local last, types = samples[#samples], environment.Enum.HumanoidStateType
	local jump = AimMotion.jumpTime(environment, state, samples, time)
	if state.ffgJump == true and last.st == types.Freefall and not last.air and not jump then
		local index = #samples
		while index > 1 and samples[index - 1].st == types.Freefall and not samples[index - 1].air do index -= 1 end
		if time - samples[index].t < (state.ffgMax or 0.3) then
			for _, variant in {{0, 0.5}, {0.05, 0.45}, {0.1, 0.35}, {0.2, 0.2}} do
				state.jForceTmp = variant[1]
				add("mdJa", delay, errorFor("md"), variant[2] * (state.ffgW or 1), true)
			end
			state.jForceTmp = nil
		end
	end
	if state.jumpAny == true and not jump and not last.air and last.st ~= types.Climbing then
		local weight = state.jumpAnyW or 0.1
		state.jForceTmp = 0.02
		add("mdJa", delay, errorFor("md"), weight, true)
		state.jForceTmp = 0.1
		add("mdJa", delay, errorFor("md"), weight * 0.7, true)
		state.jForceTmp = nil
	end
	if jump then
		add("mdNJ", delay, errorFor("md"), state.njW or 0.4, false)
		if state.jVar ~= false and not last.air then
			local variants = {{-0.1, "mdJe", 0.6}, {0.1, "mdJl", 0.45}}
			if state.jVarW ~= false then
				local multiplier = state.jvMul or 1
				variants = {{-0.18, "mdJe", 0.35 * multiplier * (state.jvEarly or 1)},
					{-0.09, "mdJe", 0.55 * multiplier * (state.jvEarly or 1)},
					{0.09, "mdJl", 0.45 * multiplier}, {0.18, "mdJl", 0.25 * multiplier}}
			end
			for _, variant in variants do
				state.jShiftTmp = variant[1]
				add(variant[2], delay, errorFor("md"), variant[3], true)
			end
			state.jShiftTmp = nil
		end
	end
	if state.airV2 and AimMotion.mode(environment, samples) ~= "nr" then
		local base = trajectories[1].traj
		for _, hypothesis in trajectories do
			if hypothesis.kind ~= "mdv" and hypothesis.kind ~= "arc" and hypothesis.kind ~= "stop" and hypothesis.kind ~= "half" then continue end
			local combined = {}
			for index = 0, math.ceil(duration * 60) do
				local elapsed = index / 60
				local point = AimMotion.trajectoryAt(hypothesis.traj, elapsed)
				table.insert(combined, {tau = elapsed, p = vector.new(point.X, AimMotion.trajectoryAt(base, elapsed).Y, point.Z)})
			end
			hypothesis.traj = combined
		end
	end
	local clusters = AimMotion.clusters(environment, state, samples, time)
	for index, cluster in clusters or {} do
		table.insert(trajectories, {kind = "cl" .. index, w = 1.5 * cluster.w,
			traj = {{tau = 0, p = cluster.p}, {tau = duration, p = cluster.p}}})
	end
	local oscillation = AimMotion.oscillation(environment, state, samples, time)
	if oscillation then
		state.oscMode = environment.os.clock()
		for index, point in oscillation do
			table.insert(trajectories, {kind = "osc" .. index, w = 1.6 / #oscillation,
				traj = {{tau = 0, p = point}, {tau = duration, p = point}}})
		end
	end
	local repeated = state.repInfo
	if state.repOn ~= false and repeated and math.abs(time - repeated.tEnd) < 0.35 and repeated.fit < (state.repFitMax or 1.2) then
		local trajectory = AimMotion.repeatTrajectory(samples, time, repeated.T, duration)
		if trajectory then
			local error = errorFor("rep", 3)
			table.insert(trajectories, {kind = "rep", w = (state.repW or 1.2) * math.exp(-(error * error) / scale), traj = trajectory})
		end
	end
	local total, kept, keptWeight = 0, {}, 0
	for _, hypothesis in trajectories do total += hypothesis.w end
	for _, hypothesis in trajectories do
		hypothesis.w /= math.max(total, 1e-6)
		if hypothesis.w < 0.03 then continue end
		table.insert(kept, hypothesis)
		keptWeight += hypothesis.w
	end
	for _, hypothesis in kept do hypothesis.w /= math.max(keptWeight, 1e-6) end
	return kept
end

function AimMotion.backtest(context, state, history, parameters)
	local environment = context.environment
	local samples = AimMotion.despike(environment, state, history)
	if #samples < 20 then return end
	local time, mode = samples[#samples].t, AimMotion.mode(environment, samples)
	local latency = math.clamp((mode == "nr" and state.Lnr or state.L) + state.shiftL(), 0.08, 0.45)
	local start = time - latency
	if start - 0.2 < samples[1].t then return end
	local actual = AimGeometry.posAt(samples, time)
	local previous = AimGeometry.posAt(samples, start)
	local velocity = AimGeometry.kinVelocity(environment, samples, start, 0.08) or environment.Vector3.zero
	local airborne = false
	for index = #samples, 1, -1 do
		if samples[index].t < start - 0.1 then break end
		if samples[index].air then airborne = true; break end
	end
	local factor = AimMotion.currentFactor(state, samples, start, state.k or 1)
	local kinds = {"md", "cv", "mdE", "mdL", "fly", "mdv", "arc", "stop", "half", "mdS"}
	if state.velOn ~= false then table.insert(kinds, "vel"); table.insert(kinds, "velf") end
	local errors, movementHeight = {}, nil
	local function yieldIfNeeded()
		if state.btSlice and coroutine.isyieldable() and environment.os.clock() - (state.btT or 0) > (state.btBudget or 0.0006) then coroutine.yield() end
	end
	for _, kind in kinds do
		local delay = (kind == "md" or kind == "mdv" or kind == "mdS") and state.D
			or kind == "mdE" and state.D - 0.05 or kind == "mdL" and state.D + 0.05 or 0
		local prediction
		if kind == "stop" then prediction = previous end
		if kind == "half" then prediction = previous + velocity * (latency * 0.5) end
		if not prediction then
			local model = (kind == "cv" or kind == "fly" or kind == "mdv" or kind == "arc" or kind == "vel" or kind == "velf") and kind
				or kind == "mdS" and "mds" or "md"
			local trajectory = AimMotion.simulate(context, state, samples, {time = start, delay = delay, kind = model,
				duration = latency, factor = factor, parameters = parameters, jumps = true})
			prediction = AimMotion.trajectoryAt(trajectory, latency)
		end
		if kind == "md" then movementHeight = prediction.Y end
		local error = environment.Vector3.new(prediction.X - actual.X, 0, prediction.Z - actual.Z).Magnitude
		errors[kind] = error
		state.err[kind] = (state.err[kind] or 0.8) + (math.min(error, 6) - (state.err[kind] or 0.8)) * 0.08
		if state.airV2 then
			local height = prediction.Y
			if movementHeight and mode ~= "nr" and (kind == "mdv" or kind == "arc" or kind == "stop" or kind == "half") then height = movementHeight end
			local vertical = airborne and (height - actual.Y) * 0.45 or 0
			local combined = math.sqrt(error * error + vertical * vertical)
			local key, old = "W" .. kind, state.err["W" .. kind] or state.err[kind]
			state.err[key] = old + (math.min(combined, 6) - old) * 0.08
		end
		yieldIfNeeded()
	end
	if state.repOn ~= false then
		local repeated = state.repInfo
		if repeated and repeated.tEnd <= start + 0.001 and time - repeated.tEnd < 0.5 and repeated.fit < (state.repFitMax or 1.2) then
			local trajectory = AimMotion.repeatTrajectory(samples, start, repeated.T, latency)
			if trajectory then
				local point = AimMotion.trajectoryAt(trajectory, latency)
				local error = environment.Vector3.new(point.X - actual.X, 0, point.Z - actual.Z).Magnitude
				local vertical, old = airborne and (point.Y - actual.Y) * 0.45 or 0, state.err.rep or 3
				state.err.rep = old + (math.min(error, 6) - old) * 0.08
				local weighted = state.err.Wrep or old
				state.err.Wrep = weighted + (math.min(math.sqrt(error * error + vertical * vertical), 6) - weighted) * 0.08
			end
		end
		if not repeated or repeated.tEnd > start + 0.001 or time - repeated.tEnd >= 0.5 or repeated.fit >= (state.repFitMax or 1.2) then
			local old = state.err.rep or 3
			state.err.rep = old + (4 - old) * 0.08
			state.err.Wrep = state.err.rep
		end
		yieldIfNeeded()
		local period, error, amplitude = AimMotion.findRepeat(environment, state, samples, time)
		state.repInfo = period and {T = period, fit = error, amp = amplitude, tEnd = time} or nil
	end
	if (actual - AimGeometry.posAt(samples, start)).Magnitude > latency * 4 then
		if errors.mdE + 0.05 < errors.md and errors.mdE <= errors.mdL then state.D = math.clamp(state.D - 0.002, 0.06, 0.32) end
		if not (errors.mdE + 0.05 < errors.md and errors.mdE <= errors.mdL) and errors.mdL + 0.05 < errors.md and errors.mdL < errors.mdE then
			state.D = math.clamp(state.D + 0.002, 0.06, 0.32)
		end
	end
	if state.dPrior ~= false and state.modern ~= false then state.D = math.clamp(state.D + ((state.dCenter or 0.158) - state.D) * 0.01, 0.1, 0.24) end
end

local AimPlan = {}
local AIM_AIR_MIX = {{0.8, 0.15}, {-0.8, 0.15}, {2.4, 0.1}, {-2.4, 0.1}, {4.5, 0.05}, {-4.5, 0.05}, {7, 0.025}, {-7, 0.025}}
local AIM_TAKEOFF_MIX = {{-1, 0.08}, {1.5, 0.17}, {3, 0.22}, {4.5, 0.15}, {6.5, 0.13}}

function AimPlan.points(context, state, options)
	local environment, samples = context.environment, options.samples
	local vector, types = environment.Vector3, environment.Enum.HumanoidStateType
	local time, mode, hypotheses, taus = options.time, options.mode, options.hypotheses, options.taus
	local latest = samples[#samples]
	local shift = state.useLagL and state.shift or 0
	local latency = math.clamp((mode == "nr" and state.Lnr or state.L) + shift, 0.08, 0.45)
	local jumpWeight, jumpDelay = 0, 0
	if mode == "std" and not latest.air and latest.st ~= types.Climbing and not AimMotion.jumpTime(environment, state, samples, time) then
		local jumps = 0
		for index = #samples, 2, -1 do
			if time - samples[index].t > 3 then break end
			if samples[index].air and not samples[index - 1].air then jumps += 1 end
		end
		jumpWeight = latest.jump and 0.6 or (jumps >= 2 and 0.35 or 0.1)
		if latest.st == types.Freefall then
			for index = #samples, 2, -1 do
				local current, previous = samples[index], samples[index - 1]
				if time - current.t > 0.3 then break end
				if current.st == types.Freefall and previous.st ~= types.Freefall and not previous.air then
					jumpWeight, jumpDelay = 0.75, math.max(0, current.t + state.J + state.shiftJ() - time)
					break
				end
			end
		end
	end
	local function heightSpan(window)
		local minimum, maximum = math.huge, -math.huge
		for index = #samples, 1, -1 do
			local sample = samples[index]
			if time - sample.t > window then break end
			minimum, maximum = math.min(minimum, sample.p.Y), math.max(maximum, sample.p.Y)
		end
		return maximum - minimum
	end
	local stuck = false
	if state.stuckFix ~= false and latest.st == types.Freefall and not latest.air then
		local index = #samples
		while index > 1 and samples[index].st == types.Freefall and not samples[index].air do index -= 1 end
		stuck = time - samples[index].t >= 0.2 and heightSpan(0.5) < (state.stuckSpan or 1.2)
	end
	local falseAir = false
	if state.airFix ~= false and latest.air and (latest.st == types.Running or latest.st == types.RunningNoPhysics or latest.st == types.Landed) then
		local velocity = AimGeometry.kinVelocity(environment, samples, time, 0.15)
		falseAir = velocity and math.abs(velocity.Y) < (state.airFixVy or 2.5) or false
	end
	local kineticAir = false
	if state.kinAir ~= false and mode == "std" and not stuck and not latest.air and latest.st ~= types.Climbing then
		local velocity = AimGeometry.kinVelocity(environment, samples, time, 0.1)
		kineticAir = velocity and math.abs(velocity.Y) > (state.kinAirVy or 12) and heightSpan(0.4) > (state.kinAirSpan or 2) or false
	end
	state.lastKinAir = kineticAir
	local airborne = mode == "std" and not stuck and not falseAir
		and (latest.air or kineticAir or latest.st == types.Freefall or latest.st == types.Jumping or latest.st == types.Climbing)
	local points, indexed = {}, {}
	local function add(position, weight)
		local key = math.floor(position.X * 4 + 0.5) .. ":" .. math.floor(position.Y * 4 + 0.5) .. ":" .. math.floor(position.Z * 4 + 0.5)
		local point = indexed[key]
		if point then point.w += weight; return end
		point = {p = position, w = weight}
		indexed[key] = point
		table.insert(points, point)
	end
	for _, hypothesis in hypotheses do
		local error = state.err[hypothesis.kind] or state.err.md
		for index, tau in ipairs(taus) do
			local position = tau.tau >= 0 and AimMotion.trajectoryAt(hypothesis.traj, tau.tau)
				or (AimGeometry.posAt(samples, time + tau.tau) or AimMotion.trajectoryAt(hypothesis.traj, 0))
			local spread = tau.tau > 0 and math.min(error * (tau.tau / latency), 6) * 0.8 * (state.spreadK or 1) or 0
			local weight = hypothesis.w * tau.w
			local jump = jumpWeight > 0 and tau.tau > 0.005 and jumpWeight or 0
			local remaining = weight * (1 - jump)
			if jump > 0 then
				local elapsed = math.max(0, tau.tau - jumpDelay)
				local height = math.max(0, (latest.jp or 50) * elapsed - 0.5 * (state.G or context.workspace.Gravity) * elapsed * elapsed)
				add(position + vector.new(0, height, 0), weight * jump)
			end
			if airborne and tau.tau > 0.005 then
				local scale = math.clamp(tau.tau / 0.15, 0.3, 1.5)
				local actualAir = latest.air or latest.st == types.Climbing
				local mix = actualAir and (state.airMix or AIM_AIR_MIX) or (state.takeoffMix or AIM_TAKEOFF_MIX)
				local distance = actualAir and (state.airMixK or 1) or (state.takeMixK or 1)
				local multiplier = actualAir and (state.airMixW or 1) or (state.takeMixW or 1)
				local left = 1
				for _, entry in mix do
					add(position + vector.new(0, entry[1] * scale * distance, 0), remaining * entry[2] * multiplier)
					left -= entry[2] * multiplier
				end
				remaining *= left
			end
			if spread > 0.6 and hypothesis.w > 0.12 and math.abs(index - (#taus + 1) / 2) <= 1 then
				local previous = tau.tau >= 0.01 and AimMotion.trajectoryAt(hypothesis.traj, tau.tau - 0.01) or position
				local direction = vector.new(position.X - previous.X, 0, position.Z - previous.Z)
				direction = direction.Magnitude > 0.001 and direction.Unit or vector.new(1, 0, 0)
				local side = vector.new(-direction.Z, 0, direction.X)
				add(position, remaining * 0.4)
				add(position + side * spread, remaining * 0.15)
				add(position - side * spread, remaining * 0.15)
				add(position + direction * spread, remaining * 0.15)
				add(position - direction * spread, remaining * 0.15)
				continue
			end
			add(position, remaining)
		end
	end
	local raw = state.histRaw or samples
	local jitter = state.jitOn ~= false and AimMotion.jitterAxis(context, state, raw, raw[#raw].t, latency + (options.planDelay or 0)) or nil
	state.lastJit = jitter
	if jitter then
		local share = state.jitShare or 0.85
		for _, point in points do point.w *= 1 - share end
		local along = state.jitAlong or {-2.9, -1.5, -0.45, 0.45, 1.5, 2.9}
		local across = state.jitPerp or {{0, 0.46}, {4, 0.14}, {-4, 0.14}, {11, 0.13}, {-11, 0.13}}
		local heights = state.jitDy or {-1.25, 0.57, 4.2}
		local weight = share / (#along * #heights)
		for _, distance in along do
			local center = jitter.c + jitter.u * (distance * jitter.half)
			for _, offset in across do
				local position = center + jitter.nv * offset[1]
				for _, height in heights do add(position + vector.new(0, height, 0), weight * offset[2]) end
			end
		end
	end
	local total = 0
	for _, point in points do total += point.w end
	local kept, center, weight = {}, vector.zero, 0
	for _, point in points do
		if point.w < total * 0.0015 then continue end
		table.insert(kept, point)
		center, weight = center + point.p * point.w, weight + point.w
	end
	center /= math.max(weight, 1e-6)
	return {points = kept, center = center, weight = weight, airborne = airborne, stuck = stuck, jitter = jitter, latency = latency}
end

local AimTarget = {}

function AimTarget.murderer(context, roles)
	local globals = context.environment.getgenv and context.environment.getgenv()
	local duel = globals and rawget(globals, "__pulseDuel")
	if duel and duel.on and duel.pick then return duel.pick() end
	for _, player in context.players:GetPlayers() do
		local role = roles[player.Name]
		if player == context.player or not role or role.Role ~= "Murderer" or role.Dead == true or role.Killed == true then continue end
		local character = player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		if humanoid and humanoid.Health > 0 and character:FindFirstChild("HumanoidRootPart") then return player, character end
	end
	return nil
end

function AimTarget.sample(context, state, character)
	local root = character:FindFirstChild("HumanoidRootPart")
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not root or not humanoid then return nil end
	local jumpPower = humanoid.UseJumpPower and humanoid.JumpPower or math.sqrt(2 * (state.G or context.workspace.Gravity) * humanoid.JumpHeight)
	return {t = context.environment.os.clock(), p = root.Position, cf = root.CFrame, md = humanoid.MoveDirection,
		ws = humanoid.WalkSpeed, v = root.AssemblyLinearVelocity, st = humanoid:GetState(),
		air = humanoid.FloorMaterial == context.environment.Enum.Material.Air, jump = humanoid.Jump, hh = humanoid.HipHeight, jp = jumpPower}
end

function AimTarget.moveParams(context)
	local environment = context.environment
	local parameters = environment.RaycastParams.new()
	parameters.FilterType, parameters.IgnoreWater, parameters.RespectCanCollide = environment.Enum.RaycastFilterType.Exclude, true, true
	local characters = {}
	for _, player in context.players:GetPlayers() do if player.Character then table.insert(characters, player.Character) end end
	parameters.FilterDescendantsInstances = characters
	return parameters
end

function AimTarget.serverRay(context, state, origin, direction, extra, includePassthrough)
	local environment = context.environment
	local ignored = {}
	if includePassthrough then
		pcall(function() for _, object in context.collection:GetTagged("WeaponPassthrough") do table.insert(ignored, object) end end)
	end
	if context.player.Character then table.insert(ignored, context.player.Character) end
	for _, object in extra or {} do table.insert(ignored, object) end
	local parameters = environment.RaycastParams.new()
	parameters.FilterType, parameters.IgnoreWater = environment.Enum.RaycastFilterType.Exclude, true
	for _ = 1, 12 do
		parameters.FilterDescendantsInstances = ignored
		local hit = context.workspace:Raycast(origin, direction, parameters)
		if not hit then return nil end
		if not state.skipTransparent or hit.Instance.Transparency < 1 then return hit end
		table.insert(ignored, hit.Instance)
	end
	return nil
end

function AimTarget.insideSolid(context, state, position, collisionGroup)
	local environment = context.environment
	local probe = state.probePart
	if not probe then
		probe = environment.Instance.new("Part")
		probe.Size, probe.Anchored, probe.CanCollide, probe.CanQuery, probe.CanTouch = environment.Vector3.new(0.6, 0.6, 0.6), true, false, false, false
		state.probePart = probe
	end
	probe.CFrame = environment.CFrame.new(position)
	local parameters = environment.OverlapParams.new()
	parameters.FilterType, parameters.RespectCanCollide = environment.Enum.RaycastFilterType.Exclude, true
	local characters = {}
	for _, player in context.players:GetPlayers() do if player.Character then table.insert(characters, player.Character) end end
	parameters.FilterDescendantsInstances = characters
	if collisionGroup then pcall(function() parameters.CollisionGroup = collisionGroup end) end
	local ok, parts = pcall(context.workspace.GetPartsInPart, context.workspace, probe, parameters)
	if not ok then return nil end
	for _, part in parts do
		if not part.CanQuery or not part.CanCollide then continue end
		local model = part:FindFirstAncestorOfClass("Model")
		if not model or not model:FindFirstChildOfClass("Humanoid") then return part end
	end
	return nil
end

function AimTarget.track(context, state, roles)
	local environment = context.environment
	local globals = environment.getgenv and environment.getgenv()
	local world = globals and rawget(globals, "__pulseWorldBase")
	state.G = type(world) == "table" and tonumber(world.Gravity) or context.workspace.Gravity
	local _, target = AimTarget.murderer(context, roles)
	if state.forced and state.forced.Parent then target = state.forced end
	if state.queued and state.queued.c and state.queued.c.Parent then target = state.queued.c end
	if target ~= state.target then state.target, state.hist = target, {} end
	local now = environment.os.clock()
	for _, player in context.players:GetPlayers() do
		if player == context.player then continue end
		local character = player.Character
		local root = character and character:FindFirstChild("HumanoidRootPart")
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		if root and humanoid then
			AimCalibration.watchLag(context, state, {player = player, character = character, root = root, humanoid = humanoid, time = now})
		end
	end
	state.shift = target and AimCalibration.lagOf(context, state, target) or 0
	if target then
		local sample = AimTarget.sample(context, state, target)
		local position = sample and sample.p
		local far = position and (position.X ~= position.X or position.Y ~= position.Y or position.Z ~= position.Z
			or position.Magnitude > (state.dsFar or 1000000))
		if far then
			local desync = state.ds
			if not desync or desync.c ~= target then desync = {c = target}; state.ds = desync end
			if not desync.farAt or desync.backAt then
				table.clear(state.hist)
				state.hcut, desync.since = environment.os.clock(), sample.t
			end
			desync.farAt, desync.backAt, sample = sample.t, nil, nil
		end
		if sample and state.ds and state.ds.c == target and state.ds.farAt and not state.ds.backAt then
			state.ds.backAt = sample.t
			table.clear(state.hist)
			if state.say then state.say(string.format("DESYNC back %s after %.2fs", target.Name, sample.t - (state.ds.since or state.ds.farAt))) end
		end
		if sample then
			local history, previous = state.hist, state.hist[#state.hist]
			if not previous or sample.t - previous.t >= 1 / 240 then
				if previous and (sample.p - previous.p).Magnitude > 8 then state.jumps = (state.jumps or 0) + 1 end
				if previous and (sample.p - previous.p).Magnitude > 60 and state.spikeOn ~= true then
					table.clear(history)
					state.hcut = environment.os.clock()
				end
				table.insert(history, sample)
				if state.spoofOn ~= false and environment.typeof(sample.v) == "Vector3" and previous
					and sample.v.Magnitude >= (state.spoofMin or 400) then
					local ok, estimated = pcall(AimGeometry.kinVelocity, environment, history, sample.t, 0.1)
					if not ok or environment.typeof(estimated) ~= "Vector3" then
						estimated = (sample.p - previous.p) / math.max(sample.t - previous.t, 1 / 240)
					end
					if estimated.Magnitude < sample.v.Magnitude * (state.spoofK or 0.25) then sample.vr, sample.v = sample.v, estimated end
				end
				while #history > 2 and sample.t - history[1].t > 3 do table.remove(history, 1) end
			end
		end
	end
	state.othersC, state.othersData, state.othersDirty = target, roles, true
end

function AimTarget.others(context, state)
	if not state.othersDirty then return state.others or {} end
	state.othersDirty = false
	local globals = context.environment.getgenv and context.environment.getgenv()
	local duel = globals and rawget(globals, "__pulseDuel")
	if not duel or not duel.on or not duel.pick then duel = nil end
	local others = {}
	for _, player in context.players:GetPlayers() do
		local character = player.Character
		if player == context.player or character == state.othersC or duel and duel.foes[player.Name] then continue end
		local role = (state.othersData or {})[player.Name]
		local root = character and character:FindFirstChild("HumanoidRootPart")
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		if not root or not humanoid or not (humanoid.Health > 0) or role and (role.Dead or role.Killed) then continue end
		table.insert(others, {c = character, p = root.Position, md = humanoid.MoveDirection, ws = humanoid.WalkSpeed})
	end
	state.others = others
	return others
end

function AimTarget.innocentRisk(context, state, options)
	local vector = context.environment.Vector3
	for _, other in AimTarget.others(context, state) do
		if other.c == options.character then continue end
		local predicted = other.p + vector.new(other.md.X, 0, other.md.Z) * (other.ws * math.max(state.L, 0.15))
		for _, position in {other.p, predicted, (other.p + predicted) * 0.5} do
			local closest = options.origin + options.direction * math.clamp((position - options.origin):Dot(options.direction), 0, options.distance)
			local height = position.Y - closest.Y
			if vector.new(position.X - closest.X, 0, position.Z - closest.Z).Magnitude < (state.innocentMargin or 3.5)
				and height > -3.8 and height < 4.8 then return true end
		end
	end
	return false
end

function AimTarget.selfShift(context, state)
	local environment = context.environment
	local globals = environment.getgenv and environment.getgenv()
	local invisible = globals and rawget(globals, "__pulseInvis")
	if type(invisible) ~= "table" or not invisible.active or environment.typeof(invisible.real) ~= "CFrame" then return environment.Vector3.zero end
	if environment.os.clock() - (tonumber(invisible.realAt) or 0) > (state.shiftFresh or 0.3) then return environment.Vector3.zero end
	local character = context.player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	if not root then return environment.Vector3.zero end
	local shift = invisible.real.Position - root.Position
	if shift.Magnitude < 190000 or shift.Magnitude > 210000 then return environment.Vector3.zero end
	return shift
end

function AimTarget.visible(context, state, character)
	local own = context.player.Character
	local head = own and own:FindFirstChild("Head")
	if not head then return false end
	local origin = head.Position + AimTarget.selfShift(context, state)
	for _, name in {"HumanoidRootPart", "Head", "UpperTorso", "Torso"} do
		local part = character:FindFirstChild(name)
		if part and not AimTarget.serverRay(context, state, origin, part.Position - origin, {character}) then return true end
	end
	return false
end

function AimTarget.gun(context, state)
	local character = context.player.Character
	local gun = character and character:FindFirstChild("Gun")
	if gun then return gun end
	local backpack = context.player:FindFirstChild("Backpack")
	return state.backpackOk and backpack and backpack:FindFirstChild("Gun") or nil
end

function AimTarget.muzzleState(context, state)
	local character = context.player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	local head = character and character:FindFirstChild("Head")
	local attachment = root and root:FindFirstChild("GunRaycastAttachment")
	if not root or not head or not attachment then return false end
	local parameters = context.environment.RaycastParams.new()
	parameters.FilterDescendantsInstances = {character}
	local blocked = context.workspace:Raycast(head.Position + AimTarget.selfShift(context, state), attachment.WorldPosition - head.Position, parameters) ~= nil
	return blocked, root, head, attachment, parameters
end

function AimTarget.guardInfo(context)
	local environment = context.environment
	local globals = environment.getgenv and environment.getgenv()
	local guard = globals and rawget(globals, "__pulseMuzzle")
	if type(guard) ~= "table" or not guard.alive or not guard.at or environment.os.clock() - guard.at > 0.15 then return nil end
	return guard
end

function AimTarget.clearMuzzle(context, state)
	if state.mzGate == false then return true end
	local guard = AimTarget.guardInfo(context)
	if not guard then return true end
	local now = context.environment.os.clock()
	local margin = math.clamp((state.mzBase or 0.21) + context.player:GetNetworkPing() * (state.mzPingK or 0.5), 0.1, 0.35)
	if not guard.blocked and now - (guard.clearSince or 0) >= margin then return true end
	if guard.request then guard.request(state.mzHoldFor or 0.3) end
	if state.mzSaidAt == nil or now - state.mzSaidAt > 1 then
		state.mzSaidAt = now
		if state.say then
			state.say(string.format("muzzle %s, stepping out (clear %.0fms, need %.0fms)", guard.blocked and "blocked" or "fresh",
				guard.blocked and 0 or (now - (guard.clearSince or now)) * 1000, margin * 1000))
		end
	end
	if guard.blocked then return false, now + 0.02 end
	return false, guard.clearSince + margin
end

function AimPlan.scoreLine(context, state, options)
	local line, body, points = options.line, options.body, options.points
	local detachedWeight = options.detachedWeight
	local path = line.e - line.o
	line.len, line.d = path.Magnitude, path / path.Magnitude
	local hits, score = {}, 0
	if detachedWeight < 1 then
		for index, point in points do
			local hit, core = AimGeometry.hitsBody(context.environment, {body = body, position = point.p,
				origin = line.o, direction = line.d, distance = line.len + 400})
			if not hit then continue end
			hits[index] = {hit, core}
			score += core and point.w or point.w * 0.3
		end
		score *= 1 - detachedWeight
	end
	local detached
	if detachedWeight > 0 then
		detached = AimGeometry.hitsDetached(body, line.o, line.d, line.len + 400)
		if detached then score += detachedWeight end
	end
	if options.threshold and score <= options.threshold then
		line.score, line.pruned = score, true
		return score
	end
	local obstruction = AimTarget.serverRay(context, state, line.o, line.d * (line.len + 0.1), {options.character})
	local distance = obstruction and (obstruction.Position - line.o).Magnitude or math.huge
	local function between(near, far)
		local result = 0
		if detachedWeight < 1 then
			for index, point in points do
				local hit = hits[index]
				if not hit then continue end
				if hit[2] and hit[2] < far and hit[2] > near then result += point.w; continue end
				if hit[1] < far and hit[1] > near then result += point.w * 0.3 end
			end
			result *= 1 - detachedWeight
		end
		if detached and detached < far and detached > near then result += detachedWeight end
		return result
	end
	score = between(0, distance)
	if obstruction and state.wallStart == true and not line.o2 then
		local origin = obstruction.Position + line.d * 0.08
		local traveled = (origin - line.o).Magnitude
		local remaining = line.len + 0.1 - traveled
		if remaining > 0.5 then
			local second = AimTarget.serverRay(context, state, origin, line.d * remaining, {options.character})
			local secondDistance = second and (second.Position - line.o).Magnitude or math.huge
			if secondDistance > distance + 0.3 then
				local afterWall = between(traveled, secondDistance)
				if afterWall > score + 0.001 then line.o2, score = origin, afterWall end
			end
		end
	end
	line.score = score
	return score
end

function AimPlan.candidates(context, state, options)
	local environment, samples, cloud = context.environment, options.samples, options.cloud
	local vector, types = environment.Vector3, environment.Enum.HumanoidStateType
	local time, mode, hypotheses, taus = options.time, options.mode, options.hypotheses, options.taus
	local points, center, weight = cloud.points, cloud.center, cloud.weight
	local latest, jitter, anomaly = samples[#samples], cloud.jitter, options.anomaly
	local detachedWeight = #options.body.detached > 0 and (#options.body.parts == 0 and 1 or 0.5) or 0
	local targetCenter = center
	if detachedWeight >= 1 then
		targetCenter = vector.zero
		for _, part in options.body.detached do targetCenter += part.cf.Position end
		targetCenter /= #options.body.detached
	end
	local candidates = {}
	local function add(origin, endpoint, kind)
		if (endpoint - origin).Magnitude >= 0.5 then table.insert(candidates, {o = origin, e = endpoint, kind = kind}) end
	end
	local firstTau, lastTau = taus[1].tau, taus[#taus].tau
	local sorted = table.clone(hypotheses)
	table.sort(sorted, function(first, second) return first.w > second.w end)
	local best = {}
	for index = 1, math.min(3, #sorted) do if sorted[index].w > 0.07 then table.insert(best, sorted[index]) end end
	local velocity = AimGeometry.kinVelocity(environment, samples, time, 0.08) or vector.zero
	local fast = mode == "nr" or vector.new(velocity.X, 0, velocity.Z).Magnitude > 22
	local speed = vector.new(velocity.X, 0, velocity.Z).Magnitude
	if state.fastWin then speed = math.max(speed, AimMotion.recentSpeed(environment, samples, time, state.fastWin)) end
	local preferVelocityFit = false
	if state.velfPref == true and speed > (state.velfPrefV or 60) then
		for _, hypothesis in hypotheses do
			if hypothesis.kind ~= "velf" or not hypothesis.better then continue end
			preferVelocityFit = true
			if not table.find(best, hypothesis) then table.insert(best, hypothesis) end
			break
		end
	end
	state.lastVelfMode = preferVelocityFit
	local function endpoints(hypothesis)
		local first = firstTau >= 0 and AimMotion.trajectoryAt(hypothesis.traj, firstTau)
			or (AimGeometry.posAt(samples, time + firstTau) or AimMotion.trajectoryAt(hypothesis.traj, 0))
		return first, AimMotion.trajectoryAt(hypothesis.traj, lastTau)
	end
	if fast then
		for _, hypothesis in best do
			local first, last = endpoints(hypothesis)
			local delta = last - first
			if delta.Magnitude <= 1 or math.abs(delta.Unit.Y) <= 0.25 then continue end
			local direction = delta.Unit
			for _, height in {0.35, 0.9} do
				local offset = vector.new(0, height, 0)
				add(first + offset - direction * 2.5, last + offset + direction * 1.5, "chord:" .. hypothesis.kind .. ":" .. height)
			end
		end
	end
	local takeoff = cloud.airborne and not (latest.air or latest.st == types.Climbing)
	if takeoff and state.pca3 then
		local xx, xy, xz, yy, yz, zz = 0, 0, 0, 0, 0, 0
		for _, point in points do
			local dx, dy, dz = point.p.X - center.X, point.p.Y - center.Y, point.p.Z - center.Z
			xx, xy, xz = xx + point.w * dx * dx, xy + point.w * dx * dy, xz + point.w * dx * dz
			yy, yz, zz = yy + point.w * dy * dy, yz + point.w * dy * dz, zz + point.w * dz * dz
		end
		xx, xy, xz, yy, yz, zz = xx / weight, xy / weight, xz / weight, yy / weight, yz / weight, zz / weight
		local function multiply(axis)
			return vector.new(xx * axis.X + xy * axis.Y + xz * axis.Z,
				xy * axis.X + yy * axis.Y + yz * axis.Z, xz * axis.X + yz * axis.Y + zz * axis.Z)
		end
		local axis = vector.new(0.3, 1, 0.2)
		for _ = 1, 24 do
			local nextAxis = multiply(axis)
			if nextAxis.Magnitude < 1e-9 then break end
			axis = nextAxis.Unit
		end
		local extent = math.sqrt(math.max(axis:Dot(multiply(axis)), 0)) * 2.2 + 2
		for _, height in {0.35, 0.9} do
			for _, sign in {1, -1} do
				local direction, middle = axis * sign, center + vector.new(0, height, 0)
				add(middle - direction * (extent + 1.5), middle + direction * (extent + 1), "pca3:" .. height)
			end
		end
	end
	local slopes = takeoff and state.pca3 and {0, 0.35, 0.7, -0.35, -0.7, 1.4, -1.4} or {0, 0.35, 0.7}
	for _, hypothesis in best do
		local first, last = endpoints(hypothesis)
		local delta = vector.new(last.X - first.X, 0, last.Z - first.Z)
		if delta.Magnitude <= 0.6 then continue end
		local direction, middle, half = delta.Unit, (first + last) * 0.5, delta.Magnitude * 0.5 + 1.2
		for _, height in {0.35, 0.9} do
			for _, slope in slopes do
				local centerAtHeight = middle + vector.new(0, height, 0)
				local axis = (direction + vector.new(0, -slope, 0)).Unit
				local extent = half / math.max(0.35, axis:Dot(direction))
				add(centerAtHeight - axis * (extent + 1.5), centerAtHeight + axis * (extent + 1),
					"rail:" .. hypothesis.kind .. ":" .. height .. ":" .. slope)
			end
		end
	end
	if not (takeoff and state.noPcaTake == true) then
		local xx, xz, zz = 0, 0, 0
		for _, point in points do
			local dx, dz = point.p.X - center.X, point.p.Z - center.Z
			xx, xz, zz = xx + point.w * dx * dx, xz + point.w * dx * dz, zz + point.w * dz * dz
		end
		local sum, determinant = xx + zz, xx * zz - xz * xz
		local eigenvalue = sum / 2 + math.sqrt(math.max(0, sum * sum / 4 - determinant))
		local axis = math.abs(xz) > 1e-6 and vector.new(eigenvalue - zz, 0, xz)
			or (xx >= zz and vector.new(1, 0, 0) or vector.new(0, 0, 1))
		if axis.Magnitude > 1e-6 then
			axis = axis.Unit
			local extent = math.sqrt(math.max(eigenvalue, 0)) * 2 + 1.5
			for _, height in state.pcaOffs or {0, 0.5, 0.9} do
				for _, slope in {0, 0.2, 0.55} do
					for _, sign in {1, -1} do
						local direction, middle = (axis * sign + vector.new(0, -slope, 0)).Unit, center + vector.new(0, height, 0)
						add(middle - direction * (extent + 1.5), middle + direction * (extent + 1), "pca")
					end
				end
			end
		end
	end
	local clusters = AimMotion.clusters(environment, state, samples, time)
	if clusters then
		local first, second = clusters[1].p + vector.new(0, 0.5, 0), clusters[2].p + vector.new(0, 0.5, 0)
		if (second - first).Magnitude > 1 then
			local direction = (second - first).Unit
			add(first - direction * 2.5, second + direction * 2, "cluster")
			add(second + direction * 2.5, first - direction * 2, "clusterR")
		end
	end
	local oscillation = AimMotion.oscillation(environment, state, samples, time)
	if oscillation and #oscillation >= 2 then
		local largest, first, second = -1, oscillation[1], oscillation[2]
		for left = 1, #oscillation do
			for right = left + 1, #oscillation do
				local distance = (oscillation[left] - oscillation[right]).Magnitude
				if distance > largest then largest, first, second = distance, oscillation[left], oscillation[right] end
			end
		end
		if largest > 1 then
			local direction = (second - first).Unit
			for _, height in {0.35, 0.9} do
				local offset = vector.new(0, height, 0)
				add(first + offset - direction * 2.5, second + offset + direction * 2, "osc")
				add(second + offset + direction * 2.5, first + offset - direction * 2, "oscR")
			end
		end
	end
	if jitter then
		local extent = jitter.half * (state.jitExtK or 3.2) + (state.jitExt or 6)
		for _, height in state.jitOffs or {0, 0.8, 1.6, 2.4} do
			local middle = jitter.c + vector.new(0, height, 0)
			add(middle - jitter.u * extent, middle + jitter.u * extent, "jit")
			add(middle + jitter.u * extent, middle - jitter.u * extent, "jitR")
		end
	end
	if anomaly and anomaly.holdFrac > 0.2 and state.holdLines ~= false then
		local direction
		for _, hypothesis in hypotheses do
			if hypothesis.kind ~= "md" then continue end
			local first, last = AimMotion.trajectoryAt(hypothesis.traj, 0), AimMotion.trajectoryAt(hypothesis.traj, math.min(options.duration, 0.3))
			local delta = vector.new(last.X - first.X, 0, last.Z - first.Z)
			if delta.Magnitude > 0.5 and (state.lastTrust or 1) >= 0.6 then direction = delta.Unit end
		end
		local movement = anomaly.vel and vector.new(anomaly.vel.X, 0, anomaly.vel.Z)
		if not direction and movement and movement.Magnitude > 2 then direction = movement.Unit end
		if direction then
			local extent = (movement and movement.Magnitude or 16) * (lastTau + 0.25) + 2
			for _, height in {0.35, 0.9} do
				local middle = latest.p + vector.new(0, height, 0)
				add(middle - direction * 4, middle + direction * extent, "hold")
				add(middle + direction * (extent + 3), middle - direction * 3, "holdR")
			end
		end
	end
	local frozen = options.frozen
	if frozen then
		local extent = frozen.ws * (state.k or 1) * (frozen.T + lastTau + 0.3) + 4
		for _, height in {0.35, 0.9} do
			local middle = frozen.p + vector.new(0, height, 0)
			add(middle - frozen.dir * 4, middle + frozen.dir * extent, "frz")
		end
	end
	local middleTau = taus[state.tMidL == true and taus.mid or math.ceil(#taus / 2)].tau
	if anomaly and (state.reslAll == true or (anomaly.jumps or 0) >= 5) then
		local positions = {}
		for _, hypothesis in hypotheses do
			if hypothesis.kind == "res" then table.insert(positions, AimMotion.trajectoryAt(hypothesis.traj, middleTau)) end
		end
		local count = 0
		for left = 1, #positions do
			for right = left + 1, #positions do
				if count >= 18 then break end
				local first, last = positions[left], positions[right]
				if vector.new(last.X - first.X, 0, last.Z - first.Z).Magnitude <= 1.2 or (left + right) % 3 ~= 0 then continue end
				local direction = (last - first).Unit
				for _, height in {0.35, 0.9} do
					local offset = vector.new(0, height, 0)
					add(first + offset - direction * 3, last + offset + direction * 3, "resl")
				end
				count += 1
			end
		end
	end
	local function clearance(position, height)
		local hit = AimTarget.serverRay(context, state, position, vector.new(0, height, 0), {options.character})
		if hit then return math.max(1.5, (hit.Position - position).Magnitude - 0.3) end
		return height
	end
	if not (state.colFastMax and state.fastWin and speed > state.colFastMax) then
		add(targetCenter + vector.new(0.02, clearance(targetCenter, 9), 0.02), targetCenter + vector.new(0, -5, 0), "column")
	end
	if cloud.airborne then
		for _, hypothesis in best do
			local position = middleTau >= 0 and AimMotion.trajectoryAt(hypothesis.traj, middleTau) or targetCenter
			local height = clearance(position, 10)
			if not (state.colFastMax and speed > state.colFastMax) then
				add(position + vector.new(0.02, height, 0.02), position + vector.new(0, -7, 0), "colH:" .. hypothesis.kind)
			end
			if state.tiltCols ~= false and mode ~= "nr" and not (state.tiltFastMax and speed > state.tiltFastMax) then
				local before = AimMotion.trajectoryAt(hypothesis.traj, math.max(0, middleTau - 0.05))
				local after = AimMotion.trajectoryAt(hypothesis.traj, middleTau + 0.05)
				local movement = vector.new(after.X - before.X, 0, after.Z - before.Z)
				if movement.Magnitude > 0.2 then
					for _, degrees in {30, 55} do
						local angle = math.rad(degrees)
						for _, sign in {1, -1} do
							local direction = (movement.Unit * (math.sin(angle) * sign) + vector.new(0, math.cos(angle), 0)).Unit
							add(position + direction * 8, position - direction * 7, "tilt:" .. degrees)
						end
					end
				end
			end
		end
	end
	local character = context.player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	local attachment = root and root:FindFirstChild("GunRaycastAttachment")
	local muzzle = attachment and attachment.WorldPosition or root and root.Position
	if muzzle then
		muzzle += AimTarget.selfShift(context, state)
		local endpoint = targetCenter + vector.new(0, 0.5, 0)
		local delta = endpoint - muzzle
		if delta.Magnitude > 1 then
			if delta.Magnitude < (state.losMax or 900) then add(muzzle, endpoint + delta.Unit * 3, "los") end
			add(endpoint - delta.Unit * 6, endpoint + delta.Unit * 3, "losNear")
		end
	end
	return {lines = candidates, detachedWeight = detachedWeight, targetCenter = targetCenter, speed = speed, muzzle = muzzle,
		preferVelocityFit = preferVelocityFit}
end

function AimPlan.select(context, state, options)
	local environment, samples, cloud, candidates = context.environment, options.samples, options.cloud, options.candidates
	local vector, latest = environment.Vector3, samples[#samples]
	local velocity = AimGeometry.kinVelocity(environment, samples, options.time, 0.08) or vector.zero
	local groundMoving = not cloud.airborne and vector.new(velocity.X, 0, velocity.Z).Magnitude > (state.gmSpeed or 6)
	local bouncy = false
	if state.bounceK then
		local low, high = math.huge, -math.huge
		for index = #samples, 1, -1 do
			if options.time - samples[index].t > (state.bounceT or 1.5) then break end
			low, high = math.min(low, samples[index].p.Y), math.max(high, samples[index].p.Y)
		end
		bouncy = high - low > (state.bounceY or 4)
	end
	state.lastBouncy = bouncy
	local best
	local function consider(line)
		local column = line.kind:find("^column") or line.kind:find("^colH")
		local tilt = line.kind:find("^tilt")
		local airPenalty = (latest.air or latest.st == environment.Enum.HumanoidStateType.Climbing) and (state.airPen or 0.8) or (state.takePen or 0.65)
		local threshold = best and best.score + 0.001 or nil
		if threshold and cloud.airborne and not column and not tilt then threshold /= airPenalty end
		local score = AimPlan.scoreLine(context, state, {line = line, body = options.body, points = cloud.points,
			detachedWeight = candidates.detachedWeight, character = options.character, threshold = threshold})
		if line.pruned then return end
		local skipPenalty = cloud.jitter and state.jitNoPen and line.kind:find("^jit") ~= nil
		if line.o2 then line.o, line.len, line.kind = line.o2, (line.e - line.o2).Magnitude, line.kind .. "+w" end
		if bouncy and not skipPenalty and not column then score *= state.bounceK end
		if state.tiltK and tilt then score *= state.tiltK end
		if cloud.airborne and state.airNonColK and not skipPenalty and not column and not tilt then score *= state.airNonColK end
		if groundMoving and state.railPref ~= false and column then score *= state.railK or 0.72 end
		if candidates.preferVelocityFit and not skipPenalty and not line.kind:find("^rail:velf") and not line.kind:find("^chord:velf") then
			score *= state.velfOtherK or 0.75
		end
		if cloud.airborne and not skipPenalty and not column and not tilt then score *= airPenalty end
		line.score = score
		if score <= 0.02 then return end
		local hit = AimTarget.serverRay(context, state, line.o, line.d * 600, {options.character}, true)
		line.reach = hit and (hit.Position - line.o).Magnitude or 600
		if AimTarget.innocentRisk(context, state, {origin = line.o, direction = line.d, distance = line.reach, character = options.character}) then
			line.risky, state.riskyN = true, (state.riskyN or 0) + 1
			return
		end
		if not best or score > best.score + 0.001 then best = line end
	end
	local lines = candidates.lines
	for _, line in lines do if line.kind == "los" or line.kind == "losNear" then consider(line) end end
	if cloud.jitter and state.jitOnly then
		local filtered = {}
		for _, line in lines do
			if line.kind:find("^jit") or line.kind == "los" or line.kind == "losNear" then table.insert(filtered, line) end
		end
		lines = filtered
	end
	if cloud.jitter and state.jitFirst ~= false then
		for _, line in lines do
			if not line.kind:find("^jit") then continue end
			consider(line)
			line.done = true
		end
	end
	if state.colFirst then
		local order = {column = 1, colH = 2, tilt = 2, hold = 2, holdR = 2, resl = 3, pca3 = 3, chord = 4, rail = 5, pca = 6}
		for index, line in lines do line.ord = (order[line.kind:match("^[^:+]+") ] or 7) * 1000 + index end
		table.sort(lines, function(first, second) return first.ord < second.ord end)
	end
	state.riskyN = 0
	local deadline, evaluated = environment.os.clock() + (state.candBudget or 0.012), 0
	for _, line in lines do
		if line.kind == "los" or line.kind == "losNear" or line.done then continue end
		consider(line)
		evaluated += 1
		if best and (best.score > 0.995 or environment.os.clock() > deadline) then break end
	end
	state.lastEval = evaluated .. "/" .. #lines
	if best and best.score < 0.995 then
		local direction = best.d
		local vertical = math.abs(direction.Y) > 0.9 and vector.new(1, 0, 0) or vector.new(0, 1, 0)
		local first = direction:Cross(vertical).Unit
		local second, original = direction:Cross(first).Unit, best
		local column = original.kind:find("^column") or original.kind:find("^colH")
		local skip = column and state.colRefine ~= true or state.rRefine ~= true
		local multiplier = column and state.colRefineK or 1
		for _, offset in {first * 0.3, -first * 0.3, second * 0.3, -second * 0.3, first * 0.6, -first * 0.6} do
			if skip or environment.os.clock() > deadline + 0.003 then break end
			consider({o = original.o + offset * multiplier, e = original.e + offset * multiplier, kind = original.kind .. "+r"})
		end
		if state.refineRot == true and best and best.score < 0.995 then
			local original = best
			local axis = original.d or (original.e - original.o).Unit
			local pivot = original.o + axis * (candidates.targetCenter - original.o):Dot(axis)
			local backward, forward = (pivot - original.o).Magnitude, (original.e - pivot).Magnitude
			local basis = axis:Cross(math.abs(axis.Y) > 0.9 and vector.new(1, 0, 0) or vector.new(0, 1, 0)).Unit
			for _, rotationAxis in {basis, axis:Cross(basis).Unit} do
				for _, degrees in {6, -6, 12, -12} do
					if environment.os.clock() > deadline + 0.006 then break end
					local angle = math.rad(degrees)
					local rotated = (axis * math.cos(angle) + rotationAxis:Cross(axis) * math.sin(angle)
						+ rotationAxis * rotationAxis:Dot(axis) * (1 - math.cos(angle))).Unit
					consider({o = pivot - rotated * backward, e = pivot + rotated * forward, kind = original.kind .. "+a"})
				end
			end
		end
	end
	if (not best or best.score < (state.pbBelow or 0.3)) and state.pbFallback ~= false then
		for _, direction in {vector.new(0, 1, 0), vector.new(0, -1, 0), vector.new(1, 0, 0), vector.new(-1, 0, 0),
			vector.new(0, 0, 1), vector.new(0, 0, -1), vector.new(0.7, 0.7, 0), vector.new(-0.7, 0.7, 0),
			vector.new(0, 0.7, 0.7), vector.new(0, 0.7, -0.7)} do
			direction = direction.Unit
			consider({o = candidates.targetCenter - direction * (state.pbBack or 2.2),
				e = candidates.targetCenter + direction * (state.pbFwd or 3), kind = "pb"})
		end
	end
	if best then
		for _, kind in {"los", "losNear"} do
			if kind == "losNear" and best.kind == "los" then break end
			for _, line in lines do
				if line.kind == kind and line.score and not line.risky and line.reach and line.score >= 0.985 and line.score >= best.score - 0.005 then
					best = line
					break
				end
			end
		end
	end
	if state.inWall ~= false and vector.new(velocity.X, 0, velocity.Z).Magnitude < 4 then
		local root = options.character:FindFirstChild("HumanoidRootPart")
		local wall = root and AimTarget.insideSolid(context, state, root.Position, root.CollisionGroup)
		if wall then
			if state.lastInWall ~= wall then
				state.lastInWall = wall
				if state.say then state.say("target inside " .. wall:GetFullName()) end
			end
			local center = candidates.targetCenter
			local muzzle = candidates.muzzle or center + vector.new(0, 0, 5)
			local direction = vector.new(center.X - muzzle.X, 0, center.Z - muzzle.Z)
			direction = direction.Magnitude > 0.1 and direction.Unit or vector.new(0, 0, 1)
			local height = vector.new(0, 0.5, 0)
			local inside = {{o = center + height - direction * 2.2, e = center + height + direction * 3, kind = "inW"},
				{o = center + vector.new(0.02, 3.3, 0.02), e = center - vector.new(0, 4, 0), kind = "inWc"}}
			if state.inWallInside ~= false then
				local half, frame = wall.Size * 0.5, wall.CFrame
				local function contains(position)
					local point = frame:PointToObjectSpace(position)
					return math.abs(point.X) < half.X - 0.05 and math.abs(point.Y) < half.Y - 0.05 and math.abs(point.Z) < half.Z - 0.05
				end
				local side = vector.new(-direction.Z, 0, direction.X)
				for _, axis in {direction, side, -side, -direction} do
					for _, distance in {0.5, 1, 1.6, 2.4} do
						local origin = center + height - axis * distance
						if not contains(origin) then continue end
						table.insert(inside, {o = origin, e = center + height + axis * 3, kind = "inWi"})
						break
					end
				end
				for _, distance in {1.2, 2.2, 3.2} do
					local origin = center + vector.new(0.02, distance, 0.02)
					if not contains(origin) then continue end
					table.insert(inside, {o = origin, e = center - vector.new(0, 4, 0), kind = "inWci"})
					break
				end
			end
			for _, line in inside do
				local previous = best
				best = nil
				consider(line)
				local found = best
				best = previous
				if found and (not best or found.score >= best.score - 0.02) then best = found end
			end
		end
	end
	return best, lines
end

function AimPlan.resolve(environment, state, samples, options)
	local time, duration, hypotheses, anomaly = options.time, options.duration, options.hypotheses, options.anomaly
	local vector = environment.Vector3
	local function recent(window)
		local first, indices = #samples, {}
		while first > 1 and time - samples[first - 1].t <= window do first -= 1 end
		for index = first, #samples do table.insert(indices, index) end
		return indices
	end
	local function fit(indices)
		local sum, weighted, sumTime, sumSquared = vector.zero, vector.zero, 0, 0
		for _, index in indices do
			local sample, elapsed = samples[index], samples[index].t - time
			sumTime, sumSquared = sumTime + elapsed, sumSquared + elapsed * elapsed
			sum, weighted = sum + sample.p, weighted + sample.p * elapsed
		end
		local count = #indices
		local denominator = count * sumSquared - sumTime * sumTime
		local velocity = denominator > 1e-9 and (weighted * count - sum * sumTime) / denominator or vector.zero
		return (sum - velocity * sumTime) / count, velocity
	end
	local indices = recent(anomaly.jumps >= 6 and 1 or 0.7)
	if #indices < 6 then return hypotheses end
	local base, velocity = fit(indices)
	local errors = {}
	for _, index in indices do table.insert(errors, (samples[index].p - (base + velocity * (samples[index].t - time))).Magnitude) end
	local median, filtered = AimCalibration.median(errors) or 0, {}
	for index, sampleIndex in indices do if errors[index] <= median * 3 + 3 then table.insert(filtered, sampleIndex) end end
	if #filtered >= 6 then base, velocity = fit(filtered) end
	local horizontal = vector.new(velocity.X, 0, velocity.Z)
	if horizontal.Magnitude > 150 then horizontal = horizontal.Unit * 150 end
	local movement = vector.new(horizontal.X, samples[#samples].air and math.clamp(velocity.Y, -60, 60) or 0, horizontal.Z)
	local offsets = {}
	for _, index in recent(0.6) do
		local offset = samples[index].p - (base + velocity * (samples[index].t - time))
		if offset.Magnitude <= 25 then table.insert(offsets, offset) end
	end
	if #offsets == 0 then offsets[1] = vector.zero end
	local motion
	for _, hypothesis in hypotheses do if hypothesis.kind == "md" then motion = hypothesis end end
	local models = {}
	if motion and (state.lastTrust or 1) >= 0.6 and anomaly.holdFrac > 0.2 and state.resMd ~= false then
		local start = AimMotion.trajectoryAt(motion.traj, 0)
		table.insert(models, {w = 0.6, at = function(elapsed) return AimMotion.trajectoryAt(motion.traj, elapsed) - start + base end})
		table.insert(models, {w = 0.4, at = function(elapsed) return base + movement * elapsed end})
	end
	if #models == 0 then table.insert(models, {w = 1, at = function(elapsed) return base + movement * elapsed end}) end
	local result = {}
	for _, hypothesis in hypotheses do hypothesis.w *= 0.25; table.insert(result, hypothesis) end
	local count = math.min(10, #offsets)
	for _, model in models do
		local steps, trajectory = math.max(2, math.ceil(duration * 30)), {}
		for index = 0, steps do
			local elapsed = duration * index / steps
			table.insert(trajectory, {tau = elapsed, p = model.at(elapsed)})
		end
		for index = 1, count do
			local offset = offsets[math.clamp(math.floor((index - 0.5) / count * #offsets) + 1, 1, #offsets)]
			local shifted = {}
			for _, point in trajectory do table.insert(shifted, {tau = point.tau, p = point.p + offset}) end
			table.insert(result, {kind = "res", w = 0.75 * model.w / count, traj = shifted})
		end
	end
	anomaly.vel, anomaly.base = movement, base
	return result
end

function AimPlan.prepare(context, state, options)
	local samples, time, duration, parameters = options.samples, options.time, options.duration, options.parameters
	local environment, mode = context.environment, options.mode
	local vector = environment.Vector3
	local hypotheses = AimMotion.hypotheses(context, state, samples, {time = time, duration = duration, parameters = parameters})
	local anomaly = AimMotion.anomaly(state, samples, time)
	local trust = state.useTrust == false and 1 or AimMotion.directionTrust(environment, state, samples, time)
	state.lastTrust = trust
	if trust < 0.98 then
		local affected = {md = true, mdE = true, mdL = true, mdNJ = true, mdv = true, mdJe = true, mdJl = true, mdJa = true}
		for _, hypothesis in hypotheses do if affected[hypothesis.kind] then hypothesis.w *= math.max(trust, 0.03) end end
	end
	local velocity = AimGeometry.kinVelocity(environment, samples, time, 0.08) or vector.zero
	if mode == "nr" or velocity.Magnitude > 28 then
		local trajectory = AimMotion.arcTrajectory(environment, samples, time, duration)
		if mode ~= "nr" and not AimMotion.flightLike(context, state, samples, time) then
			trajectory = AimMotion.clampTrajectory(context.workspace, state, trajectory, parameters)
		end
		if trajectory then
			local maximum, hasVelocity, betterFit = 0, false, false
			for _, hypothesis in hypotheses do
				maximum = math.max(maximum, hypothesis.w)
				if hypothesis.kind == "vel" or hypothesis.kind == "velf" then hasVelocity = true end
				if hypothesis.kind == "velf" and hypothesis.better then betterFit = true end
			end
			local factor = hasVelocity and state.arc3VelK or 0.8
			if betterFit and state.arc3VelfK then factor = state.arc3VelfK end
			local floor = betterFit and state.arc3FloorF ~= false and 0 or 0.15
			table.insert(hypotheses, {kind = "arc3", w = math.max(maximum * factor, floor), traj = trajectory})
		end
	end
	local accelerated = false
	if (state.caW or 0) > 0 and (mode == "nr" or AimMotion.flightLike(context, state, samples, time)) then
		local trajectory = AimMotion.accelerationTrajectory(context, state, samples, time, duration)
		if trajectory then
			local maximum = 0
			for _, hypothesis in hypotheses do
				maximum = math.max(maximum, hypothesis.w)
				if hypothesis.kind ~= "arc3" then hypothesis.w *= state.caOthers or 1 end
			end
			table.insert(hypotheses, {kind = "ca", w = math.max(maximum * state.caW, 0.15), traj = trajectory})
			state.lastCA, accelerated = environment.os.clock(), true
		end
	end
	local canOrbit = not accelerated and mode ~= "nr" and not samples[#samples].air
	if canOrbit and state.orbitFastMax and AimMotion.recentSpeed(environment, samples, time, state.fastWin or 1) > state.orbitFastMax then canOrbit = false end
	local orbit = canOrbit and AimMotion.orbitFit(environment, state, samples, time, duration) or nil
	if orbit then
		local maximum = 0
		for _, hypothesis in hypotheses do
			maximum = math.max(maximum, hypothesis.w)
			hypothesis.w *= 0.35
		end
		table.insert(hypotheses, {kind = "orbit", w = math.max(maximum, 0.3) * 1.5, traj = orbit})
		state.lastOrbit = environment.os.clock()
	end
	local frozen
	if state.frz == true then
		local last, index = samples[#samples], #samples
		while index > 1 and (samples[index - 1].p - last.p).Magnitude < 0.001 do index -= 1 end
		local elapsed, movement = time - samples[index].t, vector.new(last.md.X, 0, last.md.Z)
		if elapsed >= (state.frzMin or 0.04) and elapsed < 1.5 and movement.Magnitude > 0.5 and not last.air then
			local direction, speed = movement.Unit, (last.ws or 16) * (state.k or 1)
			local maximum, moving, stationary = 0, {}, {}
			for _, hypothesis in hypotheses do maximum = math.max(maximum, hypothesis.w) end
			for step = 0, math.ceil(duration * 60) do
				local offset = step / 60
				table.insert(moving, {tau = offset, p = last.p + direction * (speed * (elapsed + offset))})
				table.insert(stationary, {tau = offset, p = last.p})
			end
			table.insert(hypotheses, {kind = "frzGo", w = math.max(maximum, 0.2) * (state.frzW or 1), traj = moving})
			table.insert(hypotheses, {kind = "frzStay", w = math.max(maximum, 0.2) * (state.frzStayW or 0.5), traj = stationary})
			state.lastFrz = environment.os.clock()
			frozen = {T = elapsed, dir = direction, p = last.p, ws = last.ws or 16}
		end
	end
	if state.airRev == true and mode == "std" and samples[#samples].air then
		local movement
		for _, hypothesis in hypotheses do if hypothesis.kind == "md" then movement = hypothesis end end
		if movement and #movement.traj >= 3 then
			local stop, reverse = {}, {}
			local pivot = AimMotion.trajectoryAt(movement.traj, 0.05)
			for _, point in movement.traj do
				if point.tau <= 0.05 then table.insert(stop, point); table.insert(reverse, point); continue end
				local position = AimMotion.trajectoryAt(movement.traj, point.tau)
				local dx, dz = position.X - pivot.X, position.Z - pivot.Z
				table.insert(stop, {tau = point.tau, p = vector.new(pivot.X, position.Y, pivot.Z)})
				table.insert(reverse, {tau = point.tau, p = vector.new(pivot.X - dx, position.Y, pivot.Z - dz)})
			end
			local total = 0
			for _, hypothesis in hypotheses do total += hypothesis.w end
			table.insert(hypotheses, {kind = "airStop", w = total * 0.06, traj = stop})
			table.insert(hypotheses, {kind = "airRev", w = total * 0.06, traj = reverse})
		end
	end
	if anomaly and state.resolver == true and not (state.resSpikeOff == true and (anomaly.jumps or 0) >= 5) then
		hypotheses = AimPlan.resolve(environment, state, samples,
			{time = time, duration = duration, hypotheses = hypotheses, anomaly = anomaly})
	end
	state.lastAnom = anomaly
	local total = 0
	for _, hypothesis in hypotheses do total += hypothesis.w end
	if total > 0 then for _, hypothesis in hypotheses do hypothesis.w /= total end end
	return {hypotheses = hypotheses, anomaly = anomaly, frozen = frozen}
end

function AimPlan.raw(context, state, character, requireSight)
	local samples, environment = state.hist, context.environment
	if #samples < 8 or not character or not character.Parent then return nil end
	if requireSight and not AimTarget.visible(context, state, character) then return nil, nil, "no line of sight" end
	state.shift = AimCalibration.lagOf(context, state, character)
	local body = AimGeometry.bodyModel(environment, character)
	if not body then return nil end
	local parameters, time = AimTarget.moveParams(context), samples[#samples].t
	local latest, mode = samples[#samples], AimMotion.mode(environment, samples)
	local types, vector = environment.Enum.HumanoidStateType, environment.Vector3
	local airborne = latest.air or latest.st == types.Freefall or latest.st == types.Jumping
	local scale = airborne and not latest.air and state.tauSdKTake or nil
	if state.tauFastK then
		local speed = environment.typeof(latest.v) == "Vector3" and vector.new(latest.v.X, 0, latest.v.Z).Magnitude or 0
		local velocity = AimGeometry.kinVelocity(environment, samples, time, 0.08)
		if velocity then speed = math.max(speed, vector.new(velocity.X, 0, velocity.Z).Magnitude) end
		if speed > (state.tauFastV or 60) then scale = state.tauFastK end
	end
	local taus, started = AimCalibration.taus(context, state, mode, airborne, scale), environment.os.clock()
	local planDelay = math.clamp(started - time + (state.planDur or 0.01), 0, 0.15)
	for _, tau in ipairs(taus) do tau.tau += planDelay end
	local duration = taus[#taus].tau + 0.05
	local prepared = AimPlan.prepare(context, state, {samples = samples, time = time, duration = duration, parameters = parameters, mode = mode})
	local cloud = AimPlan.points(context, state, {samples = samples, time = time, mode = mode, hypotheses = prepared.hypotheses,
		taus = taus, planDelay = planDelay})
	local candidates = AimPlan.candidates(context, state, {samples = samples, time = time, mode = mode, hypotheses = prepared.hypotheses,
		taus = taus, cloud = cloud, body = body, anomaly = prepared.anomaly, frozen = prepared.frozen, duration = duration, character = character})
	local best, lines = AimPlan.select(context, state, {samples = samples, time = time, body = body, cloud = cloud,
		candidates = candidates, character = character})
	state.planDur = (state.planDur or 0.01) + (math.clamp(environment.os.clock() - started, 0, 0.1) - (state.planDur or 0.01)) * 0.3
	if not best then return nil, lines end
	best.hyps, best.samples, best.tNow, best.mode, best.anom = prepared.hypotheses, #cloud.points, time, mode, prepared.anomaly ~= nil
	best.spike = prepared.anomaly ~= nil and (prepared.anomaly.jumps or 0) >= 5
	local velocity = AimGeometry.kinVelocity(environment, samples, time, 0.08) or vector.zero
	local flatSpeed = 0
	if state.fastFlat ~= false and time - samples[1].t >= 0.3 then
		local previous = AimGeometry.posAt(samples, time - 0.3)
		for index = 1, 18 do
			local position = AimGeometry.posAt(samples, time - 0.3 + index / 60)
			if position and previous then flatSpeed += vector.new(position.X - previous.X, 0, position.Z - previous.Z).Magnitude end
			previous = position
		end
		flatSpeed /= 0.3
	end
	best.fast = mode == "nr" or velocity.Magnitude > 60 or flatSpeed > 26
	best.fresh = #samples < 2 or samples[#samples].t - samples[1].t < (state.freshSpan or 2.5)
	best.lowTrust = (state.lastTrust or 1) < (state.trustScaleT or 0.6)
	local index = #samples
	while index > 1 and (samples[index - 1].p - latest.p).Magnitude < 0.001 do index -= 1 end
	local frozen = time - samples[index].t
	if frozen >= 0.04 and frozen < 1.5 and vector.new(latest.md.X, 0, latest.md.Z).Magnitude > 0.5 then best.frozenT = frozen end
	best.air = (cloud.airborne or latest.st == types.Jumping or AimMotion.jumpTime(environment, state, samples, time) ~= nil) and not cloud.stuck
	best.takeoff = best.air and not (latest.air or latest.st == types.Climbing)
	return best, lines
end

function AimPlan.plan(context, state, character, requireSight)
	local original = state.hist
	local filtered = AimMotion.despike(context.environment, state, original)
	if filtered == original then return AimPlan.raw(context, state, character, requireSight) end
	if state.jitNoDespike ~= false and state.jitOn ~= false and #original > 0
		and AimMotion.jitterAxis(context, state, original, original[#original].t, state.L) then
		return AimPlan.raw(context, state, character, requireSight)
	end
	state.hist, state.histRaw = filtered, original
	local ok, result = pcall(AimPlan.raw, context, state, character, requireSight)
	state.hist, state.histRaw = original, nil
	if not ok then error(result, 0) end
	return result
end

local AimTiming = {}

function AimTiming.configure(state, settings)
	state.window, state.airPatT2, state.maxThink = 0.95, 0.9, 0.1
	state.quick100, state.q100Need, state.q100Floor = true, 0, 0
	state.hpMdS, state.mdsW, state.jFlagLead, state.landExtra = 0.6, 0.2, 0.05, 0.06
	state.spikeOn, state.spikeNeed, state.caW, state.caOthers = true, 2, 1, 0.25
	state.orbit, state.kinClamp, state.colFastMax, state.tiltFastMax = true, true, 40, 40
	state.fastWin, state.arc3VelK, state.tauFastK, state.oscMinRev = 1, 0.3, 0.75, 2
	state.jitOn, state.jitNoDespike, state.autoCalmV, state.autoCalmMax = true, false, 40, 4
	state.orbitFastMax, state.clusterRev = 60, 1
	AimTiming.apply(state, settings)
end

function AimTiming.apply(state, settings)
	state.quick100 = settings.shotTiming == "Instant"
	state.maxThink = state.quick100 and 0.1 or 0.8
end

function AimTiming.need(state, plan, elapsed)
	if state.quick100 == true then
		if plan and (plan.fast or plan.anom) and elapsed < (state.cheatWait or 0) then return state.cheatNeed or 0.6 end
		local initial = state.q100Need or 0.6
		return initial - (initial - (state.q100Floor or 0.15)) * math.clamp(elapsed / (state.q100Ramp or 0.07), 0, 1)
	end
	local initial = 0.55 - 0.3 * math.clamp(elapsed / 0.6, 0, 1) ^ 1.5
	if plan and plan.spike and state.spikeImm == true then return math.min(initial, 0.15) end
	if plan and (plan.anom or plan.fast) and state.anomImm == true then return math.min(initial, 0.15) end
	if plan and plan.takeoff and state.takeHold and elapsed < state.takeHold and not plan.fast and not plan.fresh then return 2 end
	if plan and plan.air and state.airPat ~= false and state.modern ~= false and not plan.fast and not plan.fresh then
		if elapsed < (plan.takeoff and state.takeT or state.airPatT or 0.8) then return state.airPatNeed or 0.95 end
		local second = state.airPatT2 == nil and 1.1 or state.airPatT2
		if second and elapsed < second then return state.airPatNeed2 or 0.6 end
		return 0
	end
	if plan and plan.frozenT and state.frzWait == true then
		if elapsed < (state.frzWaitT or 0.6) then return state.frzNeed or 0.95 end
		return 0
	end
	if plan and state.quick ~= false and not plan.fast and not plan.fresh then
		local need = state.modern == false and 0.6 or (state.quickNeed or 0.7)
		if plan.lowTrust and state.trustScale ~= false then need = state.quickNeedLie or 0.6 end
		if elapsed < (state.quickWin or 0.3) then return need end
		local second = state.quickWin2 == nil and 0.8 or state.quickWin2
		if second and elapsed < second then
			local value = math.min(need, state.quickNeed2 or 0.7)
			local target = state.quickRampTo == nil and 0.55 or state.quickRampTo
			if target then
				local first = state.quickWin or 0.3
				value += (math.min(target, value) - value) * math.clamp((elapsed - first) / math.max(second - first, 0.001), 0, 1)
			end
			return value
		end
		return 0
	end
	if plan and plan.air and state.airWait ~= false then
		if elapsed < 0.75 then return 0.9 end
		if elapsed < 0.9 then return 0.6 end
		return 0.25
	end
	if state.groundWait ~= false then
		if elapsed < 0.35 then return 0.9 end
		if elapsed < 0.7 then return 0.75 end
		if elapsed < 0.9 then return 0.5 end
		return 0.25
	end
	return initial
end

function AimTiming.pathSpeed(environment, samples, time, window)
	local previous, distance, first = nil, 0, nil
	for index = #samples, 1, -1 do
		local sample = samples[index]
		if time - sample.t > window then break end
		if previous then distance += environment.Vector3.new(previous.X - sample.p.X, 0, previous.Z - sample.p.Z).Magnitude end
		previous, first = sample.p, sample.t
	end
	if not first or time - first < 0.05 then return 0 end
	return distance / (time - first)
end

function AimTiming.autoNeed(context, state, plan, elapsed)
	local now, target = context.environment.os.clock(), state.target
	local desync = state.ds
	if desync and desync.backAt and (not target or desync.c == target) and now - desync.backAt < (state.dsWinT or 0.9) then
		return math.max(state.dsNeedMin or 0.3, (state.dsNeed or 0.6) - 2 * (now - desync.backAt))
	end
	local spoof = state.sp
	if spoof and not spoof.on and spoof.c == target and spoof.endAt and now - spoof.endAt < (state.spWinT or 0.9) then
		return math.max(state.dsNeedMin or 0.3, (state.dsNeed or 0.6) - 2 * (now - spoof.endAt))
	end
	if state.autoPatient == false then return AimTiming.need(state, plan, elapsed) end
	local quick = state.quick100
	state.quick100 = false
	local ok, need = pcall(AimTiming.need, state, plan, elapsed)
	state.quick100 = quick
	need = ok and need or 0.5
	if plan and (plan.fast or plan.anom) then need = math.max(need, state.autoFastNeed or 0.45) end
	if state.autoCalmV and elapsed < (state.autoCalmMax or 2.5) then
		local history = state.hist
		if history and #history > 1 and AimTiming.pathSpeed(context.environment, history, history[#history].t, state.autoCalmWin or 0.2) > state.autoCalmV then
			need = math.max(need, state.autoCalmNeed or 2)
		end
	end
	return need
end

local AimEngine = {}

function AimEngine.spoofVelocity(environment, state, character)
	if state.spoofOn == false or state.target ~= character or #state.hist < 4 then return nil end
	local samples, count, spoofed, velocity = state.hist, 0, 0, nil
	local time = samples[#samples].t
	for index = #samples, 1, -1 do
		local sample = samples[index]
		if time - sample.t > 0.35 then break end
		count += 1
		if sample.vr then spoofed, velocity = spoofed + 1, velocity or sample.vr end
	end
	if count < 4 or spoofed < count * 0.6 or not velocity then return nil end
	return velocity
end

function AimEngine.spoofPlan(context, state, plan, character)
	local velocity = AimEngine.spoofVelocity(context.environment, state, character)
	if not velocity or velocity.Magnitude < 1 then return nil end
	local highest, position
	for _, hypothesis in plan.hyps or {} do if not highest or hypothesis.w > highest.w then highest = hypothesis end end
	if highest and highest.traj then
		local ok, predicted = pcall(AimMotion.trajectoryAt, highest.traj, state.L + state.shiftL())
		if ok and context.environment.typeof(predicted) == "Vector3" then position = predicted end
	end
	if not position then
		local root = character:FindFirstChild("HumanoidRootPart")
		local direction = (plan.e - plan.o).Unit
		if root then position = plan.o + direction * (root.Position - plan.o):Dot(direction) end
	end
	if not position or position.X ~= position.X then return nil end
	local sign, distance = state.vsLock or state.vsNext or 1, math.clamp(velocity.Magnitude * (state.spoofT or 0.1), 80, 9000)
	local origin, direction = position + velocity.Unit * (distance * sign), velocity.Unit * -sign
	if AimTarget.innocentRisk(context, state, {origin = origin, direction = direction, distance = distance + 250, character = character}) then
		if state.say then state.say("SPOOF skip: innocent on line") end
		return nil
	end
	local result = table.clone(plan)
	result.o, result.e, result.kind, result.anom = origin, position + direction * 2, tostring(plan.kind) .. (sign > 0 and "+vs" or "-vs"), true
	if state.say then
		state.say(string.format("SPOOF v=(%.0f,%.0f,%.0f) sign=%d D=%.0f p=(%.1f,%.1f,%.1f)",
			velocity.X, velocity.Y, velocity.Z, sign, distance, position.X, position.Y, position.Z))
	end
	return result, sign
end

function AimEngine.send(context, state, plan, character, reason, clickTime)
	local environment = context.environment
	local gun = AimTarget.gun(context, state)
	local shoot = gun and gun:FindFirstChild("Shoot")
	if not shoot or plan.o.X ~= plan.o.X or plan.e.X ~= plan.e.X or (plan.e - plan.o).Magnitude < 0.3 then return false end
	local sign
	if context.configuration.sheriff.wallbang and state.spoofOn ~= false and state.spShots == true then
		local ok, alternate, alternateSign = pcall(AimEngine.spoofPlan, context, state, plan, character)
		if ok and alternate then plan, sign = alternate, alternateSign end
	end
	local globals, now = environment.getgenv(), environment.os.clock()
	pcall(function() globals.__pulseShotHold = now + (state.holdFor or 0.35) end)
	pcall(function()
		local antiAim = rawget(globals, "__pulseAntiAim")
		if antiAim and antiAim.desyncAt and environment.os.clock() - antiAim.desyncAt < 0.2 then
			globals.__pulseShotHold = math.max(tonumber(globals.__pulseShotHold) or 0, environment.os.clock() + 0.45)
		end
	end)
	local root, previous = character:FindFirstChild("HumanoidRootPart"), state.pending
	state.pending = {t = environment.os.clock(), o = plan.o, e = plan.e, kind = plan.kind, score = plan.score, c = character,
		reason = reason, tp = root and root.Position, manual = reason ~= "auto",
		tries = previous and previous.c == character and previous.retry and previous.tries + 1 or 0,
		mode = plan.mode or "std", shift = state.shiftL(), lag = state.shift, anom = plan.anom, vsSign = sign}
	local blocked = false
	pcall(function() blocked = AimTarget.muzzleState(context, state) end)
	local guard, geometry = AimTarget.guardInfo(context), nil
	pcall(function()
		local own = context.player.Character
		local ownRoot, head = own and own:FindFirstChild("HumanoidRootPart"), own and own:FindFirstChild("Head")
		local muzzle = ownRoot and ownRoot:FindFirstChild("GunRaycastAttachment")
		local camera, shift = context.workspace.CurrentCamera, AimTarget.selfShift(context, state)
		geometry = {hp = ownRoot and ownRoot.Position + shift, hl = ownRoot and ownRoot.CFrame.LookVector,
			hd = head and head.Position + shift, mz = muzzle and muzzle.WorldPosition + shift,
			cp = camera and camera.CFrame.Position, cl = camera and camera.CFrame.LookVector}
	end)
	local hook = globals.__pulseAim
	if hook then hook.bypass = true end
	local sent = pcall(function() shoot:FireServer(environment.CFrame.new(plan.o, plan.e), environment.CFrame.new(plan.e)) end)
	if hook then hook.bypass = false end
	if guard and guard.request and (guard.guarding or guard.blocked) then guard.request(state.mzAfter or 0.2) end
	if sent then
		state.stats.fired += 1
		local pending = state.pending
		if pending then
			pending.mz = guard and (guard.blocked and "noPose" or (guard.guarding and "guard" or "clear")) or (blocked and "blocked" or "clear")
			if guard and not guard.blocked then pending.mzAge = environment.os.clock() - (guard.clearSince or 0) end
			pending.me, pending.pre, pending.post = geometry, state.target == character and table.clone(state.hist) or {}, {}
			pending.pingS = AimCalibration.pingShift(context, state)
			pcall(function() pending.body = AimGeometry.bodyModel(environment, character, true) end)
			local connection
			connection = context.runService.Heartbeat:Connect(function()
				if environment.os.clock() - pending.t > 0.66 or not character.Parent then connection:Disconnect(); return end
				local sample = AimTarget.sample(context, state, character)
				if sample then table.insert(pending.post, sample) end
			end)
		end
	end
	pcall(function()
		local highest
		for _, hypothesis in plan.hyps or {} do if not highest or hypothesis.w > highest.w then highest = hypothesis end end
		if not highest or not state.pending then return end
		state.pending.top = highest.kind
		state.pending.pred = {AimMotion.trajectoryAt(highest.traj, 0), AimMotion.trajectoryAt(highest.traj, 0.05), AimMotion.trajectoryAt(highest.traj, 0.1)}
		local sample = state.hist[#state.hist]
		state.pending.info = sample and string.format("st=%s md=(%.2f,%.2f) ws=%.0f air=%s", tostring(sample.st.Name),
			sample.md.X, sample.md.Z, sample.ws, tostring(sample.air)) or "?"
	end)
	local path = -1
	pcall(function()
		local history = state.hist
		if state.target ~= character or #history <= 2 then return end
		local time, previous, distance = history[#history].t, nil, 0
		for index = 0, 30 do
			local position = AimGeometry.posAt(history, time - 3 + index * 0.1)
			if position and previous then distance += environment.Vector3.new(position.X - previous.X, 0, position.Z - previous.Z).Magnitude end
			previous = position
		end
		path = distance
	end)
	if state.pending then state.pending.mvPath = path end
	return sent
end

function AimEngine.fireFor(context, state, character, manual, requireSight)
	if not character or not character.Parent then return false end
	local environment, now = context.environment, context.environment.os.clock()
	local root = character:FindFirstChild("HumanoidRootPart")
	local own = context.player.Character
	local ownRoot = own and own:FindFirstChild("HumanoidRootPart")
	if root and ownRoot then
		local position = root.Position
		local far = position.X ~= position.X or position.Y ~= position.Y or position.Z ~= position.Z or position.Magnitude > (state.dsFar or 1000000)
		if far then
			if manual then state.dsArm = {c = character, untilT = now + (state.dsArmT or 1.2), needLos = requireSight} end
			return manual and true or false
		end
		local spoof = state.sp
		if spoof and spoof.on and spoof.c == character and now >= (state.spPassUntil or 0) then
			if manual then
				state.spArm = {c = character, untilT = now + (state.spArmT or 2), needLos = requireSight}
				return true
			end
			if now - spoof.since < (state.spAutoWait or 2.5) then return false end
		end
		if ((position - ownRoot.Position - AimTarget.selfShift(context, state)).Magnitude > (state.maxDist or 3000)
			or position.Y < context.workspace.FallenPartsDestroyHeight + 20) and root.AssemblyLinearVelocity.Magnitude > (state.flungSpeed or 350) then
			return false
		end
	end
	if state.pending and not state.pending.dead and now - state.pending.t < 0.5 or state.queued then return manual and true or false end
	local function queue()
		state.queued = {c = character, needLos = requireSight, clickAt = now, waitFrom = now, manual = true}
	end
	if state.target ~= character then
		state.forced, state.forcedAt = character, now
		if manual then queue(); return true end
		return false
	end
	if now - state.lastAccepted < state.cooldown then
		if manual then
			local remaining = state.cooldown - (now - state.lastAccepted)
			if remaining < (state.maxThink or 0.1) - 0.02 then queue() end
			return true
		end
		return false
	end
	local clear, retry = AimTarget.clearMuzzle(context, state)
	if not clear and retry then
		if manual then queue() end
		return manual
	end
	local plan = state.plan(character, requireSight)
	if not plan then
		if manual then queue() end
		return manual and true or false
	end
	local need = state.minAuto
	if not manual then
		if not state.autoFrom or now - (state.autoLast or 0) > 0.5 then state.autoFrom = now end
		state.autoLast = now
		need = AimTiming.autoNeed(context, state, plan, now - state.autoFrom)
	end
	if plan.score >= (manual and AimTiming.need(state, plan, 0) or need) then
		state.autoFrom = nil
		return state.send(plan, character, manual and "manual" or "auto", now)
	end
	if not manual then return false end
	queue()
	state.queued.plans, state.queued.bestScore = 1, plan.score
	return true
end

function AimEngine.flushQueue(context, state)
	local queued = state.queued
	if not queued then return end
	local now = context.environment.os.clock()
	queued.clickAt, queued.waitFrom = queued.clickAt or now, queued.waitFrom or queued.clickAt or now
	local deadline = queued.clickAt + (state.maxThink or 0.1)
	if queued.hardUntil then deadline = math.max(deadline, queued.hardUntil) end
	if now > deadline + (state.capGrace or 0.03) or not (queued.c and queued.c.Parent) or not AimTarget.gun(context, state) then
		state.queued = nil
		return
	end
	if now - state.lastAccepted < state.cooldown then queued.nr = (queued.nr or 0) + 1; return end
	if state.target ~= queued.c then queued.tm = (queued.tm or 0) + 1; return end
	local root = queued.c:FindFirstChild("HumanoidRootPart")
	if root then
		local position = root.Position
		if position.X ~= position.X or position.Y ~= position.Y or position.Z ~= position.Z or position.Magnitude > (state.dsFar or 1000000) then
			state.dsArm, state.queued = {c = queued.c, untilT = now + (state.dsArmT or 1.2), needLos = queued.needLos}, nil
			return
		end
	end
	local spoof = state.sp
	if spoof and spoof.on and spoof.c == queued.c and now >= (state.spPassUntil or 0) then
		state.spArm, state.queued = {c = queued.c, untilT = now + (state.spArmT or 2), needLos = queued.needLos}, nil
		return
	end
	local cost = (state.frameDt or 0.017) * 1.2 + (state.planCost or 0.006)
	local lastChance = now + cost > deadline
	if not AimTarget.clearMuzzle(context, state) then
		queued.mzWait = true
		local guard = AimTarget.guardInfo(context)
		local hardAt, hardAge = guard and guard.hardAt, state.mzHardAge or 0.13
		if hardAt and now - hardAt < hardAge then
			local hardUntil = math.min(queued.clickAt + (state.mzHardCap or 0.22), hardAt + hardAge)
			if not queued.hardUntil or hardUntil > queued.hardUntil then
				queued.hardUntil = hardUntil
				deadline = math.max(deadline, hardUntil)
				lastChance = now + cost > deadline
			end
		end
		if not lastChance then queued.mzw = (queued.mzw or 0) + 1; return end
		queued.mzForced = true
	end
	if queued.nextPlan and now < queued.nextPlan and not lastChance then return end
	queued.nextPlan = now + (state.planGap or 0.035)
	local started = context.environment.os.clock()
	local plan = state.plan(queued.c, queued.needLos)
	state.planCost = (state.planCost or 0.006) + (math.min(context.environment.os.clock() - started, 0.05) - (state.planCost or 0.006)) * 0.2
	queued.plans = (queued.plans or 0) + 1
	if not plan then return end
	queued.bestScore = queued.bestScore and math.max(queued.bestScore, plan.score) or plan.score
	local elapsed = now - queued.clickAt
	if lastChance or plan.score >= AimTiming.need(state, plan, elapsed)
		or now + (state.planGap or 0.035) > deadline and plan.score >= (state.lastChance or 0.1) then
		state.queued = nil
		state.send(plan, queued.c, queued.mzForced and "queue-mz" or "queue", queued.clickAt)
	end
end

function AimEngine.record(context, state, shot, result, endpoint, part)
	local environment = context.environment
	environment.task.delay(0.7, function()
		pcall(function()
			local function jsonVector(value)
				if environment.typeof(value) ~= "Vector3" then return "null" end
				return string.format("[%.3f,%.3f,%.3f]", value.X, value.Y, value.Z)
			end
			local geometry = shot.me
			local me = "null"
			if geometry then
				me = string.format('{"hp":%s,"hl":%s,"hd":%s,"mz":%s,"cp":%s,"cl":%s}',
					jsonVector(geometry.hp), jsonVector(geometry.hl), jsonVector(geometry.hd),
					jsonVector(geometry.mz), jsonVector(geometry.cp), jsonVector(geometry.cl))
			end
			local rows = {}
			for _, sample in AimCalibration.shotTrajectory(shot) do
				if sample.t < shot.t - 0.6 or sample.t > shot.t + 0.65 then continue end
				local velocity = environment.typeof(sample.v) == "Vector3" and sample.v or environment.Vector3.zero
				table.insert(rows, string.format("[%.4f,%.3f,%.3f,%.3f,%.2f,%.2f,%.1f,%d,%s,%s,%.1f,%.1f,%.1f]",
					sample.t - shot.t, sample.p.X, sample.p.Y, sample.p.Z, sample.md.X, sample.md.Z, sample.ws,
					sample.st.Value, sample.air and "1" or "0", sample.jump and "1" or "0", velocity.X, velocity.Y, velocity.Z))
			end
			local line = string.format('{"t":%.3f,"kind":"%s","score":%.3f,"res":"%s","part":"%s","o":[%.3f,%.3f,%.3f],"e":[%.3f,%.3f,%.3f],"end":%s,"L":%.3f,"D":%.3f,"k":%.2f,"emd":%.2f,"ecv":%.2f,"reason":"%s","ping":%.3f,"mz":"%s","mv":%.1f,"me":%s,"rows":[%s]}',
				shot.t, tostring(shot.kind), shot.score or -1, result, tostring(part), shot.o.X, shot.o.Y, shot.o.Z,
				shot.e.X, shot.e.Y, shot.e.Z, jsonVector(endpoint), state.L, state.D, state.k, state.err.md, state.err.cv,
				tostring(shot.reason), context.player:GetNetworkPing(), tostring(shot.mz), shot.mvPath or -1, me, table.concat(rows, ","))
			local file = "PulseHub/aimx_shots.jsonl"
			if environment.isfile(file) then environment.appendfile(file, line .. "\n"); return end
			environment.writefile(file, line .. "\n")
		end)
	end)
end

function AimEngine.onFired(context, state, tool, start, endpoint, hit)
	local environment = context.environment
	local shot, own = state.pending, false
	pcall(function()
		own = environment.typeof(tool) == "Instance" and context.player.Character ~= nil and tool:IsDescendantOf(context.player.Character)
	end)
	if not own then return end
	local previous = state.trueAccepted or 0
	state.lastAccepted = environment.os.clock() - state.echo * 0.5
	state.trueAccepted = state.lastAccepted
	if not shot then return end
	state.pending = nil
	local latency = environment.os.clock() - shot.t
	state.echo += (latency - state.echo) * 0.3
	state.lastAccepted, state.trueAccepted = shot.t, shot.t
	local gap = shot.t - previous
	pcall(function()
		if environment.typeof(start) ~= "Vector3" or environment.typeof(endpoint) ~= "Vector3" or not shot.o or not shot.e then return end
		local path = shot.e - shot.o
		if path.Magnitude <= 0.01 then return end
		local direction, delta = path.Unit, endpoint - shot.o
		if (delta - direction * delta:Dot(direction)).Magnitude > 1.5 or (start - shot.o).Magnitude > 0.5 then shot.offLine = true end
	end)
	if gap > 2.4 and gap < state.cooldown + 0.25 then
		state.cdLo = math.max((state.cdLo or 0) - 0.001, 0)
		state.cooldown = math.clamp(math.max(math.min(state.cooldown, gap) - 0.04, state.cdLo), state.cdFloor, 5.5)
	end
	local success = environment.typeof(hit) == "Instance" and shot.c and hit:IsDescendantOf(shot.c)
	if success then
		state.stats.hit += 1
		local history, direction = state.hist, (shot.e - shot.o).Unit
		local velocity = AimGeometry.kinVelocity(environment, history, math.min(shot.t, history[#history] and history[#history].t or shot.t), 0.1)
			or environment.Vector3.zero
		local horizontal = environment.Vector3.new(velocity.X, 0, velocity.Z)
		local core = hit.Name == "HumanoidRootPart" or hit.Name == "UpperTorso" or hit.Name == "LowerTorso" or hit.Name == "Torso"
		local estimate, error
		if (core and horizontal.Magnitude > 8 or velocity.Magnitude > 40) and velocity.Magnitude > 1
			and math.abs(velocity.Unit:Dot(direction)) > 0.6 and environment.typeof(endpoint) == "Vector3" and #history > 0 then
			local last = history[#history]
			for offset = -0.15, 0.45, 0.005 do
				local time = shot.t + offset
				local position = time <= last.t and AimGeometry.posAt(history, time) or last.p + velocity * (time - last.t)
				if not position then continue end
				local delta = endpoint + direction * 0.5 - position
				local perpendicular = (delta - direction * delta:Dot(direction)).Magnitude
				local current = math.abs(delta:Dot(direction)) + perpendicular * 0.25
				if not error or current < error then error, estimate = current, offset end
			end
		end
		local accepted = estimate and error and error < (core and 2 or 3) and estimate or nil
		environment.task.delay(0.7, function()
			if shot.anom or (shot.mvPath or 0) >= (state.tauMissMaxV or 60) and state.tauFastHits ~= true then return end
			local ok = pcall(AimCalibration.tauEvidence, context, state, shot, true, endpoint)
			if ok and state.tauShots then AimCalibration.save(context, state) end
		end)
		if accepted and shot.mode == "nr" then
			state.nrCalib = (state.nrCalib or 0) + 1
			local difference = estimate - state.Lnr - AimCalibration.pingShift(context, state) - (shot.shift or 0)
			state.Lnr = math.clamp(state.Lnr + difference * (state.nrCalib < 4 and 0.4 or 0.15), 0, 0.5)
			state.Lnrsd = math.clamp(state.Lnrsd + (math.abs(difference) - state.Lnrsd) * 0.2, 0.03, 0.12)
		end
		if accepted and shot.mode ~= "nr" and not shot.anom then state.calib = (state.calib or 0) + 1 end
		if shot.vsSign then state.vsLock, state.vsNext, state.vsMiss = shot.vsSign, nil, 0 end
		AimCalibration.save(context, state)
		AimEngine.record(context, state, shot, "hit", endpoint, hit.Name)
		return
	end
	state.stats.miss += 1
	if shot.vsSign then
		if state.vsLock then
			state.vsMiss = (state.vsMiss or 0) + 1
			if state.vsMiss >= 3 then
				state.vsLock, state.vsMiss, state.vsNext = nil, 0, -shot.vsSign
				AimCalibration.save(context, state)
			end
		end
		if not state.vsLock then state.vsNext = -shot.vsSign end
	end
	AimEngine.record(context, state, shot, "miss", endpoint, environment.typeof(hit) == "Instance" and hit:GetFullName() or "nil")
	environment.task.delay(0.7, function()
		if shot.anom or (shot.mvPath or 0) >= (state.tauMissMaxV or 60) then return end
		local ok = pcall(AimCalibration.tauEvidence, context, state, shot, false, endpoint)
		if ok and state.tauShots then AimCalibration.save(context, state) end
	end)
end

function AimEngine.tick(context, state, runtime)
	if not state.alive then return end
	local environment = context.environment
	local now = environment.os.clock()
	if not state.forced and not state.queued then
		local gun = false
		pcall(function()
			local character, backpack = context.player.Character, context.player:FindFirstChildOfClass("Backpack")
			gun = character ~= nil and character:FindFirstChild("Gun") ~= nil or backpack ~= nil and backpack:FindFirstChild("Gun") ~= nil
		end)
		if not gun then
			if now - runtime.lastNoGun < 0.1 then return end
			runtime.lastNoGun = now
		end
	end
	pcall(function() AimTarget.track(context, state, context.roles()) end)
	now = environment.os.clock()
	local spoof = state.sp
	if spoof.c ~= state.target then spoof.c, spoof.on, spoof.since, spoof.endAt = state.target, false, 0, nil end
	if state.target then
		local ok, velocity = pcall(AimEngine.spoofVelocity, environment, state, state.target)
		local active = ok and environment.typeof(velocity) == "Vector3"
		if active and not spoof.on then spoof.on, spoof.since = true, now end
		if not active and spoof.on then spoof.on, spoof.endAt = false, now end
	end
	local armed = state.spArm
	if armed then
		if not armed.c or not armed.c.Parent then state.spArm = nil end
		if armed.c and armed.c.Parent and state.target == armed.c and not state.queued and #state.hist >= 8 then
			local active = spoof.on and spoof.c == armed.c
			if active and now > armed.untilT or not active then
				state.spArm, state.spPassUntil = nil, now + 0.6
				pcall(state.fireFor, armed.c, true, armed.needLos)
			end
		end
	end
	armed = state.dsArm
	if armed then
		if now > armed.untilT or not armed.c or not armed.c.Parent then state.dsArm = nil end
		local desync = state.ds
		if state.dsArm and state.target == armed.c and desync and desync.backAt and desync.c == armed.c
			and now - desync.backAt < (state.dsWinT or 0.9) and #state.hist >= 8 and not state.queued then
			state.dsArm = nil
			pcall(state.fireFor, armed.c, true, armed.needLos)
		end
	end
	if state.hbLast then state.frameDt = (state.frameDt or 0.017) + (math.clamp(now - state.hbLast, 0.004, 0.2) - (state.frameDt or 0.017)) * 0.1 end
	state.hbLast = now
	if not state.btCo and state.target and now - runtime.lastBacktest > 0.1 then
		runtime.lastBacktest = now
		local samples, parameters = table.clone(state.hist), AimTarget.moveParams(context)
		state.btCo = coroutine.create(function()
			state.btSlice = true
			pcall(AimMotion.backtest, context, state, samples, parameters)
			state.btSlice = false
		end)
	end
	if state.btCo then
		state.btT = environment.os.clock()
		local ok = coroutine.resume(state.btCo)
		if not ok or coroutine.status(state.btCo) == "dead" then state.btCo, state.btSlice = nil, false end
	end
	if state.target and now - runtime.lastFactor > 0.5 then
		runtime.lastFactor = now
		local ok, factor = pcall(AimMotion.speedFactor, environment, state, state.hist)
		if ok then state.k = factor end
	end
	if state.queued then pcall(AimEngine.flushQueue, context, state) end
	if state.autoRotateRestore and now > (state.autoRotateUntil or 0) then
		pcall(function() state.autoRotateRestore.AutoRotate = true end)
		state.autoRotateRestore = nil
	end
	local pending = state.pending
	if pending and not pending.dead and now - pending.t > math.max(0.28, state.echo * 2.5, context.player:GetNetworkPing() * 2.2 + 0.15) then
		pending.dead = true
		local gap, character = pending.t - (state.trueAccepted or 0), context.player.Character
		local holding = character ~= nil and character:FindFirstChild("Gun") ~= nil
		if holding and gap < state.cooldown + 0.6 and not pending.anom then
			if gap > 2.4 then state.cdLo = math.max(state.cdLo or 0, gap + 0.03) end
			state.cooldown = math.clamp(math.max(state.cooldown + 0.1, state.cdLo or 0), state.cdFloor, 5.5)
			state.lastAccepted = math.max(state.lastAccepted, now - state.cooldown + 0.12)
		end
		if pending.manual and pending.tries < 3 and pending.c and pending.c.Parent then
			pending.retry = true
			state.queued = {c = pending.c, needLos = false, clickAt = now, waitFrom = now, manual = true, hardUntil = now + 0.25}
		end
	end
	if state.forced and (not state.forced.Parent or now - (state.forcedAt or now) > 5) then state.forced = nil end
end

function AimEngine.install(context)
	local environment = context.environment
	local globals = environment.getgenv()
	local previous = rawget(globals, "__pulseAimXPrev")
	if type(previous) == "table" and type(previous.stop) == "function" then pcall(previous.stop) end
	local state, runtime = AimCalibration.create(), {lastNoGun = 0, lastBacktest = 0, lastFactor = 0}
	AimTiming.configure(state, context.configuration.sheriff)
	state.sp = {c = nil, on = false, since = 0, endAt = nil}
	state.posAt = AimGeometry.posAt
	state.hitsBody = function(body, position, origin, direction, distance)
		return AimGeometry.hitsBody(environment, {body = body, position = position, origin = origin, direction = direction, distance = distance})
	end
	state.say = function(message)
		table.insert(state.log, environment.os.date("%H:%M:%S") .. " " .. message)
		if #state.log > 300 then table.remove(state.log, 1) end
		if not state.logFile then return end
		pcall(function()
			local line = state.log[#state.log] .. "\n"
			if environment.isfile(state.logFile) then environment.appendfile(state.logFile, line); return end
			environment.writefile(state.logFile, line)
		end)
	end
	state.stop = function()
		state.alive = false
		if state.stopInventory then state.stopInventory() end
		for _, connection in state.conns do pcall(function() connection:Disconnect() end) end
	end
	state.plan = function(character, requireSight) return AimPlan.plan(context, state, character, requireSight) end
	state.send = function(plan, character, reason, clickTime) return AimEngine.send(context, state, plan, character, reason, clickTime) end
	state.fireFor = function(character, manual, requireSight) return AimEngine.fireFor(context, state, character, manual, requireSight) end
	state.applyTiming = function() AimTiming.apply(state, context.configuration.sheriff) end
	state.ready = function() return environment.os.clock() - state.lastAccepted >= state.cooldown end
	state.muzzleState = function() return AimTarget.muzzleState(context, state) end
	state.spActive = function(character) return state.sp.on and state.sp.c == character end
	state.spHot = function(character)
		return not state.sp.on and state.sp.c == character and state.sp.endAt ~= nil and environment.os.clock() - state.sp.endAt < (state.spWinT or 0.9)
	end
	state.dsHot = function(character)
		return state.ds ~= nil and state.ds.backAt ~= nil and (not character or state.ds.c == character)
			and environment.os.clock() - state.ds.backAt < (state.dsWinT or 0.9)
	end
	globals.__pulseAimX, globals.__pulseAimXPrev = state, state
	AimCalibration.load(context, state)
	pcall(function()
		if environment.isfolder and environment.makefolder and not environment.isfolder("PulseHub") then environment.makefolder("PulseHub") end
	end)
	state.logFile = "PulseHub/aimx_log.txt"
	environment.task.spawn(function()
		local services = context.storage:WaitForChild("ClientServices", 10)
		local weapons = services and services:WaitForChild("WeaponService", 10)
		local fired = weapons and weapons:WaitForChild("GunFired", 10)
		if not fired or not state.alive then return end
		table.insert(state.conns, context.connect(fired.OnClientEvent, function(...)
			if state.alive then pcall(AimEngine.onFired, context, state, ...) end
		end))
	end)
	table.insert(state.conns, context.connect(context.runService.Heartbeat, function() AimEngine.tick(context, state, runtime) end))
	pcall(state.say, string.format("start L=%.3f sd=%.3f D=%.3f", state.L, state.Lsd, state.D))
	return state
end

local SheriffRuntime = {}

function SheriffRuntime.target(context)
	local globals = context.environment.getgenv()
	local duel = rawget(globals, "__pulseDuel")
	if duel and duel.on then return duel.pick() end
	local roles = context.roles()
	for _, player in context.players:GetPlayers() do
		local role, character = roles[player.Name], player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		if role and role.Role == "Murderer" and role.Dead ~= true and role.Killed ~= true and humanoid and humanoid.Health > 0 then
			return player, character
		end
	end
	return nil
end

function SheriffRuntime.gun(context)
	local character = context.player.Character
	local equipped = character and character:FindFirstChild("Gun")
	if equipped then return equipped end
	local backpack = context.player:FindFirstChild("Backpack")
	return backpack and backpack:FindFirstChild("Gun") or nil
end

function SheriffRuntime.flick(context, character, manual)
	if context.flicking or not SheriffRuntime.gun(context) then return end
	local environment, settings = context.environment, context.configuration.sheriff
	local camera = context.workspace.CurrentCamera
	local function target() return character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Head") end
	if not camera or not target() then return end
	context.flicking = true
	local own = context.player.Character
	local root = own and own:FindFirstChild("HumanoidRootPart")
	local cameraStart, rootStart = camera.CFrame, root and root.CFrame
	local duration = math.clamp(0.35 - (settings.flickSpeed or 6) * 0.032, 0.02, 0.35)
	if settings.flickMouse ~= false and type(environment.mousemoverel) == "function" then
		local inset = environment.Vector2.new(0, 36)
		pcall(function() inset = context.guiService:GetGuiInset() end)
		local steps, movedX, movedY = math.max(1, math.floor(duration / 0.008)), 0, 0
		for index = 1, steps do
			if not context.isAlive() then context.flicking = false; return end
			local part = target()
			if not part then break end
			local position, visible = camera:WorldToViewportPoint(part.Position)
			if not visible then break end
			local x, y
			if context.inputService.MouseBehavior == environment.Enum.MouseBehavior.LockCenter then
				x, y = camera.ViewportSize.X * 0.5, camera.ViewportSize.Y * 0.5
			end
			if not x then
				local mouse = context.inputService:GetMouseLocation()
				x, y = mouse.X, mouse.Y - inset.Y
			end
			local remaining = steps - index + 1
			local dx, dy = (position.X - x) / remaining, (position.Y - y) / remaining
			movedX, movedY = movedX + dx, movedY + dy
			pcall(environment.mousemoverel, dx, dy)
			context.runService.RenderStepped:Wait()
		end
		if not context.isAlive() then context.flicking = false; return end
		pcall(context.aim.fireFor, character, manual, not settings.wallbang)
		local fraction = math.clamp((settings.flickReturn or 100) / 100, 0, 1)
		if fraction > 0 and (movedX ~= 0 or movedY ~= 0) then
			for _ = 1, steps do
				if not context.isAlive() then context.flicking = false; return end
				pcall(environment.mousemoverel, -movedX * fraction / steps, -movedY * fraction / steps)
				context.runService.RenderStepped:Wait()
			end
		end
		context.flicking = false
		return
	end
	local started = environment.os.clock()
	while environment.os.clock() - started < duration do
		if not context.isAlive() then context.flicking = false; return end
		local alpha, part = (environment.os.clock() - started) / duration, target()
		if part then
			pcall(function() camera.CFrame = cameraStart:Lerp(environment.CFrame.lookAt(cameraStart.Position, part.Position), alpha) end)
			if root and root.Parent then
				pcall(function()
					local point = environment.Vector3.new(part.Position.X, root.Position.Y, part.Position.Z)
					root.CFrame = root.CFrame:Lerp(environment.CFrame.lookAt(root.Position, point), alpha)
				end)
			end
		end
		context.runService.RenderStepped:Wait()
	end
	local part = target()
	if part then
		pcall(function() camera.CFrame = environment.CFrame.lookAt(cameraStart.Position, part.Position) end)
		if root and root.Parent then
			pcall(function() root.CFrame = environment.CFrame.lookAt(root.Position, environment.Vector3.new(part.Position.X, root.Position.Y, part.Position.Z)) end)
		end
	end
	if not context.isAlive() then context.flicking = false; return end
	pcall(context.aim.fireFor, character, manual, not settings.wallbang)
	local fraction, current = math.clamp((settings.flickReturn or 100) / 100, 0, 1), camera.CFrame
	local destination, rootCurrent = current:Lerp(cameraStart, fraction), root and root.CFrame
	local rootDestination = rootStart and rootCurrent and rootCurrent:Lerp(rootStart, fraction) or nil
	started = environment.os.clock()
	while environment.os.clock() - started < duration do
		if not context.isAlive() then context.flicking = false; return end
		local alpha = (environment.os.clock() - started) / duration
		pcall(function() camera.CFrame = current:Lerp(destination, alpha) end)
		if root and root.Parent and rootDestination and rootCurrent then pcall(function() root.CFrame = rootCurrent:Lerp(rootDestination, alpha) end) end
		context.runService.RenderStepped:Wait()
	end
	pcall(function() camera.CFrame = destination end)
	context.flicking = false
end

function SheriffRuntime.installHooks(context)
	local environment, settings = context.environment, context.configuration.sheriff
	local globals = environment.getgenv()
	globals.__pulseAim = globals.__pulseAim or {}
	local aim = globals.__pulseAim
	aim.on, aim.silent = not not settings.silentAim, false
	if not aim.origFire then
		pcall(function()
			local remote = environment.Instance.new("RemoteEvent")
			aim.origFire = remote.FireServer
			remote:Destroy()
		end)
	end
	local function targetPart()
		local duel = rawget(globals, "__pulseDuel")
		if duel and duel.on then
			local _, character = duel.pick(globals.__pulseFovOK)
			if not character then _, character = duel.pick() end
			return character and (character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Head")) or nil
		end
		local _, character = SheriffRuntime.target(context)
		return character and (character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Head")) or nil
	end
	aim.fire = function()
		local part = targetPart()
		local character = part and part.Parent
		if not character then return false end
		if globals.__pulseFovOK and not globals.__pulseFovOK(character) then return false end
		if settings.aimType == "Flick Shot" then
			if context.flicking then return true end
			if not SheriffRuntime.gun(context) or not context.workspace.CurrentCamera then return true end
			environment.task.spawn(SheriffRuntime.flick, context, character, true)
			return true
		end
		context.aim.fireFor(character, true, not settings.wallbang)
		return true
	end
	local pointInstalled = false
	local function installPointHook()
		if pointInstalled then return true end
		pcall(function()
			local services = context.storage:FindFirstChild("ClientServices")
			local module = services and services:FindFirstChild("WeaponService")
			local weapons = module and environment.require(module)
			if not weapons then return end
			local saved = globals.__pulseAimWS
			if saved and weapons.GetTargetPosition == saved.tgt and weapons.GetMouseTargetCFrame == saved.mouse then pointInstalled = true; return end
			local oldMouse, oldTarget = weapons.GetMouseTargetCFrame, weapons.GetTargetPosition
			local function position()
				if globals.__pulseAim and globals.__pulseAim.bypass then return nil end
				if not settings.silentAim and settings.aimType ~= "Flick Shot" then return nil end
				local own = context.player.Character
				if not own or not own:FindFirstChild("Gun") then return nil end
				local part = targetPart()
				if not part or globals.__pulseFovOK and not globals.__pulseFovOK(part.Parent) then return nil end
				return part.Position
			end
			local function mouse(self, ...)
				local point = position()
				if point then return environment.CFrame.new(point) end
				return oldMouse(self, ...)
			end
			local function target(self, ...)
				local point = position()
				if point then return environment.CFrame.new(point) end
				return oldTarget(self, ...)
			end
			weapons.GetMouseTargetCFrame, weapons.GetTargetPosition = mouse, target
			if weapons.GetMouseTargetCFrame == mouse and weapons.GetTargetPosition == target then
				globals.__pulseAimWS, pointInstalled = {mouse = mouse, tgt = target, origMouse = oldMouse, origTarget = oldTarget}, true
			end
		end)
		return pointInstalled
	end
	aim.installAimPointHook = installPointHook
	globals.__pulseAimGen = (globals.__pulseAimGen or 0) + 1
	local generation, hooked, notifiedLimited, notifiedMissing = globals.__pulseAimGen, false, false, false
	globals.__pulseHookCapOK = function()
		if type(environment.newcclosure) ~= "function" then return false end
		local original = function(value) return value end
		local ok, wrapper = pcall(environment.newcclosure, original)
		if not ok or type(wrapper) ~= "function" or wrapper == original then return false end
		local success, slow = pcall(function()
			local started = environment.os.clock()
			for index = 1, 3000 do wrapper(index) end
			return environment.os.clock() - started > 0.15
		end)
		return not (success and slow)
	end
	local function installNamecall()
		if hooked then return true end
		if type(environment.hookmetamethod) ~= "function" or type(environment.getnamecallmethod) ~= "function"
			or type(environment.newcclosure) ~= "function" or not globals.__pulseHookCapOK() then return false end
		local original
		local function intercept(remote, ...)
			local ok, method = pcall(environment.getnamecallmethod)
			local current = globals.__pulseAim
			if not ok or method ~= "FireServer" or not current or not current.on or current.bypass or globals.__pulseAimGen ~= generation then
				return original(remote, ...)
			end
			if environment.checkcaller then
				local success, caller = pcall(environment.checkcaller)
				if success and caller then return original(remote, ...) end
			end
			local handled = false
			pcall(function()
				if remote.Name == "Shoot" and remote.Parent and remote.Parent.Name == "Gun" and current.fire then handled = current.fire(remote) end
			end)
			if handled then return end
			return original(remote, ...)
		end
		local ok = pcall(function() original = environment.hookmetamethod(environment.__hostGame or environment.game, "__namecall", environment.newcclosure(intercept)) end)
		hooked = hooked or ok
		return hooked
	end
	local function notify(message)
		pcall(function() environment.game:GetService("StarterGui"):SetCore("SendNotification", {Title = "MYTHIC HUB", Text = message, Duration = 7}) end)
	end
	aim.installHook = function()
		if installNamecall() then return true end
		if installPointHook() then
			if not notifiedLimited then notifiedLimited = true; notify("Silent Aim: limited mode (V2 only, no Wallbang)") end
			return true
		end
		if not notifiedMissing then notifiedMissing = true; notify("Silent Aim: executor has no hooks. Use Auto Shoot.") end
		return false
	end
	if settings.silentAim then aim.installHook() end
end

function SheriffRuntime.install(context)
	local environment, settings = context.environment, context.configuration.sheriff
	local globals = environment.getgenv()
	local function allowed(character) return FovTarget.isInside({configuration = context.configuration, workspace = context.workspace}, character) end
	local function shoot(character, manual)
		if not character or not allowed(character) then return end
		if settings.aimType == "Flick Shot" then environment.task.spawn(SheriffRuntime.flick, context, character, manual); return end
		context.aim.fireFor(character, manual, not settings.wallbang)
	end
	globals.__pulseShotMurder = function()
		if not SheriffRuntime.gun(context) then return false end
		local player, character = SheriffRuntime.target(context)
		if not player or not character or not allowed(character) then return false end
		local own = context.player.Character
		local humanoid = own and own:FindFirstChildOfClass("Humanoid")
		if humanoid and not own:FindFirstChild("Gun") then
			local backpack = context.player:FindFirstChild("Backpack")
			local gun = backpack and backpack:FindFirstChild("Gun")
			if gun then
				pcall(function() humanoid:EquipTool(gun) end)
				local started = environment.os.clock()
				while not own:FindFirstChild("Gun") and environment.os.clock() - started < 0.5 do context.runService.Heartbeat:Wait() end
			end
		end
		if settings.aimType == "Universal" then
			local camera = context.workspace.CurrentCamera
			local target = character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Head")
			if camera and target then pcall(function() camera.CFrame = environment.CFrame.lookAt(camera.CFrame.Position, target.Position) end) end
			context.aim.fireFor(character, true, not settings.wallbang)
			return true
		end
		shoot(character, true)
		return true
	end
	context.connect(context.runService.RenderStepped, function()
		local own = context.player.Character
		if settings.aimType ~= "Universal" or not own or not own:FindFirstChild("Gun") then return end
		local _, character = SheriffRuntime.target(context)
		local target = character and (character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Head"))
		local camera = context.workspace.CurrentCamera
		if target and camera then pcall(function() camera.CFrame = camera.CFrame:Lerp(environment.CFrame.lookAt(camera.CFrame.Position, target.Position), 0.65) end) end
	end)
	local lastAuto, lastEquip = 0, 0
	context.connect(context.runService.Heartbeat, function()
		local aim, now = context.aim, environment.os.clock()
		local rapid = aim.alive and (aim.dsHot(aim.target) or aim.spHot(aim.target) or aim.spActive(aim.target))
		local interval = rapid and (aim.dsGap or 0.04) or 0.2
		if not settings.autoShoot or now - lastAuto <= interval or not SheriffRuntime.gun(context) then return end
		local player, character = SheriffRuntime.target(context)
		if not player or not character then return end
		pcall(function()
			if not allowed(character) then return end
			local own = context.player.Character
			local humanoid = own and own:FindFirstChildOfClass("Humanoid")
			if not humanoid or not (humanoid.Health > 0) then return end
			if not own:FindFirstChild("Gun") then
				if now - lastEquip < 1 then return end
				if not settings.wallbang then
					local head, root = own:FindFirstChild("Head"), character:FindFirstChild("HumanoidRootPart")
					if not head or not root then return end
					local parameters = environment.RaycastParams.new()
					parameters.FilterType, parameters.FilterDescendantsInstances = environment.Enum.RaycastFilterType.Exclude, {own, character}
					local hit = context.workspace:Raycast(head.Position, root.Position - head.Position, parameters)
					if hit and hit.Instance.Transparency < 0.9 then return end
				end
				local backpack = context.player:FindFirstChild("Backpack")
				local gun = backpack and backpack:FindFirstChild("Gun")
				if not gun then return end
				lastEquip = environment.os.clock()
				humanoid:EquipTool(gun)
				return
			end
			shoot(character, false)
		end)
		lastAuto = now
	end)
	SheriffRuntime.installHooks(context)
end

local FarmState = {}
local FARM_ROLES = {Zombie = true, Survivor = true, Vampire = true, Hunter = true, Villager = true, Freezer = true, Runner = true}
local FARM_OPTIONS = {"respawnOnFull", "antiFling", "avoidMurder", "autoFling", "killAura", "pickGun", "shootOnFull"}

function FarmState.create(context)
	local state = {enabled = false, round = false, active = false, session = 0, total = 0, rate = 0,
		respawnOnFull = false, antiFling = false, avoidMurder = false, autoFling = false, killAura = true,
		pickGun = true, shootOnFull = true, collected = {}, full = {}, skipped = {}, containers = {}, containersAt = 0,
		bagItems = {}, bagAt = 0, coinsStarted = 0, knownRound = false, forcedFull = false}
	local environment = context.environment
	pcall(function()
		if not environment.isfile or not environment.readfile or not environment.isfile("PulseFarmMM2_config.json") then return end
		local saved = context.http:JSONDecode(environment.readfile("PulseFarmMM2_config.json"))
		if type(saved) ~= "table" then return end
		for _, key in FARM_OPTIONS do if type(saved[key]) == "boolean" then state[key] = saved[key] end end
		if type(saved.totalCoins) == "number" then state.total = saved.totalCoins end
	end)
	return state
end

function FarmState.save(context, state)
	if type(context.environment.writefile) ~= "function" then return end
	pcall(function()
		local saved = {totalCoins = state.total}
		for _, key in FARM_OPTIONS do saved[key] = state[key] end
		context.environment.writefile("PulseFarmMM2_config.json", context.http:JSONEncode(saved))
	end)
end

function FarmState.roundStart(state, now, coinsStarted)
	state.round, state.skipped = true, {}
	if not coinsStarted then state.collected = {} end
	table.clear(state.full)
	state.forcedFull, state.coinsStarted, state.knownRound = false, coinsStarted and now or 0, true
	if not coinsStarted then state.active = false end
end

function FarmState.roundEnd(state)
	state.round, state.active, state.collected, state.forcedFull, state.coinsStarted = false, false, {}, false, 0
end

function FarmState.coinCollected(state, now, kind, count, maximum)
	if state.coinsStarted == 0 then state.coinsStarted = now end
	if type(kind) == "string" and type(count) == "number" and type(maximum) == "number" then state.full[kind] = count >= maximum end
	local name = type(kind) == "string" and kind or "Coin"
	local previous = state.collected[name] or 0
	if not state.knownRound and state.collected[name] == nil and type(count) == "number" and count > 1 then previous = count - 1 end
	local added = 1
	if type(count) == "number" then
		added = count >= previous and count - previous or count
		state.collected[name] = count
	end
	if added > 0 then state.session, state.total, state.dirty = state.session + added, state.total + added, true end
	return added
end

function FarmState.refreshBag(context, state)
	local gui = context.player:FindFirstChild("PlayerGui")
	local main = gui and gui:FindFirstChild("MainGUI")
	local game = main and main:FindFirstChild("Game")
	state.bag, state.bagItems = game and game:FindFirstChild("CoinBags") or nil, {}
	if not state.bag then return end
	for _, item in state.bag:GetDescendants() do
		if item.Name == "Full" and item:IsA("GuiObject") and not item:FindFirstAncestorWhichIsA("LuaSourceContainer") then
			table.insert(state.bagItems, item)
		end
	end
end

function FarmState.isFull(context, state, now)
	if state.forcedFull then return true end
	if state.coinsStarted == 0 then return false end
	local any, all = false, true
	for _, full in state.full do any = true; if not full then all = false end end
	if any and all then return true end
	if now - state.coinsStarted < 2 then return false end
	if not state.bag or not state.bag.Parent or now - state.bagAt > 2 then
		state.bagAt = now
		FarmState.refreshBag(context, state)
	end
	if not state.bag or not state.bag.Visible then return false end
	for _, item in state.bagItems do if item.Parent and item.Visible then return true end end
	return false
end

function FarmState.validCoin(context, state, coin)
	if context.environment.getgenv().__pulseAntiCoin or not coin or not coin.Parent or not coin:IsA("BasePart") then return false end
	if not coin:FindFirstChild("TouchInterest") or coin:GetAttribute("Collected") or state.skipped[coin] then return false end
	local id = coin:GetAttribute("CoinID")
	if id ~= nil and state.full[id] then return false end
	return true
end

function FarmState.containers(context, state, now)
	if now - state.containersAt < 0.5 then return state.containers end
	state.containersAt = now
	local containers, seen = {}, {}
	for _, object in context.workspace:GetChildren() do
		local container = object:FindFirstChild("CoinContainer")
		if container and not seen[container] then seen[container] = true; table.insert(containers, container) end
	end
	for _, coin in context.collection:GetTagged("ServerCoinPart") do
		local container = coin.Parent
		if container and not seen[container] and container ~= context.workspace and container:IsDescendantOf(context.workspace) then
			seen[container] = true
			table.insert(containers, container)
		end
	end
	state.containers = containers
	return containers
end

function FarmState.canFarm(context, state, roles)
	if context.environment.getgenv().__pulseRoleFarmHold or type(roles) ~= "table" then return false end
	local role = roles[context.player.Name]
	if not role or not role.Role then return false end
	if role.Dead ~= true and role.Killed ~= true then return true end
	local character = context.player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	return FARM_ROLES[tostring(role.Role)] == true and humanoid ~= nil and humanoid.Health > 0
end

local FarmMotion = {}

function FarmMotion.create()
	return {phase = "idle", gotT = 0, learn = {}, tries = 0, got = 0, fails = 0, info = {}, skip = {}, pending = {},
		att = {}, gapT = 4.5, tp = {}, containerAt = 0, containers = {}, filterAt = 0}
end

function FarmMotion.parts(character)
	local parts = {}
	if not character then return parts end
	local head = character:FindFirstChild("Head")
	if head then table.insert(parts, head) end
	for _, child in character:GetChildren() do
		if not child:IsA("Accessory") then continue end
		local handle = child:FindFirstChild("Handle")
		if handle then table.insert(parts, handle) end
	end
	for _, name in {"UpperTorso", "Torso", "HumanoidRootPart", "LowerTorso"} do
		local part = character:FindFirstChild(name)
		if part then table.insert(parts, part) end
	end
	return parts
end

function FarmMotion.touch(context, character, coin)
	local environment = context.environment
	if type(environment.firetouchinterest) ~= "function" or not character or not coin or not coin.Parent then return end
	local globals = environment.getgenv()
	if globals.__pulseInvis and globals.__pulseInvis.active then globals.__pulseInvisPause = environment.os.clock() + 0.35 end
	for _, part in FarmMotion.parts(character) do
		pcall(function() environment.firetouchinterest(part, coin, 0); environment.firetouchinterest(part, coin, 1) end)
	end
end

function FarmMotion.containers(context, state, motion, now)
	if now - motion.containerAt > 1 then
		motion.containerAt = now
		motion.containers = FarmState.containers(context, state, now)
	end
	return motion.containers
end

function FarmMotion.rayParams(context, state, motion, now)
	if not motion.rayParams then
		motion.rayParams = context.environment.RaycastParams.new()
		motion.rayParams.FilterType = context.environment.Enum.RaycastFilterType.Exclude
		pcall(function() motion.rayParams.RespectCanCollide = true end)
	end
	if now - motion.filterAt > 1 then
		motion.filterAt = now
		local ignored = {}
		for _, player in context.players:GetPlayers() do if player.Character then table.insert(ignored, player.Character) end end
		for _, container in FarmMotion.containers(context, state, motion, now) do table.insert(ignored, container) end
		motion.rayParams.FilterDescendantsInstances = ignored
	end
	return motion.rayParams
end

function FarmMotion.floor(context, state, motion, position, offset, depth, now)
	local vector = context.environment.Vector3
	local hit = context.workspace:Raycast(position + vector.new(0, offset or 0.5, 0), vector.new(0, -(depth or 50), 0),
		FarmMotion.rayParams(context, state, motion, now))
	return hit and hit.Position.Y or nil
end

function FarmMotion.coinInfo(context, state, motion, coin, now)
	if motion.info[coin] then return motion.info[coin] end
	local position = coin.Position
	local surface = FarmMotion.floor(context, state, motion, position, 0.5, 40, now) or position.Y - 3
	local hide = context.environment.Vector3.new(position.X, surface - 3.8, position.Z)
	local info = {surf = surface, hide = hide, exposed = FarmMotion.floor(context, state, motion, hide, 0, 7, now) ~= nil}
	motion.info[coin] = info
	return info
end

function FarmMotion.murderPenalty(state, position, murderer)
	if not murderer then return 0 end
	local distance = (position - murderer).Magnitude
	if distance < 24 then return state.avoidMurder and 1000000 or 300 end
	if distance < 48 then return (48 - distance) * 2 end
	return 0
end

function FarmMotion.choose(context, state, motion, position, murderer, now)
	local chosen, lowest
	for _, container in FarmMotion.containers(context, state, motion, now) do
		if not container.Parent then continue end
		for _, coin in container:GetChildren() do
			if not FarmState.validCoin(context, state, coin) or (motion.skip[coin] or 0) >= now or motion.pending[coin] then continue end
			local info = FarmMotion.coinInfo(context, state, motion, coin, now)
			local score = (info.hide - position).Magnitude + math.max(0, coin.Position.Y - info.hide.Y) * 1.2
			if info.exposed then score += 25 end
			score += FarmMotion.murderPenalty(state, coin.Position, murderer)
			if not lowest or score < lowest then chosen, lowest = coin, score end
		end
	end
	if chosen and state.avoidMurder and lowest >= 1000000 then return nil end
	return chosen, lowest
end

function FarmMotion.jump(context, motion, root, destination, forced, now)
	if not forced and (destination - root.Position).Magnitude <= 1000 then return false end
	root.CFrame = context.environment.CFrame.new(destination)
	root.AssemblyLinearVelocity, root.AssemblyAngularVelocity = context.environment.Vector3.zero, context.environment.Vector3.zero
	motion.goT, motion.jumps = now, (motion.jumps or 0) + 1
	return true
end

function FarmMotion.hideAt(context, state, motion, root, now)
	local floor = FarmMotion.floor(context, state, motion, root.Position, 9, 30, now)
	if not floor then return root.Position end
	local position = root.Position
	return context.environment.Vector3.new(position.X, math.min(position.Y, floor - 3.8), position.Z)
end

function FarmMotion.canShootAt(context, position, root)
	local environment, workspace = context.environment, context.workspace
	local character = root.Parent
	local head = character and character:FindFirstChild("Head")
	if not head then return true end
	local attachment = root:FindFirstChild("GunRaycastAttachment")
	local ray = environment.RaycastParams.new()
	ray.FilterType, ray.FilterDescendantsInstances = environment.Enum.RaycastFilterType.Exclude, {character}
	if not attachment then
		local overlap = environment.OverlapParams.new()
		overlap.FilterType, overlap.FilterDescendantsInstances = environment.Enum.RaycastFilterType.Exclude, {character}
		return #workspace:GetPartBoundsInBox(environment.CFrame.new(position + environment.Vector3.new(0, 1, 0)),
			environment.Vector3.new(5, 4.5, 5), overlap) == 0
	end
	local vector, frame = environment.Vector3, environment.CFrame
	local headLocal, muzzle = root.CFrame:PointToObjectSpace(head.Position), attachment.CFrame
	local offsets = {vector.zero, vector.new(0.2, 0, 0), vector.new(-0.2, 0, 0), vector.new(0, 0.2, 0), vector.new(0, -0.2, 0)}
	local function clear(pose)
		local endpoint = (pose * muzzle).Position
		for _, offset in offsets do
			local origin = pose * (headLocal + offset)
			local direction = endpoint - origin
			if direction.Magnitude > 0.001 and workspace:Raycast(origin, direction * ((direction.Magnitude + 0.6) / direction.Magnitude), ray) then return false end
		end
		return true
	end
	for index = 0, 7 do
		local pose = frame.new(position) * frame.Angles(0, index * math.pi / 4, 0)
		if clear(pose) and clear(pose * frame.Angles(0, math.rad(25), 0)) and clear(pose * frame.Angles(0, -math.rad(25), 0)) then return true end
	end
	return false
end

function FarmMotion.findShootSpot(context, root, position, startDepth)
	local vector = context.environment.Vector3
	local minimum = context.workspace.FallenPartsDestroyHeight + 25
	local depth = startDepth or 0
	for offset = depth, depth + 48, 2 do
		local candidate = position - vector.new(0, offset, 0)
		if candidate.Y < minimum then break end
		if FarmMotion.canShootAt(context, candidate, root) then return candidate, offset end
	end
	return position - vector.new(0, depth, 0), depth
end

function FarmMotion.shootSpot(context, state, motion, root, now)
	local spot = motion.spot
	local vector = context.environment.Vector3
	if not spot or vector.new(spot.X - root.Position.X, 0, spot.Z - root.Position.Z).Magnitude > 10 then
		motion.spotBase = FarmMotion.hideAt(context, state, motion, root, now)
		motion.spot, motion.spotD = FarmMotion.findShootSpot(context, root, motion.spotBase, 0)
		motion.spotT = now
		return motion.spot
	end
	if (spot - root.Position).Magnitude < 1 and now - (motion.spotT or 0) > 1 then
		local guard = rawget(context.environment.getgenv(), "__pulseMuzzle")
		if type(guard) == "table" and guard.at and now - guard.at < 0.2 and guard.blocked then
			motion.spot, motion.spotD = FarmMotion.findShootSpot(context, root, motion.spotBase or spot, (motion.spotD or 0) + 2)
		end
		motion.spotT = now
	end
	return motion.spot
end

function FarmMotion.skip(context, state, coin)
	state.skipped[coin] = true
	context.environment.task.delay(30, function() state.skipped[coin] = nil end)
end

function FarmMotion.underMap(context, state, motion, options)
	local now, root, velocity, gyro, character = options.time, options.root, options.velocity, options.gyro, options.character
	local murderer, gun = options.murderer, options.gun
	local speed = math.clamp(tonumber(context.configuration.farm.speed) or 21, 8, 22)
	local coin = motion.coin
	if gun then coin = gun end
	if not gun then
		if coin and not FarmState.validCoin(context, state, coin) then
			if motion.phase == "rise" and motion.riseFrom then
				table.insert(motion.learn, coin.Position.Y - root.Position.Y)
				if #motion.learn > 60 then table.remove(motion.learn, 1) end
			end
			motion.got, motion.coin, motion.phase, coin = motion.got + 1, nil, "go", nil
		end
		if not coin or motion.phase == "go" and now - (motion.pickT or 0) > 0.3 then
			motion.pickT = now
			local candidate, score = FarmMotion.choose(context, state, motion, root.Position, murderer, now)
			if candidate and candidate ~= coin then
				local current
				if coin and FarmState.validCoin(context, state, coin) then
					local info = FarmMotion.coinInfo(context, state, motion, coin, now)
					current = (info.hide - root.Position).Magnitude + math.max(0, coin.Position.Y - info.hide.Y) * 1.2
						+ (info.exposed and 25 or 0) + FarmMotion.murderPenalty(state, coin.Position, murderer)
				end
				if not current or score + 6 < current then
					motion.coin, motion.phase, motion.goT, motion.tries, coin = candidate, "go", now, motion.tries + 1, candidate
				end
			end
		end
	end
	if not coin then
		FarmMotion.move(context, root, velocity, gyro, FarmMotion.hideAt(context, state, motion, root, now), options.delta, speed)
		motion.phase = "wait"
		return
	end
	local info = FarmMotion.coinInfo(context, state, motion, coin, now)
	if motion.phase ~= "rise" then
		if motion.phase ~= "go" then motion.phase, motion.goT = "go", now end
		local force = state.firstJump and coin ~= gun
		if force then state.firstJump = false end
		FarmMotion.jump(context, motion, root, info.hide, force, now)
		local distance = FarmMotion.move(context, root, velocity, gyro, info.hide, options.delta, speed)
		if distance < 0.7 then motion.phase, motion.riseFrom, motion.dwellFrom = "rise", now, nil; return end
		if now - (motion.goT or now) > distance / math.max(speed, 1) + 4 then
			FarmMotion.skip(context, state, coin)
			motion.coin, motion.gunTarget, motion.fails, motion.phase = nil, nil, motion.fails + 1, "go"
		end
		return
	end
	if murderer and coin ~= gun then
		local distance = context.environment.Vector3.new(coin.Position.X - murderer.X, 0, coin.Position.Z - murderer.Z).Magnitude
		if distance < 12 then motion.skip[coin], motion.coin, motion.phase = now + 3, nil, "go"; return end
	end
	local attempts = motion.att[coin] or 0
	local gap = math.max(0.6, motion.gapT - 0.45 * attempts)
	local destination = context.environment.Vector3.new(info.hide.X, math.max(coin.Position.Y - gap, info.hide.Y), info.hide.Z)
	local distance = FarmMotion.move(context, root, velocity, gyro, destination, options.delta, speed)
	if now - (motion.touchAt or 0) >= 1 / 30 then motion.touchAt = now; FarmMotion.touch(context, character, coin) end
	if distance < 0.3 then
		motion.dwellFrom = motion.dwellFrom or now
		if now - motion.dwellFrom >= 0.1 + (coin == gun and 0.25 or 0) then
			motion.att[coin] = attempts + 1
			if attempts + 1 >= 4 then FarmMotion.skip(context, state, coin); motion.fails += 1 end
			if attempts + 1 < 4 then motion.pending[coin] = {t = now + 0.9, gap = gap} end
			if coin == gun then motion.gunTarget = nil end
			motion.coin, motion.dwellFrom, motion.phase = nil, nil, "go"
			return
		end
	end
	if now - (motion.riseFrom or now) > 1.4 then
		FarmMotion.skip(context, state, coin)
		if coin == gun then motion.gunTarget = nil end
		motion.coin, motion.fails, motion.phase = nil, motion.fails + 1, "go"
	end
end

function FarmMotion.fast(context, state, motion, options)
	local now, root, velocity, gyro, character = options.time, options.root, options.velocity, options.gyro, options.character
	local gun, murderer = options.gun, options.murderer
	local speed = math.clamp(tonumber(context.configuration.farm.fastSpeed) or 24, 8, 24)
	local coin = motion.coin
	if coin and coin ~= gun and (not FarmState.validCoin(context, state, coin) or motion.pending[coin]) then
		if not FarmState.validCoin(context, state, coin) then motion.got += 1 end
		motion.coin, coin = nil, nil
	end
	local target = gun
	if not target then
		if not coin or now - (motion.pickT or 0) > 0.3 then
			motion.pickT = now
			local best, lowest
			for _, container in FarmMotion.containers(context, state, motion, now) do
				if not container.Parent then continue end
				for _, candidate in container:GetChildren() do
					if not FarmState.validCoin(context, state, candidate) or (motion.skip[candidate] or 0) >= now or motion.pending[candidate] then continue end
					local score = (candidate.Position - root.Position).Magnitude + FarmMotion.murderPenalty(state, candidate.Position, murderer)
					if not lowest or score < lowest then best, lowest = candidate, score end
				end
			end
			if best and state.avoidMurder and lowest >= 1000000 then best = nil end
			if best and best ~= coin and (not coin or lowest + 6 < (coin.Position - root.Position).Magnitude + FarmMotion.murderPenalty(state, coin.Position, murderer)) then
				motion.coin, motion.goT, motion.tries, coin = best, now, motion.tries + 1, best
			end
		end
		target = coin
	end
	if not target then velocity.Velocity, motion.phase = context.environment.Vector3.zero, "wait"; return end
	motion.phase = "go"
	local force = state.firstJump and target ~= gun
	if force then state.firstJump = false end
	FarmMotion.jump(context, motion, root, target.Position, force, now)
	local distance = FarmMotion.move(context, root, velocity, gyro, target.Position, options.delta, speed)
	if distance < 4 then FarmMotion.touch(context, character, target) end
	if distance < 1 then
		motion.dwellFrom = motion.dwellFrom or now
		if now - motion.dwellFrom >= (target == gun and 0.25 or 0.05) then
			if target == gun then motion.gunTarget = nil end
			if target ~= gun then motion.pending[target] = {t = now + 0.9} end
			motion.coin, motion.dwellFrom = nil, nil
		end
		return
	end
	motion.dwellFrom = nil
	if now - (motion.goT or now) > distance / speed + 4 then
		FarmMotion.skip(context, state, target)
		if target == gun then motion.gunTarget = nil end
		motion.coin, motion.fails = nil, motion.fails + 1
	end
end

function FarmMotion.setTouch(context, state, motion, target, now)
	local teleport = motion.tp
	teleport.ct = teleport.ct or {}
	for coin in teleport.ct do
		if coin ~= target and coin.Parent then continue end
		if coin.Parent then pcall(function() coin.CanTouch = true end) end
		teleport.ct[coin] = nil
	end
	for _, container in FarmMotion.containers(context, state, motion, now) do
		if not container.Parent then continue end
		for _, coin in container:GetChildren() do
			if coin == target or teleport.ct[coin] or not coin:IsA("BasePart") or not coin.CanTouch or not coin:FindFirstChild("TouchInterest") then continue end
			teleport.ct[coin] = true
			pcall(function() coin.CanTouch = false end)
		end
	end
end

function FarmMotion.restoreTouch(motion)
	local teleport = motion.tp
	if not teleport or not teleport.ct then return end
	for coin in teleport.ct do
		if coin.Parent and not coin:GetAttribute("Collected") then pcall(function() coin.CanTouch = true end) end
	end
	teleport.ct = nil
end

function FarmMotion.teleportPick(context, state, motion, now, murderer, previous, current)
	local coins, teleport = {}, motion.tp
	for _, container in FarmMotion.containers(context, state, motion, now) do
		if not container.Parent then continue end
		for _, coin in container:GetChildren() do
			if not FarmState.validCoin(context, state, coin) or (motion.skip[coin] or 0) >= now then continue end
			table.insert(coins, coin)
			teleport.valid = teleport.valid or setmetatable({}, {__mode = "k"})
			teleport.valid[coin] = true
		end
	end
	if #coins == 0 then return nil end
	if previous and (coins[1].Position - previous).Magnitude > 1000 then previous = nil end
	local best, lowest, currentScore
	for _, coin in coins do
		local nearest, second, third = math.huge, math.huge, math.huge
		for _, other in coins do
			if other == coin then continue end
			local distance = (other.Position - coin.Position).Magnitude
			if distance < nearest then nearest, second, third = distance, nearest, second; continue end
			if distance < second then second, third = distance, second; continue end
			if distance < third then third = distance end
		end
		local score = previous and ((coin.Position - previous).Magnitude + 0.25 * (math.min(nearest, 120) + math.min(second, 120)))
			or math.min(nearest, 300) + math.min(second, 300) + math.min(third, 300)
		score += FarmMotion.murderPenalty(state, coin.Position, murderer)
		if coin == current then currentScore = score end
		if not lowest or score < lowest then best, lowest = coin, score end
	end
	if best and state.avoidMurder and lowest >= 1000000 then return nil end
	return best, lowest, currentScore
end

function FarmMotion.home(context, motion, now)
	if motion.homeCF and now - (motion.homeAt or 0) < 5 then return motion.homeCF end
	motion.homeAt, motion.homeCF = now, nil
	local lobby = context.workspace:FindFirstChild("Lobby")
	local spawns = lobby and lobby:FindFirstChild("Spawns")
	if not spawns then return nil end
	local total, count = context.environment.Vector3.zero, 0
	for _, part in spawns:GetDescendants() do
		if part:IsA("BasePart") then total, count = total + part.Position, count + 1 end
	end
	if count > 0 then motion.homeCF = context.environment.CFrame.new(total / count + context.environment.Vector3.new(0, 4, 0)) end
	return motion.homeCF
end

function FarmMotion.teleportReference(context, state, motion, root, now, tickTime)
	local teleport = motion.tp
	if teleport.lastAt == nil then
		teleport.lastAt = motion.evAt or 0
		if motion.evAt and now - motion.evAt < 20 then teleport.lastPos = motion.evPos end
		teleport.seenEv = motion.evAt or 0
		if not motion.evAt and state.coinsStarted > 0 and tickTime - state.coinsStarted > 5 then teleport.cold = now + 2.5 end
	end
	local event = motion.evAt
	if not event or event <= (teleport.seenEv or 0) then return end
	teleport.seenEv = event
	if event <= teleport.lastAt then return end
	teleport.lastAt = event
	if teleport.touchC and event - (teleport.touchT or 0) < 1.5 then teleport.lastPos = teleport.touchC.Position; return end
	local position, nearest, selected = root and root.Position, nil, nil
	for coin in teleport.valid or {} do
		if not coin.Parent or not coin:GetAttribute("Collected") or not position then continue end
		local distance = (coin.Position - position).Magnitude
		if distance < 14 and (not nearest or distance < nearest) then nearest, selected = distance, coin end
	end
	if selected then teleport.lastPos = selected.Position; return end
	teleport.lastPos, teleport.cold = nil, event + 3
end

function FarmMotion.teleport(context, state, motion, options)
	local now, root, velocity, character, gun = options.time, options.root, options.velocity, options.character, options.gun
	local teleport = motion.tp
	local home = FarmMotion.home(context, motion, now) or teleport.start
	teleport.start = teleport.start or root.CFrame
	velocity.Velocity = context.environment.Vector3.zero
	FarmMotion.teleportReference(context, state, motion, root, now, options.tickTime)
	local function returnHome(phase, force)
		motion.phase = phase
		if home and (force or (root.Position - home.Position).Magnitude > 6) then root.CFrame = home end
		root.AssemblyLinearVelocity = context.environment.Vector3.zero
	end
	if teleport.cold and not gun and now < teleport.cold then returnHome("wait"); return end
	local coin = teleport.coin
	if coin then
		local isGun = coin == gun
		local collected = isGun and (coin.Parent == nil or options.hasGun) or not isGun and not FarmState.validCoin(context, state, coin)
		local done = false
		if collected then
			if isGun then motion.gunTarget = nil end
			if not isGun and coin:GetAttribute("Collected") then motion.got += 1 end
			if not isGun and teleport.touching then
				if now > teleport.lastAt then teleport.lastAt = now end
				teleport.lastPos = coin.Position
				local latency = now - (teleport.at or now)
				if latency > 0 and latency < 0.6 then motion.tpLat = math.clamp((motion.tpLat or 0.1) * 0.8 + latency * 0.2, 0.03, 0.3) end
			end
			done = true
		end
		if not collected then
			if teleport.touching and now - teleport.at > 0.9 then
				FarmMotion.skip(context, state, coin)
				motion.fails += 1
				if isGun then motion.gunTarget = nil end
				done = true
			end
			if gun and not isGun then done = true end
		end
		if done then teleport.coin, teleport.touching = nil, false; returnHome("home", true); return end
	end
	if options.full and not gun then returnHome("full"); return end
	if not teleport.coin then
		local target = gun or FarmMotion.teleportPick(context, state, motion, now, options.murderer, teleport.lastPos)
		if not target then returnHome("home"); return end
		teleport.coin, teleport.touching, teleport.at = target, false, now
		teleport.hide = target == gun and target.Position or FarmMotion.coinInfo(context, state, motion, target, now).hide
		motion.tries += 1
		FarmMotion.setTouch(context, state, motion, target, now)
		teleport.ctAt = now
	end
	coin = teleport.coin
	if now - (teleport.ctAt or 0) > 0.1 then
		teleport.ctAt = now
		FarmMotion.setTouch(context, state, motion, coin, now)
	end
	local isGun = coin == gun
	if not isGun and not teleport.touching and now - (teleport.rpAt or 0) > 0.2 then
		teleport.rpAt = now
		local target, score, currentScore = FarmMotion.teleportPick(context, state, motion, now, options.murderer, teleport.lastPos, coin)
		if target and target ~= coin and (not currentScore or score + 2 < currentScore) then
			coin, teleport.coin, teleport.at = target, target, now
			teleport.hide = FarmMotion.coinInfo(context, state, motion, target, now).hide
			FarmMotion.setTouch(context, state, motion, target, now)
			teleport.ctAt = now
		end
	end
	local arrival = now
	if not isGun and teleport.lastPos and (coin.Position - teleport.lastPos).Magnitude < 1000 then
		local speed = math.clamp(tonumber(context.configuration.farm.tpSpeed) or 25, 10, 26)
		arrival = math.max(teleport.lastAt, teleport.lastAt + (coin.Position - teleport.lastPos).Magnitude / speed - 0.4 * (motion.tpLat or 0.1))
	end
	if not teleport.touching and now < arrival - 0.35 then returnHome("wait"); return end
	root.CFrame = context.environment.CFrame.new(teleport.hide)
	root.AssemblyLinearVelocity = context.environment.Vector3.zero
	if teleport.touching or now >= arrival then
		if not teleport.touching then teleport.touching, teleport.at = true, now end
		teleport.touchC, teleport.touchT = coin, now
		FarmMotion.touch(context, character, coin)
		motion.phase = "tp"
		return
	end
	motion.phase = "wait"
end

function FarmMotion.ensureMover(context, root, humanoid, withoutGyro)
	local environment = context.environment
	local velocity = root:FindFirstChild("PulseFarmBV")
	if not velocity then
		velocity = environment.Instance.new("BodyVelocity")
		velocity.Name, velocity.MaxForce, velocity.P = "PulseFarmBV", environment.Vector3.new(9e9, 9e9, 9e9), 30000
		velocity.Velocity = environment.Vector3.zero
		velocity:SetAttribute("PulseMine", true)
		velocity.Parent = root
	end
	local gyro = root:FindFirstChild("PulseFarmBG")
	if withoutGyro and gyro then pcall(function() gyro:Destroy() end); gyro = nil end
	if not withoutGyro and not gyro then
		gyro = environment.Instance.new("BodyGyro")
		gyro.Name, gyro.MaxTorque, gyro.P = "PulseFarmBG", environment.Vector3.new(9e9, 9e9, 9e9), 30000
		gyro:SetAttribute("PulseMine", true)
		gyro.CFrame = environment.CFrame.lookAt(root.Position, root.Position + environment.Vector3.new(root.CFrame.LookVector.X, 0, root.CFrame.LookVector.Z + 1e-4))
		gyro.Parent = root
	end
	if humanoid and not humanoid.PlatformStand then pcall(function() humanoid.PlatformStand = true end) end
	return velocity, gyro
end

function FarmMotion.move(context, root, velocity, gyro, destination, delta, speed)
	local direction = destination - root.Position
	local distance = direction.Magnitude
	velocity.Velocity = distance < 0.05 and context.environment.Vector3.zero or direction.Unit * math.min(speed, distance / math.max(delta, 1 / 60))
	local horizontal = context.environment.Vector3.new(direction.X, 0, direction.Z)
	if gyro and horizontal.Magnitude > 0.4 then gyro.CFrame = context.environment.CFrame.lookAt(root.Position, root.Position + horizontal.Unit) end
	return distance
end

function FarmMotion.reset(context, motion, character)
	pcall(function()
		local root = character and character:FindFirstChild("HumanoidRootPart")
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		if root then
			for _, name in {"PulseFarmBV", "PulseFarmBG"} do local object = root:FindFirstChild(name); if object then object:Destroy() end end
		end
		if humanoid and humanoid.PlatformStand then
			humanoid.PlatformStand = false
			if humanoid.Health > 0 then humanoid:ChangeState(context.environment.Enum.HumanoidStateType.GettingUp) end
		end
	end)
	if motion.ctRestore then pcall(motion.ctRestore) end
	motion.phase, motion.coin, motion.hold = "idle", nil, nil
end

function FarmMotion.updatePending(context, state, motion, now)
	for coin, entry in motion.pending do
		if type(entry) ~= "table" then continue end
		if FarmState.validCoin(context, state, coin) then
			if now > entry.t then
				motion.pending[coin] = nil
				motion.gapT, motion.miss = math.max(1, motion.gapT - 0.25), (motion.miss or 0) + 1
			end
			continue
		end
		motion.pending[coin] = nil
		motion.gapT, motion.ok = math.min(6, motion.gapT + 0.08), (motion.ok or 0) + 1
	end
end

local FarmRuntime = {}

FarmRuntime.MapFarm = {}

function FarmRuntime.MapFarm.create(context)
	local environment = context.environment
	local farm = {file = "PulseHub/mm2_mapfarm.json", data = {}, dirty = false, K = 1.3, first = setmetatable({}, {__mode = "k"})}
	farm.SEED = {house2 = 20.1, hospital3 = 21.7, office3 = 23.4, mansion2 = 24.7, hotel2 = 24.8, bank2 = 25,
		policestation = 25.2, factory = 26.5, biolab = 27, manor = 27.4, barn = 29.2, researchfacility = 29.9,
		vampirescastle = 30.2, farmhouse = 30.9, spaceship = 31.9, mineshaft = 34.7, workplace = 39, milbase = 41.8}
	pcall(function()
		if type(environment.isfile) ~= "function" or not environment.isfile(farm.file) then return end
		local data = context.http:JSONDecode(environment.readfile(farm.file))
		if type(data) == "table" then farm.data = data end
	end)
	farm.alias = type(farm.data.__alias) == "table" and farm.data.__alias or {}
	farm.data.__alias = farm.alias
	if farm.alias.hotel == nil then farm.alias.hotel = "hotel2" end
	if farm.alias.vampirecastle == nil then farm.alias.vampirecastle = "vampirescastle" end
	farm.key = function(name) return (string.lower(tostring(name or "")):gsub("[^%w]", "")) end
	farm.rec = function(key)
		if type(farm.data[key]) ~= "table" then farm.data[key] = {h = 0, d = 0, nn = 0, nnN = 0} end
		return farm.data[key]
	end
	farm.merge = function(key, target)
		farm.alias[key] = target
		local source = farm.data[key]
		if type(source) == "table" then
			local result = farm.rec(target)
			for _, field in {"h", "d", "nn", "nnN"} do result[field] = (tonumber(result[field]) or 0) + (tonumber(source[field]) or 0) end
			farm.data[key] = nil
		end
		farm.dirty = true
	end
	farm.guess = function(key, exists)
		for _, suffix in {"2", "3"} do if exists(key .. suffix) then return key .. suffix end end
		local base = key:match("^(.-)%d+$")
		if base and base ~= "" and exists(base) then return base end
		return nil
	end
	farm.fix = function(exists)
		local merges = {}
		for key, record in farm.data do
			if key == "__alias" or type(record) ~= "table" or exists(key) then continue end
			local target = type(farm.alias[key]) == "string" and farm.alias[key] or farm.guess(key, exists)
			if target then table.insert(merges, {key, target}) end
		end
		for _, pair in merges do farm.merge(pair[1], pair[2]) end
	end
	farm.ckey = function(name, map)
		local key = farm.key(name)
		if type(farm.alias[key]) == "string" then return farm.alias[key] end
		local vote = rawget(environment.getgenv(), "__pulseVoteDbg")
		if not vote or type(vote.isKey) ~= "function" or vote.isKey(key) then return key end
		local guess = farm.guess(key, vote.isKey)
		if guess then farm.merge(key, guess); return guess end
		local win, first = vote.lastWin, map and farm.first[map]
		if not first or type(win) ~= "table" or type(win.k) ~= "string" or not vote.isKey(win.k) then return key end
		local time = tonumber(win.t) or 0
		if first < time - 1 or first - time >= 120 then return key end
		if farm.alias[win.k] ~= nil then return key end
		farm.merge(key, win.k)
		return win.k
	end
	farm.measured = function(key)
		local record = farm.data[key]
		return type(record) == "table" and ((tonumber(record.h) or 0) >= 12 or (tonumber(record.nnN) or 0) >= 3) or false
	end
	farm.map = function()
		for _, map in context.workspace:GetChildren() do
			if not map:IsA("Model") or not map:FindFirstChild("CoinContainer") then continue end
			if not farm.first[map] then farm.first[map] = environment.os.clock() end
			return map
		end
		return nil
	end
	farm.at = function()
		local motion = rawget(environment.getgenv(), "__pulseFarmUF")
		local teleport = motion and motion.tp
		if teleport and teleport.touchC and teleport.touchC.Parent and environment.os.clock() - (teleport.touchT or 0) < 1.5 then
			return teleport.touchC.Position
		end
		local character = context.player.Character
		local root = character and character:FindFirstChild("HumanoidRootPart")
		return root and root.Position or nil
	end
	farm.hop = function()
		local now, map, position = environment.os.clock(), farm.map(), farm.at()
		if position and map and farm.lp and farm.lk == map.Name and now - (farm.lt or 0) < 6 then
			local distance = (position - farm.lp).Magnitude
			if distance < 400 then
				local record = farm.rec(farm.ckey(map.Name, map))
				record.h = (tonumber(record.h) or 0) + 1
				record.d = (tonumber(record.d) or 0) + distance
				if record.h > 400 then record.h, record.d = record.h / 2, record.d / 2 end
				farm.dirty = true
			end
		end
		farm.lp, farm.lt, farm.lk = position, now, map and map.Name or nil
	end
	farm.calib = function()
		local hops, nearest = 0, 0
		for _, record in farm.data do
			if type(record) ~= "table" or (tonumber(record.h) or 0) < 15 or (tonumber(record.nnN) or 0) < 3 then continue end
			hops += record.d / record.h
			nearest += record.nn / record.nnN
		end
		if hops > 0 and nearest > 0 then farm.K = math.clamp(hops / nearest, 0.8, 3) end
	end
	farm.hopOf = function(key)
		local record = farm.data[key]
		if type(record) == "table" then
			if (tonumber(record.h) or 0) >= 12 then return record.d / record.h end
			if (tonumber(record.nnN) or 0) >= 3 then return record.nn / record.nnN * farm.K end
		end
		return farm.SEED[key]
	end
	farm.save = function()
		if not farm.dirty or type(environment.writefile) ~= "function" then return end
		farm.dirty = false
		pcall(function()
			if type(environment.isfolder) == "function" and not environment.isfolder("PulseHub") and type(environment.makefolder) == "function" then
				environment.makefolder("PulseHub")
			end
			environment.writefile(farm.file, context.http:JSONEncode(farm.data))
		end)
	end
	return farm
end

function FarmRuntime.MapFarm.sample(farm)
	local map = farm.map()
	local container = map and map:FindFirstChild("CoinContainer")
	if not container then return end
	local positions = {}
	for _, coin in container:GetChildren() do
		if coin:IsA("BasePart") and coin:FindFirstChild("TouchInterest") and not coin:GetAttribute("Collected") then
			table.insert(positions, coin.Position)
		end
	end
	if #positions < 8 then return end
	local total = 0
	for index, position in positions do
		local nearest = math.huge
		for other, target in positions do
			if index ~= other then nearest = math.min(nearest, (position - target).Magnitude) end
		end
		total += math.min(nearest, 300)
	end
	local record = farm.rec(farm.ckey(map.Name, map))
	record.nn = (tonumber(record.nn) or 0) + total / #positions
	record.nnN = (tonumber(record.nnN) or 0) + 1
	if record.nnN > 200 then record.nn, record.nnN = record.nn / 2, record.nnN / 2 end
	farm.dirty = true
end

function FarmRuntime.MapFarm.install(context, active)
	local farm, environment = FarmRuntime.MapFarm.create(context), context.environment
	environment.getgenv().__pulseMapFarm = farm
	environment.task.spawn(function()
		local saved = environment.os.clock()
		while active() do
			environment.task.wait(3)
			pcall(FarmRuntime.MapFarm.sample, farm)
			if environment.os.clock() - saved > 20 then
				saved = environment.os.clock()
				pcall(farm.calib)
				farm.save()
			end
		end
	end)
	return farm
end

function FarmRuntime.zombie(context)
	local duel = rawget(context.environment.getgenv(), "__pulseDuel")
	if type(duel) == "table" and duel.on and duel.mode == "Zombie" then return duel end
	return nil
end

function FarmRuntime.zombieTarget(context, motion, root, duel)
	if not duel or duel.mine ~= "prey" or context.configuration.farm.zShoot == false then return nil end
	local best, bestDistance, current, currentDistance
	for _, player in context.players:GetPlayers() do
		if not duel.isFoe(player) then continue end
		local character = duel.alive(player)
		local target = character and character:FindFirstChild("HumanoidRootPart")
		if not target or target.Position.Y <= context.workspace.FallenPartsDestroyHeight + 30 then continue end
		local distance = (target.Position - root.Position).Magnitude
		if not bestDistance or distance < bestDistance then best, bestDistance = character, distance end
		if character == motion.zCur then current, currentDistance = character, distance end
	end
	if current and best ~= current and currentDistance < bestDistance + 8 then best = current end
	motion.zCur = best
	return best
end

function FarmRuntime.zombieFling(context, state, motion, duel, root, now)
	local environment, globals = context.environment, context.environment.getgenv()
	local fling = rawget(globals, "__pulseFlingPersist")
	if state.flinging or environment.tick() - (state.lastFling or 0) < 1 or type(fling) ~= "function" then return end
	motion.zfs = motion.zfs or {}
	local best, lowest
	for _, player in context.players:GetPlayers() do
		if not duel.isFoe(player) or now < (motion.zfs[player] or 0) then continue end
		local character = duel.alive(player)
		local target = character and character:FindFirstChild("HumanoidRootPart")
		if not target or target.Position.Y <= context.workspace.FallenPartsDestroyHeight + 30 then continue end
		local backpack = player:FindFirstChildOfClass("Backpack")
		local gun = character:FindFirstChild("Gun") or backpack and backpack:FindFirstChild("Gun")
		local score = (target.Position - root.Position).Magnitude + (gun and 0 or 100000)
		if not lowest or score < lowest then best, lowest = player, score end
	end
	if not best then return end
	state.flinging, state.lastFling = true, environment.tick()
	local antiFling = state.antiFling
	state.antiFling = false
	environment.task.spawn(function()
		local ok, result = pcall(fling, best, {time = 12, stop = function()
			local character = context.player.Character
			local humanoid = character and character:FindFirstChildOfClass("Humanoid")
			return not state.enabled or not state.round or not humanoid or humanoid.Health <= 0 or not FarmRuntime.zombie(context)
				or not duel.isFoe(best) or context.configuration.farm.zFling == false
		end})
		state.antiFling, state.flinging, state.lastFling = antiFling, false, environment.tick()
		if not ok or not result then motion.zfs[best] = environment.os.clock() + 6 end
	end)
end

function FarmRuntime.killAll(context)
	local character, environment = context.player.Character, context.environment
	local root = character and character:FindFirstChild("HumanoidRootPart")
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not root or not humanoid or humanoid.Health <= 0 then return end
	local knife = character:FindFirstChild("Knife")
	if not knife then
		local backpack = context.player:FindFirstChild("Backpack")
		knife = backpack and backpack:FindFirstChild("Knife")
		if knife and knife:IsA("Tool") then knife.Parent = character end
	end
	if not knife then return end
	local handle = knife:FindFirstChild("Handle") or knife:FindFirstChildWhichIsA("BasePart")
	if not handle or not environment.firetouchinterest then return end
	for _, player in context.players:GetPlayers() do
		if player == context.player or not player.Character then continue end
		local target = player.Character
		local targetHumanoid, targetRoot = target:FindFirstChild("Humanoid"), target:FindFirstChild("HumanoidRootPart")
		if not targetHumanoid or targetHumanoid.Health <= 0 or not targetRoot then continue end
		for _, name in {"HumanoidRootPart", "UpperTorso", "Torso", "LowerTorso", "Head"} do
			local part = target:FindFirstChild(name)
			if not part then continue end
			pcall(function()
				environment.firetouchinterest(handle, part, 0)
				environment.firetouchinterest(handle, part, 1)
				environment.firetouchinterest(part, handle, 0)
				environment.firetouchinterest(part, handle, 1)
			end)
		end
		pcall(function() knife:Activate() end)
	end
end

function FarmRuntime.shoot(context, motion, target)
	local character = context.player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not target or not character or not humanoid or humanoid.Health <= 0 then return false end
	if not character:FindFirstChild("Gun") then
		local backpack = context.player:FindFirstChild("Backpack")
		local gun = backpack and backpack:FindFirstChild("Gun")
		if not gun then return false end
		pcall(function() humanoid:EquipTool(gun) end)
		return true
	end
	local root = character:FindFirstChild("HumanoidRootPart")
	local gyro = root and root:FindFirstChild("PulseFarmBG")
	if gyro then pcall(function() gyro:Destroy() end) end
	local globals = context.environment.getgenv()
	local aim = rawget(globals, "__pulseAimX")
	if aim and aim.alive and type(aim.fireFor) == "function" then
		if (type(aim.ready) ~= "function" or aim.ready()) and not aim.queued
			and not (aim.pending and not aim.pending.dead and context.environment.os.clock() - (aim.pending.t or 0) < 0.5) then
			local ok, sent = pcall(aim.fireFor, target, true, false)
			if ok and sent then motion.shots, motion.shotAt = (motion.shots or 0) + 1, context.environment.os.clock() end
		end
		return true
	end
	if type(globals.__pulseShotMurder) == "function" then pcall(globals.__pulseShotMurder) end
	return true
end

function FarmRuntime.resetMotion(motion)
	FarmMotion.restoreTouch(motion)
	motion.coin, motion.hold, motion.gunTarget = nil, nil, nil
	motion.info, motion.skip, motion.pending, motion.att, motion.tp = {}, {}, {}, {}, {}
	if motion.phase ~= "idle" then motion.phase = "go" end
end

function FarmRuntime.setEnabled(context, state, motion, enabled)
	state.enabled = enabled
	local globals = context.environment.getgenv()
	local antiCoin = globals.__pulseAntiCoinInfo
	if enabled and globals.__pulseAntiCoin and antiCoin and type(antiCoin.set) == "function" then
		state.restoreAntiCoin = true
		pcall(antiCoin.set, false)
	end
	if not enabled and state.restoreAntiCoin then
		state.restoreAntiCoin = false
		if antiCoin and type(antiCoin.set) == "function" then pcall(antiCoin.set, true) end
	end
	if enabled then state.firstJump = true; return end
	state.active = false
	FarmMotion.reset(context, motion, context.player.Character)
end

function FarmRuntime.installRejoin(context, api)
	local environment, globals = context.environment, context.environment.getgenv()
	local state = {busy = false, tries = 0}
	globals.__pulseRejoin = state
	state.key = function()
		local key = rawget(globals, "script_key")
		if type(key) ~= "string" or key == "" then key = rawget(environment._G or {}, "script_key") end
		if type(key) == "string" and key ~= "" then return key end
		return nil
	end
	state.queue = function()
		local queue = environment.syn and environment.syn.queue_on_teleport or environment.queue_on_teleport
			or environment.queueonteleport or environment.fluxus and environment.fluxus.queue_on_teleport
		if type(queue) ~= "function" then return false end
		local key = state.key()
		local prefix = key and string.format("script_key = %q\npcall(function() getgenv().script_key = %q end)\n", key, key) or ""
		local source = prefix .. 'if not game:IsLoaded() then game.Loaded:Wait() end\ntask.wait(1)\nif getgenv().PulseMM2HubState then return end\nloadstring(game:HttpGet("offline://raw.githubusercontent.com/PulseZax/Loader/refs/heads/main/.lua"))()'
		return pcall(queue, source)
	end
	state.onKick = function(message)
		pcall(function()
			local webhook = rawget(globals, "__pulseFarmWebhook")
			if type(webhook) == "table" and type(webhook.onKick) == "function" then environment.task.spawn(webhook.onKick, message) end
		end)
		if state.busy or context.configuration.farm.rejoin == false or not api.getEnabled() then return end
		local text = tostring(message or ""):lower()
		for _, blocked in {"same account", "another device", "different device", "banned", "anti stealer"} do
			if text:find(blocked, 1, true) then return end
		end
		state.busy, state.msg, state.queued = true, tostring(message or ""), state.queue()
		environment.task.spawn(function()
			while true do
				state.tries += 1
				pcall(function()
					if state.tries <= 2 then
						context.teleport:TeleportToPlaceInstance(environment.game.PlaceId, environment.game.JobId, context.player)
						return
					end
					context.teleport:Teleport(environment.game.PlaceId, context.player)
				end)
				local wait = state.tries <= 6 and 10 or (state.tries <= 12 and 30 or 60)
				environment.task.wait(wait)
			end
		end)
	end
	if context.guiService.ErrorMessageChanged then
		context.connect(context.guiService.ErrorMessageChanged, function(message)
			if type(message) == "string" and message ~= "" then state.onKick(message) end
		end)
	end
	environment.task.spawn(function()
		local overlay
		pcall(function() overlay = context.coreGui:WaitForChild("RobloxPromptGui", 20):WaitForChild("promptOverlay", 20) end)
		if not overlay or not context.isAlive() then return end
		context.connect(overlay.ChildAdded, function(prompt)
			if prompt.Name ~= "ErrorPrompt" then return end
			environment.task.wait(0.5)
			local message = ""
			pcall(function() message = context.guiService:GetErrorMessage() end)
			if message == "" then
				pcall(function()
					for _, label in prompt:GetDescendants() do
						if label:IsA("TextLabel") and #label.Text > #message then message = label.Text end
					end
				end)
			end
			state.onKick(message ~= "" and message or "disconnected")
		end)
	end)
	return state
end

function FarmRuntime.tick(context, state, motion, runtime, delta)
	local environment, globals = context.environment, context.environment.getgenv()
	local now, tickTime = environment.os.clock(), environment.tick()
	local character = context.player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	local root = character and character:FindFirstChild("HumanoidRootPart")
	local alive = humanoid and humanoid.Health > 0
	if now - runtime.roundAt >= 0.5 then
		runtime.roundAt = now
		for _, container in FarmState.containers(context, state, now) do
			if not container.Parent or #container:GetChildren() == 0 then continue end
			if state.coinsStarted == 0 then state.coinsStarted = tickTime end
			if not state.round then state.round, state.firstJump = true, true end
			break
		end
		local ok, roles = pcall(context.roles)
		state.active = state.enabled and alive and ok and FarmState.canFarm(context, state, roles) or false
	end
	if state.enabled and state.round and alive and not state.active and motion.phase ~= "idle" then
		motion.cfLost = motion.cfLost or now
		if now - motion.cfLost < 1.5 then
			if root and humanoid then
				local velocity = FarmMotion.ensureMover(context, root, humanoid, false)
				velocity.Velocity = environment.Vector3.zero
			end
			return
		end
	end
	if not state.enabled or not state.round or not alive or state.active then motion.cfLost = nil end
	if not state.enabled or not state.round or not state.active or not alive or not root or state.flinging then
		if motion.phase ~= "idle" then FarmMotion.reset(context, motion, character) end
		return
	end
	if globals.__pulseManualFling or globals.__pulseTouchFlinging then
		if motion.phase ~= "idle" then FarmMotion.reset(context, motion, character) end
		return
	end
	FarmMotion.updatePending(context, state, motion, now)
	local ok, roles = pcall(context.roles)
	local murderer
	if ok and type(roles) == "table" then
		local _, target = AimTarget.murderer(context, roles)
		local targetRoot = target and target:FindFirstChild("HumanoidRootPart")
		murderer = targetRoot and targetRoot.Position
	end
	local full = FarmState.isFull(context, state, tickTime)
	local duel = FarmRuntime.zombie(context)
	local hasGun = SheriffRuntime.gun(context) ~= nil
	local zombieTarget = hasGun and not motion.gunTarget and FarmRuntime.zombieTarget(context, motion, root, duel) or nil
	motion.zTarget = zombieTarget
	local handled = false
	if duel and duel.mine == "hunt" then
		handled = true
		if not full then motion.zkN, motion.zkT = nil, nil end
		if full then
			local keepKilling = false
			local killPass = rawget(globals, "__pulseKillPass")
			if context.configuration.farm.zKill ~= false and type(killPass) == "function" and killPass() then
				local count = 0
				for _, player in context.players:GetPlayers() do if duel.isFoe(player) and duel.alive(player) then count += 1 end end
				if count ~= motion.zkN then motion.zkN, motion.zkT = count, now end
				keepKilling = now - (motion.zkT or 0) <= 10
			end
			if not keepKilling and context.configuration.farm.zFling ~= false then FarmRuntime.zombieFling(context, state, motion, duel, root, now) end
		end
	end
	if duel and duel.mine == "prey" and not (state.respawnOnFull and full) then
		handled = true
		if zombieTarget and zombieTarget.Parent and hasGun and now - (motion.shootT or 0) >= 0.15 then
			motion.shootT = now
			pcall(FarmRuntime.shoot, context, motion, zombieTarget)
		end
	end
	if full and not handled then
		local role = ok and type(roles) == "table" and roles[context.player.Name]
		local isMurderer = role and role.Role and tostring(role.Role):lower():find("murder") ~= nil
		if isMurderer and state.killAura then
			pcall(FarmRuntime.killAll, context)
			handled = true
		end
		if not handled and state.shootOnFull and SheriffRuntime.gun(context) and murderer then
			local _, target = AimTarget.murderer(context, roles)
			if target and now - (motion.shootT or 0) >= 0.2 then motion.shootT = now; pcall(FarmRuntime.shoot, context, motion, target) end
			handled = target ~= nil
		end
		if not handled and state.respawnOnFull and not state.forcedFull and not state.respawnPending and tickTime - (state.lastRespawn or 0) > 3 then
			state.respawnPending = true
			environment.task.delay(0.4, function()
				state.respawnPending = false
				if not FarmState.isFull(context, state, environment.tick()) or state.forcedFull then return end
				local current = context.player.Character
				local currentHumanoid = current and current:FindFirstChildOfClass("Humanoid")
				if not currentHumanoid or currentHumanoid.Health <= 0 then return end
				state.active, currentHumanoid.Health, state.lastRespawn, state.forcedFull = false, 0, environment.tick(), true
				table.clear(state.full)
			end)
		end
	end
	if now - runtime.gunAt >= 0.15 then
		runtime.gunAt = now
		local drop = state.pickGun and context.drops.findDrop() or nil
		local role = ok and type(roles) == "table" and roles[context.player.Name]
		if not drop or SheriffRuntime.gun(context) or not role or tostring(role.Role) ~= "Innocent" or role.Dead == true or role.Killed == true then
			motion.gunTarget, motion.gunSince = nil, nil
		end
		if drop and not SheriffRuntime.gun(context) and role and tostring(role.Role) == "Innocent" and role.Dead ~= true and role.Killed ~= true then
			if motion.gunSeen ~= drop then motion.gunSeen, motion.gunSince = drop, now end
			FarmMotion.touch(context, character, drop)
			if now - motion.gunSince > 0.8 and motion.gunTarget ~= drop then
				motion.gunTarget, motion.info[drop], motion.phase, motion.goT = drop, nil, "go", now
			end
		end
	end
	local gun = motion.gunTarget
	if gun and (not gun.Parent or not state.pickGun) then motion.gunTarget, gun = nil, nil end
	local method = context.configuration.farm.method
	if method ~= "Teleport" and method ~= "Fast" then method = "Under Map" end
	local aim = zombieTarget and rawget(globals, "__pulseAimX")
	local zombieReady = zombieTarget and type(aim) == "table" and aim.alive and type(aim.ready) == "function" and aim.ready() and now >= (motion.zSkip or 0)
	if zombieReady then
		motion.zSince = motion.zSince or now
		if now - motion.zSince > 3 then motion.zSkip, motion.zSince, zombieReady = now + 2, nil, false end
	end
	if not zombieReady and motion.zSince then
		motion.zSince = nil
		if motion.phase == "zshoot" then motion.phase, motion.goT = "go", now end
	end
	local zombieShooting = zombieReady and method == "Under Map"
	local shooting = full and not gun and state.shootOnFull and hasGun and murderer ~= nil
	local velocity, gyro = FarmMotion.ensureMover(context, root, humanoid, shooting)
	if zombieShooting and gyro then pcall(function() gyro:Destroy() end); gyro = nil end
	if method ~= motion.lastMethod then
		motion.lastMethod, motion.coin, motion.hold, motion.dwellFrom = method, nil, nil, nil
		FarmMotion.restoreTouch(motion)
		motion.tp = {}
	end
	if method == "Teleport" and shooting then
		if (root.Position - murderer).Magnitude > 120 then
			local near = murderer + environment.Vector3.new(14, 0, 0)
			local floor = FarmMotion.floor(context, state, motion, near, 6, 40, now)
			if floor then
				root.CFrame = environment.CFrame.new(environment.Vector3.new(near.X, floor - 3.8, near.Z))
				root.AssemblyLinearVelocity, motion.spot = environment.Vector3.zero, nil
			end
		end
		motion.phase, motion.coin, motion.tp.coin = "full", nil, nil
		FarmMotion.move(context, root, velocity, gyro, FarmMotion.shootSpot(context, state, motion, root, now), delta,
			math.clamp(tonumber(context.configuration.farm.speed) or 21, 8, 22))
		return
	end
	if full and not gun and method ~= "Teleport" then
		motion.phase, motion.coin = "full", nil
		if shooting or zombieTarget then
			FarmMotion.move(context, root, velocity, gyro, FarmMotion.shootSpot(context, state, motion, root, now), delta,
				math.clamp(tonumber(context.configuration.farm.speed) or 21, 8, 22))
			return
		end
		if not motion.hold or (motion.hold - root.Position).Magnitude > 60 then motion.hold = FarmMotion.hideAt(context, state, motion, root, now) end
		FarmMotion.move(context, root, velocity, gyro, motion.hold, delta, math.clamp(tonumber(context.configuration.farm.speed) or 21, 8, 22))
		return
	end
	if zombieShooting then
		motion.phase = "zshoot"
		FarmMotion.move(context, root, velocity, gyro, FarmMotion.shootSpot(context, state, motion, root, now), delta,
			math.clamp(tonumber(context.configuration.farm.speed) or 21, 8, 22))
		return
	end
	motion.hold, motion.spot = nil, nil
	local options = {time = now, tickTime = tickTime, root = root, velocity = velocity, gyro = gyro, character = character,
		delta = delta, gun = gun, murderer = murderer, full = full, hasGun = SheriffRuntime.gun(context) ~= nil}
	if method == "Teleport" then state.firstJump = false; FarmMotion.teleport(context, state, motion, options); return end
	if method == "Fast" then FarmMotion.fast(context, state, motion, options); return end
	FarmMotion.underMap(context, state, motion, options)
end

function FarmRuntime.install(context)
	local environment, globals = context.environment, context.environment.getgenv()
	globals.PULSE_FARM_MM2_VERSION = (globals.PULSE_FARM_MM2_VERSION or 0) + 1
	local generation = globals.PULSE_FARM_MM2_VERSION
	globals.PULSE_FARM_MM2_HEARTBEAT = true
	local state, motion = FarmState.create(context), FarmMotion.create()
	motion.ctRestore = function() FarmMotion.restoreTouch(motion) end
	globals.__pulseFarmUF = motion
	local api = {}
	api.setEnabled = function(value) FarmRuntime.setEnabled(context, state, motion, value) end
	api.getEnabled = function() return state.enabled end
	local setters = {Respawn = "respawnOnFull", AntiFling = "antiFling", Avoid = "avoidMurder", AutoFling = "autoFling",
		KillAura = "killAura", PickGun = "pickGun", ShootFull = "shootOnFull"}
	for name, key in setters do
		api["set" .. name] = function(value) state[key] = value; FarmState.save(context, state) end
		api["get" .. name] = function() return state[key] end
	end
	api.getSession = function() return state.session end
	api.getTotal = function() return state.total end
	api.getRate = function() return state.rate end
	api.getStatus = function()
		if not state.round then return "waiting" end
		if not state.active then return "spectating" end
		return FarmState.isFull(context, state, environment.tick()) and "full" or "farming"
	end
	globals.__pulseFarmAPI = api
	FarmRuntime.installRejoin(context, api)
	local function active() return globals.PULSE_FARM_MM2_HEARTBEAT and globals.PULSE_FARM_MM2_VERSION == generation and context.isAlive() end
	local mapFarm = FarmRuntime.MapFarm.install(context, active)
	local runtime = {roundAt = 0, gunAt = 0}
	context.connect(context.runService.Heartbeat, function(delta)
		if not active() then return end
		local ok, message = pcall(FarmRuntime.tick, context, state, motion, runtime, delta)
		if not ok then motion.err = tostring(message); FarmMotion.reset(context, motion, context.player.Character) end
	end)
	context.connect(context.runService.Stepped, function()
		if not active() or motion.phase == "idle" then return end
		local character = context.player.Character
		if not character then return end
		local parts
		if type(globals.__pulseNoclipParts) == "function" then
			local ok, result = pcall(globals.__pulseNoclipParts, character)
			if ok and type(result) == "table" then parts = result end
		end
		if not parts then
			parts = {}
			for _, part in character:GetDescendants() do if part:IsA("BasePart") then table.insert(parts, part) end end
		end
		for _, part in parts do if part.Parent and part.CanCollide then part.CanCollide = false end end
	end)
	environment.task.spawn(function()
		local remotes = context.storage:WaitForChild("Remotes", 20)
		local gameplay = remotes and remotes:WaitForChild("Gameplay", 20)
		if not gameplay or not active() then return end
		local function listen(name, callback)
			local remote = gameplay:FindFirstChild(name)
			if remote then context.connect(remote.OnClientEvent, callback) end
		end
		listen("RoundStart", function()
			FarmState.roundStart(state, environment.tick(), false)
			state.firstJump = true
			FarmRuntime.resetMotion(motion)
		end)
		listen("CoinsStarted", function()
			FarmState.roundStart(state, environment.tick(), true)
			state.firstJump = true
			FarmRuntime.resetMotion(motion)
		end)
		listen("RoundEndFade", function() FarmState.roundEnd(state) end)
		listen("CoinCollected", function(kind, count, maximum)
			if state.enabled then pcall(mapFarm.hop) end
			motion.evAt = environment.os.clock()
			local character = context.player.Character
			local root = character and character:FindFirstChild("HumanoidRootPart")
			motion.evPos = motion.tp.touchC and motion.tp.touchC.Parent and motion.evAt - (motion.tp.touchT or 0) < 1.5
				and motion.tp.touchC.Position or root and root.Position or nil
			FarmState.coinCollected(state, environment.tick(), kind, count, maximum)
		end)
	end)
	environment.task.spawn(function()
		local history, elapsed, last = {}, 0, environment.tick()
		while active() do
			local now = environment.tick()
			if state.enabled then
				elapsed += math.min(now - last, 10)
				table.insert(history, {elapsed, state.session})
				while #history > 2 and elapsed - history[1][1] > 3600 do table.remove(history, 1) end
				local interval = history[#history][1] - history[1][1]
				if interval >= 30 then state.rate = math.max(0, math.floor((history[#history][2] - history[1][2]) / interval * 60 + 0.5)) end
			end
			environment.task.wait(2)
			last = now
		end
	end)
	environment.task.spawn(function()
		while active() do
			environment.task.wait(5)
			if state.dirty then state.dirty = false; FarmState.save(context, state) end
		end
	end)
	local function stop()
		FarmRuntime.setEnabled(context, state, motion, false)
		if globals.PULSE_FARM_MM2_VERSION == generation then globals.PULSE_FARM_MM2_HEARTBEAT = false end
	end
	return state, motion, api, stop
end

	mm2 = "offline://pulsehub.gg/assets/mm2wh/mm2.png"}

function FarmRuntime.Webhook.valid(url)
	return type(url) == "string" and (url:match("^offline://[%w%.]*discord%.com/api/webhooks/%d+/[%w%-_]+") ~= nil
		or url:match("^offline://[%w%.]*discordapp%.com/api/webhooks/%d+/[%w%-_]+") ~= nil)
end

function FarmRuntime.Webhook.number(value)
	local number = math.floor(tonumber(value) or 0)
	local text = tostring(math.abs(number)):reverse():gsub("(%d%d%d)", "%1,"):reverse()
	if text:sub(1, 1) == "," then text = text:sub(2) end
	return (number < 0 and "-" or "") .. text
end

function FarmRuntime.Webhook.duration(value)
	local seconds = math.max(0, math.floor(value))
	local hours, minutes = math.floor(seconds / 3600), math.floor(seconds % 3600 / 60)
	if hours > 0 then return string.format("%dh %02dm", hours, minutes) end
	return string.format("%dm %02ds", minutes, seconds % 60)
end

function FarmRuntime.Webhook.load(context)
	local environment = context.environment
	local settings = {url = "", on = false, every = 10, kick = true}
	pcall(function()
		if not environment.isfile or not environment.isfile("PulseHub/mm2_webhook.json") then return end
		local data = context.http:JSONDecode(environment.readfile("PulseHub/mm2_webhook.json"))
		if type(data) ~= "table" then return end

		if type(data.url) == "string" then settings.url = data.url end
		settings.on = data.on == true
		settings.every = math.clamp(tonumber(data.every) or 10, 1, 60)
		settings.kick = data.kick ~= false
	end)
	return settings
end

function FarmRuntime.Webhook.save(state)
	local environment = state.context.environment
	pcall(function()
		if environment.isfolder and environment.makefolder and not environment.isfolder("PulseHub") then environment.makefolder("PulseHub") end
		environment.writefile("PulseHub/mm2_webhook.json", state.context.http:JSONEncode(state.cfg))
	end)
end

function FarmRuntime.Webhook.getIcons(state)
	if state.icons then return state.icons end

	local context, environment = state.context, state.context.environment
	state.icons = FarmRuntime.Webhook.icons
	pcall(function()
		local response = context.http:JSONDecode(environment.game:HttpGet("offline://thumbnails.roblox.com/v1/assets?assetIds=3150319038,121588790682191&returnPolicy=PlaceHolder&size=250x250&format=Png&isCircular=false"))
		local icons = table.clone(FarmRuntime.Webhook.icons)
		for _, item in response.data or {} do
			if item.state ~= "Completed" or type(item.imageUrl) ~= "string" then continue end
			if item.targetId == 3150319038 then icons.coin = item.imageUrl end
			if item.targetId == 121588790682191 then icons.candy = item.imageUrl end
		end

		response = context.http:JSONDecode(environment.game:HttpGet("offline://thumbnails.roblox.com/v1/places/gameicons?placeIds=142823291&returnPolicy=PlaceHolder&size=256x256&format=Png&isCircular=false"))
		local item = response.data and response.data[1]
		if item and item.state == "Completed" and type(item.imageUrl) == "string" then icons.mm2 = item.imageUrl end
		state.icons = icons
	end)
	return state.icons
end

function FarmRuntime.Webhook.profile(state)
	local context, environment = state.context, state.context.environment
	local result = {}
	pcall(function()
		local modules = context.storage:FindFirstChild("Modules")
		local profile = environment.require(modules:FindFirstChild("ProfileData"))
		local levels = environment.require(modules:FindFirstChild("LevelModule"))
		local materials = profile.Materials and profile.Materials.Owned or {}
		result.coins = tonumber(materials.Coins)

		local latest, candy
		for name, value in materials do
			local year = type(name) == "string" and tonumber(name:match("^Candies(%d+)$"))
			if year and tonumber(value) and (not latest or year > latest) then latest, candy = year, name end
		end
		if candy then result.candies = tonumber(materials[candy]) end
		result.level = tonumber(levels.GetLevel(profile.NewXP))
		result.prestige = tonumber(profile.Prestige) or 0
	end)
	return result
end

		footer = {text = "MYTHIC HUB", icon_url = icons.coin}, timestamp = environment.DateTime.now():ToIsoDate()}
end

function FarmRuntime.Webhook.post(state, payload)
	local environment = state.context.environment
	local request = environment.syn and environment.syn.request or environment.http_request or environment.request or environment.http and environment.http.request
	if type(request) ~= "function" then return false, "no http" end

	local body = state.context.http:JSONEncode(payload)
	local ok, response = pcall(request, {Url = state.cfg.url, Method = "POST", Headers = {["Content-Type"] = "application/json"}, Body = body})
	if not ok then return false, tostring(response) end

	local status = type(response) == "table" and tonumber(response.StatusCode or response.status) or 0
	return status >= 200 and status < 300, status
end

function FarmRuntime.Webhook.send(state, kind, reason)
	if not FarmRuntime.Webhook.valid(state.cfg.url) then return false, "url" end

	local embed = FarmRuntime.Webhook.embed(state, kind, reason)
	local success, code = FarmRuntime.Webhook.post(state, {username = "MYTHIC HUB", avatar_url = (state.icons or FarmRuntime.Webhook.icons).mm2, embeds = {embed}})
	if success then
		state.sent += 1
		if kind ~= "kick" then
			local api = state.context.environment.getgenv().__pulseFarmAPI
			state.lastSess = api and api.getSession and api.getSession() or state.lastSess
		end
	end
	state.lastCode = code
	return success, code
end

function FarmRuntime.Webhook.onKick(state, reason)
	if not state.cfg.kick or not FarmRuntime.Webhook.valid(state.cfg.url) then return end

	local now = state.context.environment.os.clock()
	if now - state.kickAt < 60 then return end
	state.kickAt = now
	FarmRuntime.Webhook.send(state, "kick", reason)
end

function FarmRuntime.Webhook.step(state)
	if not state.cfg.on or not FarmRuntime.Webhook.valid(state.cfg.url) then return end

	local environment = state.context.environment
	local api = environment.getgenv().__pulseFarmAPI
	local now = environment.os.clock()
	if not (api and api.getEnabled and api.getEnabled()) then state.lastSent = now; return end
	if now - state.lastSent < state.cfg.every * 60 then return end

	state.lastSent = now
	FarmRuntime.Webhook.send(state, "report")
end

local RoleFarm = {}

function RoleFarm.hold(context, state, value)
	value = not not value
	if value then state.acted = true end
	if state.holding == value then return end
	state.holding = value
	context.environment.getgenv().__pulseRoleFarmHold = value or nil
end

function RoleFarm.mine(context)
	local ok, roles = pcall(context.roles)
	if not ok or type(roles) ~= "table" then return nil, {} end
	local role = roles[context.player.Name]
	if type(role) ~= "table" or not role.Role or role.Dead == true or role.Killed == true then return nil, roles end
	return tostring(role.Role), roles
end

function RoleFarm.who(context, roles, roleName)
	for name, role in roles do
		if type(role) ~= "table" or not role.Role or role.Dead == true or role.Killed == true
			or not tostring(role.Role):lower():find(roleName) then continue end
		local player = context.players:FindFirstChild(name)
		local character = player and player ~= context.player and player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		if humanoid and humanoid.Health > 0 then return player, character end
	end
	return nil, nil
end

function RoleFarm.lobby(context)
	local lobby = context.workspace:FindFirstChild("RegularLobby") or context.workspace:FindFirstChild("Lobby")
	if not lobby then return nil end
	local spawns = lobby:FindFirstChild("Spawns")
	local spawn = spawns and spawns:FindFirstChildWhichIsA("BasePart") or lobby:FindFirstChildWhichIsA("SpawnLocation", true)
	return spawn and spawn.CFrame + context.environment.Vector3.new(0, 4, 0) or nil
end

function RoleFarm.toLobby(context, state)
	local character = context.player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	local destination = RoleFarm.lobby(context)
	if not root or not destination then return false end
	if (root.Position - destination.Position).Magnitude < 300 then return true end
	pcall(function()
		root.AssemblyLinearVelocity, root.AssemblyAngularVelocity = context.environment.Vector3.zero, context.environment.Vector3.zero
		root.CFrame = destination
	end)
	state.lastTp = context.environment.os.clock()
	return true
end

function RoleFarm.gun(context)
	local character = context.player.Character
	local gun = character and character:FindFirstChild("Gun")
	if gun then return gun, true end
	local backpack = context.player:FindFirstChildOfClass("Backpack")
	return backpack and backpack:FindFirstChild("Gun"), false
end

function RoleFarm.shoot(context, state, roles)
	local _, target = RoleFarm.who(context, roles, "murder")
	if not target then return false end
	local gun, equipped = RoleFarm.gun(context)
	if not gun then return false end
	if not equipped then
		local character = context.player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		if humanoid then pcall(function() humanoid:EquipTool(gun) end) end
		return true
	end
	state.target = target.Name
	local globals = context.environment.getgenv()
	local aim = rawget(globals, "__pulseAimX")
	if type(aim) == "table" and aim.alive and type(aim.fireFor) == "function" then
		pcall(aim.fireFor, target, false, false)
		return true
	end
	if type(globals.__pulseShotMurder) == "function" then pcall(globals.__pulseShotMurder) end
	return true
end

function RoleFarm.stabAll(context, roles)
	local environment, character = context.environment, context.player.Character
	local knife = character and character:FindFirstChild("Knife")
	if not knife then
		local backpack = context.player:FindFirstChildOfClass("Backpack")
		local stored = backpack and backpack:FindFirstChild("Knife")
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		if stored and stored:IsA("Tool") and humanoid then pcall(function() humanoid:EquipTool(stored) end) end
		knife = character and character:FindFirstChild("Knife")
	end
	if not knife or not environment.firetouchinterest then return 0 end
	local handle = knife:FindFirstChild("Handle") or knife:FindFirstChildWhichIsA("BasePart")
	if not handle then return 0 end
	local count = 0
	for _, player in context.players:GetPlayers() do
		local role = player ~= context.player and roles[player.Name]
		if type(role) ~= "table" or role.Dead == true or role.Killed == true then continue end
		local target = player.Character
		local humanoid = target and target:FindFirstChildOfClass("Humanoid")
		if not humanoid or humanoid.Health <= 0 then continue end
		count += 1
		for _, name in {"HumanoidRootPart", "UpperTorso", "Torso", "LowerTorso", "Head"} do
			local part = target:FindFirstChild(name)
			if not part then continue end
			pcall(function()
				environment.firetouchinterest(handle, part, 0)
				environment.firetouchinterest(handle, part, 1)
				environment.firetouchinterest(part, handle, 0)
				environment.firetouchinterest(part, handle, 1)
			end)
		end
	end
	if count > 0 then pcall(function() knife:Activate() end) end
	return count
end

function RoleFarm.touch(context, drop)
	local character, environment = context.player.Character, context.environment
	if not character or not drop or not environment.firetouchinterest then return end
	for _, part in character:GetChildren() do
		if not part:IsA("BasePart") then continue end
		pcall(function() environment.firetouchinterest(part, drop, 0); environment.firetouchinterest(part, drop, 1) end)
	end
end

function RoleFarm.grab(context, state)
	local drop = context.drops.findDrop()
	if not drop then return false end
	local function alive()
		local character = context.player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		return humanoid and humanoid.Health > 0
	end
	local started = context.environment.os.clock()
	while context.environment.os.clock() - started < 0.6 and drop.Parent and not RoleFarm.gun(context) and alive() do
		RoleFarm.touch(context, drop)
		context.runService.Heartbeat:Wait()
	end
	if RoleFarm.gun(context) then return true end
	if not drop.Parent then return false end
	local character = context.player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	if not root then return false end
	started = context.environment.os.clock()
	while context.environment.os.clock() - started < 1.2 and drop.Parent and not RoleFarm.gun(context) and alive() do
		pcall(function()
			root.AssemblyLinearVelocity = context.environment.Vector3.zero
			root.CFrame = context.environment.CFrame.new(drop.Position + context.environment.Vector3.new(0, 2.5, 0))
		end)
		RoleFarm.touch(context, drop)
		context.runService.Heartbeat:Wait()
	end
	local picked = RoleFarm.gun(context) ~= nil
	RoleFarm.toLobby(context, state)
	return picked
end

function RoleFarm.count(context, state, roles)
	for name, role in roles do
		if name == context.player.Name or type(role) ~= "table" then continue end
		if role.Dead ~= true and role.Killed ~= true then state.seen[name] = true; continue end
		if not state.seen[name] then continue end
		state.seen[name] = nil
		if state.state == "Killing" or state.state == "Shooting" and state.target == name then state.kills += 1 end
	end
end

function RoleFarm.expInnocent(context, state, roles)
	if RoleFarm.gun(context) then
		RoleFarm.toLobby(context, state)
		state.state = "Shooting"
		RoleFarm.shoot(context, state, roles)
		return
	end
	local _, sheriff = RoleFarm.who(context, roles, "sheriff")
	if not sheriff then _, sheriff = RoleFarm.who(context, roles, "hero") end
	local now = context.environment.os.clock()
	if sheriff and now >= state.flingWait and (state.flung[sheriff] or 0) < 4 then
		state.flung[sheriff] = (state.flung[sheriff] or 0) + 1
		state.state = "Flinging"
		local fling = rawget(context.environment.getgenv(), "__pulseFlingDown")
		local ok, result = false, nil
		if type(fling) == "function" then ok, result = pcall(fling, sheriff, 0.8, {home = RoleFarm.lobby(context)}) end
		state.flingWait = context.environment.os.clock() + (ok and result == true and 2.5 or 0.5)
		RoleFarm.toLobby(context, state)
		return
	end
	if context.drops.findDrop() then state.state = "Picking Gun"; RoleFarm.grab(context, state); return end
	RoleFarm.toLobby(context, state)
	state.state = "Surviving"
end

function RoleFarm.step(context, state)
	local enabled = state.on
	if not (enabled.inno or enabled.sheriff or enabled.murder or enabled.exp) then
		state.state, state.role = "Off", nil
		RoleFarm.hold(context, state, false)
		return
	end
	local role, roles = RoleFarm.mine(context)
	state.role = role
	local character = context.player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not role or not humanoid or humanoid.Health <= 0 then
		state.state = "Waiting"
		RoleFarm.hold(context, state, false)
		if not role then table.clear(state.seen) end
		return
	end
	RoleFarm.count(context, state, roles)
	local name = role:lower()
	if name:find("murder") and (enabled.murder or enabled.exp) then
		RoleFarm.hold(context, state, true)
		state.state = "Killing"
		RoleFarm.stabAll(context, roles)
		return
	end
	if name == "sheriff" and (enabled.sheriff or enabled.exp) or name == "hero" and enabled.exp then
		RoleFarm.hold(context, state, true)
		RoleFarm.toLobby(context, state)
		state.state = "Shooting"
		RoleFarm.shoot(context, state, roles)
		return
	end
	if name == "hero" and enabled.inno then
		RoleFarm.hold(context, state, true)
		RoleFarm.toLobby(context, state)
		state.state = "Surviving"
		return
	end
	if name ~= "hero" and not name:find("murder") and name ~= "sheriff" and enabled.exp then
		RoleFarm.hold(context, state, true)
		RoleFarm.expInnocent(context, state, roles)
		return
	end
	if name ~= "hero" and not name:find("murder") and name ~= "sheriff" and enabled.inno then
		RoleFarm.hold(context, state, true)
		RoleFarm.toLobby(context, state)
		state.state = "Surviving"
		return
	end
	RoleFarm.hold(context, state, false)
	state.state = "Waiting"
end

function RoleFarm.install(context)
	local environment, globals = context.environment, context.environment.getgenv()
	local state = {on = {inno = false, sheriff = false, murder = false, exp = false}, state = "Off", role = nil,
		log = {}, drops = {}, holding = false, acted = false, wins = 0, kills = 0, seen = {},
		flung = setmetatable({}, {__mode = "k"}), flingWait = 0, lastTp = 0, conns = {}}
	state.say = function(message)
		table.insert(state.log, string.format("%s %s", environment.os.date("%H:%M:%S"), tostring(message)))
		if #state.log > 40 then table.remove(state.log, 1) end
	end
	state.mine = function() return RoleFarm.mine(context) end
	state.hold = function(value) RoleFarm.hold(context, state, value) end
	state.step = function() return RoleFarm.step(context, state) end
	state.toLobby = function() return RoleFarm.toLobby(context, state) end
	state.gun = function() return RoleFarm.gun(context) end
	state.shoot = function(roles) return RoleFarm.shoot(context, state, roles) end
	state.stabAll = function(roles) return RoleFarm.stabAll(context, roles) end
	globals.__pulseRoleFarm = state
	environment.task.spawn(function()
		while context.isAlive() do
			environment.task.wait(0.1)
			local ok, message = pcall(RoleFarm.step, context, state)
			if not ok then pcall(state.say, "ERR " .. tostring(message)); environment.task.wait(1) end
		end
		RoleFarm.hold(context, state, false)
	end)
	environment.task.spawn(function()
		local remotes = context.storage:WaitForChild("Remotes", 30)
		local gameplay = remotes and remotes:WaitForChild("Gameplay", 30)
		local victory = gameplay and gameplay:WaitForChild("VictoryScreen", 30)
		if not victory or not context.isAlive() then return end
		table.insert(state.conns, context.connect(victory.OnClientEvent, function(_, mode, winner)
			local sounds = rawget(globals, "__pulseGameSounds")
			local winners = type(sounds) == "table" and sounds.WIN and sounds.WIN[tostring(mode)]
			if state.acted and winners and winners[tostring(winner)] == true then state.wins += 1 end
			pcall(state.say, "END " .. tostring(mode) .. " " .. tostring(winner))
			state.acted = false
		end))
	end)
	local function stop()
		state.on.inno, state.on.sheriff, state.on.murder, state.on.exp = false, false, false, false
		RoleFarm.hold(context, state, false)
		for _, connection in state.conns do pcall(function() connection:Disconnect() end) end
		globals.__pulseRoleFarmUI = nil
	end
	return state, stop
end

function RoleFarm.controls(context, state)
	local section, environment = context.section, context.environment
	section:Sub("Role")
	section:Header("Stats")
	local grid, cards = section:Grid({Columns = 2, Height = 66, MinWidth = 150}), {}
	for _, options in {{"Role", "Accent", "user-round"}, {"Status", "Success", "activity"},

	end

	state.prestige = prestige
	section:Toggle({Text = "Auto Prestige", Default = false, Callback = function(value) prestige.on = not not value end})
	local numerals = {[0] = "0", "I", "II", "III", "IV", "V", "VI", "VII", "VIII", "IX", "X"}
	local function level()
		if prestige.pd == nil then
			prestige.pd = false
			pcall(function()
				local modules = context.storage:WaitForChild("Modules", 5)
				prestige.lm = environment.require(modules:WaitForChild("LevelModule", 5))
				prestige.pd = environment.require(modules:WaitForChild("ProfileData", 5))
			end)
		end
		if not prestige.pd or not prestige.lm then return nil, nil end
		local ok, value = pcall(prestige.lm.GetLevel, prestige.pd.NewXP)
		return ok and tonumber(value) or nil, tonumber(prestige.pd.Prestige) or 0
	end
	environment.task.spawn(function()
		while context.isAlive() do
			environment.task.wait(2)
			pcall(function()
				local current, rank = level()
				if not current then return end
				prestige.level, prestige.pres = current, rank
				if not prestige.on or current < 100 or rank >= 10 or environment.os.clock() - prestige.last <= 15 then return end
				local remotes = context.storage:FindFirstChild("Remotes")
				local inventory = remotes and remotes:FindFirstChild("Inventory")
				local remote = inventory and inventory:FindFirstChild("Prestige")
				if not remote or not remote:IsA("RemoteEvent") then return end
				prestige.last = environment.os.clock()
				remote:FireServer()
				prestige.done += 1
			end)
		end
	end)
	environment.task.spawn(function()
		while context.isAlive() do
			environment.task.wait(0.4)
			pcall(function()
				local values = {Role = state.role and context.translation.tr(state.role) or "-", Status = context.translation.tr(state.state),
					Wins = state.wins, Kills = state.kills, Level = prestige.level or "-",
					Prestige = prestige.pres and (numerals[prestige.pres] or tostring(prestige.pres)) or "-"}
				for name, card in cards do if card then card:Set(tostring(values[name])) end end
			end)
		end
	end)
end

local BoxFarm = {}
local BOX_CURRENCIES = {"HalloweenKey2026", "Key", "Candies2026", "Coins"}
local BOX_LABELS = {Candies2026 = "Candies", HalloweenKey2026 = "Halloween Key", Coins = "Coins", Key = "Key", Gems = "Gems"}

function BoxFarm.data(context, state)
	if not state.database then
		pcall(function()
			local database = context.storage:FindFirstChild("Database")
			state.database = context.environment.require(database:FindFirstChild("Sync"))
		end)
	end
	if not state.profile then
		pcall(function()
			local modules = context.storage:FindFirstChild("Modules")
			state.profile = context.environment.require(modules:FindFirstChild("ProfileData"))
		end)
	end
	return state.database, state.profile
end

function BoxFarm.catalog(context, state)
	local database = BoxFarm.data(context, state)
	local boxes = {}
	if not database then return boxes end
	for id, item in database.NewShop or {} do
		if type(item) ~= "table" or item.Type ~= "MysteryBox" or type(item.Price) ~= "table" or item.Price.Robux ~= nil or next(item.Price) == nil then continue end
		local description = database.MysteryBox and database.MysteryBox[id]
		table.insert(boxes, {id = id, name = tostring(description and (description.ItemName or description.Name) or id),
			price = item.Price, order = tonumber(item.LayoutOrder) or 999})
	end
	if #boxes == 0 then
		for id, item in database.Shop and database.Shop.Weapons or {} do
			local description = database.MysteryBox and database.MysteryBox[id]
			if not description or type(item) ~= "table" or type(item.Price) ~= "table" or item.Price.Robux ~= nil then continue end
			table.insert(boxes, {id = id, name = tostring(description.ItemName or description.Name or id), price = item.Price, order = 999})
		end
	end
	table.sort(boxes, function(first, second)
		if first.order ~= second.order then return first.order < second.order end
		return first.name < second.name
	end)
	return boxes
end

function BoxFarm.balance(context, state, currency)
	local _, profile = BoxFarm.data(context, state)
	if not profile then return 0 end
	local value
	pcall(function() value = profile[currency] end)
	if type(value) ~= "number" then pcall(function() value = profile.Materials.Owned[currency] end) end
	if type(value) ~= "number" then pcall(function() value = profile.Weapons.Owned[currency] end) end
	return tonumber(value) or 0
end

function BoxFarm.currency(context, state, box)
	if not box then return nil end
	local selected = context.configuration.boxes.cur
	if selected ~= "Auto" then
		local price = box.price[selected]
		if price and BoxFarm.balance(context, state, selected) >= price then return selected, price end
		return nil, price, selected
	end
	local first
	for _, currency in BOX_CURRENCIES do
		local price = box.price[currency]
		if not price then continue end
		first = first or currency
		if BoxFarm.balance(context, state, currency) >= price then return currency, price end
	end
	return nil, nil, first
end

function BoxFarm.find(context, state, id)
	for _, box in BoxFarm.catalog(context, state) do if box.id == id then return box end end
	return nil
end

function BoxFarm.open(context, state)
	local settings = context.configuration.boxes
	local box = BoxFarm.find(context, state, settings.id)
	if not box then state.state = "Off"; return false end
	local currency = BoxFarm.currency(context, state, box)
	if not currency then state.state = "Not enough"; return false end
	state.state = "Opening"
	local remotes = context.storage:FindFirstChild("Remotes")
	local shop = remotes and remotes:FindFirstChild("Shop")
	local open = shop and shop:FindFirstChild("OpenCrate")
	if not open then state.state = "Off"; return false end
	local ok, reward = pcall(function() return open:InvokeServer(box.id, "MysteryBox", currency) end)
	if not ok or not reward then state.state = "Not enough"; return false end
	state.opened, state.last = state.opened + 1, reward
	local database = BoxFarm.data(context, state)
	local weapon = database and database.Weapons and database.Weapons[reward]
	state.lastName, state.lastRarity = tostring(weapon and (weapon.ItemName or weapon.Name) or reward), weapon and weapon.Rarity or nil
	if settings.anim then
		pcall(function()
			local controller = shop:FindFirstChild("BoxController")
			if controller then controller:Fire({{MysteryBoxId = box.id, RewardedItemId = reward}}) end
		end)
	end
	local rare = {Legendary = true, Godly = true, Ancient = true, Unique = true, Vintage = true, Chroma = true}
	if state.lastRarity and rare[tostring(state.lastRarity)] then
		pcall(function()
			local nova = context.environment.getgenv().__pulseMM2Nova
			if nova and nova.Notify then nova:Notify({Title = tostring(state.lastRarity) .. ": " .. state.lastName, Kind = "success"}) end
		end)
	end
	return true
end

function BoxFarm.install(context)
	local settings, environment = context.configuration, context.environment
	settings.boxes = type(settings.boxes) == "table" and settings.boxes or {}
	if type(settings.boxes.id) ~= "string" then settings.boxes.id = "Halloween2026Box" end
	if type(settings.boxes.cur) ~= "string" then settings.boxes.cur = "Auto" end
	if tonumber(settings.boxes.delay) == nil then settings.boxes.delay = 1 end
	local state = {on = false, opened = 0, state = "Off", busy = false}
	state.openOnce = function() return BoxFarm.open(context, state) end
	environment.getgenv().__pulseBoxes = state
	environment.task.spawn(function()
		while context.isAlive() do
			if state.on and not state.busy then
				state.busy = true
				local ok, opened = pcall(BoxFarm.open, context, state)
				state.busy = false
				environment.task.wait(ok and opened and math.clamp(tonumber(settings.boxes.delay) or 1, 0.5, 10) or 2.5)
				continue
			end
			if not state.on and not state.busy then state.state = "Off" end
			environment.task.wait(0.3)
		end
	end)
	return state, function() state.on = false; environment.getgenv().__pulseBoxUI = nil end
end

local FlingFeature = {}

function FlingFeature.installTool(context)
	local environment, player = context.environment, context.player
	local enabled, respawn = false, nil
	local function remove()
		for _, container in {player:FindFirstChild("Backpack"), player.Character} do
			if not container then continue end
			for _, tool in container:GetChildren() do
				if tool:IsA("Tool") and tool.Name == "Jerk Off R15" then tool:Destroy() end
			end
		end
	end
	local function create()
		if not enabled then return end
		local backpack = player:FindFirstChild("Backpack") or player:WaitForChild("Backpack", 5)
		if not backpack then return end
		remove()
		local tool = environment.Instance.new("Tool")
		tool.Name, tool.RequiresHandle = "Jerk Off R15", false
		local animation
		context.connect(tool.Equipped, function()
			local character = player.Character or player.CharacterAdded:Wait()
			local humanoid = character:FindFirstChildOfClass("Humanoid")
			if not humanoid then return end
			local asset = environment.Instance.new("Animation")
			asset.AnimationId = "rbxassetid://698251653"
			animation = humanoid:LoadAnimation(asset)
			animation.Looped = false
			animation:Play()
			animation.TimePosition = 0.58
			animation:AdjustSpeed(0.4)
			environment.task.spawn(function()
				while animation and animation.IsPlaying do
					if animation.TimePosition >= 0.63 then animation.TimePosition = 0.58 end
					environment.task.wait()
				end
			end)
		end)
		context.connect(tool.Unequipped, function()
			if animation then animation:Stop(); animation = nil end
		end)
		tool.Parent = backpack
	end
	local function setEnabled(value)
		enabled = value
		if enabled then
			create()
			respawn = respawn or context.connect(player.CharacterAdded, function()
				environment.task.wait(0.5)
				create()
			end)
			return
		end
		if respawn then respawn:Disconnect(); respawn = nil end
		remove()
	end
	context.section:Toggle({Text = "Jerk Off Tool (R15)", Default = false, Callback = setEnabled})
	return function() setEnabled(false) end
end

function FlingFeature.down(context, state, target, duration, options)
	options = options or {}
	if state.busy then return false, "busy" end
	if not target or not target.Parent then return false, "no-target" end
	local targetHumanoid = target:FindFirstChildOfClass("Humanoid")
	local targetRoot = target:FindFirstChild("HumanoidRootPart") or target:FindFirstChild("Head")
	if not targetHumanoid or not targetRoot or targetHumanoid.Health <= 0 then return false, "dead" end
	local character = context.player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	local root = character and character:FindFirstChild("HumanoidRootPart")
	if not humanoid or not root or humanoid.Health <= 0 then return false, "me" end
	local environment, workspace, run = context.environment, context.workspace, context.runService
	local vector, frame, globals = environment.Vector3, environment.CFrame, environment.getgenv()
	local now = environment.os.clock
	state.busy = true
	duration = tonumber(duration) or 0.8
	globals.__pulseManualFling, globals.__pulseTouchFlinging = true, true
	globals.__pulseInvisPause = now() + duration + 1.5
	if state.fpdh == nil then state.fpdh = workspace.FallenPartsDestroyHeight end
	local home, camera = options.home or root.CFrame, workspace.CurrentCamera
	local subject, seated, platform = camera and camera.CameraSubject, true, false
	pcall(function() seated = humanoid:GetStateEnabled(environment.Enum.HumanoidStateType.Seated) end)
	pcall(function() platform = humanoid.PlatformStand end)
	pcall(function() workspace.FallenPartsDestroyHeight = 0 / 0 end)
	pcall(function() humanoid:SetStateEnabled(environment.Enum.HumanoidStateType.Seated, false) end)
	if options.cam ~= false then pcall(function() if camera then camera.CameraSubject = targetHumanoid end end) end
	local parts = {}
	pcall(function()
		for _, part in target:GetDescendants() do if part:IsA("BasePart") then table.insert(parts, {part, part.CanCollide}) end end
	end)
	local function disableCollision()
		for _, entry in parts do
			local part = entry[1]
			if part.Parent and part.CanCollide then pcall(function() part.CanCollide = false end) end
		end
	end
	disableCollision()
	local downward, startY = tonumber(rawget(globals, "__pulseFlingDownV")) or 120000, targetRoot.Position.Y
	local aim = rawget(globals, "__pulseAimX")
	local pingOk, ping = pcall(function() return context.player:GetNetworkPing() end)
	local lead = tonumber(rawget(globals, "__pulseFlingLead")) or math.clamp((aim and tonumber(aim.L) or 0.16)
		+ (pingOk and tonumber(ping) or 0.08) * 2 + 0.05, 0.15, 0.6)
	local speeds = {1.4, 0.5, 2.3, 0.9}
	local moveOk, moving = pcall(function() return targetHumanoid.MoveDirection.Magnitude * targetHumanoid.WalkSpeed end)
	local velocity, spoofCount = targetRoot.AssemblyLinearVelocity, 0
	if velocity.Magnitude > 1500 then velocity, spoofCount = vector.zero, 1 end
	local initialSpeed = vector.new(velocity.X, 0, velocity.Z).Magnitude
	if moveOk and moving > 2 or initialSpeed > 3 or spoofCount > 0 then duration = math.max(duration, 1.2) end
	local targetPlayer = context.players:GetPlayerFromCharacter(target)
	local userId = targetPlayer and targetPlayer.UserId
	state.leadMem = state.leadMem or {}
	local multiplier = 1
	local positions, candidates, offsets = {}, {}, {0, 0.3, 0.55, 0.8}
	local started, frames, done, result, minimumVelocity, minimumY = now(), 0, false, "timeout", 0, startY
	local lastTarget, teleported, lastSelf, firstPosition, filtered, filteredAt = targetRoot.Position, false, nil, nil, nil, nil
	local predicted, predictedAt, teleportAt = nil, nil, nil
	local params = environment.RaycastParams.new()
	params.FilterType, params.FilterDescendantsInstances = environment.Enum.RaycastFilterType.Exclude, {character, target}
	pcall(function() params.RespectCanCollide = true end)
	local stepped = run.Stepped:Connect(function()
		if done then return end
		pcall(function() root.AssemblyLinearVelocity, root.AssemblyAngularVelocity = vector.zero, vector.zero end)
		disableCollision()
	end)
	local heartbeat = run.Heartbeat:Connect(function()
		if done then return end
		frames += 1
		if not target.Parent or not targetRoot.Parent or targetHumanoid.Health <= 0 then done, result = true, "gone"; return end
		if not root.Parent or humanoid.Health <= 0 then done, result = true, "me"; return end
		local position, actualVelocity, time = targetRoot.Position, targetRoot.AssemblyLinearVelocity, now()
		if actualVelocity.Magnitude > 1500 then spoofCount += 1 end
		local observed
		if (position - lastTarget).Magnitude > 60 then
			if spoofCount == 0 then done, result = true, "tp-target"; return end
			teleportAt = teleportAt or time
			if time - teleportAt > 0.35 then
				if position.Y < startY - 25 then done, result = true, "down"; return end
				if (position - lastTarget).Magnitude > 300 then done, result = true, "tp-target"; return end
				lastTarget, teleportAt, observed = position, nil, position
			end
			if not observed then observed = lastTarget end
		end
		if not observed then lastTarget, teleportAt, observed = position, nil, position end
		table.insert(positions, {time, observed})
		while #positions > 2 and time - positions[1][1] > 0.25 do table.remove(positions, 1) end
		local measured = actualVelocity
		if spoofCount > 0 then
			local elapsed = time - positions[1][1]
			measured = elapsed > 0.06 and (observed - positions[1][2]) / elapsed or vector.zero
		end
		if lastSelf and (root.Position - lastSelf).Magnitude > 30 then done, result, teleported = true, "tp-me", true; return end
		minimumVelocity, minimumY = math.min(minimumVelocity, measured.Y), math.min(minimumY, observed.Y)
		if observed.Y < startY - 25 then done, result = true, "down"; return end
		if measured.Y < -250 then done, result = true, "falling"; return end
		if now() - started > duration then done = true; return end
		if firstPosition and (root.Position - firstPosition).Magnitude > 60 and (observed - firstPosition).Magnitude > 55 then done, result = true, "far"; return end
		local horizontal = vector.new(measured.X, 0, measured.Z)
		local movementOk, direction, speed = pcall(function() return targetHumanoid.MoveDirection, targetHumanoid.WalkSpeed end)
		local movement = horizontal
		if movementOk and direction and direction.Magnitude > 0.1 then
			local desired = vector.new(direction.X, 0, direction.Z).Unit * math.clamp(tonumber(speed) or 16, 0, 60)
			movement = horizontal.Magnitude > 40 and horizontal or desired:Lerp(horizontal, 0.25)
		end
		local movingNow = movement.Magnitude > 2
		local timeNow, targetPosition, height = now(), observed, nil
		if movingNow then
			if filtered and filteredAt then
				local limit = 30 * math.clamp(timeNow - filteredAt, 1 / 240, 0.05)
				local delta = vector.new(observed.X - filtered.X, 0, observed.Z - filtered.Z)
				if delta.Magnitude > limit then delta = delta.Unit * limit end
				filtered = vector.new(filtered.X + delta.X, observed.Y, filtered.Z + delta.Z)
			end
			if not filtered then filtered = observed end
			local cycle = frames % 8
			multiplier = offsets[cycle % 4 + 1]
			local delta = movement * multiplier
			if delta.Magnitude > 25 then delta = delta.Unit * 25 end
			if delta.Magnitude > 0.1 then
				local ok, hit = pcall(function() return workspace:Raycast(filtered, delta, params) end)
				if ok and hit then delta = delta.Unit * math.max(0, hit.Distance - 1.5) end
			end
			table.insert(candidates, {timeNow, multiplier})
			while #candidates > 0 and timeNow - candidates[1][1] > 1.2 do table.remove(candidates, 1) end
			targetPosition, height = filtered + delta, cycle < 4 and 1.4 or -0.6
		end
		if not movingNow then height = speeds[frames % #speeds + 1] end
		pcall(function()
			humanoid.PlatformStand = true
			local position = vector.new(targetPosition.X, math.max(targetPosition.Y, startY - 3) + height, targetPosition.Z)
			if predicted and predictedAt and not movingNow then
				local limit = (tonumber(rawget(globals, "__pulseFlingMaxSpeed")) or 20) * math.clamp(timeNow - predictedAt, 1 / 240, 0.05)
				local delta = vector.new(position.X - predicted.X, 0, position.Z - predicted.Z)
				if delta.Magnitude > limit then
					local reduced = predicted + delta.Unit * limit
					position = vector.new(reduced.X, position.Y, reduced.Z)
				end
			end
			local grounded = workspace:Raycast(position, vector.new(0, -15, 0), params) ~= nil
			if not grounded then
				local fallback = vector.new(lastTarget.X, math.max(lastTarget.Y, startY - 3) + height, lastTarget.Z)
				if (movingNow or not predicted or vector.new(fallback.X - predicted.X, 0, fallback.Z - predicted.Z).Magnitude
					<= (tonumber(rawget(globals, "__pulseFlingMaxSpeed")) or 20) * 0.05)
					and workspace:Raycast(fallback, vector.new(0, -15, 0), params) then position, grounded = fallback, true end
			end
			if movingNow then
				local direction = movement.Unit
				root.CFrame = frame.lookAt(position, position + vector.new(-direction.Z, 0, direction.X))
			end
			if not movingNow then root.CFrame = frame.new(position) end
			root.AssemblyAngularVelocity = vector.zero
			if grounded and movingNow then
				root.AssemblyLinearVelocity = vector.new(movement.X, -downward, movement.Z)
				local spin = tonumber(rawget(globals, "__pulseFlingSpin")) or 60000
				root.AssemblyAngularVelocity = vector.new(spin, spin, spin)
			end
			if grounded and not movingNow then root.AssemblyLinearVelocity = vector.new(0, -downward, 0) end
			if not grounded then root.AssemblyLinearVelocity = vector.zero end
			lastSelf, predicted, predictedAt, filteredAt = root.Position, position, timeNow, timeNow
			firstPosition = firstPosition or position
		end)
	end)
	while not done and now() - started < duration + 0.5 do run.Heartbeat:Wait() end
	done = true
	pcall(function() heartbeat:Disconnect() end)
	pcall(function() stepped:Disconnect() end)
	local function ground(position)
		local ok, hit = pcall(function()
			local filter = environment.RaycastParams.new()
			filter.FilterType, filter.FilterDescendantsInstances = environment.Enum.RaycastFilterType.Exclude, {character, target}
			return workspace:Raycast(position + vector.new(0, 2, 0), vector.new(0, -90, 0), filter) ~= nil
		end)
		return ok and hit
	end
	local ok, position = pcall(function() return root.Position end)
	local restore = not teleported and not (ok and (position - home.Position).Magnitude > 35 and ground(position))
		and (ground(home.Position) or not (ok and ground(position)))
	local restoreAt, stable = now(), 0
	while restore and now() - restoreAt < 1.5 do
		if not root.Parent then break end
		pcall(function() root.AssemblyLinearVelocity, root.AssemblyAngularVelocity, root.CFrame = vector.zero, vector.zero, home end)
		run.Heartbeat:Wait()
		local valid, distance = pcall(function() return (root.Position - home.Position).Magnitude end)
		if valid and distance > 20 then break end
		stable = valid and distance < 3 and stable + 1 or 0
		if stable >= 3 then break end
	end
	pcall(function() root.AssemblyLinearVelocity, root.AssemblyAngularVelocity = vector.zero, vector.zero end)
	pcall(function() humanoid.PlatformStand = platform end)
	pcall(function() if humanoid.Health > 0 then humanoid:ChangeState(environment.Enum.HumanoidStateType.GettingUp) end end)
	pcall(function() humanoid:SetStateEnabled(environment.Enum.HumanoidStateType.Seated, seated) end)
	pcall(function() if camera and subject then camera.CameraSubject = subject end end)
	for _, entry in parts do
		if entry[1].Parent and entry[1].CanCollide ~= entry[2] then pcall(function() entry[1].CanCollide = entry[2] end) end
	end
	if state.fpdh ~= nil then
		local height = state.fpdh
		state.fpdh = nil
		pcall(function() workspace.FallenPartsDestroyHeight = height end)
	end
	globals.__pulseInvisPause = now() + 0.2
	if userId and (result == "down" or result == "falling") and initialSpeed > 3 then
		local best, distance = nil, math.huge
		for index = #candidates, 1, -1 do
			local difference = math.abs(candidates[index][1] - (now() - lead))
			if difference < distance then best, distance = candidates[index][2], difference end
		end
		if best then state.leadMem[userId] = best end
	end
	globals.__pulseFlingLast = string.format("down res=%s frames=%d dy=%.1f vmin=%.0f V=%d", result, frames, minimumY - startY, minimumVelocity, downward)
	globals.__pulseFlingLog = type(globals.__pulseFlingLog) == "table" and globals.__pulseFlingLog or {}
	table.insert(globals.__pulseFlingLog, string.format("%s %s spd=%.0f lead=%.2f mult=%.2f mem=%s res=%s fr=%d t=%.2f dy=%.1f vmin=%.0f spoof=%d",
		environment.os.date("%H:%M:%S"), tostring(target.Name), initialSpeed, lead, multiplier, tostring(userId and state.leadMem[userId]),
		result, frames, now() - started, minimumY - startY, minimumVelocity, spoofCount))
	while #globals.__pulseFlingLog > 10 do table.remove(globals.__pulseFlingLog, 1) end
	globals.__pulseManualFling, globals.__pulseTouchFlinging, state.busy = false, false, false
	return result == "down" or result == "falling" or result == "gone", result
end

function FlingFeature.persist(context, state, player, options)
	options = type(options) == "table" and options or {}
	if not player or player == context.player or not player.Parent then return false, "bad" end
	local character = player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	if not root then return false, "no-target" end
	state.token += 1
	local token, startY, duration, started = state.token, root.Position.Y, tonumber(options.time) or 10, context.environment.os.clock()
	local environment, globals = context.environment, context.environment.getgenv()
	local result, attempts, low = "timeout", 0, nil
	state.target = player
	local function validTarget()
		local current = player.Character
		local humanoid = current and current:FindFirstChildOfClass("Humanoid")
		local currentRoot = current and current:FindFirstChild("HumanoidRootPart")
		if not player.Parent or not current or not humanoid or not currentRoot or humanoid.Health <= 0 or current ~= character then return nil, "gone" end
		local roles = context.roles()
		local role = roles and roles[player.Name]
		if type(role) == "table" and (role.Dead == true or role.Killed == true) then return nil, "dead" end
		if currentRoot.Position.Y < startY - 25 then
			low = low or environment.os.clock()
			if environment.os.clock() - low >= 0.25 then return nil, "down" end
		end
		if currentRoot.Position.Y >= startY - 25 then low = nil end
		return current
	end
	while environment.os.clock() - started < duration do
		if token ~= state.token then result = "cancel"; break end
		if not context.isAlive() then result = "unload"; break end
		if type(options.stop) == "function" then
			local ok, stop = pcall(options.stop)
			if ok and stop then result = "stop"; break end
		end
		local target, reason = validTarget()
		if not target then result = reason; break end
		local own = context.player.Character
		local humanoid = own and own:FindFirstChildOfClass("Humanoid")
		if not humanoid or humanoid.Health <= 0 then result = "me"; break end
		if state.active or state.busy then environment.task.wait(0.05); continue end
		attempts, state.active, globals.__pulseManualFling = attempts + 1, true, true
		local ok, error = pcall(FlingFeature.down, context, state, target, tonumber(options.dur) or 0.8)
		if not ok then globals.__pulseFlingLast = "down err " .. tostring(error) end
		globals.__pulseManualFling, globals.__pulseTouchFlinging, state.active = false, false, false
		local finished = environment.os.clock()
		while environment.os.clock() - finished < 0.3 do
			environment.task.wait(0.05)
			if not validTarget() then break end
		end
	end
	if token == state.token then state.target = nil end
	globals.__pulseFlingLog = type(globals.__pulseFlingLog) == "table" and globals.__pulseFlingLog or {}
	table.insert(globals.__pulseFlingLog, string.format("%s persist %s tries=%d t=%.1f res=%s",
		environment.os.date("%H:%M:%S"), player.Name, attempts, environment.os.clock() - started, result))
	while #globals.__pulseFlingLog > 10 do table.remove(globals.__pulseFlingLog, 1) end
	return result == "gone" or result == "dead" or result == "down", result
end

function FlingFeature.player(context, state, player)
	if state.active or state.busy or not player or player == context.player or not player.Parent then return end
	local character = player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not humanoid or humanoid.Health <= 0 then return end
	state.active, context.environment.getgenv().__pulseManualFling = true, true
	context.environment.task.spawn(function()
		local ok, error = pcall(FlingFeature.down, context, state, character, 0.8)
		local globals = context.environment.getgenv()
		if not ok then globals.__pulseFlingLast = "down err " .. tostring(error) end
		globals.__pulseManualFling, globals.__pulseTouchFlinging, state.active = false, false, false
	end)
end

function FlingFeature.installTouch(context)
	local environment = context.environment
	local enabled, busy, bound = false, false, nil
	local function bind(character)
		if not character or character == bound then return end
		bound = character
		for _, part in character:GetChildren() do
			if not part:IsA("BasePart") then continue end
			context.connect(part.Touched, function(hit)
				if not enabled or busy then return end
				local target = hit and hit:FindFirstAncestorWhichIsA("Model")
				local player = target and context.players:GetPlayerFromCharacter(target)
				local fling = rawget(environment.getgenv(), "__pulseFlingDown")
				if not player or player == context.player or type(fling) ~= "function" then return end
				busy = true
				environment.task.spawn(function()
					pcall(fling, target, 0.4, {cam = false})
					busy = false
				end)
			end)
		end
	end
	bind(context.player.Character)
	context.connect(context.player.CharacterAdded, function(character)
		environment.task.wait(0.4)
		bind(character)
	end)
	context.touchSection:Toggle({Text = "Touch Fling", Default = false, Callback = function(value) enabled = value end})
end

function FlingFeature.hump(context, state, target)
	local environment, globals = context.environment, context.environment.getgenv()
	if state.humping then
		state.humping = false
		return
	end

	local character = context.player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	local root = character and character:FindFirstChild("HumanoidRootPart")
	local targetRoot = target:FindFirstChild("HumanoidRootPart")

	if not humanoid or not root or not targetRoot or humanoid.Health <= 0 then return end

	state.humping = true
	globals.__pulseTouchFlinging, globals.__pulseManualFling, globals.__pulseFlingTargetChar = true, true, target

	local seated, animation = true, nil
	pcall(function() seated = humanoid:GetStateEnabled(environment.Enum.HumanoidStateType.Seated) end)
	pcall(function() humanoid:SetStateEnabled(environment.Enum.HumanoidStateType.Seated, false) end)

	pcall(function()
		local asset = environment.Instance.new("Animation")
		asset.AnimationId = "rbxassetid://698251653"
		animation = humanoid:LoadAnimation(asset)
		animation.Looped = false
		animation:Play()
		animation.TimePosition = 0.58
		animation:AdjustSpeed(0.5)
	end)

	environment.task.spawn(function()
		local started = environment.tick()

		while state.humping and state.humpEnabled and target.Parent and targetRoot.Parent and root.Parent and humanoid.Health > 0 do
			if state.humpDuration > 0 and environment.tick() - started >= state.humpDuration then break end

			pcall(function()
				if animation and animation.IsPlaying and animation.TimePosition >= 0.63 then animation.TimePosition = 0.58 end

				local offset = 1.6 + 0.35 * math.sin(environment.tick() * 16)
				local position = (targetRoot.CFrame * environment.CFrame.new(0, 0, offset)).Position
				root.CFrame = environment.CFrame.new(position, targetRoot.Position)
			end)
			context.runService.Heartbeat:Wait()
		end

		if animation then pcall(function() animation:Stop(0.1) end) end
		pcall(function() humanoid:SetStateEnabled(environment.Enum.HumanoidStateType.Seated, seated) end)
		pcall(function() if humanoid.Health > 0 then humanoid:ChangeState(environment.Enum.HumanoidStateType.GettingUp) end end)

		globals.__pulseTouchFlinging, globals.__pulseManualFling, globals.__pulseFlingTargetChar = false, false, nil
		state.humping = false
	end)
end

function FlingFeature.stopOrbit(state)
	state.orbitToken += 1
	state.orbitTarget = nil
end

function FlingFeature.orbit(context, state, target)
	local environment, globals = context.environment, context.environment.getgenv()

	state.orbitToken += 1
	state.orbitTarget = target
	local token = state.orbitToken

	environment.task.spawn(function()
		local character = context.player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		local root = character and character:FindFirstChild("HumanoidRootPart")

		if not character or not humanoid or not root or humanoid.Health <= 0 then return end

		globals.__pulseTouchFlinging, globals.__pulseManualFling = true, true
		local home, height, seated = root.CFrame, context.workspace.FallenPartsDestroyHeight, true
		pcall(function() seated = humanoid:GetStateEnabled(environment.Enum.HumanoidStateType.Seated) end)

		local velocity
		pcall(function()
			context.workspace.FallenPartsDestroyHeight = 0 / 0
			humanoid:SetStateEnabled(environment.Enum.HumanoidStateType.Seated, false)
			velocity = environment.Instance.new("BodyVelocity")
			velocity.Velocity = environment.Vector3.new(0, 0, 0)
			velocity.MaxForce = environment.Vector3.new(9000000000, 9000000000, 9000000000)
			velocity:SetAttribute("PulseMine", true)
			velocity.Parent = root
		end)

		local angle = 0
		while token == state.orbitToken and state.orbitEnabled do
			local targetPart = target:FindFirstChild("Head") or target:FindFirstChild("HumanoidRootPart")
			local targetHumanoid = target:FindFirstChildOfClass("Humanoid")

			if not targetPart or not targetHumanoid or targetHumanoid.Health <= 0 or not root.Parent or humanoid.Health <= 0 then break end

			angle += 0.5
			local center = targetPart.Position + environment.Vector3.new(0, 1.4, 0)
			local position = center + environment.Vector3.new(math.cos(angle) * 1.1, 0, math.sin(angle) * 1.1)

			pcall(function()
				root.CFrame = environment.CFrame.new(position, center)
				root.AssemblyAngularVelocity = environment.Vector3.zero
				root.AssemblyLinearVelocity = environment.Vector3.new(0, -(tonumber(rawget(globals, "__pulseFlingDownV")) or 120000), 0)
			end)
			context.runService.Heartbeat:Wait()
		end

		pcall(function() if velocity then velocity:Destroy() end end)
		pcall(function() humanoid:SetStateEnabled(environment.Enum.HumanoidStateType.Seated, seated) end)

		for _ = 1, 6 do
			if not root.Parent then break end
			pcall(function()
				root.CFrame = home
				for _, part in character:GetChildren() do
					if part:IsA("BasePart") then part.AssemblyLinearVelocity, part.AssemblyAngularVelocity = environment.Vector3.zero, environment.Vector3.zero end
				end
				if humanoid.Health > 0 then humanoid:ChangeState(environment.Enum.HumanoidStateType.GettingUp) end
			end)
			context.runService.Heartbeat:Wait()
		end

		pcall(function() context.workspace.FallenPartsDestroyHeight = height end)
		globals.__pulseTouchFlinging, globals.__pulseManualFling = false, false
	end)
end
