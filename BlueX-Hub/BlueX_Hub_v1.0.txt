-- BlueX Hub v1.0 | Autocontido; somente APIs oficiais do Roblox.
-- INSTALAÇÃO: em uma experiência que você possa editar, cole este arquivo inteiro
-- em um LocalScript de StarterPlayer > StarterPlayerScripts. Um .txt não executa
-- diretamente. Não exige executor, HTTP, loadstring, Drawing, arquivos ou módulos.
--
-- LIMITAÇÃO IMPORTANTE: não há API pública de MM2 confirmada nesta entrega.
-- A UI e o ESP visual funcionam independentemente do contrato abaixo; dados
-- privados ficam desconhecidos. Farm, postura, velocidade, noclip e ações exigem
-- autorização explícita da experiência, sem sondar remotes/armas/GUI de MM2.
-- Contrato OPCIONAL para o desenvolvedor da própria experiência, não uma API MM2:
-- ReplicatedStorage/BlueXIntegrationV1 (Folder) com atributos replicados:
--   ProtocolVersion=1, Enabled=true, PlaceId=game.PlaceId,
--   RoundId="id-único-por-rodada", RoundState="Active"|"Lobby"|"Ended".
-- Cada Player publica um atributo String JSON BlueX_StateV1, por exemplo:
-- {"protocol":1,"roundId":"r1","characterId":"spawn-unico-1","role":"Innocent",
--  "bag":{"count":0,"capacity":50,"sequence":1},
--  "receipt":{"coinId":"coin-1","sequence":1},
--  "permissions":{"fast":true,"underMap":false,"setSpeed":true,
--                 "noclip":false,"reset":false,"combat":false},
--  "combatTarget":false}
-- O servidor deve publicar snapshots/recibos autoritativos APÓS validar coleta.
-- Cada Character tem atributo BlueX_CharacterId único por spawn; characterId no
-- snapshot deve coincidir. Isso rejeita bolsa/cargo antigos durante o respawn.
-- Não publique recibo fictício: é exigido coinId e sequence nova para confirmar
-- uma moeda específica. Ausência de recibo não significa coleta.
-- Moedas: BasePart ou Model com tag CollectionService "BlueXCoin", atributos
-- BlueX_RoundId e BlueX_CoinId (strings). O servidor define capacidade real.
-- UnderMap: cada moeda precisa adicionalmente BlueX_UnderCollectAllowed=true e
-- BlueX_UnderPosition=Vector3 num piso normal acessível abaixo do mapa. Não há
-- teleporte, perfuração do piso nem noclip automático; o motor pode impedir
-- caminhada horizontal. Isso é falha de movimento, nunca coleta simulada.
-- Combate opcional: Tool legítima própria com tag "BlueXAuthorizedMelee",
-- BlueX_RoundId, BlueX_AutoUseAllowed=true e BlueX_Range (0..20 studs).
-- Somente alvos com combatTarget=true e permissão combat são considerados.
-- Usa EquipTool/Activate normais; dano continua exclusivamente a cargo do jogo.
-- Reset opcional: RequestReset (RemoteEvent) filho da integração, implementado
-- pelo servidor da experiência para receber APENAS roundId e validar bolsa,
-- cargo e limite de uma solicitação/rodada. Não é um remote presumido de MM2.
-- Sem esse remote e permissão reset, Auto Reset fica indisponível.
-- Preferências: JSON versionado em atributo LOCAL do Player, só nesta sessão.
-- Cliente Roblox não oferece arquivo local/DataStore. Persistência entre sessões
-- exige implementação própria no servidor; este script não a simula.
-- Farm e noclip sempre desligados em nova execução. Respawn preserva intenção
-- de Farm na sessão; UnderMap nunca escolhe Fast por recuperação.
-- TESTES: consulte BlueX_Hub_v1.0_LEIA-ME.md. Sem teste real em MM2/Android.

local Services = {}

    "HttpService", "ReplicatedStorage", "CollectionService", "PathfindingService"}) do
    Services[name] = game:GetService(name)

    return
end
local LocalPlayer = Services.Players.LocalPlayer

if not PlayerGui then warn("BlueX Hub: PlayerGui indisponível."); return end

-- Protocolo síncrono de substituição: a execução antiga encerra e restaura
-- recursos antes de esta execução criar controladores. X não usa esse caminho.
do
    local old = PlayerGui:FindFirstChild("BlueXHub_Runtime")
    if old then
        local stop = old:FindFirstChild("Shutdown")
        if not (stop and stop:IsA("BindableFunction")) then
            warn("BlueX Hub: instância anterior sem protocolo de limpeza; inicialização cancelada.")
            return
        end
        local ok, result = pcall(function() return stop:Invoke() end)
        if not ok or result ~= true or old.Parent ~= nil then
            warn("BlueX Hub: limpeza anterior não foi confirmada; inicialização cancelada.")
            return
        end
    end
end


-- Diagnostics Manager e Settings Manager: apenas APIs oficiais do cliente Roblox.
local DiagnosticsManager = {}
DiagnosticsManager.__index = DiagnosticsManager

do
	local MAX_ENTRIES = 80
	local MAX_MESSAGE = 220

	local function bounded(value, limit)
		local result = tostring(value)
		result = string.gsub(result, "[%c]", " ")
		if #result > limit then
			result = string.sub(result, 1, limit) .. "..."
		end
		return result
	end

	function DiagnosticsManager.new()
		return setmetatable({ _entries = {}, _last = {}, _total = 0 }, DiagnosticsManager)
	end

	function DiagnosticsManager:Log(level, code, message)
		local now = os.clock()
		local safeLevel = bounded(level, 12)
		local safeCode = bounded(code, 48)
		local safeMessage = bounded(message, MAX_MESSAGE)
		local key = safeLevel .. ":" .. safeCode .. ":" .. safeMessage
		local previous = self._last[key]
		if previous and now - previous < 5 then
			return
		end
		-- A tabela de supressão também é limitada: entradas de mais de 5 s saem.
		for oldKey, at in pairs(self._last) do
			if now - at >= 5 then
				self._last[oldKey] = nil
			end
		end
		self._last[key] = now
		self._total += 1
		table.insert(self._entries, {
			at = now,
			level = safeLevel,
			code = safeCode,
			message = safeMessage,
		})
		if #self._entries > MAX_ENTRIES then
			table.remove(self._entries, 1)
		end
		-- Proteção adicional contra muitas mensagens diferentes no mesmo instante.
		if self._total % MAX_ENTRIES == 0 then
			self._last = { [key] = now }
		end
	end

	function DiagnosticsManager:Snapshot()
		local lines = { string.format("Diagnóstico local | %d evento(s) registrados", self._total) }
		local first = math.max(1, #self._entries - 29)
		for index = first, #self._entries do
			local item = self._entries[index]
			table.insert(lines, string.format("[%.1fs] %s / %s: %s", item.at, item.level, item.code, item.message))
		end
		if #self._entries == 0 then
			table.insert(lines, "Nenhum evento registrado.")
		end
		return table.concat(lines, "\n")
	end

	function DiagnosticsManager:Count()
		return self._total
	end
end

local SettingsManager = {}
SettingsManager.__index = SettingsManager

do
	local VERSION = 1
	local ATTRIBUTE = "BlueXHub_Settings_v1"
	local DEFAULTS = {
		fastSpeed = 24,
		underSpeed = 24,
		manualSpeed = 16,
		farmMode = "Fast",
		autoReset = false,
		autoKill = false,
		espRoles = true,
		espHighlight = true,
		espTracers = true,
		espDistance = true,
		premiumIntro = true,
		visualEffects = true,
		performanceMode = false,
		selectedTab = "Farm",
		windowPosition = { x = 0.5, y = 0.5 },
		bubblePosition = { x = 0.9, y = 0.5 },
		minimized = false,
		noclip = false,
	}
	local BOOLEAN_KEYS = {
		autoReset = true, autoKill = true,
		espRoles = true, espHighlight = true, espTracers = true, espDistance = true,
		premiumIntro = true, visualEffects = true, performanceMode = true,
		minimized = true, noclip = true,
	}
	local SPEED_KEYS = { fastSpeed = true, underSpeed = true, manualSpeed = true }
	local TABS = { Farm = true, ESP = true, Player = true, Config = true }

	local function finite(value)
		return type(value) == "number" and value == value and value > -math.huge and value < math.huge
	end

	local function cloneDefaults()
		local result = {}
		for key, value in pairs(DEFAULTS) do
			if type(value) == "table" then
				result[key] = { x = value.x, y = value.y }
			else
				result[key] = value
			end
		end
		return result
	end

	local function validate(key, value)
		if BOOLEAN_KEYS[key] then
			return type(value) == "boolean", value
		elseif SPEED_KEYS[key] then
			if not finite(value) then
				return false, nil
			end
			return true, math.clamp(math.floor(value + 0.5), 16, 30)
		elseif key == "farmMode" then
			return value == "Fast" or value == "UnderMap", value
		elseif key == "selectedTab" then
			return type(value) == "string" and TABS[value] == true, value
		elseif key == "windowPosition" or key == "bubblePosition" then
			if type(value) ~= "table" or not finite(value.x) or not finite(value.y) then
				return false, nil
			end
			return true, { x = math.clamp(value.x, 0, 1), y = math.clamp(value.y, 0, 1) }
		end
		return false, nil
	end

	local function sameValue(first, second)
		if type(first) == "table" and type(second) == "table" then
			return first.x == second.x and first.y == second.y
		end
		return first == second
	end

	function SettingsManager.new(ctx)
		local self = setmetatable({
			values = cloneDefaults(),
			_player = ctx.player,
			_http = ctx.services.HttpService,
			_diag = ctx.diag,
			_destroyed = false,
			_dirty = false,
			_saveGeneration = 0,
			_saveTask = nil,
		}, SettingsManager)
		local readOK, raw = pcall(function()
			return self._player:GetAttribute(ATTRIBUTE)
		end)
		if not readOK then
			self._diag:Log("WARN", "SETTINGS_READ", "Armazenamento de sessão indisponível; usando preferências padrão.")
		elseif raw ~= nil then
			if type(raw) ~= "string" or #raw > 16384 then
				self._diag:Log("WARN", "SETTINGS_INVALID", "Configuração ausente ou inválida: preferências padrão restauradas.")
			else
				local decodedOK, decoded = pcall(function()
					return self._http:JSONDecode(raw)
				end)
				if not decodedOK or type(decoded) ~= "table" or type(decoded.values) ~= "table" then
					self._diag:Log("WARN", "SETTINGS_INVALID", "JSON de preferências inválido: usando valores padrão.")
				elseif decoded.version ~= VERSION then
					self._diag:Log("WARN", "SETTINGS_VERSION", "Versão de preferências incompatível: usando valores padrão.")
				else
					local rejected = 0
					for key, value in pairs(decoded.values) do
						-- Noclip é uma autorização manual da sessão, nunca uma preferência restaurada.
						if key ~= "noclip" then
							local valid, clean = validate(key, value)
							if valid then
								self.values[key] = clean
							else
								rejected += 1
							end
						end
					end
					self._diag:Log("INFO", "SETTINGS_LOADED", "Preferências recuperadas da memória desta sessão Roblox.")
					if rejected > 0 then
						self._diag:Log("WARN", "SETTINGS_FIELDS", tostring(rejected) .. " campo(s) inválido(s) ignorado(s).")
					end
				end
			end
		end
		self.values.noclip = false
		self._diag:Log("INFO", "SETTINGS_SCOPE", "Salvamento em memória da sessão; persistência entre sessões exige implementação no servidor da experiência.")
		return self
	end

	function SettingsManager:Set(key, value)
		if self._destroyed then
			return false
		end
		local valid, clean = validate(key, value)
		if not valid then
			self._diag:Log("WARN", "SETTINGS_SET", "Preferência inválida ignorada: " .. tostring(key))
			return false
		end
		if sameValue(self.values[key], clean) then
			return false
		end
		self.values[key] = clean
		if key == "noclip" then
			return true
		end
		self._dirty = true
		self._saveGeneration += 1
		local generation = self._saveGeneration
		if self._saveTask then
			pcall(task.cancel, self._saveTask)
		end
		self._saveTask = task.delay(0.4, function()
			if not self._destroyed and self._saveGeneration == generation then
				self._saveTask = nil
				self:Flush()
			end
		end)
		return true
	end

	function SettingsManager:Flush()
		if self._destroyed then
			return false
		end
		if not self._dirty then
			return true
		end
		local serializable = {}
		for key, value in pairs(self.values) do
			if key ~= "noclip" then
				serializable[key] = value
			end
		end
		local ok, err = pcall(function()
			local encoded = self._http:JSONEncode({ version = VERSION, values = serializable })
			self._player:SetAttribute(ATTRIBUTE, encoded)
		end)
		if ok then
			self._dirty = false
		else
			self._diag:Log("WARN", "SETTINGS_WRITE", "Falha ao salvar preferências de sessão: " .. tostring(err))
		end
		return ok
	end

	function SettingsManager:Destroy()
		if self._destroyed then
			return
		end
		self:Flush()
		self._destroyed = true
		self._saveGeneration += 1
		if self._saveTask then
			pcall(task.cancel, self._saveTask)
			self._saveTask = nil
		end
	end
end


-- Adaptador de dados: lê somente o contrato documentado, sem descobrir remotes.
local Integration = {}
do
    Integration.__index = Integration
    local function number(value)
        return type(value) == "number" and value == value and math.abs(value) < math.huge
    end
    local function integer(value)
        return number(value) and value >= 0 and value % 1 == 0
    end
    function Integration.new(ctx)
        return setmetatable({ctx = ctx, cache = setmetatable({}, {__mode = "k"})}, Integration)
    end
    function Integration:Folder()
        local folder = self.ctx.services.ReplicatedStorage:FindFirstChild("BlueXIntegrationV1")
        if folder and folder:IsA("Folder") and folder:GetAttribute("ProtocolVersion") == 1
            and folder:GetAttribute("Enabled") == true
            and folder:GetAttribute("PlaceId") == game.PlaceId then return folder end
        return nil
    end
    function Integration:Round()
        local folder = self:Folder()
        if not folder then return nil, "Unknown" end
        local id, state = folder:GetAttribute("RoundId"), folder:GetAttribute("RoundState")
        if type(id) ~= "string" or #id == 0 or #id > 128 then return nil, "Unknown" end
        if state ~= "Active" and state ~= "Lobby" and state ~= "Ended" then state = "Unknown" end
        return id, state
    end
    function Integration:State(player)
        local roundId = self:Round()
        if not roundId then return nil end
        local raw = player:GetAttribute("BlueX_StateV1")
        if type(raw) ~= "string" or #raw > 16384 then return nil end
        local cached = self.cache[player]
        if not cached or cached.raw ~= raw then
            local ok, decoded = pcall(function() return self.ctx.services.HttpService:JSONDecode(raw) end)
            cached = {raw = raw, value = ok and type(decoded) == "table" and decoded or false}
            self.cache[player] = cached
            if not cached.value then self.ctx.diag:Log("WARN", "state_json", "Snapshot de integração inválido.") end
        end
        local state = cached.value
        if not state or state.protocol ~= 1 or state.roundId ~= roundId then return nil end
        local character = player.Character
        local characterId = character and character:GetAttribute("BlueX_CharacterId")
        if type(characterId) ~= "string" or #characterId == 0 or #characterId > 128
            or state.characterId ~= characterId then return nil end
        return state
    end
    function Integration:Allowed(capability)
        local state = self:State(self.ctx.player)
        return state ~= nil and type(state.permissions) == "table" and state.permissions[capability] == true
    end
    function Integration:Bag()
        local state = self:State(self.ctx.player)
        local bag = state and state.bag
        if type(bag) == "table" and integer(bag.count) and integer(bag.capacity)
            and bag.capacity > 0 and bag.count <= bag.capacity and integer(bag.sequence) then
            return bag.count, bag.capacity, bag.sequence
        end
        return nil, nil, nil
    end
    function Integration:Receipt()
        local state = self:State(self.ctx.player)
        local receipt = state and state.receipt
        if type(receipt) == "table" and type(receipt.coinId) == "string"
            and #receipt.coinId > 0 and #receipt.coinId <= 128 and integer(receipt.sequence) then
            return receipt.coinId, receipt.sequence
        end
        return nil, nil
    end
    function Integration:Reset(roundId)
        local folder = self:Folder()
        local remote = folder and folder:FindFirstChild("RequestReset")
        if not self:Allowed("reset") or not (remote and remote:IsA("RemoteEvent")) then
            return false, "Reset indisponível: requer RequestReset e autorização do servidor."
        end
        -- Solicitação, nunca confirmação de morte/respawn; servidor deve validar.
        local ok, failure = pcall(function() remote:FireServer(roundId) end)
        return ok, ok and "Solicitação de reset enviada; aguardando respawn." or tostring(failure)
    end
    Integration.IsFinite = number
end

local RoleTracker = {}
do
    RoleTracker.__index = RoleTracker
    function RoleTracker.new(ctx) return setmetatable({ctx = ctx}, RoleTracker) end
    function RoleTracker:Get(player)
        local state = self.ctx.integration:State(player)
        local role = state and state.role
        if role == "Murderer" or role == "Sheriff" or role == "Innocent" then return role end
        return "Unknown"
    end
    function RoleTracker:MurdererPositions()
        local positions = {}
        if self:Get(self.ctx.player) == "Murderer" then return positions end
        for _, player in ipairs(self.ctx.services.Players:GetPlayers()) do
            if player ~= self.ctx.player and self:Get(player) == "Murderer" then
                local character = player.Character
                local root = character and character:FindFirstChild("HumanoidRootPart")
                local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                if root and root:IsA("BasePart") and humanoid and humanoid.Health > 0 then
                    table.insert(positions, root.Position)
                end
            end
        end
        return positions
    end
end

-- Coin Tracker: cache limitado, IDs de rodada, duas tentativas, recibos exatos.
local CoinTracker = {}
do
    CoinTracker.__index = CoinTracker
    function CoinTracker.new(ctx)
        return setmetatable({ctx = ctx, coins = {}, attempts = {}, verified = {}, failedUntil = {}, connections = {}, active = false}, CoinTracker)
    end
    function CoinTracker:Start()
        if self.active then return end
        self.active = true
        local service = self.ctx.services.CollectionService
        self.coins = {}
        for _, item in ipairs(service:GetTagged("BlueXCoin")) do self.coins[item] = true end
        table.insert(self.connections, service:GetInstanceAddedSignal("BlueXCoin"):Connect(function(item)
            self.coins[item] = true
        end))
        table.insert(self.connections, service:GetInstanceRemovedSignal("BlueXCoin"):Connect(function(item)
            self.coins[item] = nil
        end))
    end
    function CoinTracker:Stop()
        self.active = false
        for _, connection in ipairs(self.connections) do connection:Disconnect() end
        table.clear(self.connections); table.clear(self.coins)
    end
    function CoinTracker:Round(id)
        if self.round ~= id then
            self.round = id; table.clear(self.attempts); table.clear(self.verified); table.clear(self.failedUntil)
        end
    end
    function CoinTracker:ProgressText()
        local count, capacity = self.ctx.integration:Bag()
        return count and string.format("Moedas: %d/%d", count, capacity) or "Moedas: ?/? (desconhecido)"
    end
    function CoinTracker:Confirmed(id, baseline)
        local coinId, sequence = self.ctx.integration:Receipt()
        if coinId == id and sequence and sequence > baseline then
            self.verified[id] = true
            return true
        end
        return false
    end
    function CoinTracker:Candidates(origin, mode)
        local candidates, seen, records = {}, {}, 0
        for item in pairs(self.coins) do
            if item:IsDescendantOf(workspace) and item:GetAttribute("BlueX_RoundId") == self.round then
                local id = item:GetAttribute("BlueX_CoinId")
                if type(id) == "string" and #id > 0 and #id <= 128 and not seen[id] then
                    seen[id] = true
                    if not self.verified[id] and (self.attempts[id] or 0) < 2
                        and (self.failedUntil[id] or 0) <= os.clock() then
                        local position
                        if mode == "UnderMap" then
                            local under = item:GetAttribute("BlueX_UnderPosition")
                            if item:GetAttribute("BlueX_UnderCollectAllowed") == true and typeof(under) == "Vector3" then position = under end
                        elseif item:IsA("BasePart") then position = item.Position
                        elseif item:IsA("Model") then position = item:GetPivot().Position end
                        if position and Integration.IsFinite(position.X) and Integration.IsFinite(position.Y)
                            and Integration.IsFinite(position.Z) then
                            table.insert(candidates, {id = id, instance = item, position = position, distance = (position - origin).Magnitude})
                        end
                    end
                end
            end
            records += 1
            -- O contrato deve limitar moedas replicadas; evita trabalho sem limite.
            if records >= 2048 then break end
        end
        table.sort(candidates, function(a, b) return a.distance < b.distance end)
        while #candidates > 6 do table.remove(candidates) end
        return candidates
    end
end

-- Player Controller: só altera propriedades mediante permissão explícita.
local PlayerController = {}
do
    PlayerController.__index = PlayerController
    function PlayerController.new(ctx)
        return setmetatable({ctx = ctx, collisions = {}, collisionConnections = {}, owner = nil, noclipActive = false, alive = true}, PlayerController)
    end
    function PlayerController:Character()
        local character = self.ctx.player.Character
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if character and character:IsDescendantOf(workspace) and humanoid and humanoid.Health > 0
            and root and root:IsA("BasePart") and not root.Anchored then return character, humanoid, root end
        return nil, nil, nil
    end
    function PlayerController:StopMovement()
        local _, humanoid, root = self:Character()
        if self.owner and humanoid and root then
            humanoid:MoveTo(root.Position)
            humanoid:Move(Vector3.zero, false)
        end
    end
    function PlayerController:Release()
        self:StopMovement()
        self.owner = nil
        if self.orientation then self.orientation:Destroy(); self.orientation = nil end
        if self.attachment then self.attachment:Destroy(); self.attachment = nil end
        local snapshot = self.posture
        self.posture = nil
        if snapshot and snapshot.humanoid.Parent then
            snapshot.humanoid.AutoRotate = snapshot.autoRotate
            local root = snapshot.root
            if root.Parent then
                -- Restaura só orientação, mantendo posição; sem teleporte.
                root.AssemblyAngularVelocity = Vector3.zero
                root.CFrame = CFrame.new(root.Position) * CFrame.Angles(0, snapshot.yaw, 0)
            end
        end
    end
    function PlayerController:Acquire(mode, speed)
        local _, humanoid, root = self:Character()
        if not humanoid then return false, "Personagem indisponível." end
        if not self.ctx.integration:Allowed("setSpeed") then return false, "Velocidade não autorizada pelo ambiente." end
        if self.owner ~= mode then self:Release() end
        if self.speedHumanoid ~= humanoid then
            self:RestoreSpeed()
            self.speedHumanoid, self.originalSpeed = humanoid, humanoid.WalkSpeed
        end
        if self.owner ~= mode or self.appliedSpeed ~= speed then humanoid.WalkSpeed = speed end
        self.appliedSpeed = speed
        self.appliedManualSpeed = nil
        self.owner = mode
        if mode == "UnderMap" and not self.orientation then
            local _, yaw = root.CFrame:ToOrientation()
            self.posture = {humanoid = humanoid, root = root, autoRotate = humanoid.AutoRotate, yaw = yaw}
            humanoid.AutoRotate = false
            local attachment = Instance.new("Attachment")
            attachment.Name = "BlueX_UnderAttachment"; attachment.Parent = root
            self.attachment = attachment
            local orientation = Instance.new("AlignOrientation")
            orientation.Name = "BlueX_UnderOrientation"
            orientation.Mode = Enum.OrientationAlignmentMode.OneAttachment
            orientation.Attachment0 = attachment
            -- LookVector aponta +Y: face voltada para cima, costas para o chão.
            orientation.CFrame = CFrame.Angles(0, yaw, 0) * CFrame.Angles(math.pi / 2, 0, 0)
            orientation.MaxTorque = 20000; orientation.Responsiveness = 20
            orientation.RigidityEnabled = false; orientation.Parent = root
            self.orientation = orientation
        end
        return true
    end
    function PlayerController:RestoreSpeed()
        if self.speedHumanoid and self.speedHumanoid.Parent and self.originalSpeed then
            self.speedHumanoid.WalkSpeed = self.originalSpeed
        end
        self.speedHumanoid, self.originalSpeed = nil, nil
        self.appliedManualSpeed, self.appliedSpeed = nil, nil
    end
    function PlayerController:ManualSpeed()
        if self.owner then return end
        local _, humanoid = self:Character()
        if not humanoid or not self.ctx.integration:Allowed("setSpeed") then self:RestoreSpeed(); return end
        if self.speedHumanoid ~= humanoid then
            self:RestoreSpeed()
            self.speedHumanoid, self.originalSpeed = humanoid, humanoid.WalkSpeed
        end
        if self.appliedManualSpeed ~= self.ctx.settings.manualSpeed then
            humanoid.WalkSpeed = self.ctx.settings.manualSpeed
            self.appliedManualSpeed = self.ctx.settings.manualSpeed
        end
    end
    function PlayerController:NoclipOff()
        self.noclipActive = false
        for _, connection in pairs(self.collisionConnections) do connection:Disconnect() end
        table.clear(self.collisionConnections)
        if self.descendantConnection then self.descendantConnection:Disconnect(); self.descendantConnection = nil end
        if self.removingConnection then self.removingConnection:Disconnect(); self.removingConnection = nil end
        for part, original in pairs(self.collisions) do
            if part.Parent then part.CanCollide = original end
        end
        table.clear(self.collisions); self.noclipActive = false; self.noclipCharacter = nil
    end
    function PlayerController:ReconcileNoclip()
        local character = self:Character()
        local allowed = self.ctx.settings.noclip and self.ctx.integration:Allowed("noclip") and character ~= nil
        if not allowed then
            if self.noclipActive then self:NoclipOff() end
            if self.ctx.settings.noclip and not self.ctx.integration:Allowed("noclip") then
                self.ctx.diag:Log("INFO", "noclip_unavailable", "Noclip manual sem autorização; colisões preservadas.")
            end
            return
        end
        if self.noclipActive and self.noclipCharacter == character then return end
        self:NoclipOff()
        self.noclipActive, self.noclipCharacter = true, character
        local function track(part)
            if not part:IsA("BasePart") or self.collisions[part] ~= nil then return end
            self.collisions[part] = part.CanCollide
            part.CanCollide = false -- Uma aplicação, sem disputar correções.
            self.collisionConnections[part] = part:GetPropertyChangedSignal("CanCollide"):Connect(function()
                if self.noclipActive and part.CanCollide then
                    -- Mantém a correção recebida e encerra o pedido manual.
                    self.collisions[part] = nil
                    self:NoclipOff()
                    self.ctx.settingsManager:Set("noclip", false)
                    self.ctx.diag:Log("WARN", "noclip_corrected", "Colisão corrigida pelo ambiente; noclip desligado sem insistência.")
                end
            end)
        end
        for _, part in ipairs(character:GetDescendants()) do track(part) end
        self.descendantConnection = character.DescendantAdded:Connect(track)
        self.removingConnection = character.DescendantRemoving:Connect(function(part)
            local original = self.collisions[part]
            if original ~= nil then
                local connection = self.collisionConnections[part]
                if connection then connection:Disconnect(); self.collisionConnections[part] = nil end
                self.collisions[part] = nil
                part.CanCollide = original
            end
        end)
    end
    function PlayerController:Destroy()
        self.alive = false; self:Release(); self:NoclipOff(); self:RestoreSpeed()
    end
end

-- Fast Mode / Under Map Mode: estratégias exclusivas; nenhum fallback implícito.
local FastMode = {name = "Fast", capability = "fast", speedKey = "fastSpeed"}
local UnderMapMode = {name = "UnderMap", capability = "underMap", speedKey = "underSpeed"}
local FarmController = {}
do
    FarmController.__index = FarmController
    function FarmController.new(ctx)
        return setmetatable({ctx = ctx, enabled = false, state = "OFF", token = 0,
            worker = false, alive = true, resetRounds = {}, combatAttempts = {}, status = "Farm desligado."}, FarmController)
    end
    function FarmController:SetState(state, message)
        if self.state ~= state or self.status ~= message then
            self.ctx.diag:Log(state == "ERROR" and "WARN" or "INFO", "farm_" .. state, message)
        end
        self.state, self.status = state, message
    end
    function FarmController:Strategy()
        return self.ctx.settings.farmMode == "UnderMap" and UnderMapMode or FastMode
    end
    function FarmController:Cancel(reason)
        self.token += 1
        local tool = self.activeTool
        self.activeTool = nil
        if tool and tool.Parent then pcall(function() tool:Deactivate() end) end
        self.ctx.playerController:Release()
        self:SetState(self.enabled and "RECOVERING" or "OFF", reason)
    end
    function FarmController:SetEnabled(enabled)
        enabled = enabled == true
        if self.enabled == enabled then return end
        self.enabled = enabled
        self:Cancel(enabled and "Aguardando condições verificadas." or "Farm desligado.")
        if enabled then self.ctx.coins:Start() else self.ctx.coins:Stop(); self.ctx.playerController:ManualSpeed() end
    end
    function FarmController:SetMode(mode)
        if mode ~= "Fast" and mode ~= "UnderMap" then return end
        if self.ctx.settings.farmMode == mode then return end
        self.ctx.settingsManager:Set("farmMode", mode)
        self:Cancel("Modo alterado manualmente para " .. mode .. ".")
        self.nextAttempt = 0
    end
    function FarmController:Valid(token, round, character)
        local currentRound, roundState = self.ctx.integration:Round()
        local mode = self:Strategy()
        return self.alive and self.enabled and self.token == token and currentRound == round
            and roundState == "Active" and self.ctx.player.Character == character
            and self.ctx.integration:Allowed(mode.capability) and self.ctx.integration:Allowed("setSpeed")
            and not self.ctx.idle
    end
    function FarmController:Path(origin, destination, token, round, character)
        local path = self.ctx.services.PathfindingService:CreatePath({AgentRadius = 2, AgentHeight = 5,
            AgentCanJump = true, WaypointSpacing = 4})
        local ok, failure = pcall(function() path:ComputeAsync(origin, destination) end)
        if not self:Valid(token, round, character) then path:Destroy(); return nil, "cancelled" end
        if not ok or path.Status ~= Enum.PathStatus.Success then
            path:Destroy()
            return nil, ok and "Sem caminho navegável." or tostring(failure)
        end
        local points = path:GetWaypoints()
        path:Destroy()
        if #points == 0 then return nil, "Caminho vazio." end
        return points
    end
    function FarmController:RouteCost(points, threats, origin)
        local cost, previous = 0, origin
        for _, point in ipairs(points) do
            local distance = (point.Position - previous).Magnitude
            cost += distance
            for _, threat in ipairs(threats) do
                local segment = point.Position - previous
                local t = segment:Dot(segment) > 0 and math.clamp((threat - previous):Dot(segment) / segment:Dot(segment), 0, 1) or 0
                local separation = (threat - (previous + segment * t)).Magnitude
                if separation < 28 then cost += (28 - separation) * (distance / 8 + 1) end
            end
            previous = point.Position
        end
        return cost
    end
    function FarmController:Select(token, round, character, origin)
        local candidates = self.ctx.coins:Candidates(origin, self.ctx.settings.farmMode)
        local threats, best, bestCost = self.ctx.roles:MurdererPositions(), nil, math.huge
        for index, candidate in ipairs(candidates) do
            if index > 3 then break end -- Máximo de três consultas por seleção.
            local points, failure = self:Path(origin, candidate.position, token, round, character)
            if not self:Valid(token, round, character) then return nil end
            if points then
                local cost = self:RouteCost(points, threats, origin)
                if cost < bestCost then candidate.points = points; best, bestCost = candidate, cost end
            else
                self.ctx.coins.attempts[candidate.id] = (self.ctx.coins.attempts[candidate.id] or 0) + 1
                self.ctx.coins.failedUntil[candidate.id] = os.clock() + 4
                self.ctx.diag:Log("WARN", "coin_path", candidate.id .. ": " .. tostring(failure))
            end
        end
        return best
    end
    function FarmController:Walk(points, token, round, character, coinId, receiptSequence, coinInstance, actionGuard)
        local _, humanoid, root = self.ctx.playerController:Character()
        if not humanoid then return false, "Personagem indisponível." end
        for _, point in ipairs(points) do
            if not self:Valid(token, round, character) then return false, "cancelled" end
            if actionGuard and not actionGuard() then return false, "Condição de combate alterada." end
            if humanoid.Health <= 0 or not root.Parent then return false, "Personagem morto ou removido." end
            if coinId and self.ctx.coins:Confirmed(coinId, receiptSequence) then return true, "receipt" end
            if coinInstance and not coinInstance:IsDescendantOf(workspace) then return false, "Moeda saiu da área replicada; coleta não confirmada." end
            -- Velocidade só muda em Acquire ou por escolha manual na UI;
            -- não reimpõe propriedades corrigidas pelo servidor por waypoint.
            if point.Action == Enum.PathWaypointAction.Jump then humanoid.Jump = true end
            humanoid:MoveTo(point.Position)
            local start, progressAt, lastPosition = os.clock(), os.clock(), root.Position
            local deadline = math.clamp((root.Position - point.Position).Magnitude / math.max(humanoid.WalkSpeed, 1) * 3 + 2, 3, 12)
            while (root.Position - point.Position).Magnitude > 2.5 do
                task.wait(0.12)
                if not self:Valid(token, round, character) then return false, "cancelled" end
                if actionGuard and not actionGuard() then return false, "Condição de combate alterada." end
                if humanoid.Health <= 0 or not root.Parent then return false, "Personagem indisponível." end
                if coinId and self.ctx.coins:Confirmed(coinId, receiptSequence) then return true, "receipt" end
                if coinInstance and not coinInstance:IsDescendantOf(workspace) then return false, "Moeda indisponível; sem presumir coleta." end
                if (root.Position - lastPosition).Magnitude > 0.6 then lastPosition, progressAt = root.Position, os.clock() end
                if os.clock() - progressAt > 2.5 then return false, "Sem progresso por 2,5 segundos." end
                if os.clock() - start > deadline then return false, "Tempo de deslocamento excedido." end
            end
        end
        return true, "arrived"
    end
    function FarmController:CombatAllowed(token, round, character, target, targetCharacter, tool)
        if not self:Valid(token, round, character) or not self.ctx.settings.autoKill
            or not self.ctx.integration:Allowed("combat") or self.ctx.roles:Get(self.ctx.player) ~= "Murderer" then return false end
        local count, capacity = self.ctx.integration:Bag()
        if not count or count ~= capacity then return false end
        local targetState = self.ctx.integration:State(target)
        local targetHumanoid = targetCharacter:FindFirstChildOfClass("Humanoid")
        if target.Character ~= targetCharacter or not targetState or targetState.combatTarget ~= true
            or not targetHumanoid or targetHumanoid.Health <= 0 then return false end
        local backpack = self.ctx.player:FindFirstChildOfClass("Backpack")
        local range = tool:GetAttribute("BlueX_Range")
        return tool:IsA("Tool") and (tool.Parent == character or (backpack ~= nil and tool.Parent == backpack))
            and self.ctx.services.CollectionService:HasTag(tool, "BlueXAuthorizedMelee")
            and tool:GetAttribute("BlueX_RoundId") == round and tool:GetAttribute("BlueX_AutoUseAllowed") == true
            and Integration.IsFinite(range) and range > 0 and range <= 20
    end
    function FarmController:Combat(token, round, character)
        if not self.ctx.integration:Allowed("combat") then return false, "Combate automático não autorizado." end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local root = character:FindFirstChild("HumanoidRootPart")
        local tool, range
        for _, candidate in ipairs(self.ctx.services.CollectionService:GetTagged("BlueXAuthorizedMelee")) do
            local backpack = self.ctx.player:FindFirstChildOfClass("Backpack")
            local attackRange = candidate:GetAttribute("BlueX_Range")
            if candidate:IsA("Tool") and (candidate.Parent == character or (backpack ~= nil and candidate.Parent == backpack))
                and candidate:GetAttribute("BlueX_RoundId") == round and candidate:GetAttribute("BlueX_AutoUseAllowed") == true
                and Integration.IsFinite(attackRange) and attackRange > 0 and attackRange <= 20 then
                tool, range = candidate, attackRange; break
            end
        end
        if not tool then return false, "Nenhuma Tool de combate automático autorizada." end
        local target, nearest = nil, math.huge
        for _, player in ipairs(self.ctx.services.Players:GetPlayers()) do
            local state = self.ctx.integration:State(player)
            local targetCharacter = player.Character
            local targetRoot = targetCharacter and targetCharacter:FindFirstChild("HumanoidRootPart")
            local targetHumanoid = targetCharacter and targetCharacter:FindFirstChildOfClass("Humanoid")
            if player ~= self.ctx.player and state and state.combatTarget == true and targetRoot
                and targetRoot:IsA("BasePart") and targetHumanoid and targetHumanoid.Health > 0
                and (self.combatAttempts[player.UserId] or 0) < 2 then
                local distance = (targetRoot.Position - root.Position).Magnitude
                if distance < nearest then target, nearest = player, distance end
            end
        end
        if not target then return false, "Sem alvo autorizado ou limite de tentativas atingido." end
        self.combatAttempts[target.UserId] = (self.combatAttempts[target.UserId] or 0) + 1
        local targetCharacter = target.Character
        local targetRoot = targetCharacter:FindFirstChild("HumanoidRootPart")
        local points, failure = self:Path(root.Position, targetRoot.Position, token, round, character)
        if not points then return false, failure end
        local function guard() return self:CombatAllowed(token, round, character, target, targetCharacter, tool) end
        if not guard() then return false, "Condição de combate alterada." end
        local strategy = self:Strategy()
        local acquired, acquireFailure = self.ctx.playerController:Acquire(strategy.name, self.ctx.settings[strategy.speedKey])
        if not acquired then return false, acquireFailure end
        local moved, reason = self:Walk(points, token, round, character, nil, nil, nil, guard)
        if not moved then return false, reason end
        if not guard() then return false, "cancelled" end
        range = tool:GetAttribute("BlueX_Range")
        if not targetRoot.Parent or (targetRoot.Position - root.Position).Magnitude > range then return false, "Alvo fora do alcance normal da Tool." end
        humanoid:EquipTool(tool)
        if tool.Parent ~= character or not guard() then return false, "Não foi possível equipar/autorizar a Tool." end
        -- Sem remote de dano; Activate só solicita a interação normal da Tool.
        self.activeTool = tool
        tool:Activate()
        task.wait(0.2)
        if self.activeTool == tool and self.token == token then
            self.activeTool = nil
            if tool.Parent then tool:Deactivate() end
        end
        return true, "Tool ativada normalmente; dano e eliminação não confirmados."
    end
    function FarmController:Work(token, round, character)
        local strategy = self:Strategy()
        local count, capacity = self.ctx.integration:Bag()
        if count and count == capacity then
            local role = self.ctx.roles:Get(self.ctx.player)
            if role == "Murderer" and self.ctx.settings.autoKill then
                self:SetState("RUNNING", "Bolsa cheia: avaliando combate autorizado.")
                local ok, reason = self:Combat(token, round, character)
                if self:Valid(token, round, character) then self:SetState(ok and "WAITING" or "ERROR", tostring(reason)) end
            elseif (role == "Innocent" or role == "Sheriff") and self.ctx.settings.autoReset then
                if self.resetRounds[round] then self:SetState("WAITING", "Reset já solicitado nesta rodada; aguardando nova rodada."); return end
                local ok, reason = self.ctx.integration:Reset(round)
                if ok then self.resetRounds[round] = true; self.awaitingReset = {character = character, at = os.clock(), round = round} end
                self:SetState("WAITING", reason)
            else
                self:SetState("WAITING", role == "Unknown" and "Bolsa cheia; cargo desconhecido, sem ação automática." or "Bolsa cheia; aguardando condições da rodada.")
            end
            return
        end
        local target = self:Select(token, round, character, select(3, self.ctx.playerController:Character()).Position)
        if not self:Valid(token, round, character) then return end
        if not target then
            self:SetState("WAITING", strategy.name == "UnderMap"
                and "Under Map sem rota/moeda autorizada. Sem troca de modo; validação de coleta depende do servidor."
                or "Sem moeda/rota autorizada disponível; limite de duas tentativas por moeda.")
            return
        end
        local acquired, failure = self.ctx.playerController:Acquire(strategy.name, self.ctx.settings[strategy.speedKey])
        if not acquired then self:SetState("WAITING", failure); return end
        self.ctx.coins.attempts[target.id] = (self.ctx.coins.attempts[target.id] or 0) + 1
        local _, baseline = self.ctx.integration:Receipt()
        baseline = baseline or -1
        self:SetState("RUNNING", "Percorrendo rota: " .. target.id)
        local moved, reason = self:Walk(target.points, token, round, character, target.id, baseline, target.instance)
        if not self:Valid(token, round, character) then return end
        self.ctx.playerController:StopMovement()
        if moved then
            local untilAt = os.clock() + 1.25
            repeat
                if self.ctx.coins:Confirmed(target.id, baseline) then
                    self.ctx.diag:Log("INFO", "coin_confirmed", "Recibo de coleta confirmado: " .. target.id)
                    return
                end
                task.wait(0.12)
            until os.clock() >= untilAt or not self:Valid(token, round, character)
            if not self:Valid(token, round, character) then return end
            reason = "Sem recibo autoritativo de coleta; desaparecimento visual ignorado."
        end
        self.ctx.diag:Log("WARN", strategy.name == "UnderMap" and "under_failure" or "move_failure", target.id .. ": " .. tostring(reason))
        self.ctx.coins.failedUntil[target.id] = os.clock() + 8
        self:SetState("RECOVERING", "Tentativa encerrada; avaliando outra moeda no mesmo modo.")
    end
    function FarmController:Tick()
        if not self.enabled or not self.alive then return end
        local round, roundState = self.ctx.integration:Round()
        -- Uma lacuna de replicação NÃO constitui uma rodada nova e não reinicia
        -- os limites de tentativas/reset da última rodada confirmada.
        if round and self.round ~= round then
            self:Cancel("Estado da rodada alterado.")
            self.round = round; self.ctx.coins:Round(round); self.combatAttempts = {}
            self.awaitingReset = nil
            -- Apenas o ID atual é necessário; não cresce a cada rodada.
            self.resetRounds = round and {[round] = self.resetRounds[round]} or {}
        end
        local strategy = self:Strategy()
        local character = self.ctx.playerController:Character()
        local unavailable
        if self.ctx.idle then unavailable = "Inatividade detectada; use Retomar manualmente."
        elseif not round or roundState ~= "Active" then unavailable = "Rodada desconhecida/inativa; integração do jogo necessária."
        elseif not character then unavailable = "Aguardando personagem e respawn."
        elseif not self.ctx.integration:Allowed(strategy.capability) then unavailable = strategy.name .. " não autorizado pelo ambiente."
        elseif not self.ctx.integration:Allowed("setSpeed") then unavailable = "Movimentação sem autorização de velocidade."
        end
        if unavailable then
            if self.ctx.playerController.owner then self:Cancel(unavailable) end
            self:SetState("WAITING", unavailable)
            return
        end
        if self.awaitingReset then
            if self.awaitingReset.character == character then
                self:SetState("WAITING", os.clock() - self.awaitingReset.at > 12
                    and "Reset não confirmado; sem repetir nesta rodada." or "Aguardando confirmação de respawn.")
                return
            end
            self.awaitingReset = nil
        end
        if self.worker or (self.nextAttempt and os.clock() < self.nextAttempt) then return end
        self.worker = true
        local token = self.token
        task.spawn(function()
            local ok, failure = xpcall(function() self:Work(token, round, character) end, debug.traceback)
            if not ok and self.alive then
                self.ctx.diag:Log("ERROR", "farm_exception", tostring(failure))
                if self.token == token then self:SetState("ERROR", "Falha isolada; consultar diagnóstico.") end
            end
            if self.token == token then
                -- Também limpa uma Tool cuja Activate lançou erro, sem deixar
                -- a ação parcialmente inicializada após uma falha isolada.
                local tool = self.activeTool
                self.activeTool = nil
                if tool and tool.Parent then pcall(function() tool:Deactivate() end) end
                self.ctx.playerController:StopMovement()
                -- Mantém a orientação fixa entre moedas do mesmo percurso.
                -- Espera/falha terminal libera o controlador, nunca troca o modo.
                if self.state == "WAITING" or self.state == "ERROR" then
                    self.ctx.playerController:Release(); self.ctx.playerController:ManualSpeed()
                end
            end
            self.worker = false
            self.nextAttempt = os.clock() + (self.state == "ERROR" and 4 or self.state == "WAITING" and 2 or 0.4)
        end)
    end
    function FarmController:Destroy()
        self.alive = false; self.enabled = false; self:Cancel("Controlador encerrado."); self.ctx.coins:Stop()
    end
end

-- Recovery Manager: nenhuma nova thread de Farm em CharacterAdded.
local RecoveryManager = {}
do
    RecoveryManager.__index = RecoveryManager
    function RecoveryManager.new(ctx)
        local self = setmetatable({ctx = ctx, connections = {}}, RecoveryManager)
        table.insert(self.connections, ctx.player.CharacterRemoving:Connect(function()
            ctx.farm:Cancel("Personagem removido; aguardando respawn no modo selecionado.")
            ctx.playerController:NoclipOff(); ctx.playerController:RestoreSpeed()
            ctx.diag:Log("INFO", "respawn_remove", "Referências de movimento e colisão liberadas.")
        end))
        table.insert(self.connections, ctx.player.CharacterAdded:Connect(function()
            ctx.diag:Log("INFO", "respawn_added", "Novo personagem; validação pelo coordenador único.")
            ctx.farm.nextAttempt = 0
        end))
        table.insert(self.connections, ctx.player.Idled:Connect(function(seconds)
            ctx.idle = true
            ctx.farm:Cancel("Inatividade detectada; retomada manual disponível.")
            ctx.diag:Log("INFO", "idle", "Inatividade: " .. tostring(math.floor(seconds)) .. "s; nenhuma entrada simulada.")
        end))
        return self
    end
    function RecoveryManager:Destroy()
        for _, connection in ipairs(self.connections) do connection:Disconnect() end
        table.clear(self.connections)
    end
end


-- UI Manager: interface nativa Roblox, sem bibliotecas ou imagens externas.
local UIManager = {}
do
	local COLORS = {
		background = Color3.fromRGB(8, 11, 18), panel = Color3.fromRGB(14, 20, 31),
		card = Color3.fromRGB(20, 29, 43), accent = Color3.fromRGB(24, 160, 255),
		muted = Color3.fromRGB(139, 158, 184), text = Color3.fromRGB(232, 243, 255),
		line = Color3.fromRGB(29, 70, 107), inactive = Color3.fromRGB(48, 59, 75),
	}
	local function make(class, properties, parent)
		local object = Instance.new(class)
		for key, value in pairs(properties or {}) do object[key] = value end
		object.Parent = parent
		return object
	end
	local function round(parent, radius)
		return make("UICorner", {CornerRadius = UDim.new(0, radius or 9)}, parent)
	end
	local function outline(parent, color, transparency)
		return make("UIStroke", {Color = color or COLORS.line, Thickness = 1,
			Transparency = transparency or 0}, parent)
	end
	local function text(parent, label, size)
		return make("TextLabel", {BackgroundTransparency = 1, Text = label, Font = Enum.Font.Gotham,
			TextSize = size or 13, TextColor3 = COLORS.text, TextXAlignment = Enum.TextXAlignment.Left,
			TextYAlignment = Enum.TextYAlignment.Center, TextWrapped = true}, parent)
	end
	local function button(parent, label, size)
		local result = make("TextButton", {BackgroundColor3 = COLORS.card, BorderSizePixel = 0,
			Text = label, Font = Enum.Font.GothamMedium, TextSize = size or 13,
			TextColor3 = COLORS.text, AutoButtonColor = false}, parent)
		round(result, 8)
		return result
	end
	local function drawIcon(parent, name)
		-- Ícones vetoriais nativos: não dependem de imagens nem de glifos da fonte.
		local canvas = make("Frame", {BackgroundTransparency = 1, Size = UDim2.fromOffset(16, 16),
			AnchorPoint = Vector2.new(0.5, 0.5)}, parent)
		local paints = {}
		local function shape(size, position, radius, border, rotation)
			local object = make("Frame", {Size = UDim2.fromOffset(size.X, size.Y),
				Position = UDim2.fromOffset(position.X, position.Y), AnchorPoint = Vector2.new(0.5, 0.5),
				BorderSizePixel = 0, BackgroundColor3 = COLORS.muted,
				BackgroundTransparency = border and 1 or 0, Rotation = rotation or 0}, canvas)
			round(object, radius)
			if border then
				table.insert(paints, {object = outline(object, COLORS.muted), property = "Color"})
			else
				table.insert(paints, {object = object, property = "BackgroundColor3"})
			end
			return object
		end
		if name == "Farm" then
			shape(Vector2.new(13, 13), Vector2.new(8, 8), 8, true)
			shape(Vector2.new(2, 8), Vector2.new(8, 8), 1)

		local function act(callback)
			if not self.alive then return end
			local ok, failure = pcall(callback)
			if not ok then ctx.diag:Log("ERROR", "ui_action", tostring(failure)) end
		end
		local function animate(control)
			if not self.alive or not settings.visualEffects or settings.performanceMode then return end
			local previous = self.tweens[control]
			if previous then previous:Cancel() end
			control.BackgroundColor3 = COLORS.line
			local target = control == self.Bubble and COLORS.accent or COLORS.card
			local tween = ctx.services.TweenService:Create(control, TweenInfo.new(0.22), {BackgroundColor3 = target})
			self.tweens[control] = tween
			tween:Play()
		end

		local function beginDrag(target, key, input)
			if input.UserInputType ~= Enum.UserInputType.Touch and input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
			local center = target.AbsolutePosition - self.Root.AbsolutePosition + target.AbsoluteSize * 0.5
			self.dragging = {target = target, key = key, input = input, start = Vector2.new(input.Position.X, input.Position.Y),
				center = center, moved = false}
		end
		connect(top.InputBegan, function(input) beginDrag(self.Window, "windowPosition", input) end)
		connect(self.Bubble.InputBegan, function(input) beginDrag(self.Bubble, "bubblePosition", input) end)
		connect(ctx.services.UserInputService.InputChanged, function(input)
			local drag = self.dragging
			if not drag then return end
			local accepted = input == drag.input or (drag.input.UserInputType == Enum.UserInputType.MouseButton1
				and input.UserInputType == Enum.UserInputType.MouseMovement)
			if not accepted then return end
			local delta = Vector2.new(input.Position.X, input.Position.Y) - drag.start
			if delta.Magnitude > 7 then drag.moved = true end
			if not drag.moved then return end
			if drag.target == self.Bubble then self.suppressBubbleUntil = os.clock() + 0.35 end
			local center = self:ClampCenter(drag.target, drag.center + delta)
			drag.target.Position = UDim2.fromOffset(center.X, center.Y)
			if drag.target == self.Window then self.Shadow.Position = UDim2.fromOffset(center.X + 3, center.Y + 6) end
		end)
		connect(ctx.services.UserInputService.InputEnded, function(input)
			local drag = self.dragging
			if not drag or input ~= drag.input then return end
			self.dragging = nil
			if drag.moved then
				local size = self.Root.AbsoluteSize
				local center = drag.target.AbsolutePosition - self.Root.AbsolutePosition + drag.target.AbsoluteSize * 0.5
				act(function() save(drag.key, {x = center.X / math.max(1, size.X), y = center.Y / math.max(1, size.Y)}) end)
			end
		end)

	function UIManager:ClampCenter(target, center)
		local available, half = self.Root.AbsoluteSize, target.AbsoluteSize * 0.5
		return Vector2.new(math.clamp(center.X, half.X + 4, math.max(half.X + 4, available.X - half.X - 4)),
			math.clamp(center.Y, half.Y + 4, math.max(half.Y + 4, available.Y - half.Y - 4)))
	end
	function UIManager:Layout()
		if not self.alive then return end
		local available = self.Root.AbsoluteSize
		if available.X < 1 or available.Y < 1 then return end
		local width, height = math.min(700, math.max(220, available.X - 20)), math.min(510, math.max(180, available.Y - 20))
		self.Window.Size = UDim2.fromOffset(width, height)
		self.Shadow.Size = UDim2.fromOffset(width + 4, height + 4)
		local sideWidth = width < 420 and 66 or 90
		self.Sidebar.Size = UDim2.new(0, sideWidth, 1, -49)
		self.Content.Position = UDim2.fromOffset(sideWidth + 8, 56)
		self.Content.Size = UDim2.new(1, -sideWidth - 16, 1, -64)
		for name, tab in pairs(self.tabs) do
			local narrow = width < 420
			self.tabIcons[name].root.Position = narrow and UDim2.new(0.5, 0, 0, 12) or UDim2.fromOffset(14, 22)
			local label = self.tabLabels[name]
			label.Position = narrow and UDim2.fromOffset(0, 26) or UDim2.fromOffset(27, 0)
			label.Size = narrow and UDim2.new(1, 0, 0, 16) or UDim2.new(1, -29, 1, 0)
			label.TextXAlignment = narrow and Enum.TextXAlignment.Center or Enum.TextXAlignment.Left
		end
		local position = self.ctx.settings.windowPosition or {x = 0.5, y = 0.5}
		local center = self:ClampCenter(self.Window, Vector2.new(position.x * available.X, position.y * available.Y))
		self.Window.Position = UDim2.fromOffset(center.X, center.Y)
		self.Shadow.Position = UDim2.fromOffset(center.X + 3, center.Y + 6)
		local bubblePosition = self.ctx.settings.bubblePosition or {x = 0.9, y = 0.5}
		local bubbleCenter = self:ClampCenter(self.Bubble, Vector2.new(bubblePosition.x * available.X, bubblePosition.y * available.Y))
		self.Bubble.Position = UDim2.fromOffset(bubbleCenter.X, bubbleCenter.Y)
	end
	function UIManager:ApplyVisibility()
		if not self.alive then return end
		local hidden = self.ctx.settings.minimized == true or self.introVisible == true
		self.Window.Visible = not hidden
		self.Shadow.Visible = not hidden and not self.ctx.settings.performanceMode
		self.Bubble.Visible = self.ctx.settings.minimized == true and not self.introVisible
		self:UpdateEffects()
	end
	function UIManager:UpdateEffects()
		local enabled = self.ctx.settings.visualEffects and not self.ctx.settings.performanceMode and self.Window.Visible
		if enabled == self.effectsActive then return end
		self.effectsActive = enabled
		for _, animation in ipairs(self.effectTweens) do animation:Cancel() end
		table.clear(self.effectTweens)
		self.TitleGradient.Enabled = enabled
		self.TitleGradient.Offset = Vector2.new(-0.5, 0)
		self.WindowStroke.Transparency = 0.25
		if not enabled then return end
		-- Dois tweens do motor, suspensos ao ocultar ou desativar efeitos; nenhum loop Lua.
		local tweenService = self.ctx.services.TweenService
		table.insert(self.effectTweens, tweenService:Create(self.TitleGradient,
			TweenInfo.new(2.8, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true),
			{Offset = Vector2.new(0.5, 0)}))
		table.insert(self.effectTweens, tweenService:Create(self.WindowStroke,
			TweenInfo.new(1.8, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true),
			{Transparency = 0.42}))
		for _, animation in ipairs(self.effectTweens) do animation:Play() end
	end
	function UIManager:Reveal()
		if not self.alive then return end
		self.ctx.actions.SetSetting("minimized", false)
		self:ApplyVisibility()
	end
	function UIManager:Refresh(snapshot)
		if not self.alive then return end
		self.snapshot = snapshot or self.snapshot or {}
		local settings = self.ctx.settings
		local mode = settings.farmMode
		self.Selector.Text = (mode == "UnderMap" and "Auto Farm Under Map" or "Auto Farm Fast") .. "  ▾"
		self.FastControl.Visible = mode == "Fast"
		self.UnderControl.Visible = mode == "UnderMap"
		self.ModeDescription.Text = mode == "UnderMap"
			and "Experimental: percurso inferior e postura precisam de autorização. Falhas mantêm Under Map selecionado."
			or "Movimentação normal por caminhos alcançáveis. Velocidade e coleta estão sujeitas ao servidor."
		for _, binding in ipairs(self.bindings) do binding() end
		local selected = self.pages[settings.selectedTab] and settings.selectedTab or "Farm"
		for name, page in pairs(self.pages) do
			page.Visible = name == selected
			local color = name == selected and COLORS.accent or COLORS.muted
			self.tabLabels[name].TextColor3 = color
			for _, paint in ipairs(self.tabIcons[name].paints) do paint.object[paint.property] = color end
		end
		self.Bag.Text = self.snapshot.bagText or "Moedas: desconhecido"
		self.FarmState.Text = "Farm: " .. tostring(self.snapshot.farmState or "OFF")
		self.FarmStatus.Text = tostring(self.snapshot.statusText or "Aguardando dados confiáveis.")
		self.IdleText.Text = self.snapshot.idle and "Inatividade detectada · toque para retomar" or "Ativo · retomada manual disponível"
		if self.pages.Config.Visible and not self.ctx.settings.minimized then
			local ok, report = pcall(function() return self.ctx.diag:Snapshot() end)
			if ok then self.Diagnostics.Text = tostring(report) end
		end
		self:ApplyVisibility()
	end

		logo.Size = UDim2.new(1, -30, 0, 72)
		logo.Position = UDim2.fromScale(0.5, 0.46)
		logo.TextTransparency = 1
		logo.ZIndex = 22
		local glow = make("UIStroke", {Color = COLORS.accent, Thickness = 2.5, Transparency = 1}, logo)
		local subtitle = text(splash, "BLUEX HUB  •  v1.0", 12)
		subtitle.Size = UDim2.new(1, -20, 0, 30)
		subtitle.Position = UDim2.new(0, 10, 0.46, 42)
		subtitle.TextXAlignment = Enum.TextXAlignment.Center
		subtitle.TextColor3 = COLORS.muted
		subtitle.TextTransparency = 1
		subtitle.ZIndex = 22
		local particles = {}
		if self.ctx.settings.visualEffects and not self.ctx.settings.performanceMode then
			for index = 1, 8 do
				local dot = make("Frame", {BackgroundColor3 = COLORS.accent, BackgroundTransparency = 0.7,
					BorderSizePixel = 0, Size = UDim2.fromOffset(index % 2 + 2, index % 2 + 2),
					Position = UDim2.fromScale(0.14 + index * 0.08, 0.3 + (index % 3) * 0.16), ZIndex = 21}, splash)
				round(dot, 8)
				table.insert(particles, dot)
			end
		end
		local function tween(object, duration, properties)
			local animation = self.ctx.services.TweenService:Create(object, TweenInfo.new(duration, Enum.EasingStyle.Quad), properties)
			self.tweens[object] = animation
			animation:Play()
		end
		tween(logo, 0.35, {TextTransparency = 0})
		tween(subtitle, 0.5, {TextTransparency = 0})
		tween(glow, 0.7, {Transparency = 0.42})
		for _, dot in ipairs(particles) do
			tween(dot, 1.6, {Position = dot.Position - UDim2.fromScale(0, 0.12), BackgroundTransparency = 0.96})
		end
		task.delay(1.5, function()
			if not self.alive or not splash.Parent then return end
			-- A janela aparece sob a abertura durante o fade final.
			self.introVisible = false
			self:ApplyVisibility()
			tween(logo, 0.45, {TextTransparency = 1})
			tween(subtitle, 0.45, {TextTransparency = 1})
			tween(glow, 0.45, {Transparency = 1})
			tween(splash, 0.45, {BackgroundTransparency = 1})
		end)
		task.delay(2.0, function()
			if not self.alive then return end
			for object, animation in pairs(self.tweens) do
				if object == splash or object:IsDescendantOf(splash) then
					animation:Cancel()
					self.tweens[object] = nil
				end
			end
			splash:Destroy()
			self.Splash = nil
			self.introVisible = false
			self:ApplyVisibility()
		end)
	end
	function UIManager:Destroy()
		if not self.alive then return end
		self.alive = false
		for _, connection in ipairs(self.connections) do connection:Disconnect() end
		for _, tween in pairs(self.tweens) do tween:Cancel() end
		for _, tween in ipairs(self.effectTweens) do tween:Cancel() end
		table.clear(self.connections)
		table.clear(self.tweens)
		table.clear(self.effectTweens)
		self.dragging = nil
		if self.Gui then self.Gui:Destroy() end
	end
end


-- ESP Controller: APIs visuais nativas; nenhum cargo inferido de armas ou aparência.
local ESPController = {}
do
    ESPController.__index = ESPController
    local COLORS = {
        Murderer = Color3.fromRGB(255, 67, 91),
        Sheriff = Color3.fromRGB(50, 151, 255),
        Innocent = Color3.fromRGB(72, 227, 143),
        Unknown = Color3.fromRGB(157, 166, 181),
    }
    local ROLE_LABELS = {
        Murderer = "Murderer", Sheriff = "Sheriff", Innocent = "Innocent", Unknown = "Desconhecido",
    }
    local function discard(record)
        for _, key in ipairs({"highlight", "billboard", "tracer"}) do
            if record[key] then record[key]:Destroy(); record[key] = nil end
        end
    end
    local function anyEnabled(settings)
        return settings.espRoles or settings.espHighlight or settings.espTracers or settings.espDistance
    end
    function ESPController.new(ctx)
        local self = setmetatable({ctx = ctx, records = {}, elapsed = 0, destroyed = false}, ESPController)
        ctx.pendingESP = self
        local gui = Instance.new("ScreenGui")
        gui.Name = "BlueXHub_ESP"
        gui.ResetOnSpawn = false
        gui.IgnoreGuiInset = true -- Os tracers usam coordenadas de WorldToViewportPoint.
        gui.DisplayOrder = 5
        gui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
        gui.Parent = ctx.gui.Parent
        self.gui = gui
        self:Reconcile()
        return self
    end
    function ESPController:_clear()
        for player, record in pairs(self.records) do
            discard(record)
            self.records[player] = nil
        end
    end
    function ESPController:_record(player, character, adornee)
        local record = self.records[player]
        if record and record.character ~= character then
            discard(record)
            self.records[player] = nil
            record = nil
        end
        if not record then
            record = {character = character}
            self.records[player] = record
        end
        record.adornee = adornee
        return record
    end
    function ESPController:_update()
        if self.destroyed then return end
        local settings, players = self.ctx.settings, self.ctx.services.Players
        local camera = workspace.CurrentCamera
        local myCharacter = self.ctx.player.Character
        local myRoot = myCharacter and myCharacter:FindFirstChild("HumanoidRootPart")
        local live, highlightCount, capped = {}, 0, false
        for _, player in ipairs(players:GetPlayers()) do
            local character = player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            if player ~= self.ctx.player and root and root:IsA("BasePart")
                and character:IsDescendantOf(workspace) and humanoid and humanoid.Health > 0 then
                live[player] = true
                local head = character:FindFirstChild("Head")
                local adornee = head and head:IsA("BasePart") and head or root
                local record = self:_record(player, character, adornee)
                local role = self.ctx.roles:Get(player)
                if not COLORS[role] then role = "Unknown" end
                local color = COLORS[role]
                if settings.espHighlight then
                    highlightCount += 1
                    if highlightCount <= 30 then -- Limite voluntário de custo visual em celulares.
                        if not record.highlight then
                            local highlight = Instance.new("Highlight")
                            highlight.Name = "BlueX_ESP_Highlight"
                            highlight.Adornee = character
                            highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                            highlight.FillTransparency = 0.70
                            highlight.OutlineTransparency = 0.05
                            highlight.Parent = character
                            record.highlight = highlight
                        end
                        record.highlight.FillColor = color
                        record.highlight.OutlineColor = color
                    else
                        capped = true
                        if record.highlight then record.highlight:Destroy(); record.highlight = nil end
                    end
                elseif record.highlight then record.highlight:Destroy(); record.highlight = nil end
                if settings.espRoles or settings.espDistance then
                    if not record.billboard then
                        local billboard = Instance.new("BillboardGui")
                        billboard.Name = "BlueX_ESP_Label"
                        billboard.ResetOnSpawn = false
                        billboard.Size = UDim2.fromOffset(220, 44)
                        billboard.StudsOffsetWorldSpace = Vector3.new(0, 2.7, 0)
                        billboard.AlwaysOnTop = true
                        billboard.MaxDistance = 2000
                        billboard.Parent = self.gui.Parent
                        local label = Instance.new("TextLabel")
                        label.Size = UDim2.fromScale(1, 1)
                        label.BackgroundTransparency = 1
                        label.Font = Enum.Font.GothamMedium
                        label.TextSize = 12
                        label.TextStrokeColor3 = Color3.new(0, 0, 0)
                        label.TextStrokeTransparency = 0.25
                        label.TextWrapped = true
                        label.Parent = billboard
                        record.billboard, record.label = billboard, label
                    end
                    record.billboard.Adornee = adornee
                    local parts = {player.DisplayName}
                    if settings.espRoles then table.insert(parts, ROLE_LABELS[role]) end
                    if settings.espDistance then
                        local distance = myRoot and myRoot:IsA("BasePart")
                            and tostring(math.floor((myRoot.Position - root.Position).Magnitude + 0.5)) .. " studs"
                            or "Distância desconhecida"
                        table.insert(parts, distance)
                    end
                    record.label.Text = table.concat(parts, " · ")
                    record.label.TextColor3 = color
                elseif record.billboard then
                    record.billboard:Destroy(); record.billboard, record.label = nil, nil
                end
                if settings.espTracers then
                    if not record.tracer then
                        local line = Instance.new("Frame")
                        line.Name = "BlueX_Tracer"
                        line.AnchorPoint = Vector2.new(0.5, 0.5)
                        line.BorderSizePixel = 0
                        line.BackgroundTransparency = 0.15
                        line.ZIndex = 1
                        line.Parent = self.gui
                        record.tracer = line
                    end
                    local line = record.tracer
                    line.BackgroundColor3 = color
                    line.Visible = false
                    if camera then
                        local projected, onScreen = camera:WorldToViewportPoint(root.Position)
                        local viewport = camera.ViewportSize
                        if onScreen and projected.Z > 0 and viewport.X > 0 and viewport.Y > 0 then
                            local from = Vector2.new(viewport.X * 0.5, viewport.Y - 8)
                            local target = Vector2.new(projected.X, projected.Y)
                            local delta = target - from
                            line.Position = UDim2.fromOffset((from.X + target.X) * 0.5, (from.Y + target.Y) * 0.5)
                            line.Size = UDim2.fromOffset(delta.Magnitude, settings.performanceMode and 1 or 2)
                            line.Rotation = math.deg(math.atan2(delta.Y, delta.X))
                            line.Visible = true
                        end
                    end
                elseif record.tracer then record.tracer:Destroy(); record.tracer = nil end
            end
        end
        for player, record in pairs(self.records) do
            if not live[player] then discard(record); self.records[player] = nil end
        end
        if capped and not self.capReported then
            self.capReported = true
            self.ctx.diag:Log("INFO", "ESP_CAP", "Highlights limitados a 30 por orçamento visual; rótulos e tracers continuam independentes.")
        elseif not capped then self.capReported = false end
    end
    function ESPController:Reconcile()
        if self.destroyed then return end
        local active = anyEnabled(self.ctx.settings)
        self.gui.Enabled = active
        if not active then
            if self.connection then self.connection:Disconnect(); self.connection = nil end
            self:_clear()
            self.elapsed = 0
            return
        end
        if not self.connection then
            self.connection = self.ctx.services.RunService.Heartbeat:Connect(function(delta)
                self.elapsed += delta
                local interval = self.ctx.settings.performanceMode and 0.25 or 0.10
                if self.elapsed < interval then return end
                self.elapsed = 0
                local ok, err = pcall(function() self:_update() end)
                if not ok and (not self.lastError or os.clock() - self.lastError > 10) then
                    self.lastError = os.clock()
                    self.ctx.diag:Log("ERROR", "ESP_UPDATE", tostring(err))
                end
            end)
        end
        local ok, err = pcall(function() self:_update() end)
        if not ok then self.ctx.diag:Log("ERROR", "ESP_INIT", tostring(err)) end
    end
    function ESPController:Count()
        local count = 0
        for _ in pairs(self.records) do count += 1 end
        return count
    end
    function ESPController:Destroy()
        if self.destroyed then return end
        self.destroyed = true
        if self.connection then self.connection:Disconnect(); self.connection = nil end
        self:_clear()
        if self.gui then self.gui:Destroy() end
        self.ctx.diag:Log("INFO", "ESP_CLEANUP", "Heartbeat e todos os efeitos ESP removidos.")
    end
end


-- Coordenador: um único laço leve, funções isoladas, encerramento determinístico.
local Context = {services = Services, player = LocalPlayer, alive = true, idle = false}
Context.diag = DiagnosticsManager.new()
Context.settingsManager = SettingsManager.new(Context)
Context.settings = Context.settingsManager.values
Context.integration = Integration.new(Context)
Context.roles = RoleTracker.new(Context)
Context.coins = CoinTracker.new(Context)
Context.playerController = PlayerController.new(Context)
Context.farm = FarmController.new(Context) -- Sempre OFF; não é preferência salva.

-- Invoke pode ceder a execução: uma segunda inicialização concorrente deve
-- observar a propriedade já adquirida e sair, sem duplicar a interface.
if PlayerGui:FindFirstChild("BlueXHub_Runtime") then
    Context.settingsManager:Destroy()
    warn("BlueX Hub: outra inicialização já adquiriu o controlador.")
    return
end
local RuntimeMarker = Instance.new("Folder")
RuntimeMarker.Name = "BlueXHub_Runtime"
RuntimeMarker.Parent = PlayerGui
local ShutdownBinding = Instance.new("BindableFunction")
ShutdownBinding.Name = "Shutdown"
ShutdownBinding.Parent = RuntimeMarker

local function Shutdown()
    if not Context.alive then return RuntimeMarker.Parent == nil end
    Context.alive = false
    if Context.loop then pcall(task.cancel, Context.loop); Context.loop = nil end
    local clean = true
    for _, pair in ipairs({
        {"Farm", Context.farm}, {"Recovery", Context.recovery},
        {"ESP", Context.esp or Context.pendingESP}, {"Player", Context.playerController},
        {"UI", Context.ui or Context.pendingUI}, {"Settings", Context.settingsManager},
    }) do
        if pair[2] then
            local ok, failure = pcall(function() pair[2]:Destroy() end)
            if not ok then clean = false; warn("BlueX limpeza " .. pair[1] .. ": " .. tostring(failure)) end
        end
    end
    Context.diag:Log("INFO", "cleanup", "Conexões, personagens, efeitos, tarefas e interface liberados.")
    if clean then RuntimeMarker:Destroy() end
    return clean
end
ShutdownBinding.OnInvoke = Shutdown

-- Acrescenta estado atual ao histórico, sem spammar eventos a cada atualização.
do
    local history = Context.diag.Snapshot
    function Context.diag:Snapshot()
        local roundId, roundState = Context.integration:Round()
        local strategy = Context.farm:Strategy()
        return table.concat({
            "BlueX Hub v1.0 | diagnóstico local",
            "Farm: " .. Context.farm.state .. " | solicitado: " .. tostring(Context.farm.enabled),
            "Modo: " .. strategy.name .. " | velocidade: " .. tostring(Context.settings[strategy.speedKey]),
            "Speed manual: " .. tostring(Context.settings.manualSpeed),
            "Rodada: " .. tostring(roundId or "desconhecida") .. " / " .. roundState,
            Context.coins:ProgressText() .. " | cargo: " .. Context.roles:Get(LocalPlayer),
            "Movimento: " .. tostring(Context.playerController.owner or "manual") .. " | worker: " .. tostring(Context.farm.worker),
            "Estado: " .. Context.farm.status,
            "Integração: " .. (Context.integration:Folder() and "contrato v1 disponível" or "indisponível; MM2 sem API validada"),
            "Armazenamento: atributo local; somente a sessão Roblox atual.",
            "Reset/combate dependem de autorização. Under Map físico não testado.",
            history(self),
        }, "\n")
    end
end

local function RefreshUI()
    if not Context.ui or not Context.alive then return end
    local roundId, roundState = Context.integration:Round()
    Context.ui:Refresh({farmState = Context.farm.state, farmEnabled = Context.farm.enabled,
        mode = Context.settings.farmMode,
        speed = Context.settings[Context.farm:Strategy().speedKey],
        bagText = Context.coins:ProgressText(),
        roundText = tostring(roundId or "desconhecida") .. " / " .. roundState,
        statusText = Context.farm.status, idle = Context.idle})
end

Context.actions = {
    SetSetting = function(key, value)
        if not Context.alive then return end
        -- Mudança de modo sempre passa pelo controlador exclusivo.
        if key == "farmMode" then Context.farm:SetMode(value); RefreshUI(); return end
        if not Context.settingsManager:Set(key, value) then return end
        if key == "noclip" then Context.playerController:ReconcileNoclip()
        elseif key == "manualSpeed" then Context.playerController:ManualSpeed()
        elseif key == Context.farm:Strategy().speedKey and Context.playerController.owner then
            Context.playerController:Acquire(Context.farm:Strategy().name, value)
        end
        if Context.esp and (string.sub(key, 1, 3) == "esp" or key == "performanceMode") then Context.esp:Reconcile() end
        if Context.ui and (key == "windowPosition" or key == "bubblePosition") then Context.ui:Layout() end
        RefreshUI()
    end,
    SetFarmEnabled = function(enabled)
        if not Context.alive then return end
        Context.farm:SetEnabled(enabled); Context.farm.nextAttempt = 0
        RefreshUI()
    end,
    SetMode = function(mode)
        if not Context.alive then return end
        Context.farm:SetMode(mode); RefreshUI()
    end,
    ManualResume = function()
        if not Context.alive then return end
        Context.idle = false; Context.farm.nextAttempt = 0
        Context.diag:Log("INFO", "manual_resume", "Retomada escolhida pelo usuário; sem simulação de entrada.")
        RefreshUI()
    end,
    Shutdown = Shutdown,
}

local initialized, initFailure = xpcall(function()
    Context.ui = UIManager.new(Context)
    Context.gui = Context.ui.Gui
    Context.esp = ESPController.new(Context)
    Context.recovery = RecoveryManager.new(Context)
    Context.playerController:ManualSpeed()
    Context.diag:Log("INFO", "startup", "Inicialização única; Farm OFF; noclip OFF; preferências validadas.")
    if not Context.integration:Folder() then
        Context.diag:Log("INFO", "game_integration", "MM2 não validado. Sem contrato autorizado: cargos/bolsa desconhecidos; automações de jogo indisponíveis.")
    end
    Context.diag:Log("INFO", "unverified", "Interface, física, coleta e combate não executados no Roblox/Android nesta entrega.")
    RefreshUI()
end, debug.traceback)
if not initialized then
    Context.diag:Log("ERROR", "initialization", tostring(initFailure))
    warn("BlueX Hub: inicialização falhou: " .. tostring(initFailure))
    Shutdown()
    return
end

Context.loop = task.spawn(function()
    while Context.alive do
        local ok, failure = xpcall(function()
            -- A lógica de cada função desligada não executa seu controlador.
            if Context.farm.enabled then Context.farm:Tick() end
            if Context.settings.noclip or Context.playerController.noclipActive then Context.playerController:ReconcileNoclip() end
            if not Context.playerController.owner then Context.playerController:ManualSpeed() end
            if not Context.settings.minimized then RefreshUI() end
        end, debug.traceback)
        if not ok then Context.diag:Log("ERROR", "coordinator", tostring(failure)) end
        task.wait(Context.settings.performanceMode and 0.5 or (Context.farm.enabled and 0.25 or 0.5))
    end
end)
